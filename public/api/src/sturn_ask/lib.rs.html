1//! Asks — pending *responses* folded out of the WAL. Claims are
2//! pending ownership; asks are pending answers. Pure functions over
3//! the event log: same log, same state, deterministic everywhere.
4//!
5//! An `Ask` event opens a question under its own event id (the blob
6//! sha). `Answer` and `Cancel` close it — first terminal op wins.
7//! `ttl_secs == 0` means the ask never expires on its own.
8
9use std::collections::HashMap;
10use sturn_core::{Event, Op};
11
12/// An ask recorded in the WAL.
13#[derive(Debug, Clone)]
14pub struct Open {
15    /// The Ask event's id — what answers reference.
16    pub id: String,
17    /// Workspace that asked.
18    pub ws: String,
19    pub actor: String,
20    pub subject: String,
21    pub target: String,
22    pub options: Vec<String>,
23    /// Unix seconds the ask was made.
24    pub asked: u64,
25    /// Unix seconds past which the ask is gone — None = no expiry.
26    pub expires: Option<u64>,
27}
28
29/// Terminal state of an ask.
30#[derive(Debug, Clone, PartialEq, Eq)]
31pub enum Status {
32    /// No terminal op yet and ttl hasn't run out.
33    Open,
34    /// An `Answer` landed — carries the verdict + who gave it.
35    Answered { verdict: String, by: String },
36    /// A `Cancel` withdrew it.
37    Cancelled,
38    /// Its ttl ran out unanswered — expiry is derived, not an op.
39    Expired,
40}
41
42enum Fold {
43    Live,
44    Done(Status),
45}
46
47/// Fold `events` into `(ask, status)` pairs in append order at
48/// `now`. Cross-workspace by nature — answers can land on any ws's
49/// WAL, so callers fold *all* workspaces' events. `Answer`/`Cancel`
50/// for an unknown or already-closed ask is ignored.
51fn fold(events: &[Event], now: u64) -> Vec<(Open, Status)> {
52    let mut asks: HashMap<String, (Open, Fold)> = HashMap::new();
53    let mut order: Vec<String> = vec![];
54    for ev in events {
55        match &ev.op {
56            Op::Ask {
57                subject,
58                target,
59                options,
60                ttl_secs,
61            } => {
62                asks.entry(ev.id.clone()).or_insert_with(|| {
63                    order.push(ev.id.clone());
64                    (
65                        Open {
66                            id: ev.id.clone(),
67                            ws: ev.ws.clone(),
68                            actor: ev.actor.clone(),
69                            subject: subject.clone(),
70                            target: target.clone(),
71                            options: options.clone(),
72                            asked: ev.ts,
73                            expires: if *ttl_secs == 0 {
74                                None
75                            } else {
76                                Some(ev.ts + ttl_secs)
77                            },
78                        },
79                        Fold::Live,
80                    )
81                });
82            }
83            Op::Answer { ask, verdict } => {
84                if let Some((_, f)) = asks.get_mut(ask) {
85                    if matches!(f, Fold::Live) {
86                        *f = Fold::Done(Status::Answered {
87                            verdict: verdict.clone(),
88                            by: ev.actor.clone(),
89                        });
90                    }
91                }
92            }
93            Op::Cancel { ask } => {
94                if let Some((_, f)) = asks.get_mut(ask) {
95                    if matches!(f, Fold::Live) {
96                        *f = Fold::Done(Status::Cancelled);
97                    }
98                }
99            }
100            _ => {}
101        }
102    }
103    order
104        .into_iter()
105        .filter_map(|id| {
106            let (o, f) = asks.remove(&id)?;
107            let st = match f {
108                Fold::Done(s) => s,
109                Fold::Live => match o.expires {
110                    Some(e) if e <= now => Status::Expired,
111                    _ => Status::Open,
112                },
113            };
114            Some((o, st))
115        })
116        .collect()
117}
118
119/// Every ask and its state, oldest first.
120pub fn all(events: &[Event], now: u64) -> Vec<(Open, Status)> {
121    fold(events, now)
122}
123
124/// Asks still open at `now` — everything a responder could pick up.
125pub fn open(events: &[Event], now: u64) -> Vec<Open> {
126    fold(events, now)
127        .into_iter()
128        .filter(|(_, s)| matches!(s, Status::Open))
129        .map(|(o, _)| o)
130        .collect()
131}
132
133/// One ask's state — None if the id never was an Ask.
134pub fn status(events: &[Event], now: u64, id: &str) -> Option<Status> {
135    fold(events, now)
136        .into_iter()
137        .find(|(o, _)| o.id == id)
138        .map(|(_, s)| s)
139}
