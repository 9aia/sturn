1//! Vcs port — the host-side VCS operations sturn still needs.
2//! Remotes stay the backend's transport; the WAL rides along on
3//! push. Impl: `sturn-git`.
4//!
5//! The port speaks git's vocabulary (refs, sha, commit) as the
6//! contract — a non-git impl maps equivalent concepts (bookmark,
7//! change-id, object id) onto the same semantics.
8
9use std::path::Path;
10
11/// All methods return bool success — callers decide whether a
12/// failure is retryable (offline remote) or fatal.
13pub trait Vcs: Send + Sync {
14    /// Fetch + prune so `remote`'s refs are current.
15    fn sync(&self, repo: &Path, remote: &str) -> bool;
16    /// Does the object exist locally?
17    fn has_commit(&self, repo: &Path, sha: &str) -> bool;
18    /// Resolve a ref/revision to an object id — `None` when
19    /// unresolvable.
20    fn resolve(&self, repo: &Path, rev: &str) -> Option<String>;
21    /// Publish local branches and the coordination namespace —
22    /// state replicates for free on the same remote (git impl:
23    /// heads + `refs/sturn/*`).
24    fn push(&self, repo: &Path, remote: &str) -> bool;
25}
