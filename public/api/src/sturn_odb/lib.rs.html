1//! Direct git ODB access — loose-object write/read, ref resolution,
2//! tree lookup — so hot paths (`wal.append`, `ws.store`, `ws.read`)
3//! skip `git` process spawns entirely.
4//!
5//! Scope is deliberately narrow: loose objects and refs only. Anything
6//! the fast path can't see — packed objects, alternates, unusual revs,
7//! corrupt files — reports `None`/errs and the caller falls back to
8//! plumbing, so on-disk semantics stay identical to spawning `git`.
9
10use std::fs;
11use std::io::{self, Read, Write};
12use std::path::{Path, PathBuf};
13use std::sync::atomic::{AtomicU64, Ordering};

15/// `.git` dir for `repo` — a plain repo, a linked worktree (`.git`
16/// file), or a bare repo (has HEAD + objects/). Does *not* walk up
17/// to parents or honor `GIT_DIR`; callers fall back to plumbing.
18pub fn git_dir(repo: &Path) -> io::Result<PathBuf> {
19    let dot = repo.join(".git");
20    if dot.is_dir() {
21        return Ok(dot);
22    }
23    if dot.is_file() {
24        let s = fs::read_to_string(&dot)?;
25        let target = s
26            .trim()
27            .strip_prefix("gitdir:")
28            .map(str::trim)
29            .ok_or_else(|| io::Error::other("bad .git file"))?;
30        let p = PathBuf::from(target);
31        return Ok(if p.is_absolute() { p } else { repo.join(p) });
32    }
33    if repo.join("HEAD").exists() && repo.join("objects").is_dir() {
34        return Ok(repo.to_path_buf());
35    }
36    Err(io::Error::other(format!(
37        "{}: not a git repo",
38        repo.display()
39    )))
40}

42/// Dir holding shared objects/refs — follows `commondir` for linked
43/// worktrees, else the gitdir itself.
44pub fn common_dir(gitdir: &Path) -> PathBuf {
45    match fs::read_to_string(gitdir.join("commondir")) {
46        Ok(s) => {
47            let p = PathBuf::from(s.trim());
48            if p.is_absolute() {
49                p
50            } else {
51                gitdir.join(p)
52            }
53        }
54        Err(_) => gitdir.to_path_buf(),
55    }
56}

58fn objects_dir(repo: &Path) -> io::Result<PathBuf> {
59    Ok(common_dir(&git_dir(repo)?).join("objects"))

62/// Fast paths assume SHA-1 ids. On a sha256 repo
63/// (`extensions.objectformat`) every entry point must defer to
64/// plumbing — a sha1-written object is unresolvable there.
65fn sha1_repo(repo: &Path) -> io::Result<bool> {
66    let cfg = match fs::read_to_string(common_dir(&git_dir(repo)?).join("config")) {
67        Ok(c) => c,
68        Err(e) if e.kind() == io::ErrorKind::NotFound => return Ok(true),
69        Err(e) => return Err(e),
70    };
71    let mut ext = false;
72    for line in cfg.lines() {
73        let l = line.trim();
74        if l.starts_with('[') {
75            ext = l.trim_matches(|c| c == '[' || c == ']').eq_ignore_ascii_case("extensions");
76            continue;
77        }
78        if ext {
79            if let Some((k, v)) = l.split_once('=') {
80                if k.trim().eq_ignore_ascii_case("objectformat") {
81                    return Ok(v.trim().eq_ignore_ascii_case("sha1"));
82                }
83            }
84        }
85    }
86    Ok(true)
87}

89fn hex(bytes: &[u8]) -> String {
90    let mut s = String::with_capacity(bytes.len() * 2);
91    for b in bytes {
92        s.push_str(&format!("{b:02x}"));
93    }
94    s
95}

97fn unhex(s: &str) -> Option<Vec<u8>> {
98    if !s.len().is_multiple_of(2) || !s.bytes().all(|b| b.is_ascii_hexdigit()) {
99        return None;
100    }
101    (0..s.len())
102        .step_by(2)
103        .map(|i| u8::from_str_radix(&s[i..i + 2], 16).ok())
104        .collect()
105}

107/// Content address: sha1("<kind> <len>\0<payload>") — the same id
108/// `git hash-object` computes.
109pub fn object_id(kind: &str, payload: &[u8]) -> String {
110    use sha1::Digest;
111    let mut h = sha1::Sha1::new();
112    h.update(kind.as_bytes());
113    h.update(b" ");
114    h.update(payload.len().to_string().as_bytes());
115    h.update(b"\0");
116    h.update(payload);
117    hex(&h.finalize())
118}

120fn deflate(raw: &[u8]) -> Vec<u8> {
121    let mut e = flate2::write::ZlibEncoder::new(Vec::new(), flate2::Compression::fast());
122    e.write_all(raw).expect("zlib into Vec cannot fail");
123    e.finish().expect("zlib into Vec cannot fail")
124}

126fn inflate(z: &[u8]) -> io::Result<Vec<u8>> {
127    let mut out = Vec::new();
128    flate2::read::ZlibDecoder::new(z).read_to_end(&mut out)?;
129    Ok(out)
130}

132/// Write `payload` as a loose object; returns its sha. A `Ok` sha is
133/// identical to `git hash-object -w`. Errors mean the ODB couldn't be
134/// found or written — callers may retry via plumbing.
135pub fn write_object(repo: &Path, kind: &str, payload: &[u8]) -> io::Result<String> {
136    if !sha1_repo(repo)? {
137        return Err(io::Error::other("sha256 odb — use plumbing"));
138    }
139    let objects = objects_dir(repo)?;
140    let sha = object_id(kind, payload);
141    let dir = objects.join(&sha[..2]);
142    let dst = dir.join(&sha[2..]);
143    if dst.exists() {
144        return Ok(sha); // content-addressed: already stored
145    }
146    fs::create_dir_all(&dir)?;
147    let mut raw = Vec::with_capacity(payload.len() + 32);
148    raw.extend_from_slice(kind.as_bytes());
149    raw.push(b' ');
150    raw.extend_from_slice(payload.len().to_string().as_bytes());
151    raw.push(0);
152    raw.extend_from_slice(payload);
153    let z = deflate(&raw);
154    // tmp + rename: readers never see a partial object, and racing
155    // writers of the same content write identical bytes anyway. No
156    // fsync — matches git, which doesn't fsync loose objects unless
157    // core.fsyncObjectFiles is set.
158    static N: AtomicU64 = AtomicU64::new(0);
159    let tmp = dir.join(format!(
160        "tmp_obj_{}_{}",
161        std::process::id(),
162        N.fetch_add(1, Ordering::Relaxed)
163    ));
164    match fs::write(&tmp, &z).and_then(|()| fs::rename(&tmp, &dst)) {
165        Ok(()) => Ok(sha),
166        Err(e) => {
167            let _ = fs::remove_file(&tmp);
168            // Lost a same-sha race — the object landed regardless.
169            if dst.exists() {
170                Ok(sha)
171            } else {
172                Err(e)
173            }
174        }
175    }
176}

178/// Loose-object existence check only — packs and alternates are the
179/// caller's fallback (`git cat-file -e`).
180pub fn object_exists(repo: &Path, sha: &str) -> bool {
181    if !sha1_repo(repo).unwrap_or(false) {
182        return false;
183    }
184    let Ok(objects) = objects_dir(repo) else {
185        return false;
186    };
187    sha.len() == 40
188        && sha.bytes().all(|b| b.is_ascii_hexdigit())
189        && objects.join(&sha[..2]).join(&sha[2..]).is_file()
190}

192/// Read a loose object → (kind, payload). `None` when the object
193/// isn't a loose file (packed, alternate, or absent) — fall back to
194/// `git cat-file`.
195pub fn read_object(repo: &Path, sha: &str) -> io::Result<Option<(String, Vec<u8>)>> {
196    if !sha1_repo(repo)? || sha.len() != 40 || !sha.bytes().all(|b| b.is_ascii_hexdigit()) {
197        return Ok(None);
198    }
199    let objects = objects_dir(repo)?;
200    let z = match fs::read(objects.join(&sha[..2]).join(&sha[2..])) {
201        Ok(z) => z,
202        Err(e) if e.kind() == io::ErrorKind::NotFound => return Ok(None),
203        Err(e) => return Err(e),
204    };
205    let raw = inflate(&z)?;
206    let nul = raw
207        .iter()
208        .position(|&b| b == 0)
209        .ok_or_else(|| io::Error::other("corrupt loose object"))?;
210    let hdr = std::str::from_utf8(&raw[..nul])
211        .map_err(|_| io::Error::other("corrupt loose object header"))?;
212    let (kind, _len) = hdr
213        .split_once(' ')
214        .ok_or_else(|| io::Error::other("corrupt loose object header"))?;
215    Ok(Some((kind.to_string(), raw[nul + 1..].to_vec())))
216}

218fn loose_ref(dir: &Path, refname: &str, depth: u8) -> io::Result<Option<String>> {
219    if depth > 4 {
220        return Ok(None);
221    }
222    match fs::read_to_string(dir.join(refname)) {
223        Err(e) if e.kind() == io::ErrorKind::NotFound => Ok(None),
224        Err(e) => Err(e),
225        Ok(s) => {
226            let s = s.trim();
227            if let Some(target) = s.strip_prefix("ref:") {
228                return loose_ref(dir, target.trim(), depth + 1);
229            }
230            Ok(Some(s.to_string()))
231        }
232    }
233}

235fn packed_ref(common: &Path, refname: &str) -> io::Result<Option<String>> {
236    let s = match fs::read_to_string(common.join("packed-refs")) {
237        Ok(s) => s,
238        Err(e) if e.kind() == io::ErrorKind::NotFound => return Ok(None),
239        Err(e) => return Err(e),
240    };
241    for line in s.lines() {
242        if line.starts_with('#') || line.starts_with('^') || line.is_empty() {
243            continue;
244        }
245        if let Some((sha, r)) = line.split_once(' ') {
246            if r == refname {
247                return Ok(Some(sha.to_string()));
248            }
249        }
250    }
251    Ok(None)
252}

254/// Resolve a ref to its sha — loose ref beats packed, worktree gitdir
255/// beats commondir. `None` = the ref does not exist.
256pub fn read_ref(repo: &Path, refname: &str) -> io::Result<Option<String>> {
257    let gitdir = git_dir(repo)?;
258    let common = common_dir(&gitdir);
259    if let Some(v) = loose_ref(&gitdir, refname, 0)? {
260        return Ok(Some(v));
261    }
262    if common != gitdir {
263        if let Some(v) = loose_ref(&common, refname, 0)? {
264            return Ok(Some(v));
265        }
266    }
267    packed_ref(&common, refname)
268}

270/// `rev-parse`-lite: a full sha, or a ref under the standard prefixes
271/// (as given, `refs/*`, heads, tags). `None` → caller falls back to
272/// `git rev-parse` for the long tail of rev syntax.
273pub fn resolve(repo: &Path, rev: &str) -> io::Result<Option<String>> {
274    if !sha1_repo(repo)? {
275        return Ok(None);
276    }
277    if rev.len() == 40 && rev.bytes().all(|b| b.is_ascii_hexdigit()) {
278        return Ok(Some(rev.to_string()));
279    }
280    for cand in [
281        rev.to_string(),
282        format!("refs/{rev}"),
283        format!("refs/heads/{rev}"),
284        format!("refs/tags/{rev}"),
285    ] {
286        if let Some(sha) = read_ref(repo, &cand)? {
287            return Ok(Some(sha));
288        }
289    }
290    Ok(None)
291}

293/// Build a tree object's payload — entries are (mode, name, sha).
294/// Sorted per git rules (dirs sort as `name/`); `None` on a bad sha.
295pub fn tree_object(entries: &[(u32, &str, &str)]) -> Option<Vec<u8>> {
296    let mut bufs: Vec<(String, Vec<u8>)> = Vec::with_capacity(entries.len());
297    for (mode, name, sha) in entries {
298        let raw = unhex(sha)?;
299        if raw.len() != 20 {
300            return None;
301        }
302        let mut line = Vec::with_capacity(name.len() + 28);
303        line.extend_from_slice(format!("{mode:o}").as_bytes());
304        line.push(b' ');
305        line.extend_from_slice(name.as_bytes());
306        line.push(0);
307        line.extend_from_slice(&raw);
308        let key = if *mode == 0o40000 {
309            format!("{name}/")
310        } else {
311            (*name).to_string()
312        };
313        bufs.push((key, line));
314    }
315    bufs.sort_by(|a, b| a.0.cmp(&b.0));
316    let mut out = Vec::new();
317    for (_, line) in bufs {
318        out.extend_from_slice(&line);
319    }
320    Some(out)
321}

323/// One resolved ident line: `"Name <email>"` plus the local tz.
324type Ident = (String, String);
325
326/// Author/committer as ("Name <email>", tz) pairs — from `git var -l`,
327/// which resolves env (`GIT_AUTHOR_*`) and config through the same
328/// machinery `commit-tree` uses. Cached per gitdir: config/ident
329/// changing mid-process is out of scope.
330fn idents(repo: &Path) -> io::Result<Option<(Ident, Ident)>> {
331    use std::process::Command;
332    use std::sync::Mutex;
333    // Explicit date envs accept many formats — defer to plumbing.
334    if std::env::var_os("GIT_AUTHOR_DATE").is_some()
335        || std::env::var_os("GIT_COMMITTER_DATE").is_some()
336    {
337        return Ok(None);
338    }
339    static CACHE: Mutex<Option<std::collections::HashMap<PathBuf, (Ident, Ident)>>> =
340        Mutex::new(None);
341    let gitdir = git_dir(repo)?;
342    if let Some(hit) = CACHE
343        .lock()
344        .unwrap()
345        .get_or_insert_with(Default::default)
346        .get(&gitdir)
347    {
348        return Ok(Some(hit.clone()));
349    }
350    let o = Command::new("git")
351        .arg("-C")
352        .arg(repo)
353        .args(["var", "-l"])
354        .output()?;
355    if !o.status.success() {
356        return Ok(None); // ident unset — let commit-tree say so
357    }
358    let out = String::from_utf8_lossy(&o.stdout);
359    let parse = |key: &str| -> Option<(String, String)> {
360        let line = out.lines().find_map(|l| l.strip_prefix(key)?.strip_prefix('='))?;
361        // "Name <email> <ts> <tz>" → ("Name <email>", "<tz>")
362        let (head, tz) = line.rsplit_once(' ')?;
363        let (ident, _ts) = head.rsplit_once(' ')?;
364        Some((ident.to_string(), tz.to_string()))
365    };
366    let pair = match (parse("GIT_AUTHOR_IDENT"), parse("GIT_COMMITTER_IDENT")) {
367        (Some(a), Some(c)) => (a, c),
368        _ => return Ok(None),
369    };
370    CACHE
371        .lock()
372        .unwrap()
373        .get_or_insert_with(Default::default)
374        .insert(gitdir, pair.clone());
375    Ok(Some(pair))
376}

378/// Build a commit object (`git commit-tree` equivalent) and store
379/// it — the only difference is no spawn. `None` → fall back to
380/// `git commit-tree` (ident unresolvable or date envs set).
381pub fn commit(repo: &Path, tree: &str, parents: &[&str], message: &str) -> io::Result<String> {
382    let Some(((a_ident, a_tz), (c_ident, c_tz))) = idents(repo)? else {
383        return Err(io::Error::other("ident unavailable — use plumbing"));
384    };
385    let ts = std::time::SystemTime::now()
386        .duration_since(std::time::UNIX_EPOCH)
387        .map(|d| d.as_secs())
388        .unwrap_or(0);
389    let mut payload = format!("tree {tree}\n").into_bytes();
390    for p in parents {
391        payload.extend_from_slice(format!("parent {p}\n").as_bytes());
392    }
393    payload.extend_from_slice(format!("author {a_ident} {ts} {a_tz}\n").as_bytes());
394    payload.extend_from_slice(format!("committer {c_ident} {ts} {c_tz}\n").as_bytes());
395    payload.push(b'\n');
396    // `commit-tree` stripspaces the message — sturn messages are
397    // single-line tokens, so appending raw text is equivalent.
398    payload.extend_from_slice(message.as_bytes());
399    payload.push(b'\n');
400    write_object(repo, "commit", &payload)
401}

403fn commit_tree(data: &[u8]) -> Option<String> {
404    let line = data.split(|&b| b == b'\n').next()?;
405    let hex = std::str::from_utf8(line.strip_prefix(b"tree ")?).ok()?;
406    Some(hex.trim().to_string())
407}

409/// One entry of a decoded tree object — (mode, name, sha-hex).
410fn tree_entry<'a>(tree: &'a [u8], pos: &mut usize) -> Option<(u32, &'a [u8], String)> {
411    let rest = &tree[*pos..];
412    let sp = rest.iter().position(|&b| b == b' ')?;
413    let mode = std::str::from_utf8(&rest[..sp]).ok()?;
414    let mode = u32::from_str_radix(mode, 8).ok()?;
415    let rest = &rest[sp + 1..];
416    let nul = rest.iter().position(|&b| b == 0)?;
417    let name = &rest[..nul];
418    let raw = rest.get(nul + 1..nul + 21)?;
419    *pos = tree.len() - rest.len() + nul + 21;
420    Some((mode, name, hex(raw)))
421}

423/// `<rev>:<path>` walk → entry sha. Loose objects only; `None` when
424/// anything isn't loose, the path is absent, or rev syntax is fancy.
425fn entry_at(repo: &Path, rev: &str, path: &str) -> io::Result<Option<String>> {
426    let Some(mut sha) = resolve(repo, rev)? else {
427        return Ok(None);
428    };
429    // Peel a commit (or annotated tag) to its tree.
430    if let Some((kind, data)) = read_object(repo, &sha)? {
431        match kind.as_str() {
432            "commit" => match commit_tree(&data) {
433                Some(t) => sha = t,
434                None => return Ok(None),
435            },
436            "tree" => {}
437            _ => return Ok(None),
438        }
439    } else {
440        return Ok(None);
441    }
442    let mut it = path.split('/').filter(|c| !c.is_empty()).peekable();
443    while let Some(c) = it.next() {
444        let Some((kind, data)) = read_object(repo, &sha)? else {
445            return Ok(None);
446        };
447        if kind != "tree" {
448            return Ok(None);
449        }
450        let mut pos = 0;
451        let mut hit = None;
452        while pos < data.len() {
453            let Some((_mode, name, e)) = tree_entry(&data, &mut pos) else {
454                break;
455            };
456            if name == c.as_bytes() {
457                hit = Some(e);
458                break;
459            }
460        }
461        match hit {
462            Some(e) => sha = e,
463            None => return Ok(None),
464        }
465        if it.peek().is_none() {
466            return Ok(Some(sha));
467        }
468    }
469    Ok(None)
470}

472/// `git cat-file -p <rev>:<path>` over loose objects — returns blob
473/// bytes. `None` → fall back to plumbing (packs, weird revs, trees).
474pub fn read_at(repo: &Path, rev: &str, path: &str) -> io::Result<Option<Vec<u8>>> {
475    let Some(sha) = entry_at(repo, rev, path)? else {
476        return Ok(None);
477    };
478    match read_object(repo, &sha)? {
479        Some((kind, payload)) if kind == "blob" => Ok(Some(payload)),
480        _ => Ok(None),
481    }
482}

490/// `git ls-tree -r --name-only <rev>` over loose objects. `None` →
491/// fall back to plumbing.
492pub fn list_tree(repo: &Path, rev: &str) -> io::Result<Option<Vec<String>>> {
493    let Some(mut sha) = resolve(repo, rev)? else {
494        return Ok(None);
495    };
496    if let Some((kind, data)) = read_object(repo, &sha)? {
497        match kind.as_str() {
498            "commit" => match commit_tree(&data) {
499                Some(t) => sha = t,
500                None => return Ok(None),
501            },
502            "tree" => {}
503            _ => return Ok(None),
504        }
505    } else {
506        return Ok(None);
507    }
508    let mut out = Vec::new();
509    if walk(repo, &sha, "", &mut out)? {
510        Ok(Some(out))
511    } else {
512        Ok(None)
513    }
514}

516fn walk(repo: &Path, sha: &str, prefix: &str, out: &mut Vec<String>) -> io::Result<bool> {
517    let Some((kind, data)) = read_object(repo, sha)? else {
518        return Ok(false);
519    };
520    if kind != "tree" {
521        return Ok(false);
522    }
523    let mut pos = 0;
524    while pos < data.len() {
525        let Some((mode, name, e)) = tree_entry(&data, &mut pos) else {
526            break;
527        };
528        let name = String::from_utf8_lossy(name).into_owned();
529        if mode == 0o40000 {
530            if !walk(repo, &e, &format!("{prefix}{name}/"), out)? {
531                return Ok(false);
532            }
533        } else {
534            out.push(format!("{prefix}{name}"));
535        }
536    }
537    Ok(true)
538}

546        // `git hash-object --stdin` for "hello\n" → ce013625030ba8dba906f756967f9e9ca394464a

559    fn sha256_repo_defers_to_plumbing() {
560        let dir = std::env::temp_dir().join(format!("sturn-odb-sha256-{}", std::process::id()));
561        let _ = fs::remove_dir_all(&dir);
562        fs::create_dir_all(&dir).unwrap();
563        let ok = std::process::Command::new("git")
564            .args(["init", "-q", "--object-format=sha256"])
565            .arg(&dir)
566            .status()
567            .map(|s| s.success())
568            .unwrap_or(false);

573        assert_eq!(resolve(&dir, "HEAD").unwrap(), None);

575        assert!(!object_exists(&dir, &"0".repeat(64)));
576        let _ = fs::remove_dir_all(&dir);
577    }
