1//! Git impl of `Vcs`, plus the snap bridge: fold WAL ops into a
2//! real commit — the crossing from the evented world back to
3//! ordinary git history. `snap` is why GitHub sees a normal repo.
4
5use std::io;
6use std::io::Write;
7use std::path::{Path, PathBuf};
8use std::process::{Command, Stdio};
9use sturn_core::{Event, Op};
10use sturn_vcs::Vcs;

12pub struct GitVcs;

14fn git(repo: &Path, args: &[&str]) -> io::Result<String> {
15    let o = Command::new("git")
16        .arg("-C")
17        .arg(repo)
18        .args(args)
19        .stdout(Stdio::piped())
20        .stderr(Stdio::piped())
21        .output()?;
22    if o.status.success() {
23        Ok(String::from_utf8_lossy(&o.stdout).trim().to_string())
24    } else {
25        Err(io::Error::other(format!(
26            "git {}: {}",
27            args.join(" "),
28            String::from_utf8_lossy(&o.stderr).trim()
29        )))
30    }
31}

33impl Vcs for GitVcs {
34    fn sync(&self, repo: &Path, remote: &str) -> bool {
35        Command::new("git")
36            .arg("-C")
37            .arg(repo)
38            .args(["fetch", remote, "--prune"])
39            .status()
40            .map(|s| s.success())
41            .unwrap_or(false)
42    }
43
44    fn resolve(&self, repo: &Path, rev: &str) -> Option<String> {
45        if let Ok(Some(sha)) = sturn_odb::resolve(repo, rev) {
46            return Some(sha);
47        }
48        Command::new("git")
49            .arg("-C")
50            .arg(repo)
51            .args(["rev-parse", "-q", "--verify", rev])
52            .output()
53            .ok()
54            .filter(|o| o.status.success())
55            .map(|o| String::from_utf8_lossy(&o.stdout).trim().to_string())
56            .filter(|s| !s.is_empty())
57    }
58
59    fn has_commit(&self, repo: &Path, sha: &str) -> bool {
60        // Loose-object hit needs no spawn; packs/alternates fall
61        // through to `cat-file -e`.
62        if sturn_odb::object_exists(repo, sha) {
63            return true;
64        }
65        Command::new("git")
66            .arg("-C")
67            .arg(repo)
68            .args(["cat-file", "-e", sha])
69            .status()
70            .map(|s| s.success())
71            .unwrap_or(false)
72    }
73
74    fn push(&self, repo: &Path, remote: &str) -> bool {
75        let heads = Command::new("git")
76            .arg("-C")
77            .arg(repo)
78            .args(["push", remote, "--all"])
79            .status()
80            .map(|s| s.success())
81            .unwrap_or(false);
82        let sturn = Command::new("git")
83            .arg("-C")
84            .arg(repo)
85            .args(["push", remote, "refs/sturn/*:refs/sturn/*"])
86            .status()
87            .map(|s| s.success())
88            .unwrap_or(false);
89        heads && sturn
90    }
91}

93/// Store `content` as a blob in the ODB; returns its sha.
94pub fn store_blob(repo: &Path, content: &[u8]) -> io::Result<String> {
95    if let Ok(sha) = sturn_odb::write_object(repo, "blob", content) {
96        return Ok(sha);
97    }
98    // Plumbing fallback — layouts the direct writer can't resolve.
99    let mut c = Command::new("git")
100        .arg("-C")
101        .arg(repo)
102        .args(["hash-object", "-w", "--stdin"])
103        .stdin(Stdio::piped())
104        .stdout(Stdio::piped())
105        .stderr(Stdio::piped())
106        .spawn()?;
107    c.stdin.take().unwrap().write_all(content)?;
108    let o = c.wait_with_output()?;
109    if o.status.success() {
110        Ok(String::from_utf8_lossy(&o.stdout).trim().to_string())
111    } else {
112        Err(io::Error::other("hash-object failed"))
113    }
114}

116/// Overlay `events`' Write/Rename/Delete ops on `base`, commit to
117/// `branch`, return the new commit sha. Coordination ops (claims,
118/// intents, reviews) are metadata — they never touch the tree.
119///
120/// Uses a scratch index (`GIT_INDEX_FILE`) — no working tree, no
121/// checkout, no lock on the user's index.
122pub fn snap_commit(
123    repo: &Path,
124    base: &str,
125    events: &[Event],
126    message: &str,
127    branch: &str,
128) -> io::Result<String> {
129    // Per-call scratch index — two snaps in one process (daemon
130    // handling concurrent ws folds, parallel tests) must not share
131    // GIT_INDEX_FILE or they interleave updates and lock.
132    static SNAP_IDX: std::sync::atomic::AtomicU64 = std::sync::atomic::AtomicU64::new(0);
133    let idx: PathBuf = std::env::temp_dir().join(format!(
134        "sturn-index-{}-{}",
135        std::process::id(),
136        SNAP_IDX.fetch_add(1, std::sync::atomic::Ordering::Relaxed)
137    ));
138    let gi = |args: &[&str]| -> io::Result<String> {
139        let o = Command::new("git")
140            .arg("-C")
141            .arg(repo)
142            .env("GIT_INDEX_FILE", &idx)
143            .args(args)
144            .stdout(Stdio::piped())
145            .stderr(Stdio::piped())
146            .output()?;
147        if o.status.success() {
148            Ok(String::from_utf8_lossy(&o.stdout).trim().to_string())
149        } else {
150            Err(io::Error::other(format!(
151                "git {}: {}",
152                args.join(" "),
153                String::from_utf8_lossy(&o.stderr).trim()
154            )))
155        }
156    };
157
158    gi(&["read-tree", base])?;
159    // Track the overlay as we walk — a Rename's `from` may have been
160    // created by an earlier Write in the same batch (never in base).
161    let mut seen: std::collections::HashMap<String, String> = std::collections::HashMap::new();
162    for ev in events {
163        match &ev.op {
164            Op::Write { path, blob, .. } => {
165                gi(&["update-index", "--add", "--cacheinfo", &format!("100644,{blob},{path}")])?;
166                seen.insert(path.clone(), blob.clone());
167            }
168            Op::Delete { path } => {
169                gi(&["update-index", "--force-remove", path])?;
170                seen.remove(path.as_str());
171            }
172            Op::Rename { from, to } => {
173                let blob = match seen.get(from.as_str()) {
174                    Some(b) => b.clone(),
175                    None => git(repo, &["rev-parse", &format!("{base}:{from}")])?,
176                };
177                gi(&["update-index", "--force-remove", from])?;
178                gi(&["update-index", "--add", "--cacheinfo", &format!("100644,{blob},{to}")])?;
179                seen.remove(from.as_str());
180                seen.insert(to.clone(), blob);
181            }
182            _ => {}
183        }
184    }
185    let tree = gi(&["write-tree"])?;
186    let _ = std::fs::remove_file(&idx);
187    let commit = git(repo, &["commit-tree", &tree, "-p", base, "-m", message])?;
188    git(
189        repo,
190        &["update-ref", &format!("refs/heads/{branch}"), &commit],
191    )?;
192    Ok(commit)
193}
