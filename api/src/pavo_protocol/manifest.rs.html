14/// The driver manifest — the discoverability contract.
15///
16/// Wire shape (JSON):
17///
18/// ```json
19/// {
20///   "name": "myco",
21///   "title": "Myco",
22///   "version": "0.4.0",
23///   "verbs": ["manifest", "serve"],
24///   "capabilities": ["pavo"],
25///   "contributes": { "navbar": [...], "routes": [...] },
26///   "views": [ { "id": "board", "route": "/myco/board", "root": {…} } ]
27/// }
28/// ```
29#[derive(Clone, Debug, PartialEq, Serialize, Deserialize)]
30pub struct DriverManifest {
31    /// Driver name — the `<type>` in `tamias-driver-<type>` and the
32    /// namespace for its routes (`/<name>/<view>`).
33    pub name: String,
34    /// Human-facing title for chrome (navbar entries, window titles).
35    pub title: String,
36    /// Optional icon — an emoji, an icon name, or a path the renderer
37    /// knows how to resolve.
38    #[serde(default, skip_serializing_if = "Option::is_none")]
39    pub icon: Option<String>,
40    /// Driver version (semver string).
41    pub version: String,
42    /// Wire verbs this executable answers beyond the tamias install
43    /// verbs — e.g. `["manifest", "serve"]` for pavo drivers.
44    /// Capability-negotiated: pavo treats a driver as a view driver
45    /// iff `serve` is listed (or `capabilities` contains `"pavo"`).
46    #[serde(default)]
47    pub verbs: Vec<String>,
48    /// Capability tags — free-form strings products use to declare
49    /// roles (`"pavo"`, `"config"`, `"chrome"`).
50    #[serde(default)]
51    pub capabilities: Vec<String>,
52    /// Driver dependency declarations — which other drivers this one
53    /// expects (or optionally uses). See [`DriverDeps`].
54    #[serde(default, skip_serializing_if = "Option::is_none")]
55    pub deps: Option<DriverDeps>,
56    /// Contribution slots — where this driver's entries land in the
57    /// composed chrome. See [`Contributes`].
58    #[serde(default)]
59    pub contributes: Contributes,
60    /// Views this driver can render — each binds a route to a
61    /// declarative [`ViewNode`] tree plus the queries that feed it.
62    #[serde(default)]
63    pub views: Vec<ViewSpec>,
64    /// View-kind renderer plugins this driver ships — each declares
65    /// the node `kinds` it renders and which `surfaces` it supports.
66    /// See [`RendererDecl`]. A driver may declare renderers with zero
67    /// views (pure renderer plugin) or alongside views.
68    #[serde(default)]
69    pub renderers: Vec<RendererDecl>,
70    /// Driver-global actions — verbs offered on every view (new
71    /// item, refresh, settings, …). View/node/item-scoped actions
72    /// live on [`crate::view_node::NodeBase::actions`] and
73    /// [`crate::view_node::TreeItem::actions`]. All dispatch through
74    /// the wire `invoke` op. See [`crate::action`].
75    #[serde(default, skip_serializing_if = "Vec::is_empty")]
76    pub actions: Vec<Action>,
77}

79/// Driver dependency declarations — the ordering half of the plugin
80/// contract. Names are driver names: the `<type>` in
81/// `tamias-driver-<type>` (and `DriverManifest::name`).
82///
83/// Semantics mirror product `requiredDeps`/`optionalDeps`:
84///
85/// - `requires` — drivers that must be present and healthy for this
86///   driver to function. A missing required dep is a **discovery
87///   error**: the host refuses (or quarantines) the driver rather
88///   than half-rendering it.
89/// - `optional` — drivers whose absence merely degrades features.
90///   Missing optional deps never block discovery; the driver/host
91///   falls back gracefully.
92///
93/// Ordering is a host concern — pavo-core topo-sorts discovered
94/// drivers over this graph (`order_drivers`); the manifest only
95/// carries the declaration.
96///
97/// Wire shape:
98///
99/// ```json
100/// { "requires": ["config"], "optional": ["myco"] }
101/// ```
102#[derive(Clone, Debug, Default, PartialEq, Serialize, Deserialize)]
103pub struct DriverDeps {
104    /// Driver names this driver cannot function without — absence is
105    /// a discovery error.
106    #[serde(default)]
107    pub requires: Vec<String>,
108    /// Driver names this driver uses when present — absence degrades
109    /// gracefully.
110    #[serde(default)]
111    pub optional: Vec<String>,
112}
113
114/// One view-kind renderer a driver ships — the manifest half of the
115/// renderer-plugin contract (the wire half is the `render` op).
116///
117/// A renderer claims node `kinds` (ViewNode `kind` strings, or `"*"`
118/// for any kind the host doesn't build in). When a surface meets a
119/// node of a claimed kind, the host routes it to this driver's
120/// `render` op instead of its built-in interpretation; the driver
121/// answers with a [`crate::render_model::RenderModel`].
122///
123/// Wire shape:
124///
125/// ```json
126/// { "id": "graph", "kinds": ["graph"],
127///   "surfaces": { "cli": {"formats": ["svg", "ascii"]},
128///                 "tui": true,
129///                 "web": {"island": true} },
130///   "priority": 100 }
131/// ```
132#[derive(Clone, Debug, PartialEq, Serialize, Deserialize)]
133pub struct RendererDecl {
134    /// Renderer id — namespaced `<driver>:<id>` by convention; stable
135    /// for diagnostics and `/api/renderers` listings.
136    pub id: String,
137    /// Node kinds this renderer handles — ViewNode `kind` strings
138    /// (`"graph"`, `"timeline"`, …) or `"*"` for a catch-all. Explicit
139    /// kinds always beat `"*"` claims; among competing claims the
140    /// lowest [`RendererDecl::priority`] wins (ties: driver kind asc,
141    /// then renderer id asc — the chrome-merge ordering policy).
142    #[serde(default)]
143    pub kinds: Vec<String>,
144    /// Per-surface support. Absent surface = unsupported there.
145    #[serde(default)]
146    pub surfaces: RendererSurfaces,
147    /// Ordering hint within a kind — lower wins. Defaults to `100`
148    /// (same convention as [`Contribution::priority`]).
149    #[serde(default = "default_priority")]
150    pub priority: i32,
151    /// Optional entry-point hint — a component/module name the host
152    /// may prefer over `serve`-wire rendering (reserved; hosts
153    /// currently always use the wire).
154    #[serde(default, skip_serializing_if = "Option::is_none")]
155    pub entry: Option<String>,
156}

158/// Which surfaces a [`RendererDecl`] supports and how.
159#[derive(Clone, Debug, Default, PartialEq, Serialize, Deserialize)]
160pub struct RendererSurfaces {
161    /// CLI support — `formats` lists the `pavo view --format` values
162    /// the renderer claims beyond the built-ins (`text`/`json`/`dot`/
163    /// `mermaid`). An empty list is a wildcard: the renderer handles
164    /// every format it is asked for, including `text`.
165    #[serde(default, skip_serializing_if = "Option::is_none")]
166    pub cli: Option<CliSurface>,
167    /// TUI support — `true` renders plugin nodes on the ratatui host.
168    #[serde(default)]
169    pub tui: bool,
170    /// Web support — `component` names a web component/island the
171    /// host may mount; `island` marks the renderer as needing
172    /// client-side execution (SSR-incompatible output).
173    #[serde(default, skip_serializing_if = "Option::is_none")]
174    pub web: Option<WebSurface>,
175}

177/// CLI surface declaration for a [`RendererDecl`].
178#[derive(Clone, Debug, Default, PartialEq, Serialize, Deserialize)]
179pub struct CliSurface {
180    /// `--format` values this renderer claims. Empty = wildcard.
181    #[serde(default)]
182    pub formats: Vec<String>,
183}

185/// Web surface declaration for a [`RendererDecl`].
186#[derive(Clone, Debug, Default, PartialEq, Serialize, Deserialize)]
187pub struct WebSurface {
188    /// Optional component/island name — hint for hosts that resolve
189    /// `placeholder` models or `embed` nodes; the `render` op remains
190    /// the canonical path.
191    #[serde(default, skip_serializing_if = "Option::is_none")]
192    pub component: Option<String>,
193    /// `true` when the renderer's output requires client-side
194    /// execution (canvas, wasm, interactive svg) — hosts should treat
195    /// the node as an island.
196    #[serde(default)]
197    pub island: bool,
198}

200/// Contribution slots — the named places in pavo's chrome where
201/// drivers may register entries. A *chrome driver* owns the frame;
202/// view drivers fill slots. pavo merges contributions across drivers,
203/// resolving conflicts by driver priority.
204///
205/// The built-in slots are `navbar`, `dashboard`, `statusbar`,
206/// `routes`, `sidebar`, `headerActions`, `notifications`, and
207/// `jobs`. Unknown slots are preserved via [`Contributes::extra`]
208/// so newer drivers degrade gracefully on older hosts.
209#[derive(Clone, Debug, Default, PartialEq, Serialize, Deserialize)]
210pub struct Contributes {
211    /// Entries for the top navigation bar (web/TUI chrome).
212    #[serde(default)]
213    pub navbar: Vec<Contribution>,
214    /// Entries for the shared `dashboard` route — composable panels
215    /// from many drivers on one screen.
216    #[serde(default)]
217    pub dashboard: Vec<Contribution>,
218    /// Entries for the bottom status bar — typically one-line
219    /// summaries (staleness, counts, claims).
220    #[serde(default)]
221    pub statusbar: Vec<Contribution>,
222    /// Standalone routes the driver owns outside its views — e.g. a
223    /// settings page. Each becomes a navigable URL on the web host.
224    #[serde(default)]
225    pub routes: Vec<Contribution>,
226    /// Entries for a left sidebar shell — grouped nav entries hosts
227    /// may cluster by `group`-like semantics later. Same
228    /// [`Contribution`] shape as the other slots: `route` navigates,
229    /// `priority` orders, `icon` decorates.
230    #[serde(default)]
231    pub sidebar: Vec<Contribution>,
232    /// Top-bar actions — buttons/icons in the header that open a
233    /// popover or panel. `route` points at the view the host renders
234    /// inside that surface.
235    #[serde(default, rename = "headerActions")]
236    pub header_actions: Vec<Contribution>,
237    /// Marks the driver as a notification provider. Entries are
238    /// display metadata (label, icon, ordering); the notification
239    /// *data* flows through the `notifications` wire op as
240    /// [`crate::platform::NotificationItem`]s.
241    #[serde(default)]
242    pub notifications: Vec<Contribution>,
243    /// Marks the driver as a job/queue feed provider. Same pattern
244    /// as `notifications`: entries are display metadata, data comes
245    /// from the `jobs` wire op as [`crate::platform::JobItem`]s.
246    #[serde(default)]
247    pub jobs: Vec<Contribution>,
248    /// Command-palette/menu entries — the dmenu-style quick-launch
249    /// list hosts synthesize from `actions[]` + these declarations.
250    /// Entries either `invoke` an action or `route` to a view.
251    /// See [`MenuEntry`].
252    #[serde(default)]
253    pub menus: Vec<MenuEntry>,
254    /// Forward-compat escape hatch: any slot name not listed above
255    /// round-trips untouched. Renderers ignore slots they don't know.
256    #[serde(flatten)]
257    pub extra: BTreeMap<String, Vec<Contribution>>,
258}

260/// One contributed entry in a slot.
261///
262/// `route` is a host-relative path (`/myco/board`, `/dashboard`);
263/// renderers resolve it against the driver's views or its own
264/// routes. `priority` orders entries within a slot — lower wins;
265/// ties break by driver discovery order (stable, deterministic).
266#[derive(Clone, Debug, PartialEq, Serialize, Deserialize)]
267pub struct Contribution {
268    /// Stable id — namespaced as `<driver>:<id>` by convention so
269    /// contributions never collide across drivers.
270    pub id: String,
271    /// Label shown in chrome.
272    pub title: String,
273    /// Route the entry navigates to.
274    pub route: String,
275    /// Ordering hint within the slot; lower sorts first. Defaults to
276    /// `100` when absent so drivers can leave it unset.
277    #[serde(default = "default_priority")]
278    pub priority: i32,
279    /// Optional icon hint (same resolution rules as
280    /// [`DriverManifest::icon`]).
281    #[serde(default, skip_serializing_if = "Option::is_none")]
282    pub icon: Option<String>,
283}
284
285fn default_priority() -> i32 {
286    100
287}

289/// One entry in a [`Contributes::menus`] slot — a command-palette /
290/// quick-launch item.
291///
292/// An entry resolves to a **command** — `command` when set (the
293/// unified grammar: `action:<id>` | `route:<path>` | `host.<builtin>`
294/// | `driver:<name>:<id>`), else the legacy `action`/`route` pair —
295/// or is display-only (a section label). `keywords` feed host-side
296/// fuzzy matching (dmenu-style): users find entries by typing any
297/// keyword, not just the title.
298///
299/// Wire shape:
300///
301/// ```json
302/// { "id": "new-task", "title": "New task", "icon": "plus",
303///   "action": "todo.add", "keywords": ["create", "todo"],
304///   "section": "myco", "priority": 10 }
305/// ```
306#[derive(Clone, Debug, PartialEq, Serialize, Deserialize)]
307pub struct MenuEntry {
308    /// Stable id — namespaced `<driver>:<id>` by convention.
309    pub id: String,
310    /// Label shown in the palette/menu.
311    pub title: String,
312    /// Optional icon hint (same resolution rules as
313    /// [`DriverManifest::icon`]).
314    #[serde(default, skip_serializing_if = "Option::is_none")]
315    pub icon: Option<String>,
316    /// Action to invoke — resolved against the driver's
317    /// `actions[]`/node actions/ops-derived actions like the
318    /// `invoke` wire op. May be `"<action-id>"` or
319    /// `"<driver>:<action-id>"` for cross-driver entries.
320    #[serde(default, skip_serializing_if = "Option::is_none")]
321    pub action: Option<String>,
322    /// Route to navigate to instead of invoking — same semantics
323    /// as [`Contribution::route`]. May carry `{…}` template
324    /// placeholders (see [`crate::action::Action::route`]).
325    #[serde(default, skip_serializing_if = "Option::is_none")]
326    pub route: Option<String>,
327    /// Generic command target — the unified command grammar
328    /// (`action:<id>` | `route:<path>` | `host.<builtin>` |
329    /// `driver:<name>:<id>`; see `pavo_core::commands`). When
330    /// present it **wins** over `action`/`route`; the legacy fields
331    /// stay for compat. Entries whose command doesn't parse are
332    /// display-only.
333    #[serde(default, skip_serializing_if = "Option::is_none")]
334    pub command: Option<String>,
335    /// Suggested keyboard bindings for this entry — same role as
336    /// [`crate::action::Action::keys`]: declarations suggest, the
337    /// host's user keymap overrides by command id.
338    #[serde(default, skip_serializing_if = "Vec::is_empty")]
339    pub keys: Vec<String>,
340    /// Extra fuzzy-match terms — aliases, synonyms, related ids.
341    #[serde(default, skip_serializing_if = "Vec::is_empty")]
342    pub keywords: Vec<String>,
343    /// Optional grouping label hosts may cluster entries under.
344    #[serde(default, skip_serializing_if = "Option::is_none")]
345    pub section: Option<String>,
346    /// Ordering hint within the merged menu; lower sorts first.
347    #[serde(default = "default_priority")]
348    pub priority: i32,
349}

351/// One view a driver offers — a route plus the declarative tree and
352/// the queries that feed it.
353#[derive(Clone, Debug, PartialEq, Serialize, Deserialize)]
354pub struct ViewSpec {
355    /// View id, unique within the driver (`board`, `item`).
356    pub id: String,
357    /// Human-facing title.
358    pub title: String,
359    /// Route the view answers — `<driver>:<view>` deep links resolve
360    /// here (e.g. `myco:board` → `/myco/board`).
361    pub route: String,
362    /// Optional longer description for picker UIs.
363    #[serde(default, skip_serializing_if = "Option::is_none")]
364    pub description: Option<String>,
365    /// Query bindings — named data sources the [`ViewNode`] tree
366    /// references via `NodeBase::query`. Each `name` maps to a wire
367    /// `op` (usually `query` on the driver's `serve`) plus static
368    /// params merged with caller-supplied view params.
369    #[serde(default)]
370    pub queries: Vec<QueryBinding>,
371    /// The declarative UI tree — the renderers' input.
372    pub root: ViewNode,
373    /// Per-view row-navigation template — when set, activating a
374    /// [`crate::view_node::TreeItem`] (Enter/click) navigates the
375    /// host to this interpolated route instead of doing nothing.
376    /// Template placeholders resolve over the item:
377    /// `{item.id}`, `{item.label}`, `{item.<prop>}` (item `props`
378    /// keys are addressable directly — see
379    /// `pavo_core::commands::interpolate`). Example:
380    /// `"/myco/item?id={item.id}"`.
381    #[serde(default, skip_serializing_if = "Option::is_none")]
382    pub item_route: Option<String>,
383    /// Names this view's free-text filter query param — the
384    /// convention is `"q"`. The host's `host.filter` builtin opens a
385    /// filter dialog that re-resolves the view with
386    /// `params[<filter_param>]` set; no new wire op. When absent,
387    /// `host.filter` falls back to `"q"` if the view's query
388    /// bindings accept it, else the builtin is unavailable.
389    #[serde(default, skip_serializing_if = "Option::is_none")]
390    pub filter_param: Option<String>,
391    /// Auto-refresh hint — milliseconds between host-initiated
392    /// re-resolves of this view's data. `None` (the default) means
393    /// the view refreshes only on explicit user/driver action.
394    ///
395    /// Hosts SHOULD ignore values below 1s or above 5min: a
396    /// sub-second timer busy-loops the driver's `serve`, and a
397    /// longer one is indistinguishable from manual refresh. The TUI
398    /// enforces this range; other surfaces may clamp. Applies to
399    /// the view only — chrome/dashboard slot contributions are not
400    /// views and do not auto-refresh. Additive within v1: old hosts
401    /// ignore the field.
402    #[serde(default, skip_serializing_if = "Option::is_none")]
403    pub refresh_ms: Option<u64>,
404}

406/// A named query binding inside a [`ViewSpec`]: `name` is what
407/// ViewNodes reference; `op` is the wire op the host calls; `params`
408/// are static arguments merged under caller-supplied view params.
409#[derive(Clone, Debug, PartialEq, Serialize, Deserialize)]
410pub struct QueryBinding {
411    /// Binding name — referenced by `query` fields in the tree.
412    pub name: String,
413    /// Wire op to invoke on the driver's `serve` (typically `"query"`,
414    /// with the real product op carried in `params`).
415    pub op: String,
416    /// Optional target driver — when set, the binding is a
417    /// **cross-driver** query: the host resolves `driver` (a driver
418    /// name, the `<type>` in `tamias-driver-<type>`) and issues the
419    /// op against *that* driver's `serve` instead of the view's own.
420    /// This is the data half of the cross-driver query contract;
421    /// pavo-core resolves it when assembling view data (failures
422    /// degrade to an `{"error": …}` payload in the binding's slot).
423    /// Older hosts may ignore the field and treat the binding as local.
424    #[serde(default, skip_serializing_if = "Option::is_none")]
425    pub driver: Option<String>,
426    /// Static params merged into every invocation of this binding.
427    #[serde(default, skip_serializing_if = "Map::is_empty")]
428    pub params: Map<String, Value>,
429}

431#[cfg(test)]
432#[cfg_attr(coverage, coverage(off))]
433mod tests {
434    #![allow(clippy::unwrap_used)]
435    use super::*;
436    use serde_json::json;
437
438    /// `MenuEntry.command`/`keys` and `ViewSpec.item_route`/
439    /// `filter_param`/`refresh_ms` — additive optional fields: absent defaults,
440    /// present round-trips, unknown siblings ignored.
441    #[test]
442    fn menu_entry_and_view_spec_new_fields() {
443        let m: MenuEntry = serde_json::from_value(json!({
444            "id":"go","title":"Go","command":"route:/myco/item?id={item.id}",
445            "keys":["g i"],"future":true
446        }))
447        .unwrap();
448        assert_eq!(m.command.as_deref(), Some("route:/myco/item?id={item.id}"));
449        assert_eq!(m.keys, vec!["g i".to_string()]);
450        assert!(m.action.is_none() && m.route.is_none());
451        let v = serde_json::to_value(&m).unwrap();
452        assert_eq!(v["command"], "route:/myco/item?id={item.id}");
453        assert_eq!(v["keys"], json!(["g i"]));
454        assert!(v.get("future").is_none());
455
456        // Legacy-only entry still parses clean.
457        let legacy: MenuEntry = serde_json::from_value(json!({
458            "id":"x","title":"X","action":"todo.add","route":"/v"
459        }))
460        .unwrap();
461        assert!(legacy.command.is_none());
462        assert!(legacy.keys.is_empty());
463
464        let spec: ViewSpec = serde_json::from_value(json!({
465            "id":"board","title":"Board","route":"/myco/board",
466            "item_route":"/myco/item?id={item.id}",
467            "filter_param":"q",
468            "refresh_ms":5000,
469            "root":{"kind":"kv","pairs":[]}
470        }))
471        .unwrap();
472        assert_eq!(spec.item_route.as_deref(), Some("/myco/item?id={item.id}"));
473        assert_eq!(spec.filter_param.as_deref(), Some("q"));
474        assert_eq!(spec.refresh_ms, Some(5000));
475        let v = serde_json::to_value(&spec).unwrap();
476        assert_eq!(v["item_route"], "/myco/item?id={item.id}");
477        assert_eq!(v["filter_param"], "q");
478        assert_eq!(v["refresh_ms"], 5000);
479
480        let bare: ViewSpec = serde_json::from_value(json!({
481            "id":"v","title":"V","route":"/v","root":{"kind":"kv"}
482        }))
483        .unwrap();
484        assert!(bare.item_route.is_none());
485        assert!(bare.filter_param.is_none());
486        assert!(bare.refresh_ms.is_none());
487        assert!(serde_json::to_value(&bare)
488            .unwrap()
489            .get("item_route")
490            .is_none());
491    }
