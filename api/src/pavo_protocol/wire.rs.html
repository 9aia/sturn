1//! Wire ops — the ndjson JSON-RPC convention every sibling `serve`
2//! speaks, applied to pavo drivers.
3//!
4//! Transport: one JSON object per line on stdin/stdout (CLI/TUI) or
5//! bridged to ws/http by `pavo serve` (web). Requests are
6//! `{"op":"…", …}`; responses are `{"ok":true,"result":…}` or
7//! `{"ok":false,"error":"…"}`. This is the same shape as
8//! `tamias-serve`, `myco serve`, `cuora serve`, `sepia serve`.
9//!
10//! Ops (v1):
11//!
12//! | op        | params                              | result            |
13//! |-----------|-------------------------------------|-------------------|
14//! | `manifest`| —                                   | [`DriverManifest`]|
15//! | `views`   | —                                   | `Vec<ViewSummary>`|
16//! | `view`    | `view`, `params?`                   | [`ViewResult`]    |
17//! | `query`   | `query`, `params?`                  | [`QueryResult`]   |
18//! | `render`  | `kind`, `node`, `data`, `surface`,  | [`RenderModel`](crate::RenderModel)|
19//! |           | `format?`                           |                   |
20//! | `submit`  | `view`, `form?`, `values`           | [`SubmitResult`]  |
21//! | `invoke`  | `view?`, `action`, `target?`,       | [`ActionResult`]  |
22//! |           | `params?`                          |                   |
23//! | `jobs`    | —                                   | `Vec<JobItem>`    |
24//! | `notifications` | —                          | `Vec<NotificationItem>` |
25//!
26//! `jobs` and `notifications` are the **platform-feed ops**: drivers
27//! whose manifest declares the matching `contributes` slot
28//! (`jobs`/`notifications`) answer them with item lists — see
29//! [`crate::platform`]. Both are optional capabilities: drivers
30//! without the feed answer `ok:false` (or `[]`) per the usual
31//! convention, and hosts treat either as "no feed".
32//!
33//! `submit` is the form-interaction verb: a renderer collects the
34//! editable [`crate::view_node::FormField`] values of a `form` node
35//! and posts them here. `view` names the view the form lives in;
36//! `form` optionally names the form node's `id` when a view carries
37//! several forms (absent = the driver's default/first form);
38//! `values` maps field `key` → submitted JSON value. The driver
39//! dispatches on the form's `submit_op`; the result may carry a
40//! message, per-field validation `errors`, driver `data`, and an
41//! updated `view` for re-render.
42//!
43//! `invoke` is the general action verb: a renderer activates an
44//! [`crate::action::Action`] (declared on the manifest, a node, a
45//! tree item, or derived from the driver's ops manifest) and posts
46//! here. `view` optionally names the view context the action was
47//! declared in (same namespace as the `view` op); `action` is the
48//! action's `id`; `target` is opaque JSON — typically the selected
49//! row's raw value or an item id the action operates on;
50//! `params` maps each [`crate::action::ParamSchema`] `name` to its
51//! submitted JSON value. The driver resolves the action and
52//! dispatches; the result carries the same shape as
53//! [`SubmitResult`].
54//!
55//! `render` is the view-kind renderer-plugin verb: a driver whose
56//! manifest declares `renderers` answers it with a canonical
57//! [`crate::render_model::RenderModel`] for one node on one surface.
58//! Drivers that declare no renderers may ignore it (hosts only call
59//! it for kinds the manifest claims).
60//!
61//! Thin drivers may proxy a product's existing `serve` for `query`
62//! payloads; `manifest`/`views`/`view` are always answered by the
63//! driver itself.
64

70/// A wire request — one JSON object per line, discriminated by `op`.
71///
72/// Unknown ops fail deserialization; hosts should treat that as a
73/// protocol error and surface the driver's raw error text.
74#[derive(Clone, Debug, PartialEq, Serialize, Deserialize)]
75#[serde(tag = "op", rename_all = "lowercase")]
76pub enum Request {
77    /// `{"op":"manifest"}` → [`DriverManifest`]. Discovery verb —
78    /// cheap, must not require product state.
79    Manifest,
80    /// `{"op":"views"}` → `Vec<ViewSummary>` — view ids/titles/routes
81    /// without the full trees (lightweight listing).
82    Views,
83    /// `{"op":"view","view":"board","params":{…}}` → [`ViewResult`] —
84    /// the resolved tree plus query-fed data for one view.
85    View {
86        /// View id from the driver's manifest (`board`, `item`).
87        view: String,
88        /// Caller-supplied params merged into each query binding.
89        #[serde(default, skip_serializing_if = "Map::is_empty")]
90        params: Map<String, Value>,
91    },
92    /// `{"op":"query","query":"items","params":{…}}` → [`QueryResult`] —
93    /// execute a single named query binding (or raw product op, for
94    /// thin drivers) and return its data payload.
95    Query {
96        /// Query name from the view's `queries` bindings, or a raw
97        /// product op when the driver proxies `serve`.
98        query: String,
99        /// Params merged into the binding's static params.
100        #[serde(default, skip_serializing_if = "Map::is_empty")]
101        params: Map<String, Value>,
102    },
103    /// `{"op":"render","kind":"graph","node":{…},"data":{…},
104    /// "surface":"cli","format":"svg"}` → [`RenderModel`](crate::RenderModel) — the
105    /// renderer-plugin verb. `node` is the raw ViewNode JSON (the
106    /// host passes it untyped so unknown/future fields survive);
107    /// `data` is the view's query-binding map verbatim; `surface` is
108    /// `"cli"`, `"tui"`, or `"web"` (open string — hosts may add
109    /// surfaces); `format` is the surface's format hint (`"svg"`,
110    /// `"ascii"`, …) when one applies.
111    Render {
112        /// The node's `kind` — also the renderer claim key.
113        kind: String,
114        /// The node as raw JSON (a ViewNode object, `{"kind":…}`).
115        #[serde(default)]
116        node: Value,
117        /// The view's bound query data — the same map `view` results
118        /// carry, so the plugin can resolve `node.query` itself.
119        #[serde(default)]
120        data: Map<String, Value>,
121        /// Target surface — `"cli"`, `"tui"`, `"web"`.
122        surface: String,
123        /// Optional format hint (CLI `--format` value, etc.).
124        #[serde(default, skip_serializing_if = "Option::is_none")]
125        format: Option<String>,
126    },
127    /// `{"op":"submit","view":"settings","form":"prefs",
128    /// "values":{"theme":"dark"}}` → [`SubmitResult`] — the
129    /// form-submission verb. `view` is the view id the form node
130    /// belongs to; `form` optionally names the form node's `id`
131    /// (its `NodeBase::id`) when the view has several forms —
132    /// drivers resolve the node to read its `submit_op` and field
133    /// declarations. `values` maps each submitted field's `key` to
134    /// its JSON value (`string`/`number`/`bool` per field `kind`).
135    /// Drivers that don't implement forms answer `ok:false`.
136    Submit {
137        /// View id from the driver's manifest — same namespace as
138        /// the `view` op.
139        view: String,
140        /// Optional form-node id (`NodeBase::id` of the `form`
141        /// node); absent submits the view's default form.
142        #[serde(default, skip_serializing_if = "Option::is_none")]
143        form: Option<String>,
144        /// Submitted field values keyed by `FormField::key`. Keys
145        /// the driver doesn't know are ignored; missing `required`
146        /// fields come back in [`SubmitResult::errors`].
147        #[serde(default, skip_serializing_if = "Map::is_empty")]
148        values: Map<String, Value>,
149    },
150    /// `{"op":"invoke","view":"item","action":"todo.done",
151    /// "target":{"id":"T-012"},"params":{"note":"shipped"}}` →
152    /// [`ActionResult`] — the action verb. `view` optionally names
153    /// the view the action was declared in (node/tree-item actions
154    /// resolve against it; manifest-global actions may omit it);
155    /// `action` is the [`crate::action::Action::id`]; `target` is
156    /// opaque JSON the driver interprets (selected row, item id,
157    /// …); `params` maps [`crate::action::ParamSchema::name`] →
158    /// submitted value. Drivers that don't implement actions
159    /// answer `ok:false`.
160    Invoke {
161        /// Optional view id from the driver's manifest — the
162        /// resolution context for node/item-scoped actions.
163        #[serde(default, skip_serializing_if = "Option::is_none")]
164        view: Option<String>,
165        /// The action id to dispatch.
166        action: String,
167        /// Opaque action target — the selected row's raw JSON, an
168        /// item id string, or whatever the action's `target`
169        /// context calls for. Drivers interpret it.
170        #[serde(default, skip_serializing_if = "Option::is_none")]
171        target: Option<Value>,
172        /// Submitted param values keyed by `ParamSchema::name`.
173        /// Missing `required` params come back in
174        /// [`ActionResult::errors`].
175        #[serde(default, skip_serializing_if = "Map::is_empty")]
176        params: Map<String, Value>,
177    },
178    /// `{"op":"jobs"}` → `Vec<JobItem>` — the job/queue feed op.
179    /// Drivers that declare `contributes.jobs` answer with their
180    /// current job list; drivers without a job feed answer `ok:false`
181    /// (or an empty list) — both mean "no feed" to hosts.
182    Jobs,
183    /// `{"op":"notifications"}` → `Vec<NotificationItem>` — the
184    /// notification feed op. Same optional-capability convention as
185    /// `jobs`: `contributes.notifications` marks the provider, the
186    /// op carries the items.
187    Notifications,
188}

190/// A wire response — one JSON object per line.
191///
192/// `ok:true` carries `result`; `ok:false` carries `error`. Fields
193/// for the other arm are absent (not null) so consumers can match on
194/// `ok` alone.
195#[derive(Clone, Debug, PartialEq, Serialize, Deserialize)]
196pub struct Response {
197    /// Success flag — the only field guaranteed present.
198    pub ok: bool,
199    /// Result payload when `ok` is true.
200    #[serde(default, skip_serializing_if = "Option::is_none")]
201    pub result: Option<Value>,
202    /// Human/machine-readable error string when `ok` is false.
203    #[serde(default, skip_serializing_if = "Option::is_none")]
204    pub error: Option<String>,
205}
206
207impl Response {
208    /// Build a success response around an arbitrary JSON result.
209    #[must_use]
210    pub fn ok(result: Value) -> Self {
211        Self {
212            ok: true,
213            result: Some(result),
214            error: None,
215        }
216    }
217
218    /// Build a success response from any serializable payload.
219    ///
220    /// Serialization failure is reported as an error response rather
221    /// than panicking — a driver must never crash the wire.
222    #[must_use]
223    pub fn ok_typed<T: Serialize>(value: &T) -> Self {
224        match serde_json::to_value(value) {
225            Ok(v) => Self::ok(v),
226            Err(e) => Self::err(format!("result serialization failed: {e}")),
227        }
228    }
229
230    /// Build an error response.
231    #[must_use]
232    pub fn err(message: impl Into<String>) -> Self {
233        Self {
234            ok: false,
235            result: None,
236            error: Some(message.into()),
237        }
238    }
239
240    /// Typed accessor — the result payload deserialized as `T`.
241    ///
242    /// # Errors
243    /// Returns the serde error when `result` is absent or doesn't
244    /// match `T`.
245    pub fn result_as<T: for<'de> Deserialize<'de>>(&self) -> Result<T, serde_json::Error> {
246        match &self.result {
247            Some(v) => serde_json::from_value(v.clone()),
248            None => serde_json::from_value(Value::Null),
249        }
250    }
251}

253/// Result of the `views` op — a [`ViewSpec`] minus the `root` tree,
254/// for cheap listing. Renderers fetch full trees via `view`.
255#[derive(Clone, Debug, PartialEq, Serialize, Deserialize)]
256pub struct ViewSummary {
257    /// View id.
258    pub id: String,
259    /// Human-facing title.
260    pub title: String,
261    /// Route the view answers.
262    pub route: String,
263    /// Optional description.
264    #[serde(default, skip_serializing_if = "Option::is_none")]
265    pub description: Option<String>,
266    /// Names of the query bindings this view uses.
267    #[serde(default)]
268    pub queries: Vec<String>,
269}
270
271impl From<&ViewSpec> for ViewSummary {
272    fn from(spec: &ViewSpec) -> Self {
273        Self {
274            id: spec.id.clone(),
275            title: spec.title.clone(),
276            route: spec.route.clone(),
277            description: spec.description.clone(),
278            queries: spec.queries.iter().map(|q| q.name.clone()).collect(),
279        }
280    }
281}

283/// Result of the `view` op — the resolved view: the declarative tree
284/// plus every query binding's data, keyed by binding name.
285#[derive(Clone, Debug, PartialEq, Serialize, Deserialize)]
286pub struct ViewResult {
287    /// The view's manifest spec (tree included).
288    pub view: ViewSpec,
289    /// Query results keyed by binding name — `data["items"]` feeds
290    /// every node whose `base.query == "items"`. Bindings that failed
291    /// appear with an `{"error": "…"}` payload rather than failing the
292    /// whole view (partial degradation).
293    #[serde(default)]
294    pub data: Map<String, Value>,
295}

297/// Result of the `query` op — one binding's data payload.
298#[derive(Clone, Debug, PartialEq, Serialize, Deserialize)]
299pub struct QueryResult {
300    /// The query/binding name that produced this payload.
301    pub query: String,
302    /// The data — shape is defined by the driver/product; hosts pass
303    /// it through to renderers verbatim.
304    pub data: Value,
305}

307/// Result of the `submit` op — what a form submission produced.
308///
309/// Every field is optional so drivers answer with only what they
310/// have: a bare `{"message":"saved"}` is a complete result. Hosts
311/// treat `errors` non-empty as a validation failure (the form should
312/// re-render with the messages, not navigate away); `view`, when
313/// present, is a fresh [`ViewResult`] to swap in.
314#[derive(Clone, Debug, Default, PartialEq, Serialize, Deserialize)]
315pub struct SubmitResult {
316    /// Human/machine-facing outcome (`"saved myco.theme"`,
317    /// `"applied 2 keys"`). Renderers show it as a status line.
318    #[serde(default, skip_serializing_if = "Option::is_none")]
319    pub message: Option<String>,
320    /// Per-field validation errors keyed by `FormField::key` —
321    /// non-empty means the submit was rejected and the form should
322    /// re-render with these messages. Values are driver-defined
323    /// (usually strings).
324    #[serde(default, skip_serializing_if = "Map::is_empty")]
325    pub errors: Map<String, Value>,
326    /// Driver-defined result payload (what changed, new ids, …) —
327    /// shape is the driver's, hosts pass it through verbatim.
328    #[serde(default, skip_serializing_if = "Option::is_none")]
329    pub data: Option<Value>,
330    /// An updated [`ViewResult`] to re-render in place of the
331    /// submitted view (e.g. the form now shows the saved values).
332    #[serde(default, skip_serializing_if = "Option::is_none")]
333    pub view: Option<Box<ViewResult>>,
334    /// Forward-compat catch-all — unknown result fields a future
335    /// driver emits survive a typed round-trip.
336    #[serde(flatten)]
337    pub extra: Map<String, Value>,
338}
339
340/// Result of the `invoke` op — what an action dispatch produced.
341///
342/// Deliberately the same shape as [`SubmitResult`] (message /
343/// per-param `errors` / `data` / fresh `view` / `extra`) but a
344/// distinct type: action results key `errors` by
345/// [`crate::action::ParamSchema::name`] rather than
346/// `FormField::key`, and the two may diverge later. Every field is
347/// optional — a bare `{"message":"done"}` is a complete result.
348#[derive(Clone, Debug, Default, PartialEq, Serialize, Deserialize)]
349pub struct ActionResult {
350    /// Human/machine-facing outcome — renderers show it as a
351    /// status line.
352    #[serde(default, skip_serializing_if = "Option::is_none")]
353    pub message: Option<String>,
354    /// Per-param validation errors keyed by
355    /// [`crate::action::ParamSchema::name`] — non-empty means the
356    /// invoke was rejected; hosts re-render the form/palette with
357    /// these messages rather than navigating.
358    #[serde(default, skip_serializing_if = "Map::is_empty")]
359    pub errors: Map<String, Value>,
360    /// Driver-defined result payload — hosts pass it through
361    /// verbatim.
362    #[serde(default, skip_serializing_if = "Option::is_none")]
363    pub data: Option<Value>,
364    /// An updated [`ViewResult`] to re-render in place — the
365    /// refresh-after-mutation path.
366    #[serde(default, skip_serializing_if = "Option::is_none")]
367    pub view: Option<Box<ViewResult>>,
368    /// Driver-requested redirect — a route **template** (same
369    /// `{…}` syntax and context as [`crate::action::Action::route`])
370    /// the host should interpolate over the invoke context
371    /// (`target`/`params`/`item`/`view`) and navigate to *after*
372    /// applying `view`/`data`. Lets a driver turn "mark done" into
373    /// "navigate to the next item" without the host hard-coding
374    /// product routes. Hosts that don't implement navigation may
375    /// ignore it.
376    #[serde(default, skip_serializing_if = "Option::is_none")]
377    pub route: Option<String>,
378    /// Forward-compat catch-all.
379    #[serde(flatten)]
380    pub extra: Map<String, Value>,
381}

386/// Errors pavo-protocol itself can produce (parsing helpers, not the
387/// wire — wire errors travel as `Response::err` strings).
388#[derive(Debug, thiserror::Error)]
389pub enum WireError {
390    /// The line wasn't valid JSON for the expected type.
391    #[error("invalid wire JSON: {0}")]
392    Json(#[from] serde_json::Error),
393    /// The request `op` isn't in the v1 op set.
394    #[error("unknown op: {op}")]
395    UnknownOp {
396        /// The unrecognized op string.
397        op: String,
398    },
399}
400
401/// Parse one ndjson line into a [`Request`].
402///
403/// # Errors
404/// [`WireError::Json`] on malformed JSON; [`WireError::UnknownOp`]
405/// when `op` is missing or not a v1 op.
406pub fn parse_request(line: &str) -> Result<Request, WireError> {
407    let value: Value = serde_json::from_str(line)?;
408    let op = value
409        .get("op")
410        .and_then(Value::as_str)
411        .ok_or_else(|| WireError::UnknownOp {
412            op: "<missing>".to_string(),
413        })?;
414    match op {
415        "manifest" | "views" | "view" | "query" | "render" | "submit" | "invoke" | "jobs"
416        | "notifications" => Ok(serde_json::from_value(value)?),
417        other => Err(WireError::UnknownOp {
418            op: other.to_string(),
419        }),
420    }
421}

423/// Serialize a [`Response`] to its single-line ndjson form.
424///
425/// # Errors
426/// [`WireError::Json`] if serialization fails (should not happen for
427/// well-formed responses).
428pub fn response_line(response: &Response) -> Result<String, WireError> {
429    Ok(serde_json::to_string(response)?)
430}

440    /// `submit` parses from its wire shape and round-trips; absent
441    /// `form`/`values` default cleanly (old hosts may omit them).
442    #[test]
443    fn submit_request_roundtrip() {
444        let req = parse_request(
445            r#"{"op":"submit","view":"settings","form":"prefs","values":{"theme":"dark","zoom":2}}"#,
446        )
447        .unwrap();
448        match req {
449            Request::Submit { view, form, values } => {
450                assert_eq!(view, "settings");
451                assert_eq!(form.as_deref(), Some("prefs"));
452                assert_eq!(values["theme"], json!("dark"));
453                assert_eq!(values["zoom"], json!(2));
454            }
455            other => panic!("expected Submit, got {other:?}"),
456        }
457        // Minimal form: only `view` required.
458        let req = parse_request(r#"{"op":"submit","view":"v"}"#).unwrap();
459        match req {
460            Request::Submit { view, form, values } => {
461                assert_eq!(view, "v");
462                assert!(form.is_none());
463                assert!(values.is_empty());
464            }
465            other => panic!("expected Submit, got {other:?}"),
466        }
467        // Serialized form keeps the same shape.
468        let line = serde_json::to_string(&Request::Submit {
469            view: "v".into(),
470            form: None,
471            values: Map::new(),
472        })
473        .unwrap();
474        assert_eq!(line, r#"{"op":"submit","view":"v"}"#);
475    }
