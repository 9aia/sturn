1//! Interaction actions — the declarative verb layer of the protocol.
2//!
3//! An [`Action`] is everything a surface needs to offer one
4//! interactive operation: a stable `id`, a `label`, an optional
5//! `icon`, keyboard `keys`, typed `params` (each a
6//! [`ParamSchema`] — enough for hosts to auto-generate a form), a
7//! `danger` level telling hosts how hard to confirm, and an
8//! optional `target` description naming the context the action
9//! needs (an item id, a row, …).
10//!
11//! Actions are declared in three places, all the same type:
12//!
13//! - [`crate::manifest::DriverManifest::actions`] — driver-global
14//!   verbs (new item, refresh, …).
15//! - [`crate::view_node::NodeBase::actions`] — scoped to a view
16//!   node; every ViewNode kind inherits the field.
17//! - [`crate::view_node::TreeItem::actions`] — per-row actions on
18//!   trees (the only item type with a stable `id` today).
19//!
20//! Hosts render them as buttons (web), keys/palette entries (TUI),
21//! or subcommands/flags (CLI) and dispatch through the wire
22//! `invoke` op ([`crate::wire::Request::Invoke`]). `target` on the
23//! wire request is opaque JSON — typically the selected row's raw
24//! value or an item id — the driver interprets it.
25//!
26//! Actions may also be *derived* rather than declared: drivers
27//! whose ops manifest carries param schemas get actions +
28//! auto-generated forms for free (see `pavo-core`'s ops→actions
29//! derivation). [`Action::params`] empty means the action takes
30//! only its `target` context.

35/// How dangerous an [`Action`] is — drives host confirmation UX.
36///
37/// - `none` — fire immediately (read-ish ops, navigation).
38/// - `confirm` — hosts show a confirm dialog/modal; CLI requires
39///   `--yes` or a TTY prompt.
40/// - `type-target` — hosts require the user to type the target's
41///   id (e.g. `T-012`) before dispatch — the "type the repo name to
42///   delete" pattern for irreversible ops.
43///
44/// Derivation rule: ops-manifest actions built from `mutates: true`
45/// ops default to `confirm` when the driver doesn't declare a
46/// level; everything else defaults to `none`.
47#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, Serialize, Deserialize)]
48#[serde(rename_all = "kebab-case")]
49pub enum DangerLevel {
50    /// No confirmation — dispatch on activation.
51    #[default]
52    None,
53    /// One-step confirm dialog.
54    Confirm,
55    /// Type-the-target-id confirmation.
56    TypeTarget,
57}

59/// The input kind a [`ParamSchema`] declares — richer than the
60/// ops-manifest `arg|opt|flag` (which maps params onto argv) and
61/// wider than [`crate::view_node::FormFieldKind`] (which is a
62/// renderer hint). Hosts use it to pick widgets: `enum` → select,
63/// `bool` → toggle, `ref` → fuzzy picker over a query's data,
64/// `string[]` → token input.
65#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, Serialize, Deserialize)]
66#[serde(rename_all = "lowercase")]
67pub enum ParamKind {
68    /// Single-line string.
69    #[default]
70    String,
71    /// Multi-line text.
72    Text,
73    /// Floating-point number.
74    Number,
75    /// Integer.
76    Integer,
77    /// Boolean toggle/checkbox.
78    Bool,
79    /// One-of selection from `options`.
80    Enum,
81    /// Date — `YYYY-MM-DD` convention; hosts may offer pickers.
82    Date,
83    /// Reference picker — the host populates choices from the
84    /// view's `ref_query` binding data (objects with `id`/`label`
85    /// or `id`/`title` fields, or bare strings).
86    Ref,
87    /// List of strings — token/tag input.
88    #[serde(rename = "string[]")]
89    StringList,
90}

92/// One typed parameter of an [`Action`] — the auto-form substrate.
93///
94/// Validation (`required`, `pattern`, `min`/`max`, `enum` via
95/// `options`) is enforced host-side for UX *and* driver-side for
96/// truth; failures come back keyed by `name` in the result's
97/// `errors` map (same convention as [`crate::wire::SubmitResult`]).
98#[derive(Clone, Debug, PartialEq, Serialize, Deserialize)]
99pub struct ParamSchema {
100    /// Param name — the key submitted in the invoke `params` map.
101    pub name: String,
102    /// Input kind — selects the host's widget.
103    #[serde(default)]
104    pub kind: ParamKind,
105    /// Whether the param must be present/non-empty on invoke.
106    #[serde(default)]
107    pub required: bool,
108    /// Human-facing label; hosts default to a humanized `name`.
109    #[serde(default, skip_serializing_if = "Option::is_none")]
110    pub label: Option<String>,
111    /// Help text rendered under the field.
112    #[serde(default, skip_serializing_if = "Option::is_none")]
113    pub doc: Option<String>,
114    /// Hint text (secondary help, e.g. format examples).
115    #[serde(default, skip_serializing_if = "Option::is_none")]
116    pub hint: Option<String>,
117    /// Placeholder shown while empty.
118    #[serde(default, skip_serializing_if = "Option::is_none")]
119    pub placeholder: Option<String>,
120    /// Default value (JSON — string/number/bool/array per `kind`).
121    #[serde(default, skip_serializing_if = "Option::is_none")]
122    pub default: Option<Value>,
123    /// Regex the value must match (string kinds).
124    #[serde(default, skip_serializing_if = "Option::is_none")]
125    pub pattern: Option<String>,
126    /// Numeric lower bound (`number`/`integer` kinds).
127    #[serde(default, skip_serializing_if = "Option::is_none")]
128    pub min: Option<f64>,
129    /// Numeric upper bound (`number`/`integer` kinds).
130    #[serde(default, skip_serializing_if = "Option::is_none")]
131    pub max: Option<f64>,
132    /// Allowed values (`enum` kind; also documents `ref` choices
133    /// when the picker is static).
134    #[serde(default, skip_serializing_if = "Vec::is_empty")]
135    pub options: Vec<String>,
136    /// For `ref` kind — the view query-binding name whose data
137    /// populates the picker (e.g. `"items"`, `"actors"`).
138    #[serde(default, skip_serializing_if = "Option::is_none")]
139    pub ref_query: Option<String>,
140}

142/// One interactive operation — see the module docs for the three
143/// declaration sites and the `invoke` dispatch path.
144#[derive(Clone, Debug, PartialEq, Serialize, Deserialize)]
145pub struct Action {
146    /// Stable id — unique within its declaration scope (manifest,
147    /// node, or item). Hosts namespace display as
148    /// `<driver>:<view>:<id>` where needed; drivers resolve the
149    /// bare id against the request's `view` context first, then
150    /// manifest-global actions.
151    pub id: String,
152    /// Human-facing label for buttons/menu entries.
153    pub label: String,
154    /// Optional icon hint (same resolution rules as
155    /// [`crate::manifest::DriverManifest::icon`]).
156    #[serde(default, skip_serializing_if = "Option::is_none")]
157    pub icon: Option<String>,
158    /// Confirmation level — see [`DangerLevel`].
159    #[serde(default)]
160    pub danger: DangerLevel,
161    /// Keyboard shortcuts — single chars or chord strings
162    /// (`"dd"`, `"ctrl+k"`). TUI binds them; other hosts may ignore
163    /// or map them to CLI flags.
164    #[serde(default, skip_serializing_if = "Vec::is_empty")]
165    pub keys: Vec<String>,
166    /// Typed params — hosts auto-generate a form from these when
167    /// the action is activated without a hand-authored form.
168    #[serde(default, skip_serializing_if = "Vec::is_empty")]
169    pub params: Vec<ParamSchema>,
170    /// What context the action needs — a hint for hosts ("item",
171    /// "row", "selection") and for palette grouping. The actual
172    /// context travels in the invoke request's `target`.
173    #[serde(default, skip_serializing_if = "Option::is_none")]
174    pub target: Option<String>,
175    /// Longer description for palette detail lines / tooltips.
176    #[serde(default, skip_serializing_if = "Option::is_none")]
177    pub description: Option<String>,
178    /// Cross-driver target — reserved: when set, the host should
179    /// resolve `driver` (a `tamias-driver-<type>` name) and issue
180    /// the invoke against *that* driver's `serve`. v1 hosts may
181    /// ignore it and treat the action as local (same convention as
182    /// [`crate::manifest::QueryBinding::driver`]).
183    #[serde(default, skip_serializing_if = "Option::is_none")]
184    pub driver: Option<String>,
185    /// Navigate-instead-of-dispatch — a route **template** the host
186    /// interpolates and navigates to locally when the action is
187    /// activated, *instead of* sending `invoke` to the driver.
188    ///
189    /// Template syntax: `{path.to.value}` placeholders resolved
190    /// against the invoke context — `target`, `params`, `item`,
191    /// `view`, `driver` (see `pavo_core::commands::interpolate`
192    /// for the exact rules; missing keys interpolate to empty).
193    /// Example: `"/myco/item?id={target.id}"` or
194    /// `"/myco/item?id={item.id}&tab={params.tab}"`.
195    ///
196    /// `danger`/`params` still apply host-side when a route is set
197    /// (a form may collect params the template consumes), but no
198    /// wire round-trip happens — navigation is host-local.
199    #[serde(default, skip_serializing_if = "Option::is_none")]
200    pub route: Option<String>,
201}

203#[cfg(test)]
204#[cfg_attr(coverage, coverage(off))]
205mod tests {
206    #![allow(clippy::unwrap_used)]
207    use super::*;
208    use serde_json::json;

210    #[test]
211    fn action_roundtrips_minimal_and_full() {
212        let min: Action = serde_json::from_value(json!({
213            "id": "refresh", "label": "Refresh"
214        }))
215        .unwrap();
216        assert_eq!(min.danger, DangerLevel::None);
217        assert!(min.params.is_empty());
218
219        let full: Action = serde_json::from_value(json!({
220            "id": "todo.done", "label": "Mark done", "icon": "check",
221            "danger": "type-target", "keys": ["dd", "D"],
222            "target": "item",
223            "params": [
224                {"name": "id", "kind": "ref", "required": true,
225                 "ref_query": "items"},
226                {"name": "note", "kind": "text", "hint": "what happened"}
227            ]
228        }))
229        .unwrap();
230        assert_eq!(full.danger, DangerLevel::TypeTarget);
231        assert_eq!(full.params[0].kind, ParamKind::Ref);
232        assert_eq!(full.params[0].ref_query.as_deref(), Some("items"));
233        assert_eq!(full.params[1].kind, ParamKind::Text);
234    }
235
236    #[test]
237    fn param_kind_wire_spellings() {
238        // Mirrors the `kind` enum in protocol/v1/*.schema.json.
239        for (wire, kind) in [
240            ("string", ParamKind::String),
241            ("text", ParamKind::Text),
242            ("number", ParamKind::Number),
243            ("integer", ParamKind::Integer),
244            ("bool", ParamKind::Bool),
245            ("enum", ParamKind::Enum),
246            ("date", ParamKind::Date),
247            ("ref", ParamKind::Ref),
248            ("string[]", ParamKind::StringList),
249        ] {
250            let p: ParamSchema = serde_json::from_value(json!({"name":"x","kind":wire})).unwrap();
251            assert_eq!(p.kind, kind, "wire spelling {wire}");
252            assert_eq!(serde_json::to_value(p.kind).unwrap(), wire);
253        }
254    }
255
256    #[test]
257    fn action_route_roundtrips_and_defaults_none() {
258        // Absent → None, omitted on serialize.
259        let a: Action = serde_json::from_value(json!({"id":"x","label":"X"})).unwrap();
260        assert!(a.route.is_none());
261        assert!(serde_json::to_value(&a).unwrap().get("route").is_none());
262
263        // Present → round-trips; unknown sibling fields ignored.
264        let a: Action = serde_json::from_value(json!({
265            "id":"open","label":"Open","route":"/myco/item?id={item.id}",
266            "future_field": {"x": 1}
267        }))
268        .unwrap();
269        assert_eq!(a.route.as_deref(), Some("/myco/item?id={item.id}"));
270        let v = serde_json::to_value(&a).unwrap();
271        assert_eq!(v["route"], "/myco/item?id={item.id}");
272        assert!(v.get("future_field").is_none());
273    }
274
275    #[test]
276    fn danger_level_spellings() {
277        for (wire, level) in [
278            ("none", DangerLevel::None),
279            ("confirm", DangerLevel::Confirm),
280            ("type-target", DangerLevel::TypeTarget),
281        ] {
282            let a: Action =
283                serde_json::from_value(json!({"id":"x","label":"X","danger":wire})).unwrap();
284            assert_eq!(a.danger, level);
285        }
286    }
