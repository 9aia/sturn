1//! The unified **command layer** — everything invocable resolves to
2//! one grammar, one resolver, every host.
3//!
4//! ```text
5//! action:<id>              invoke the action (driver dispatch)
6//! route:<path>             host navigation (template over target/params)
7//! host.<builtin>           host capability (filter, help, reload, …)
8//! driver:<name>:<id>       cross-driver action (reserved; resolves locally)
9//! ```
10//!
11//! [`Command::parse`] turns a command string into a [`Command`];
12//! [`resolve_command`] maps it plus a [`ResolveContext`] (current
13//! driver, manifest, view result, sibling-driver manifests, invoke
14//! target/params) to a [`Resolved`] the host executes:
15//!
16//! - [`Resolved::Dispatch`] — send `{"op":"invoke"}` via
17//!   [`crate::Host::invoke`] (the `driver`/`view`/`action`/`target`/
18//!   `params` fields are already resolved; `declared` carries the
19//!   [`Action`] when one was found, for params/danger UX).
20//! - [`Resolved::Navigate`] — interpolate-free route string; the
21//!   host navigates locally (no wire round-trip). Produced by
22//!   `route:` commands, [`Action::route`], and
23//!   [`pavo_protocol::ViewSpec::item_route`] templates.
24//! - [`Resolved::Builtin`] — a [`HostBuiltin`] the host implements
25//!   itself (see [`BUILTINS`] for the catalog + suggested keys).
26//! - [`Resolved::Unsupported`] — the command parsed but can't
27//!   resolve here (unknown driver, …); display-only.
28//!
29//! Route **templates** (`{a.b.c}` placeholders) are interpolated by
30//! [`interpolate`] over a JSON-ish context — see
31//! [`ResolveContext::interpolate_ctx`] for the shape
32//! (`{driver, view, target, params, item, data}`). `ActionResult.route`
33//! (driver redirect post-invoke) uses the same rules: hosts
34//! interpolate it over the invoke context and navigate.
35//!
36//! **Bindings**: declarations suggest, users override. [`Action::keys`],
37//! [`MenuEntry::keys`], and [`BUILTINS`] default keys are *suggestions*;
38//! a host keymap file (e.g. `keys.toml` — host config, not protocol)
39//! remaps by command id. [`effective_bindings`] merges both into the
40//! sorted binding list hosts render in help/palette UI; specs
41//! validate via [`crate::keymap::KeySpec`].

180/// One invocable command — the unified grammar. Serializable via
181/// [`Command::id`]/[`std::fmt::Display`] (the canonical string form
182/// is what keymaps, palette rows, and `MenuEntry.command` carry).
183#[derive(Clone, Debug, PartialEq, Eq)]
184pub enum Command {
185    /// `action:<id>` — dispatch an action on the *current* driver.
186    Action {
187        /// The action id (resolved against node/item/manifest scope
188        /// like the `invoke` op).
189        id: String,
190    },
191    /// `route:<path>` — host-local navigation; `<path>` is a route
192    /// template interpolated over the invoke context.
193    Route {
194        /// The route template (may contain `{…}` placeholders).
195        template: String,
196    },
197    /// `host.<builtin>` — a host capability (see [`BUILTINS`]).
198    Host(HostBuiltin),
199    /// `driver:<name>:<id>` — an action on a *named* driver
200    /// (cross-driver; v1 resolves it against that driver's manifest
201    /// when the host knows it, else [`Resolved::Unsupported`]).
202    Driver {
203        /// Driver name — the `<type>` in `tamias-driver-<type>`.
204        driver: String,
205        /// The action id on that driver.
206        id: String,
207    },
208}
209
210impl Command {
211    /// Parse the command grammar — `None` for anything outside the
212    /// four namespaces (unknown namespaces degrade to display-only;
213    /// hosts keep the raw string for labels).
214    #[must_use]
215    pub fn parse(spec: &str) -> Option<Self> {
216        let s = spec.trim();
217        if s.is_empty() {
218            return None;
219        }
220        if let Some(rest) = s.strip_prefix("action:") {
221            return (!rest.is_empty()).then(|| Self::Action {
222                id: rest.to_string(),
223            });
224        }
225        if let Some(rest) = s.strip_prefix("route:") {
226            return (!rest.is_empty()).then(|| Self::Route {
227                template: rest.to_string(),
228            });
229        }
230        if let Some(rest) = s.strip_prefix("driver:") {
231            let (driver, id) = rest.split_once(':')?;
232            if driver.is_empty() || id.is_empty() {
233                return None;
234            }
235            return Some(Self::Driver {
236                driver: driver.to_string(),
237                id: id.to_string(),
238            });
239        }
240        if let Some(name) = s.strip_prefix("host.") {
241            return HostBuiltin::parse(name).map(Self::Host);
242        }
243        None
244    }
245
246    /// The canonical command-id string — round-trips through
247    /// [`Command::parse`]. This is the key commands are bound by in
248    /// user keymaps (`keys.toml`).
249    #[must_use]
250    pub fn id(&self) -> String {
251        match self {
252            Self::Action { id } => format!("action:{id}"),
253            Self::Route { template } => format!("route:{template}"),
254            Self::Host(b) => b.id().to_string(),
255            Self::Driver { driver, id } => format!("driver:{driver}:{id}"),
256        }
257    }
258}
259
260impl std::fmt::Display for Command {
261    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
262        f.write_str(&self.id())
263    }
264}

266/// What [`resolve_command`] needs to know — everything is borrowed
267/// so hosts can rebuild it cheaply per keypress/palette-open.
268#[derive(Clone, Copy)]
269pub struct ResolveContext<'a> {
270    /// Current driver kind (the `<type>` in `tamias-driver-<type>`)
271    /// — the default dispatch target for `action:` commands.
272    pub driver: &'a str,
273    /// Current driver's manifest — action lookup (manifest scope)
274    /// and view fallback.
275    pub manifest: &'a DriverManifest,
276    /// Current [`ViewResult`] when a view is resolved — enables
277    /// node/item-scoped action lookup and `data` in interpolation.
278    pub view: Option<&'a ViewResult>,
279    /// Manifests of *all* discovered drivers (including the current
280    /// one) — the lookup table for `driver:<name>:<id>` commands.
281    /// Hosts build this once per session from [`crate::Host`].
282    pub drivers: &'a BTreeMap<String, DriverManifest>,
283    /// The invoke `target` JSON (selected row, item id object, …) —
284    /// becomes `target` on dispatches and `{target.*}` in templates.
285    pub target: Option<&'a Value>,
286    /// Collected invoke params — `{params.*}` in templates, carried
287    /// verbatim on dispatches.
288    pub params: &'a Params,
289}

291/// The JSON-ish interpolation context — what [`interpolate`]
292/// resolves `{a.b}` paths against. Built by
293/// [`ResolveContext::interpolate_ctx`]; hosts interpolating
294/// `item_route`/`ActionResult.route` should use the same shape.
295///
296/// Keys present (when their source exists):
297///
298/// - `driver` — current driver name (string)
299/// - `view` — current view id (string)
300/// - `data` — the view's query-binding data map
301/// - `target` — the invoke target JSON (verbatim)
302/// - `params` — the invoke params map
303/// - `item` — the selected [`TreeItem`] as JSON (see
304///   [`item_value`]) — only when the caller supplies one
305#[must_use]
306pub fn interpolate_ctx_value(
307    driver: &str,
308    view: Option<&ViewResult>,
309    target: Option<&Value>,
310    params: &Params,
311    item: Option<&Value>,
312) -> Value {
313    let mut map = Map::new();
314    map.insert("driver".into(), Value::from(driver));
315    if let Some(vr) = view {
316        map.insert("view".into(), Value::from(vr.view.id.clone()));
317        map.insert("data".into(), Value::Object(vr.data.clone()));
318    }
319    if let Some(t) = target {
320        map.insert("target".into(), t.clone());
321    }
322    map.insert("params".into(), Value::Object(params.clone()));
323    if let Some(i) = item {
324        map.insert("item".into(), i.clone());
325    }
326    Value::Object(map)
327}

329impl ResolveContext<'_> {
330    /// The interpolation context for this resolve — see
331    /// [`interpolate_ctx_value`]. `item` is a [`item_value`]-shaped
332    /// JSON (or any JSON the host wants addressable as `{item.*}`).
333    #[must_use]
334    pub fn interpolate_ctx(&self, item: Option<&Value>) -> Value {
335        interpolate_ctx_value(self.driver, self.view, self.target, self.params, item)
336    }
337}

339/// A [`TreeItem`] as interpolation JSON — `{item.id}`,
340/// `{item.label}`, `{item.detail}`, `{item.icon}`, plus every
341/// `props` key flattened to top level (`{item.priority}` works when
342/// the item carries `props.priority`). Standard fields win over
343/// same-named props. `children`/`actions` are *not* included.
344#[must_use]
345pub fn item_value(item: &TreeItem) -> Value {
346    let mut map = Map::new();
347    for (k, v) in &item.props {
348        map.insert(k.clone(), v.clone());
349    }
350    if let Some(id) = &item.id {
351        map.insert("id".into(), Value::from(id.clone()));
352    }
353    map.insert("label".into(), Value::from(item.label.clone()));
354    if let Some(d) = &item.detail {
355        map.insert("detail".into(), Value::from(d.clone()));
356    }
357    if let Some(i) = &item.icon {
358        map.insert("icon".into(), Value::from(i.clone()));
359    }
360    Value::Object(map)
361}

363/// Interpolate a route template — `{path.to.value}` placeholders
364/// resolved against `ctx` (see [`interpolate_ctx_value`] for the
365/// shape). Rules:
366///
367/// - Paths are dot-separated; each segment indexes an object key or
368///   an array position (`{items.0.id}`).
369/// - Missing keys, `null`s, and non-container traversals interpolate
370///   to the empty string.
371/// - Values render as: strings verbatim, numbers/bools via their
372///   JSON spelling, arrays/objects as compact JSON.
373/// - `{{` and `}}` escape literal braces. No nesting inside `{…}`.
374/// - A lone `{` or `}` (unmatched, unescaped) stays literal.
375#[must_use]
376pub fn interpolate(template: &str, ctx: &Value) -> String {
377    let mut out = String::with_capacity(template.len());
378    let mut rest = template;
379    while let Some(pos) = rest.find(['{', '}']) {
380        out.push_str(&rest[..pos]);
381        let tail = &rest[pos..];
382        if let Some(stripped) = tail.strip_prefix("{{") {
383            out.push('{');
384            rest = stripped;
385            continue;
386        }
387        if let Some(stripped) = tail.strip_prefix("}}") {
388            out.push('}');
389            rest = stripped;
390            continue;
391        }
392        if let Some(stripped) = tail.strip_prefix('}') {
393            out.push('}');
394            rest = stripped;
395            continue;
396        }
397        // '{' — find the closing '}'.
398        match tail[1..].find('}') {
399            Some(end) => {
400                let path = tail[1..1 + end].trim();
401                out.push_str(&lookup(ctx, path));
402                rest = &tail[1 + end + 1..];
403            }
404            None => {
405                out.push('{');
406                rest = &tail[1..];
407            }
408        }
409    }
410    out.push_str(rest);
411    out
412}

414/// Dot-path lookup over a JSON value — see [`interpolate`].
415fn lookup(ctx: &Value, path: &str) -> String {
416    if path.is_empty() {
417        return String::new();
418    }
419    let mut cur = ctx;
420    for seg in path.split('.') {
421        cur = match cur {
422            Value::Object(map) => match map.get(seg) {
423                Some(v) => v,
424                None => return String::new(),
425            },
426            Value::Array(arr) => match seg.parse::<usize>().ok().and_then(|i| arr.get(i)) {
427                Some(v) => v,
428                None => return String::new(),
429            },
430            _ => return String::new(),
431        };
432    }
433    value_string(cur)
434}
435
436/// JSON value → interpolated string.
437fn value_string(v: &Value) -> String {
438    match v {
439        Value::String(s) => s.clone(),
440        Value::Null => String::new(),
441        Value::Number(n) => n.to_string(),
442        Value::Bool(b) => b.to_string(),
443        other => serde_json::to_string(other).unwrap_or_default(),
444    }
445}

447/// The outcome of [`resolve_command`] — what the host executes.
448#[derive(Clone, Debug)]
449pub enum Resolved {
450    /// Dispatch `action` on `driver` via the `invoke` wire op.
451    /// Boxed — [`ResolvedInvoke`] is large relative to the other
452    /// variants and resolutions are produced per activation.
453    Dispatch(Box<ResolvedInvoke>),
454    /// Navigate locally to `route` (already interpolated).
455    Navigate {
456        /// The interpolated route (`/myco/item?id=T-1`).
457        route: String,
458    },
459    /// A host builtin — filter/help/reload/editor/palette/quit.
460    Builtin(HostBuiltin),
461    /// Parsed but unresolvable here (unknown driver, …) — render
462    /// display-only.
463    Unsupported,
464}

466/// A resolved `invoke` dispatch — every field ready for
467/// [`crate::Host::invoke`] (or the driver's `serve` directly).
468#[derive(Clone, Debug)]
469pub struct ResolvedInvoke {
470    /// Driver kind to invoke on.
471    pub driver: String,
472    /// View context for the invoke request (`None` → host falls
473    /// back like [`crate::Host::invoke`]: target's view, then the
474    /// driver's first view).
475    pub view: Option<String>,
476    /// The action id to send.
477    pub action: String,
478    /// The declared [`Action`] when found in scope — hosts use it
479    /// for `params`/`danger` UX (auto-form, confirm gates).
480    /// `None` when the id isn't declared locally (ops-derived
481    /// actions the host didn't collect, or driver-side ids) —
482    /// dispatch anyway; the driver answers `ok:false` on unknowns.
483    pub declared: Option<Action>,
484    /// The invoke `target` JSON — `{"id": <item>}` for item-scoped
485    /// actions, else the context's target.
486    pub target: Option<Value>,
487    /// The invoke `params` map.
488    pub params: Params,
489}

491/// An action found by scope lookup, owned (collect borrows the view
492/// result; resolutions outlive the borrow).
493struct FoundAction {
494    action: Action,
495    view: Option<String>,
496    item_id: Option<String>,
497}
498
499fn scope_rank(scope: ActionScope) -> u8 {
500    match scope {
501        ActionScope::Item => 3,
502        ActionScope::Node => 2,
503        ActionScope::Manifest => 1,
504    }
505}
506
507/// Find `id` among collected scoped actions — highest scope
508/// precedence wins (item > node > manifest), first occurrence kept
509/// on ties (same policy as host-side dedupe).
510fn lookup_scoped(scoped: &[ScopedAction<'_>], id: &str) -> Option<FoundAction> {
511    let mut best: Option<&ScopedAction<'_>> = None;
512    for s in scoped {
513        if s.action.id != id {
514            continue;
515        }
516        match best {
517            Some(b) if scope_rank(s.scope) <= scope_rank(b.scope) => {}
518            _ => best = Some(s),
519        }
520    }
521    best.map(|s| FoundAction {
522        action: s.action.clone().into_owned(),
523        view: s.view.clone(),
524        item_id: s.item_id.as_deref().map(str::to_string),
525    })
526}

528/// Look up `id` in `manifest` + optional `view_result` — node and
529/// item actions need the view; manifest-only lookups cover
530/// cross-driver resolves (no foreign view result available).
531fn lookup_action(
532    manifest: &DriverManifest,
533    view: Option<&ViewResult>,
534    id: &str,
535) -> Option<FoundAction> {
536    match view {
537        Some(vr) => lookup_scoped(&collect_actions(manifest, vr), id),
538        None => manifest
539            .actions
540            .iter()
541            .find(|a| a.id == id)
542            .map(|a| FoundAction {
543                action: a.clone(),
544                view: None,
545                item_id: None,
546            }),
547    }
548}

550/// Resolve an action id on `driver`'s manifest — shared by
551/// `action:` (current driver + view) and `driver:` (named driver,
552/// manifest scope only unless it's the current one).
553fn resolve_action(
554    driver: &str,
555    manifest: &DriverManifest,
556    view: Option<&ViewResult>,
557    id: &str,
558    ctx: &ResolveContext<'_>,
559) -> Resolved {
560    let found = lookup_action(manifest, view, id);
561    // An action declaring `route` navigates instead of dispatching —
562    // interpolate over the invoke context (item = {"id":…} for
563    // item-scoped finds).
564    if let Some(f) = &found {
565        if let Some(template) = &f.action.route {
566            let item = f.item_id.as_ref().map(|i| json!({"id": i}));
567            return Resolved::Navigate {
568                route: interpolate(template, &ctx.interpolate_ctx(item.as_ref())),
569            };
570        }
571    }
572    let (declared, target, view_id) = match found {
573        Some(f) => (
574            Some(f.action),
575            f.item_id
576                .map(|i| json!({"id": i}))
577                .or_else(|| ctx.target.cloned()),
578            f.view.or_else(|| view.map(|v| v.view.id.clone())),
579        ),
580        None => (None, ctx.target.cloned(), view.map(|v| v.view.id.clone())),
581    };
582    Resolved::Dispatch(Box::new(ResolvedInvoke {
583        driver: driver.to_string(),
584        view: view_id,
585        action: id.to_string(),
586        declared,
587        target,
588        params: ctx.params.clone(),
589    }))
590}

592/// Resolve a [`Command`] against the current context — the single
593/// entry point hosts call for palette activation, keybindings, and
594/// menu entries. See the module docs for the [`Resolved`] variants.
595#[must_use]
596pub fn resolve_command(command: &Command, ctx: &ResolveContext<'_>) -> Resolved {
597    match command {
598        Command::Host(b) => Resolved::Builtin(*b),
599        Command::Route { template } => Resolved::Navigate {
600            route: interpolate(template, &ctx.interpolate_ctx(None)),
601        },
602        Command::Action { id } => resolve_action(ctx.driver, ctx.manifest, ctx.view, id, ctx),
603        Command::Driver { driver, id } => {
604            if driver == ctx.driver {
605                // Same driver — full scope (node/item actions too).
606                resolve_action(driver, ctx.manifest, ctx.view, id, ctx)
607            } else {
608                match ctx.drivers.get(driver) {
609                    // Foreign driver: manifest scope only — the
610                    // host has no ViewResult for it.
611                    Some(manifest) => resolve_action(driver, manifest, None, id, ctx),
612                    None => Resolved::Unsupported,
613                }
614            }
615        }
616    }
617}

619/// The command a [`MenuEntry`] resolves to — `command` wins when
620/// present; else the legacy pair: `action` (`"<id>"` →
621/// `action:<id>`, `"<driver>:<id>"` → `driver:<driver>:<id>`) or
622/// `route` (`route:<path>`). `None` = display-only entry.
623#[must_use]
624pub fn menu_command(entry: &MenuEntry) -> Option<Command> {
625    if let Some(raw) = &entry.command {
626        if let Some(cmd) = Command::parse(raw) {
627            return Some(cmd);
628        }
629        // Unparseable command — fall through to legacy fields rather
630        // than dropping the entry entirely.
631    }
632    if let Some(action) = &entry.action {
633        if let Some((driver, id)) = action.split_once(':') {
634            if !driver.is_empty() && !id.is_empty() {
635                return Some(Command::Driver {
636                    driver: driver.to_string(),
637                    id: id.to_string(),
638                });
639            }
640        }
641        if !action.is_empty() {
642            return Some(Command::Action { id: action.clone() });
643        }
644    }
645    entry
646        .route
647        .as_ref()
648        .and_then(|r| Command::parse(&format!("route:{r}")))
649}

658/// A bindable command + display metadata — input to
659/// [`effective_bindings`].
660#[derive(Clone, Debug)]
661pub struct BindingSource<'a> {
662    /// What activation resolves to.
663    pub command: Command,
664    /// Display label (palette/help).
665    pub label: Cow<'a, str>,
666    /// Secondary text — description or raw command id.
667    pub detail: Option<Cow<'a, str>>,
668    /// Grouping label (`"actions"`, `"view"`, `"item"`, menu
669    /// `section`, `"host"`).
670    pub section: Option<Cow<'a, str>>,
671    /// Suggested keys — overridden wholesale by a user-keymap entry
672    /// for the same command id.
673    pub keys: Vec<String>,
674}
675
676/// A [`BindingSource`] for a collected [`ScopedAction`] — the
677/// command is `action:<id>` locally, or `driver:<drv>:<id>` when the
678/// action declares a cross-driver `driver`.
679#[must_use]
680pub fn action_binding_source<'a>(scoped: &'a ScopedAction<'a>) -> BindingSource<'a> {
681    let action = scoped.action.as_ref();
682    let command = match &action.driver {
683        Some(driver) => Command::Driver {
684            driver: driver.clone(),
685            id: action.id.clone(),
686        },
687        None => Command::Action {
688            id: action.id.clone(),
689        },
690    };
691    BindingSource {
692        command,
693        label: Cow::Borrowed(action.label.as_str()),
694        detail: action
695            .description
696            .as_deref()
697            .map(Cow::Borrowed)
698            .or(Some(Cow::Borrowed(action.id.as_str()))),
699        section: Some(Cow::Owned(match scoped.scope {
700            ActionScope::Manifest => "actions".to_string(),
701            ActionScope::Node => "view".to_string(),
702            ActionScope::Item => "item".to_string(),
703        })),
704        keys: action.keys.clone(),
705    }
706}

708/// A [`BindingSource`] for a [`MenuEntry`] — `None` for
709/// display-only entries (no resolvable command).
710#[must_use]
711pub fn menu_binding_source(entry: &MenuEntry) -> Option<BindingSource<'_>> {
712    let command = menu_command(entry)?;
713    let detail = command.id();
714    Some(BindingSource {
715        command,
716        label: Cow::Borrowed(entry.title.as_str()),
717        detail: Some(Cow::Owned(detail)),
718        section: entry.section.as_deref().map(Cow::Borrowed),
719        keys: entry.keys.clone(),
720    })
721}

723/// [`BindingSource`]s for every [`BUILTINS`] row — section `"host"`.
724#[must_use]
725pub fn builtin_binding_sources() -> Vec<BindingSource<'static>> {
726    BUILTINS
727        .iter()
728        .map(|b| BindingSource {
729            command: Command::Host(b.builtin),
730            label: Cow::Borrowed(b.label),
731            detail: Some(Cow::Borrowed(b.description)),
732            section: Some(Cow::Borrowed("host")),
733            keys: b.keys.iter().map(|k| k.to_string()).collect(),
734        })
735        .collect()
736}

738/// One merged key→command binding — a row of the host's effective
739/// keymap (help screen, `?` overlay, conflict detection).
740#[derive(Clone, Debug)]
741pub struct CommandBinding {
742    /// Canonical command id (`action:x`, `host.filter`, …).
743    pub id: String,
744    /// The resolved command.
745    pub command: Command,
746    /// Display label.
747    pub label: String,
748    /// Secondary text.
749    pub detail: Option<String>,
750    /// Grouping label.
751    pub section: Option<String>,
752    /// Effective keys — the user-keymap override when present, else
753    /// the declaration's suggestions. Validated through
754    /// [`crate::keymap::KeySpec`]; invalid overlay entries drop out.
755    pub keys: Vec<String>,
756}
757
758/// Merge declared/suggested bindings with a user-keymap overlay.
759///
760/// `overlay` maps **command id → keys**: the value is a
761/// comma-separated list of [`crate::keymap::KeySpec`]s
762/// (`"ctrl+p,/"`), or a single spec. An overlay entry *replaces* the
763/// declaration's suggested keys wholesale; command ids absent from
764/// the overlay keep their suggestions. Sources producing the same
765/// command id dedupe to the first occurrence.
766///
767/// Returns the list sorted by command id — deterministic for help
768/// screens and tests.
769#[must_use]
770pub fn effective_bindings<'a>(
771    sources: impl IntoIterator<Item = BindingSource<'a>>,
772    overlay: &BTreeMap<String, String>,
773) -> Vec<CommandBinding> {
774    let mut out: Vec<CommandBinding> = Vec::new();
775    let mut seen = std::collections::BTreeSet::new();
776    for src in sources {
777        let id = src.command.id();
778        if !seen.insert(id.clone()) {
779            continue;
780        }
781        let keys = match overlay.get(&id) {
782            Some(spec) => spec
783                .split(',')
784                .filter_map(|k| {
785                    let k = k.trim();
786                    (!k.is_empty()).then(|| k.to_string())
787                })
788                .filter(|k| crate::keymap::KeySpec::parse(k).is_some())
789                .collect(),
790            None => src.keys,
791        };
792        out.push(CommandBinding {
793            id,
794            command: src.command,
795            label: src.label.into_owned(),
796            detail: src.detail.map(Cow::into_owned),
797            section: src.section.map(Cow::into_owned),
798            keys,
799        });
800    }
801    out.sort_by(|a, b| a.id.cmp(&b.id));
802    out
803}
