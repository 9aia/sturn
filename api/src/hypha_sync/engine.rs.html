1//! The sync engine — one supervised loop per node, `JoinSet`-managed,
2//! `CancellationToken`-cancellable.
3//!
4//! Per node the cycle is: `list_entities` → upsert + reconcile the
5//! projection → drain the outbox FIFO → tail `/api/events` and apply
6//! index diffs → on stream end/error, mark the node down and reconnect
7//! with capped exponential backoff + jitter. A periodic full-list
8//! reconcile catches anything the feed missed (tombstones are only
9//! visible in listings).
10

27/// Sink for every feed frame the engine applies — the hub's push

31/// Tuning knobs — production defaults; tests shrink the sleeps.
32#[derive(Clone)]
33pub struct SyncOptions {
34    /// TCP connect budget for node calls.
35    pub connect_timeout: Duration,
36    /// End-to-end budget for REST calls (list, post, refetch).
37    pub request_timeout: Duration,
38    /// End-to-end budget for the *direct-delivery* attempt in
39    /// [`SyncHandle::submit_op`] — deliberately below
40    /// `request_timeout`: when the node is unreachable but the
41    /// projection still reads `up` (the sync loop hasn't observed the
42    /// failure yet — feed silence alone takes `sse_idle_timeout` to
43    /// trip), the write must fail fast into the durable outbox rather
44    /// than hold the caller for the full REST budget (T-060). An
45    /// elapse is treated exactly like a transport error: the node is
46    /// marked `down` and the op queues for replay. The orphaned
47    /// in-flight post may still land node-side — the same ambiguity
48    /// every transport timeout already carries.
49    pub submit_timeout: Duration,
50    /// Max feed silence before the SSE subscription is declared dropped
51    /// — reconnected (and the index re-listed) when it lapses. The
52    /// node's heartbeats keep a healthy stream open indefinitely; this
53    /// also bounds how long a dead-silent SSE read can outlive engine
54    /// shutdown.
55    pub sse_idle_timeout: Duration,
56    /// First reconnect delay.
57    pub backoff_min: Duration,
58    /// Reconnect-delay cap.
59    pub backoff_max: Duration,
60    /// Periodic full `list_entities` reconcile while subscribed.
61    pub resync_interval: Duration,
62    /// How long a self-declared *ephemeral* node (its `/api/node`
63    /// `capabilities` carry `"ephemeral"` — CI runners) may stay
64    /// continuously `down` before the engine reaps it: projected
65    /// entities are dropped (a resurrected node resyncs itself) and
66    /// its queued writes dead-letter. Ordinary nodes are never
67    /// reaped — they keep retrying at `backoff_max` forever.
68    pub ephemeral_dead_after: Duration,
69    /// Capacity of the per-node SSE event channel.
70    pub feed_channel: usize,
71    /// Optional per-event hook (node id + frame). Inert `None`.
72    pub on_event: Option<FeedHook>,
73    /// Per-kind policy registry — diff-event projection, tombstone
74    /// semantics, op arbitration, and entity-scoped call params
75    /// resolve through it. Empty = fully generic behavior.
76    pub policies: KindPolicies,
77}

80    fn default() -> Self {
81        Self {
82            connect_timeout: Duration::from_secs(5),
83            request_timeout: Duration::from_secs(15),
84            // One connect budget plus slack — a node that can't accept
85            // and ack a small op POST within this window is treated as
86            // unreachable and the write queues.
87            submit_timeout: Duration::from_secs(8),
88            sse_idle_timeout: Duration::from_secs(60),
89            backoff_min: Duration::from_secs(1),
90            backoff_max: Duration::from_secs(30),

94            on_event: None,

141/// What [`SyncHandle::submit_op`] did with a write.
142#[derive(Clone, Debug)]
143pub enum SubmitOutcome {
144    /// The node was up and accepted the write — the response body.
145    Applied(Value),
146    /// The node is down (or the write hit a transport error) — queued
147    /// for replay on reconnect. `Box` keeps the enum small.
148    Queued(Box<hypha_outbox::OutboxEntry>),
149}

151/// Shared state between the supervisor, the per-node loops, and the
152/// public handle.
153struct Shared {
154    projection: ProjectionStore,
155    outbox: Outbox,
156    options: SyncOptions,
157    /// Registered nodes — also the source of per-node bearer tokens for
158    /// `submit_op`'s direct path.
159    targets: Mutex<HashMap<String, NodeTarget>>,
160    /// Live SSE pump threads — exposed via `SyncHandle::active_pumps`
161    /// so shutdown/teardown can be observed, never blocked on.
162    active_pumps: Arc<AtomicUsize>,
163}
164
165/// Messages the SSE pump thread sends back to the async node loop.
166enum PumpMsg {
167    /// `GET /api/events` returned 2xx — the node has registered the
168    /// broadcast subscription. Always the first message on success so
169    /// the loop can order "subscribe → list" without a diff gap.
170    Subscribed,
171    Event(FeedEvent),
172    /// EOF or read error — the connection is over either way; the
173    /// payload is `None` for a clean close.
174    Closed(Option<NodeError>),
175}

177/// The entry point: spawns the supervisor and returns a [`SyncHandle`].
178pub struct SyncEngine;

180impl SyncEngine {
181    /// Start the engine — one task per node plus a supervisor that owns
182    /// the [`JoinSet`]. Additional nodes can join/leave later via
183    /// [`SyncHandle::add_node`]/[`SyncHandle::remove_node`].
184    pub fn spawn(
185        projection: ProjectionStore,
186        outbox: Outbox,
187        nodes: impl IntoIterator<Item = NodeTarget>,
188        options: SyncOptions,
189    ) -> SyncHandle {
190        // One registry drives both the wire calls and the projection's
191        // column extraction — attach it here so callers can't drift.
192        let projection = projection.with_kind_policies(options.policies.clone());
193        let cancel = CancellationToken::new();
194        let (cmd_tx, cmd_rx) = mpsc::channel::<Cmd>(64);
195        let shared = Arc::new(Shared {
196            projection,
197            outbox,
198            options,
199            targets: Mutex::new(HashMap::new()),
200            active_pumps: Arc::new(AtomicUsize::new(0)),
201        });
202        {
203            let mut targets = shared
204                .targets
205                .lock()
206                .unwrap_or_else(std::sync::PoisonError::into_inner);
207            for target in nodes {
208                targets.insert(target.node.id.clone(), target);
209            }
210        }
211        let supervisor = {
212            let shared = Arc::clone(&shared);
213            let cancel = cancel.clone();
214            tokio::spawn(async move { supervise(shared, cmd_rx, cancel).await })
215        };
216        SyncHandle {
217            shared,
218            cmds: cmd_tx,
219            cancel,
220            supervisor,
221        }
222    }
223}

230/// Handle to a running engine — cloneable; `shutdown` consumes it.
231pub struct SyncHandle {
232    shared: Arc<Shared>,
233    cmds: mpsc::Sender<Cmd>,
234    cancel: CancellationToken,
235    supervisor: tokio::task::JoinHandle<()>,
236}

238impl SyncHandle {
239    /// The projection the engine maintains — safe to read from anywhere.
240    pub fn projection(&self) -> ProjectionStore {
241        self.shared.projection.clone()
242    }
243
244    /// The outbox — inspect `dead_letters`/`pending_for_node` directly.
245    pub fn outbox(&self) -> Outbox {
246        self.shared.outbox.clone()
247    }
248
249    /// Live SSE pump threads — `0` means every node loop is idle or
250    /// offline (used by tests to assert clean teardown).
251    pub fn active_pumps(&self) -> usize {
252        self.shared.active_pumps.load(Ordering::Relaxed)
253    }
254
255    /// The direct-delivery budget [`Self::submit_op`] enforces — the
256    /// hub's synchronous fast paths (entity create) share it so a
257    /// node that stops answering fails fast everywhere, not just on
258    /// op submits (T-060).
259    pub fn submit_timeout(&self) -> Duration {
260        self.shared.options.submit_timeout
261    }
262
263    /// The shared pump counter — cloneable so callers can keep
264    /// observing teardown after `shutdown()` consumes the handle.
265    pub fn pump_counter(&self) -> Arc<AtomicUsize> {
266        Arc::clone(&self.shared.active_pumps)
267    }
268
269    /// Register a node and start syncing it. Re-registering an existing
270    /// id retargets the node — the running loop is aborted and respawned
271    /// with the new target (e.g. a credential minted after boot); the
272    /// projection's node row + session index survive the swap.
273    ///
274    /// # Errors
275    /// [`SyncError::Shutdown`] if the engine is gone; sqlite failure
276    /// writing the node row.
277    pub async fn add_node(&self, target: NodeTarget) -> Result<(), SyncError> {
278        self.shared.projection.upsert_node(&target.node)?;
279        self.cmds
280            .send(Cmd::Add(target))
281            .await
282            .map_err(|_| SyncError::Shutdown)
283    }
284
285    /// Stop syncing a node and drop its projected state.
286    ///
287    /// # Errors
288    /// [`SyncError::Shutdown`] if the engine is gone; sqlite failure.
289    pub async fn remove_node(&self, id: &str) -> Result<(), SyncError> {
290        self.shared.projection.remove_node(id)?;
291        self.cmds
292            .send(Cmd::Remove(id.to_string()))
293            .await
294            .map_err(|_| SyncError::Shutdown)
295    }
296
297    /// A write to a node: posted immediately when the node is up;
298    /// appended to the durable outbox (per-entity FIFO) when it's down
299    /// or the direct post fails on transport. HTTP rejections surface as
300    /// errors — they were delivered and refused, not lost.
301    ///
302    /// # Errors
303    /// [`SyncError::UnknownNode`] for unregistered ids,
304    /// [`SyncError::Node`] when the node rejects the write, and
305    /// outbox/sqlite failures.
306    #[tracing::instrument(
307        name = "sync_submit_op",
308        skip_all,
309        fields(node.id = %write.node_id, entity = %write.entity_id, op = %write.op)
310    )]
311    pub async fn submit_op(&self, write: NodeWrite) -> Result<SubmitOutcome, SyncError> {
312        let target = {
313            self.shared
314                .targets
315                .lock()
316                .unwrap_or_else(std::sync::PoisonError::into_inner)
317                .get(&write.node_id)
318                .cloned()
319        };
320        let Some(target) = target else {
321            return Err(SyncError::UnknownNode(write.node_id.clone()));
322        };
323
324        let up = self
325            .shared
326            .projection
327            .node(&write.node_id)?
328            .is_some_and(|row| row.status == NodeStatus::Up);
329
330        if up {
331            let client = self
332                .shared
333                .options
334                .client(&target.node.url, target.token.clone())?;
335            let entity_id = write.entity_id.clone();
336            let kind = write.entity_kind.clone();
337            let op = write.op.clone();
338            let payload = write.payload.clone();
339            // Bounded by `submit_timeout`, not `request_timeout` —
340            // the direct-delivery probe must fail fast into the
341            // outbox when the node is gone but not yet marked `down`
342            // (T-060). `timeout` drops the await, not the blocking
343            // post: the orphaned call still finishes on its own
344            // `request_timeout` budget and its result is discarded.
345            let result = tokio::time::timeout(
346                self.shared.options.submit_timeout,
347                blocking(client.clone(), move |c| {
348                    c.post_op(&entity_id, kind.as_deref(), &op, &payload)
349                }),
350            )
351            .await;
352            match result {
353                Ok(Ok(value)) => return Ok(SubmitOutcome::Applied(value)),
354                Ok(Err(e)) if e.is_transport() => {
355                    // Fresh evidence the node is gone — reflect it and
356                    // queue; the loop's reconnect drains in order.
357                    let _ = self
358                        .shared
359                        .projection
360                        .set_node_status(&write.node_id, NodeStatus::Down);
361                }
362                Ok(Err(e)) => return Err(SyncError::Node(e)),
363                // The submit budget lapsed — same ruling as a
364                // transport failure: mark down, queue, replay later.
365                Err(_) => {
366                    let _ = self
367                        .shared
368                        .projection
369                        .set_node_status(&write.node_id, NodeStatus::Down);
370                }
371            }
372        }
373
374        let entry = self.shared.outbox.enqueue(
375            &write.node_id,
376            &write.entity_id,
377            &write.op,
378            write.policy,
379            write.payload,
380            &write.idempotency_key,
381            write.ttl_seconds,
382        )?;
383        // A confirmed-dead ephemeral node's writes can never replay —
384        // dead-letter them now rather than leaving them pending until
385        // the sync loop's next reap pass.
386        if ephemeral_dead(
387            &self.shared.projection,
388            &write.node_id,
389            self.shared.options.ephemeral_dead_after,
390        ) {
391            let _ = self
392                .shared
393                .outbox
394                .dead_letter_node(&write.node_id, "ephemeral node down past reap threshold");
395        }
396        Ok(SubmitOutcome::Queued(Box::new(entry)))
397    }

399    /// Stop every loop and await the supervisor. The SSE pump threads
400    /// exit at the next feed frame or the `sse_idle_timeout` read
401    /// deadline — [`Self::active_pumps`] reports stragglers.
402    pub async fn shutdown(self) {
403        self.cancel.cancel();
404        let _ = self.supervisor.await;
405    }
406}

408/// The supervisor — owns the `JoinSet` of per-node loops and the
409/// add/remove command channel.
410async fn supervise(shared: Arc<Shared>, mut rx: mpsc::Receiver<Cmd>, cancel: CancellationToken) {
411    let mut set: JoinSet<()> = JoinSet::new();
412    let mut tasks: HashMap<String, tokio::task::AbortHandle> = HashMap::new();
413    {
414        let targets: Vec<NodeTarget> = shared
415            .targets
416            .lock()
417            .unwrap_or_else(std::sync::PoisonError::into_inner)
418            .values()
419            .cloned()
420            .collect();
421        for target in targets {
422            if shared.projection.upsert_node(&target.node).is_err() {
423                tracing::warn!(node = %target.node.id, "failed to register node in projection");
424            }
425            let abort = set.spawn(node_loop(
426                target.clone(),
427                Arc::clone(&shared),
428                cancel.clone(),
429            ));
430            tasks.insert(target.node.id.clone(), abort);
431        }
432    }
433    loop {
434        tokio::select! {
435            cmd = rx.recv() => match cmd {
436                Some(Cmd::Add(target)) => {
437                    shared
438                        .targets
439                        .lock()
440                        .unwrap_or_else(std::sync::PoisonError::into_inner)
441                        .insert(target.node.id.clone(), target.clone());
442                    // A re-add retargets the node — abort the old loop
443                    // (its SSE pump exits at the next frame or the
444                    // `sse_idle_timeout` deadline) and respawn with the
445                    // new url/token. The projection keeps its rows.
446                    if let Some(prev) = tasks.remove(&target.node.id) {
447                        prev.abort();
448                    }
449                    let abort = set.spawn(node_loop(
450                        target.clone(),
451                        Arc::clone(&shared),
452                        cancel.clone(),
453                    ));
454                    tasks.insert(target.node.id.clone(), abort);
455                }
456                Some(Cmd::Remove(id)) => {
457                    shared
458                        .targets
459                        .lock()
460                        .unwrap_or_else(std::sync::PoisonError::into_inner)
461                        .remove(&id);
462                    if let Some(abort) = tasks.remove(&id) {
463                        abort.abort();
464                    }
465                }
466                None => break,
467            },
468            Some(Err(e)) = set.join_next() => report_join_error(&e),
469            () = cancel.cancelled() => break,
470        }
471    }
472    // `shutdown` aborts whatever didn't observe the token yet; the
473    // loops' own `cancelled` arms exit promptly regardless.
474    set.shutdown().await;
475}

477/// A node-loop task died — panics are the only noteworthy cause.
478/// Named so the arm is one region tests can drive directly.
479fn report_join_error(e: &tokio::task::JoinError) {
480    if e.is_panic() {
481        tracing::error!("node sync loop panicked: {e}");
482    }

487/// Why one connect-cycle ended.
488enum CycleEnd {
489    /// Engine shutdown.
490    Cancelled,
491    /// `list_entities` failed — the node is down; exponential backoff.
492    PreConnect(NodeError),
493    /// We were connected and the SSE stream ended — quick reconnect.
494    Disconnected(NodeError),
495}
496
497#[tracing::instrument(
498    name = "sync_node_loop",
499    skip_all,
500    fields(node.id = %target.node.id, node.url = %target.node.url)
501)]
502async fn node_loop(target: NodeTarget, shared: Arc<Shared>, cancel: CancellationToken) {
503    let node_id = target.node.id.clone();
504    let client = match shared
505        .options
506        .client(&target.node.url, target.token.clone())
507    {
508        Ok(client) => client,
509        Err(e) => {
510            tracing::error!(node = %node_id, "bad node config: {e}");
511            return;
512        }
513    };
514    let mut delay = shared.options.backoff_min;
515    loop {
516        if cancel.is_cancelled() {
517            break;
518        }
519        match connect_once(&target.node, &client, &shared, &cancel).await {
520            CycleEnd::Cancelled => break,
521            CycleEnd::PreConnect(e) => {
522                tracing::debug!(node = %node_id, "connect failed: {e}");
523            }
524            CycleEnd::Disconnected(e) => {
525                tracing::info!(node = %node_id, "connection lost: {e}");
526                delay = shared.options.backoff_min;
527            }
528        }
529        if shared
530            .projection
531            .set_node_status(&node_id, NodeStatus::Down)
532            .is_err()
533        {
534            tracing::warn!(node = %node_id, "failed to mark node down");
535        }
536        // Self-declared ephemeral nodes get reaped once they've been
537        // down past `ephemeral_dead_after`. The check is idempotent,
538        // so it runs on every failed cycle — cheap, and it also
539        // catches writes enqueued between reaps.
540        if ephemeral_dead(
541            &shared.projection,
542            &node_id,
543            shared.options.ephemeral_dead_after,
544        ) {
545            reap_ephemeral_node(&shared, &node_id);
546        }
547        // A previously-healthy stream that dropped retries fast;
548        // repeated connect failures back off exponentially.
549        let wait = jittered(delay);
550        delay = delay.saturating_mul(2).min(shared.options.backoff_max);
551        tokio::select! {
552            () = tokio::time::sleep(wait) => {}
553            () = cancel.cancelled() => break,
554        }
555    }
556}

558/// One connect cycle: subscribe → list → reconcile → drain → apply.
559/// Returns only when the subscription ends or the engine is cancelled.
560///
561/// Ordering: the SSE subscription opens *before* the listing so a diff
562/// emitted while the list is in flight lands on the stream — anything
563/// earlier is covered by the listing itself. The outbox drains last so
564/// the feed events a replayed write triggers are observed.
565#[tracing::instrument(name = "sync_connect", skip_all, fields(node.id = %node.id))]
566async fn connect_once(
567    node: &NodeRef,
568    client: &NodeClient,
569    shared: &Arc<Shared>,
570    cancel: &CancellationToken,
571) -> CycleEnd {
572    // The pump is a dedicated blocking thread feeding an mpsc; the
573    // async side stays cancel-instant.
574    let (tx, mut rx) = mpsc::channel::<PumpMsg>(shared.options.feed_channel);
575    let stop = Arc::new(AtomicBool::new(false));
576    let pump = {
577        let client = client.clone();
578        let stop = Arc::clone(&stop);
579        let shared = Arc::clone(shared);
580        std::thread::spawn(move || sse_pump(&client, &tx, &stop, &shared))
581    };
582
583    // Wait for the subscription handshake (or connect failure).
584    let subscribed = tokio::select! {
585        () = cancel.cancelled() => {
586            stop.store(true, Ordering::Relaxed);
587            // The pump may be parked in `blocking_send` on a full
588            // channel — dropping `rx` fails that send so the join
589            // below can't hang.
590            drop(rx);
591            let _ = tokio::task::spawn_blocking(move || pump.join()).await;
592            return CycleEnd::Cancelled;
593        }
594        msg = rx.recv() => match msg {
595            Some(PumpMsg::Subscribed) => true,
596            Some(PumpMsg::Closed(e)) => {
597                // Pre-subscribe closes always carry the connect error —
598                // `unwrap_or` keeps the fallback a plain evaluated arg
599                // instead of a dead closure.
600                return CycleEnd::PreConnect(e.unwrap_or(NodeError::Malformed(
601                    "event stream closed before subscribe".into(),
602                )));
603            }
604            Some(PumpMsg::Event(_)) | None => {
605                return CycleEnd::Disconnected(NodeError::Malformed(
606                    "pump ended before subscribe".into(),
607                ));
608            }
609        },
610    };
611    debug_assert!(subscribed);
612
613    // Listing — a success means the node is reachable *and* the
614    // projection is freshly reconciled (tombstones included).
615    let docs = match blocking(client.clone(), move |c| c.list_entities(None, true)).await {
616        Ok(s) => s,
617        Err(e) => {
618            stop.store(true, Ordering::Relaxed);
619            drop(rx);
620            let _ = tokio::task::spawn_blocking(move || pump.join()).await;
621            return CycleEnd::Disconnected(e);
622        }
623    };
624    if let Err(e) = shared.projection.apply_listing(&node.id, &docs) {
625        tracing::warn!(node = %node.id, "projection apply_listing failed: {e}");
626    }
627    // Self-reported lifecycle + fleet labels: `/api/node`'s
628    // `capabilities` carry `"ephemeral"` for disposable nodes (CI
629    // runners) plus integration-defined tags (`"runner:linux"`, …)
630    // schedulers match `runs_on` requirements against. Best-effort —
631    // a node that doesn't answer keeps its previous values.
632    if let Ok(info) = blocking(client.clone(), |c| c.get_json("/api/node")).await {
633        let capabilities: Vec<String> = info
634            .get("capabilities")
635            .and_then(Value::as_array)
636            .map(|caps| {
637                caps.iter()
638                    .filter_map(|c| c.as_str().map(str::to_string))
639                    .collect()
640            })
641            .unwrap_or_default();
642        let ephemeral = capabilities.iter().any(|c| c == "ephemeral");
643        if let Err(e) = shared.projection.set_node_ephemeral(&node.id, ephemeral) {
644            tracing::warn!(node = %node.id, "failed to record node capabilities: {e}");
645        }
646        if let Err(e) = shared
647            .projection
648            .set_node_capabilities(&node.id, &capabilities)
649        {
650            tracing::warn!(node = %node.id, "failed to record node capabilities: {e}");
651        }
652    }
653    if let Err(e) = shared.projection.set_node_status(&node.id, NodeStatus::Up) {
654        tracing::warn!(node = %node.id, "failed to mark node up: {e}");
655    }
656    drain_outbox(node, client, shared).await;
657
658    let mut resync = tokio::time::interval_at(
659        tokio::time::Instant::now() + shared.options.resync_interval,
660        shared.options.resync_interval,
661    );
662    // The stream's silence bound — re-armed by every wire frame (the
663    // node's heartbeats keep a quiet feed alive). Needed because the
664    // ureq body budget can't express a per-read idle timeout (see
665    // `open_stream`), so this deadline is enforced engine-side.
666    let mut silence = tokio::time::Instant::now() + shared.options.sse_idle_timeout;
667    let end = loop {
668        tokio::select! {
669            () = cancel.cancelled() => break CycleEnd::Cancelled,
670            () = tokio::time::sleep_until(silence) => {
671                break CycleEnd::Disconnected(NodeError::Malformed(
672                    "event stream silent".into(),
673                ));
674            }
675            msg = rx.recv() => match msg {
676                None => break CycleEnd::Disconnected(NodeError::Malformed("pump ended".into())),
677                // The pump sends Subscribed exactly once, before any
678                // frames — a duplicate is inert.
679                Some(PumpMsg::Subscribed) => {}
680                Some(PumpMsg::Closed(e)) => {
681                    break CycleEnd::Disconnected(e.unwrap_or_else(|| {
682                        NodeError::Malformed("event stream ended".into())
683                    }));
684                }
685                Some(PumpMsg::Event(event)) => {
686                    silence = tokio::time::Instant::now() + shared.options.sse_idle_timeout;
687                    if let Err(e) = apply_event(node, client, shared, event).await {
688                        tracing::warn!(node = %node.id, "apply event failed: {e}");
689                    }
690                }
691            },
692            _ = resync.tick() => {
693                match blocking(client.clone(), |c| c.list_entities(None, true)).await {
694                    Ok(docs) => {
695                        if let Err(e) = shared.projection.apply_listing(&node.id, &docs) {
696                            tracing::warn!(node = %node.id, "resync failed: {e}");
697                        }
698                    }
699                    Err(e) => break CycleEnd::Disconnected(e),
700                }
701            }
702        }
703    };
704
705    stop.store(true, Ordering::Relaxed);
706    // `rx` is dropped first — a pump parked in `blocking_send` on a
707    // full channel fails immediately instead of hanging the join.
708    // Otherwise bounded by `sse_idle_timeout` (the read deadline).
709    drop(rx);
710    let _ = tokio::task::spawn_blocking(move || pump.join()).await;
711    end
712}

714/// Blocking SSE pump — one std thread per live connection. Reads
715/// frames, forwards them to the async loop, checks `stop` between
716/// frames (the node's heartbeat cadence and the `sse_idle_timeout`
717/// silence bound cap the wakeup latency).
718fn sse_pump(
719    client: &NodeClient,
720    tx: &mpsc::Sender<PumpMsg>,
721    stop: &Arc<AtomicBool>,
722    shared: &Arc<Shared>,
723) {
724    shared.active_pumps.fetch_add(1, Ordering::Relaxed);
725    let finish = |msg: PumpMsg| {
726        let _ = tx.blocking_send(msg);
727        shared.active_pumps.fetch_sub(1, Ordering::Relaxed);
728    };
729    match client.events() {
730        Err(e) => finish(PumpMsg::Closed(Some(e))),
731        Ok(mut stream) => {
732            // Acknowledge the subscription before frames — the async
733            // side lists only after this, closing the diff gap.
734            if tx.blocking_send(PumpMsg::Subscribed).is_err() {
735                shared.active_pumps.fetch_sub(1, Ordering::Relaxed);
736                return;
737            }
738            loop {
739                if stop.load(Ordering::Relaxed) {
740                    shared.active_pumps.fetch_sub(1, Ordering::Relaxed);
741                    return;
742                }
743                match stream.next_event() {
744                    Ok(Some(event)) => {
745                        if tx.blocking_send(PumpMsg::Event(event)).is_err() {
746                            shared.active_pumps.fetch_sub(1, Ordering::Relaxed);
747                            return;
748                        }
749                    }
750                    Ok(None) => {
751                        finish(PumpMsg::Closed(None));
752                        return;
753                    }
754                    Err(e) => {
755                        finish(PumpMsg::Closed(Some(e)));
756                        return;
757                    }
758                }
759            }
760        }
761    }
762}

764/// Drain queued writes oldest-first. Per-entity FIFO is preserved by
765/// skipping the rest of an entity's entries once one fails back to
766/// `pending` (it stays head-of-queue for the next drain).
767#[tracing::instrument(name = "sync_outbox_drain", skip_all, fields(node.id = %node.id))]
768async fn drain_outbox(node: &NodeRef, client: &NodeClient, shared: &Arc<Shared>) {
769    let pending = match shared.outbox.pending_for_node(&node.id) {
770        Ok(p) => p,
771        Err(e) => {
772            tracing::warn!(node = %node.id, "outbox read failed: {e}");
773            return;
774        }
775    };
776    if pending.is_empty() {
777        return;
778    }
779    tracing::info!(node = %node.id, "draining {} queued write(s)", pending.len());
780    let mut blocked: HashSet<String> = HashSet::new();
781    for entry in pending {
782        if blocked.contains(&entry.entity_id) {
783            continue;
784        }
785        if let Err(e) = shared.outbox.mark_in_flight(&entry.id) {
786            tracing::warn!(node = %node.id, "mark_in_flight failed: {e}");
787            continue;
788        }
789        // `?kind` scope comes from the projection — the outbox only
790        // stores the entity id.
791        let kind = shared
792            .projection
793            .entity(&node.id, &entry.entity_id)
794            .ok()
795            .flatten()
796            .map(|e| e.kind)
797            .filter(|k| !k.is_empty());
798        let entity_id = entry.entity_id.clone();
799        let op = entry.op.clone();
800        let payload = entry.payload.clone();
801        let result = blocking(client.clone(), move |c| {
802            c.post_op(&entity_id, kind.as_deref(), &op, &payload)
803        })
804        .await;
805        match result {
806            Ok(_) => {
807                if let Err(e) = shared.outbox.mark_done(&entry.id) {
808                    tracing::warn!(node = %node.id, "mark_done failed: {e}");
809                }
810            }
811            Err(e) => {
812                tracing::warn!(node = %node.id, op = %entry.op, "replay failed: {e}");
813                match shared.outbox.mark_failed(&entry.id, &e.to_string()) {
814                    Ok(OutboxStatus::Pending) => {
815                        blocked.insert(entry.entity_id.clone());
816                    }
817                    Ok(_) => {}
818                    Err(err) => tracing::warn!(node = %node.id, "mark_failed failed: {err}"),
819                }
820            }
821        }
822    }
823}

825/// `true` when `node_id`'s row is a self-declared ephemeral node that
826/// has been continuously `down` longer than `after` — the reap
827/// predicate. `down_since` is an RFC 3339 stamp persisted in the
828/// projection, so the verdict survives hub restarts.
829fn ephemeral_dead(projection: &ProjectionStore, node_id: &str, after: Duration) -> bool {
830    let Ok(Some(row)) = projection.node(node_id) else {
831        return false;
832    };
833    if !row.ephemeral {
834        return false;
835    }
836    let Some(since) = row.down_since.as_deref() else {
837        return false;
838    };
839    let Ok(since) =
840        time::OffsetDateTime::parse(since, &time::format_description::well_known::Rfc3339)
841    else {
842        return false;
843    };
844    let threshold = time::Duration::seconds(i64::try_from(after.as_secs()).unwrap_or(i64::MAX));
845    // A future/garbage stamp reads as not-yet-dead — the subtraction
846    // goes negative and the comparison fails.
847    time::OffsetDateTime::now_utc() - since > threshold
848}

850/// Reap a confirmed-dead ephemeral node: drop its projected entities
851/// and resync cursor (a resurrected node resyncs itself) and
852/// dead-letter its queued writes — they can never replay. The `nodes`
853/// row stays so node status keeps reporting `down` + `ephemeral`.
854fn reap_ephemeral_node(shared: &Arc<Shared>, node_id: &str) {
855    if let Err(e) = shared.projection.drop_node_entities(node_id) {
856        tracing::warn!(node = %node_id, "ephemeral reap: entity drop failed: {e}");
857    }
858    match shared
859        .outbox
860        .dead_letter_node(node_id, "ephemeral node down past reap threshold")
861    {
862        Ok(0) => {}
863        Ok(n) => {
864            tracing::info!(node = %node_id, "ephemeral reap: dead-lettered {n} queued write(s)");
865        }
866        Err(e) => tracing::warn!(node = %node_id, "ephemeral reap: dead-letter failed: {e}"),
867    }
868}

870/// Apply one feed frame to the projection.
871#[tracing::instrument(name = "sync_apply_event", skip_all, fields(node.id = %node.id))]
872async fn apply_event(
873    node: &NodeRef,
874    client: &NodeClient,
875    shared: &Arc<Shared>,
876    event: FeedEvent,
877) -> Result<(), SyncError> {
878    if let Some(hook) = &shared.options.on_event {
879        hook(&node.id, &event);
880    }
881    match event {
882        // Any diff a registered kind policy claims projects; unknown
883        // event names stay inert (entity-detail frames never reach
884        // here as `Diff` unless they carry an `id`).
885        FeedEvent::Diff {
886            kind, id, patch, ..
887        } => {
888            let Some((_, tombstone)) = shared.options.policies.diff_policy(&kind) else {
889                return Ok(());
890            };
891            if patch.get("deleted") == Some(&Value::Bool(true)) {
892                match tombstone {
893                    DiffTombstone::DeleteEntity => {
894                        shared.projection.delete_entity(&node.id, &id)?;
895                        Ok(())
896                    }
897                    // Overlay-style tombstone — the row survives;
898                    // refetch reconciles it.
899                    DiffTombstone::Refetch => refetch(node, client, shared, &id).await,
900                }
901            } else {
902                patch_or_refetch(node, client, shared, &id, &patch).await
903            }
904        }
905        FeedEvent::Lagged => {
906            let docs = blocking(client.clone(), |c| c.list_entities(None, true)).await?;
907            shared.projection.apply_listing(&node.id, &docs)?;
908            Ok(())
909        }
910        // Heartbeats, entity-stream frames, and unknown frames are
911        // inert here.
912        _ => Ok(()),
913    }
914}

916/// Merge a diff, fetching the full entity doc when the row isn't
917/// indexed yet (a `meta` patch can precede the entity's first list).
918async fn patch_or_refetch(
919    node: &NodeRef,
920    client: &NodeClient,
921    shared: &Arc<Shared>,
922    id: &str,
923    patch: &Value,
924) -> Result<(), SyncError> {
925    match shared.projection.apply_patch(&node.id, id, patch)? {
926        PatchStatus::Applied => Ok(()),
927        PatchStatus::Missing => refetch(node, client, shared, id).await,
928    }
929}

931async fn refetch(
932    node: &NodeRef,
933    client: &NodeClient,
934    shared: &Arc<Shared>,
935    id: &str,
936) -> Result<(), SyncError> {
937    let entity_id = id.to_string();
938    let result = blocking(client.clone(), move |c| c.get_entity(&entity_id, None)).await;
939    match result {
940        Ok(doc) => {
941            shared.projection.upsert_entity(&node.id, &doc)?;
942        }
943        Err(NodeError::Http { status: 404, .. }) => {
944            shared.projection.delete_entity(&node.id, id)?;
945        }
946        Err(e) => return Err(e.into()),
947    }
948    Ok(())
949}

951/// `spawn_blocking` wrapper — `JoinError` becomes [`NodeError::Task`].
952async fn blocking<T, F>(client: NodeClient, f: F) -> Result<T, NodeError>
953where
954    T: Send + 'static,
955    F: FnOnce(&NodeClient) -> Result<T, NodeError> + Send + 'static,
956{
957    match tokio::task::spawn_blocking(move || f(&client)).await {
958        Ok(result) => result,
959        Err(e) => Err(NodeError::Task(e.to_string())),
960    }
961}

963/// Capped exponential backoff with ±25% jitter — `RandomState` provides
964/// the entropy so no `rand` dep is needed.
965fn jittered(base: Duration) -> Duration {
966    let mut hasher = std::collections::hash_map::RandomState::new().build_hasher();
967    hasher.write_u128(
968        std::time::SystemTime::now()
969            .duration_since(std::time::UNIX_EPOCH)
970            .map_or(0, |d| d.as_nanos()),
971    );
972    let pct = 75 + u32::try_from(hasher.finish() % 51).unwrap_or(25);
973    base.mul_f64(f64::from(pct) / 100.0)
974}

1204    /// A TCP endpoint that accepts and then never answers — the
1205    /// blackhole node: connects succeed, every request hangs until a
1206    /// client-side timeout fires.
1207    fn blackhole_node() -> String {
1208        let listener = std::net::TcpListener::bind(("127.0.0.1", 0)).unwrap();
1209        let port = listener.local_addr().unwrap().port();
1210        std::thread::spawn(move || {
1211            // Hold every accepted socket open without reading or
1212            // writing — requests stall until their own timeout.
1213            while let Ok((conn, _)) = listener.accept() {
1214                std::mem::forget(conn);
1215            }
1216        });
1217        format!("http://127.0.0.1:{port}")
1218    }

1220    /// T-060 — a node that stops answering while the projection still
1221    /// reads `up` (the sync loop hasn't observed the failure yet) must
1222    /// not hold `submit_op` for the full `request_timeout`: the direct
1223    /// attempt is bounded by `submit_timeout`, then the write lands in
1224    /// the outbox and the node flips `down`. The *next* submit skips
1225    /// the probe entirely.
1226    #[tokio::test(flavor = "multi_thread", worker_threads = 2)]
1227    async fn submit_op_fails_fast_into_outbox_when_node_hangs() {
1228        let mut opts = options();
1229        // A REST budget comfortably past the test's own runtime keeps
1230        // the node loop's probes from marking the node down before
1231        // the submit under test runs — the window being exercised is
1232        // "status still up, node dead". (It also bounds how long the
1233        // orphaned post lingers on a blocking thread.)
1234        opts.request_timeout = Duration::from_secs(5);
1235        let shared = shared_with(opts);
1236        let url = blackhole_node();
1237        let n = node(url);
1238        shared.projection.upsert_node(&n).unwrap();
1239        let engine = SyncEngine::spawn(
1240            shared.projection.clone(),
1241            shared.outbox.clone(),
1242            [NodeTarget::new(n.clone())],
1243            shared.options.clone(),
1244        );
1245        // Fresh evidence window: the loop hasn't run a cycle yet, so
1246        // the row is `unknown` — pretend the last connect succeeded.
1247        shared
1248            .projection
1249            .set_node_status("n", NodeStatus::Up)
1250            .unwrap();
1251
1252        let write = |key: &str| NodeWrite {
1253            node_id: "n".into(),
1254            entity_id: "e1".into(),
1255            entity_kind: None,
1256            op: "op".into(),
1257            policy: hypha_outbox::OpKind::Turn,
1258            payload: serde_json::json!({}),
1259            idempotency_key: key.to_string(),
1260            ttl_seconds: None,
1261        };
1262
1263        let start = std::time::Instant::now();
1264        let outcome = engine.submit_op(write("k1")).await.unwrap();
1265        let elapsed = start.elapsed();
1266        assert!(
1267            matches!(outcome, SubmitOutcome::Queued(_)),
1268            "hung node must queue, got {outcome:?}"
1269        );
1270        assert!(
1271            elapsed < Duration::from_secs(5),
1272            "submit blocked {elapsed:?} — the submit_timeout bound didn't hold"
1273        );
1274        assert_eq!(
1275            shared.projection.node("n").unwrap().unwrap().status,
1276            NodeStatus::Down,
1277            "the lapsed probe must mark the node down"
1278        );
1279        assert_eq!(shared.outbox.pending_for_node("n").unwrap().len(), 1);
1280
1281        // Known-down now: the next submit skips the direct attempt
1282        // outright — no probe, no wait.
1283        let start = std::time::Instant::now();
1284        let outcome = engine.submit_op(write("k2")).await.unwrap();
1285        assert!(matches!(outcome, SubmitOutcome::Queued(_)));
1286        assert!(
1287            start.elapsed() < Duration::from_secs(1),
1288            "known-down submit must not probe the node"
1289        );
1290        assert_eq!(shared.outbox.pending_for_node("n").unwrap().len(), 2);
1291
1292        engine.shutdown().await;
1293    }

1295    /// Canned node routing on the request path: `/api/events` → the
1296    /// held-open heartbeat stream, `/api/node` → `node_info`,
1297    /// everything else → `rest` (the entity listing).
1298    fn routed_node(node_info: &'static str, rest: &'static str) -> u16 {
1299        let listener = std::net::TcpListener::bind(("127.0.0.1", 0)).unwrap();
1300        let port = listener.local_addr().unwrap().port();
1301        std::thread::spawn(move || {
1302            while let Ok((mut conn, _)) = listener.accept() {
1303                std::thread::spawn(move || {
1304                    let mut buf = [0u8; 8192];
1305                    let n = conn.read(&mut buf).unwrap_or(0);
1306                    let req = String::from_utf8_lossy(&buf[..n]);
1307                    if req.contains("/api/events") {
1308                        let _ = conn.write_all(
1309                            b"HTTP/1.1 200 OK\r\nContent-Type: text/event-stream\r\n\r\n",
1310                        );
1311                        for _ in 0..100 {
1312                            if conn.write_all(b"event: heartbeat\ndata: {}\n\n").is_err() {
1313                                break;
1314                            }
1315                            let _ = conn.flush();
1316                            std::thread::sleep(Duration::from_millis(30));
1317                        }
1318                    } else {
1319                        let body = if req.contains("/api/node") {
1320                            node_info
1321                        } else {
1322                            rest
1323                        };
1324                        let _ = conn.write_all(
1325                            format!(
1326                                "HTTP/1.1 200 OK\r\nContent-Type: application/json\r\nContent-Length: {}\r\nConnection: close\r\n\r\n{body}",
1327                                body.len()
1328                            )
1329                            .as_bytes(),
1330                        );
1331                    }
1332                });
1333            }
1334        });
1335        port
1336    }

1403                    "n",
1404                    "e1",
1405                    "op",

1438    #[test]
1439    fn ephemeral_dead_predicate_arms() {
1440        // Missing row / not ephemeral / no down_since / fresh down →
1441        // all false; a stamped past down_since on an ephemeral node →
1442        // true. `down_since` is only writable via `set_node_status`,
1443        // so the "past" arm rides a zero-ish threshold instead.
1444        let shared = shared();
1445        let proj = &shared.projection;
1446        proj.upsert_node(&node("http://127.0.0.1:1".into()))
1447            .unwrap();
1448        assert!(!ephemeral_dead(proj, "n", Duration::ZERO));
1449        proj.set_node_ephemeral("n", true).unwrap();
1450        // Up → no down_since.
1451        proj.set_node_status("n", NodeStatus::Up).unwrap();
1452        assert!(!ephemeral_dead(proj, "n", Duration::ZERO));
1453        // Fresh down with a real threshold → false.
1454        proj.set_node_status("n", NodeStatus::Down).unwrap();
1455        assert!(!ephemeral_dead(proj, "n", Duration::from_secs(3600)));
1456        // Zero threshold → any elapsed down counts.
1457        assert!(ephemeral_dead(proj, "n", Duration::ZERO));
1458        // Unknown node id → false.
1459        assert!(!ephemeral_dead(proj, "ghost", Duration::ZERO));
1460        // Non-ephemeral with an old-enough down → still false.
1461        proj.set_node_ephemeral("n", false).unwrap();
1462        assert!(!ephemeral_dead(proj, "n", Duration::ZERO));
1463    }
