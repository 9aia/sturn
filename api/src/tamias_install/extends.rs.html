1//! `template.extends:` parent materialization — the install half of
2//! template-pack composition (scarab's `type: template` driver).
3//!
4//! A pack manifest may declare `template.extends:` inside its
5//! driver-owned `template:` table. The vocabulary is the agnostic
6//! composition spec (scarab's `scarab-template/COMPOSITION.md`):
7//! `extends` is a source string, a `{source, via}` table, or a list
8//! of either — tamias interprets *only* the `tamias-ref` scheme
9//! (absent `via` = the ambient default, which under tamias is
10//! `tamias-ref`); any other `via` fails the install. `source` stays
11//! an opaque locator — store name or installable ref.
12//!
13//! At install, after `include:` composition and the signature/consent
14//! gates, each parent's payload is resolved (store name → that
15//! entry's committed payload; anything else the ref grammar claims →
16//! a recursive [`crate::install_inner`], so the parent gets the full
17//! pipeline — fetch, *its own* extends, drivers, store commit) and
18//! copied into the composed tree: a single parent lands at
19//! [`PARENT_DIR`] (`.scarab-parent/`, the v1 layout); several parents
20//! land at [`PARENTS_DIR`]/`<index>/` (`.scarab-parents/0/`, `1/`, …)
21//! in declaration order — lowest precedence first.
22//!
23//! Because each parent's committed payload already carries *its*
24//! materialized parents (`.scarab-parent*/` inside it), nesting is
25//! automatic: a grandparent lands inside its parent's dir. The store
26//! commit therefore carries the whole ancestor chain, and every
27//! later driver verb — `install` (this dispatch), `deploy`,
28//! `uninstall` — sees it under `--dir <payload>` with no extra work.
29//!
30//! Recursion is bounded like `include:` ([`IncludeState`]): a depth
31//! cap ([`MAX_EXTENDS_DEPTH`]) and an in-progress label set
32//! ([`IncludeState::extends_in_progress`]) that catches
33//! `A extends B, B extends A` cycles through refs. Store-name
34//! parents can't cycle — their payloads are already committed and
35//! finite. Sibling parents resolve sequentially.
36//!
37//! The signature/consent/trust digest deliberately covers the tree
38//! *before* `.scarab-parent*/` is added: publish doesn't flatten
39//! `extends`, so a signed parent-extending pack still verifies
40//! against its published bytes, and per-package driver consent binds
41//! to the package's own payload. `Store::install` digests the final
42//! tree (parents included), so `verify` stays honest.

54/// Extends-recursion cap — an `extends` chain deeper than this fails
55/// instead of recursing forever (ref cycles are caught earlier by
56/// [`IncludeState::extends_in_progress`]; the cap bounds pathological
57/// DAGs). Same bound as [`include::MAX_INCLUDE_DEPTH`] and scarab's
58/// `DEFAULT_MAX_DEPTH`.
59pub(crate) const MAX_EXTENDS_DEPTH: usize = 8;

61/// Payload subdir a single resolved parent materializes into — the
62/// driver (`tamias-driver-template`) discovers it by this name.
63pub(crate) const PARENT_DIR: &str = ".scarab-parent";
64
65/// Payload subdir several resolved parents materialize into — one
66/// numbered child (`0/`, `1/`, …) per `extends` declaration, in
67/// declaration order (lowest precedence first).
68pub(crate) const PARENTS_DIR: &str = ".scarab-parents";

70/// `via` schemes tamias can resolve — absent `via` means the ambient
71/// default, which under a tamias install is `tamias-ref`.
72const TAMIAS_VIA: &[&str] = &["tamias-ref", "tamias"];

74/// One declared `extends` parent after spec parsing — `source` is the
75/// opaque locator, `via` the (already validated) scheme hint.
76#[derive(Clone, Debug, PartialEq, Eq)]
77pub(crate) struct ExtParent {
78    /// The opaque source locator (store name or installable ref).
79    pub source: String,
80    /// The resolution-scheme hint — always a [`TAMIAS_VIA`] value or
81    /// `None` (ambient default).
82    pub via: Option<String>,
83}

85/// `template.extends` parents declared by the manifest inside `dir` —
86/// empty when absent or JSON-null.
87///
88/// The `template:` table is driver-owned: tamias re-reads the raw
89/// manifest file (the typed [`tamias_core::Manifest`] drops unknown
90/// keys) and validates only this one field, per the agnostic
91/// composition spec (scarab `COMPOSITION.md`):
92///
93/// - `"<source>"` — one parent, bare string.
94/// - `{"source": "<source>", "via": "<scheme>"}` — one structured
95///   parent (`via` optional; must be a tamias scheme).
96/// - `[<string-or-table>, …]` — several parents, lowest precedence
97///   first. An empty list is an error.
98///
99/// Anything else — non-string sources, empty sources, unknown `via`
100/// schemes — fails.
101///
102/// # Errors
103/// Fails on manifest read/parse errors or a malformed `extends`.
104pub(crate) fn parents(dir: &Path) -> Result<Vec<ExtParent>, Error> {
105    let Some(path) = tamias_core::Manifest::path_in(dir) else {
106        return Ok(Vec::new());
107    };
108    let text =
109        std::fs::read_to_string(&path).map_err(Error::io(format!("read {}", path.display())))?;
110    // Same syntax dispatch as `Manifest::parse_at`: .yaml/.yml → YAML,
111    // everything else → TOML. Both normalize to JSON for the lookup.
112    let value: serde_json::Value = if matches!(
113        path.extension().and_then(|e| e.to_str()),
114        Some("yaml" | "yml")
115    ) {
116        let yaml: serde_yaml_ng::Value = serde_yaml_ng::from_str(&text)
117            .map_err(|e| Error::Failed(format!("parse {}: {e}", path.display())))?;
118        serde_json::to_value(yaml)
119            .map_err(|e| Error::Failed(format!("parse {}: {e}", path.display())))?
120    } else {
121        let table: toml::Table = toml::from_str(&text)
122            .map_err(|e| Error::Failed(format!("parse {}: {e}", path.display())))?;
123        serde_json::to_value(table)
124            .map_err(|e| Error::Failed(format!("parse {}: {e}", path.display())))?
125    };
126    let Some(extends) = value.get("template").and_then(|t| t.get("extends")) else {
127        return Ok(Vec::new());
128    };
129    if extends.is_null() {
130        return Ok(Vec::new());
131    }
132    let items: Vec<serde_json::Value> = match extends {
133        serde_json::Value::Array(items) => items.clone(),
134        other => vec![other.clone()],
135    };
136    if items.is_empty() {
137        return Err(Error::Failed(format!(
138            "{}: template.extends is an empty list — name at least one parent",
139            path.display()
140        )));
141    }
142    items.iter().map(|item| parse_parent(item, &path)).collect()
143}

145/// One `extends` item (string or `{source, via}` map) → [`ExtParent`].
146fn parse_parent(item: &serde_json::Value, path: &Path) -> Result<ExtParent, Error> {
147    let bad = |why: &str| Error::Failed(format!("{}: template.extends {why}", path.display()));
148    let (source, via) = match item {
149        serde_json::Value::String(s) => (s.clone(), None),
150        serde_json::Value::Object(map) => {
151            let source = map.get("source").and_then(|v| v.as_str()).ok_or_else(|| {
152                bad("parent must be a string or {source, via} with a string source")
153            })?;
154            let via = match map.get("via") {
155                None | Some(serde_json::Value::Null) => None,
156                Some(serde_json::Value::String(s)) => Some(s.clone()),
157                Some(_) => return Err(bad("parent 'via' must be a string")),
158            };
159            (source.to_string(), via)
160        }
161        _ => {
162            return Err(bad(
163                "parent must be a string or {source, via} with a string source",
164            ));
165        }
166    };
167    if source.trim().is_empty() {
168        return Err(bad("parent source must be a non-empty string"));
169    }
170    if let Some(v) = &via {
171        if !TAMIAS_VIA.contains(&v.as_str()) {
172            return Err(Error::Failed(format!(
173                "{}: template.extends via '{v}' is not a tamias composition source — \
174                 tamias resolves '{}' only",
175                path.display(),
176                TAMIAS_VIA.join("'/'")
177            )));
178        }
179    }
180    Ok(ExtParent { source, via })
181}

183/// Resolve each declared parent to a payload dir and copy it into
184/// `child_root` — `.scarab-parent/` for a single parent,
185/// `.scarab-parents/<index>/` (declaration order) for several —
186/// replacing any dirs already there (a declared `extends` is
187/// authoritative over vendored bytes).
188///
189/// `child_label` is the installing package's store label
190/// ([`include::pending_label`]) — the cycle-set key. `report` sees a
191/// [`Progress::Including`] event per parent, and a ref-parent's
192/// recursive install streams its own events like an `include:` base.
193///
194/// # Errors
195/// Fails on an ambiguous/missing store name, an unparseable ref, a
196/// cycle, a depth-cap breach, or any recursive-install/copy failure.
197#[allow(clippy::too_many_arguments)]
198pub(crate) fn materialize(
199    env: &Env,
200    child_root: &Path,
201    child_label: &str,
202    parents: &[ExtParent],
203    state: &mut IncludeState,
204    allow_drivers: bool,
205    surface: Surface,
206    report: &mut dyn FnMut(Progress),
207) -> Result<(), Error> {
208    let single = parents.len() == 1;
209    for (i, parent) in parents.iter().enumerate() {
210        let resolved = resolve_parent(
211            env,
212            child_label,
213            &parent.source,
214            state,
215            allow_drivers,
216            surface,
217            report,
218        )?;
219        let dest = if single {
220            child_root.join(PARENT_DIR)
221        } else {
222            child_root.join(PARENTS_DIR).join(i.to_string())
223        };
224        if dest.exists() {
225            std::fs::remove_dir_all(&dest)
226                .map_err(Error::io(format!("clear {}", dest.display())))?;
227        }
228        // create_dir_all first: an empty parent payload must still
229        // leave a dir the driver's locator can descend into (its
230        // manifest is what the chain walk reads — a payload always
231        // has one).
232        std::fs::create_dir_all(&dest).map_err(Error::io(format!("create {}", dest.display())))?;
233        copy_tree(&resolved, &dest)?;
234    }
235    Ok(())
236}

238/// One `extends` source → one parent payload dir. Mirrors
239/// [`include::resolve_base`]'s store-name-then-ref order; see the
240/// module docs for the recursion bounds.
241#[allow(clippy::too_many_arguments)]
242fn resolve_parent(
243    env: &Env,
244    child_label: &str,
245    parent_ref: &str,
246    state: &mut IncludeState,
247    allow_drivers: bool,
248    surface: Surface,
249    report: &mut dyn FnMut(Progress),
250) -> Result<PathBuf, Error> {
251    // 1. Store name — label / `@scope/name` / bare `name`. The stored
252    //    payload already carries its own `.scarab-parent*/` chain.
253    let entries = Store::new(&env.store_root).entries()?;
254    if let Some(entry) = find_store_entry(&entries, parent_ref)? {
255        report(Progress::Including {
256            base: entry.label(),
257        });
258        return Ok(entry.dir);
259    }
260    // `@scope/name` is a *name*, never a path.
261    if parent_ref.starts_with('@') {
262        return Err(missing_parent(parent_ref));
263    }
264    // 2. Installable ref — recurse through the full pipeline.
265    let spec = crate::refs()
266        .parse(parent_ref)
267        .map_err(|_| missing_parent(parent_ref))?;
268    if let Some(reason) = include::unpinned_remote_ref(&spec) {
269        eprintln!("warning: template.extends '{parent_ref}' is an unpinned remote ref — {reason}");
270    }
271    if spec.is::<FsSpec>() && !expand_path(Path::new(parent_ref)).exists() {
272        return Err(missing_parent(parent_ref));
273    }
274    if state.extends_in_progress.contains(child_label) {
275        return Err(Error::Failed(format!(
276            "template.extends cycle: {child_label} is already being installed as a parent"
277        )));
278    }
279    if state.extends_depth >= MAX_EXTENDS_DEPTH {
280        return Err(Error::Failed(format!(
281            "template.extends '{parent_ref}' exceeds the {MAX_EXTENDS_DEPTH}-deep nesting cap"
282        )));
283    }
284    report(Progress::Including {
285        base: parent_ref.to_string(),
286    });
287    state.extends_in_progress.insert(child_label.to_string());
288    state.extends_depth += 1;
289    let result = install_inner(env, parent_ref, allow_drivers, surface, report, state);
290    state.extends_depth -= 1;
291    state.extends_in_progress.remove(child_label);
292    Ok(result?.dir)
293}

295/// The hard error for an extends ref that is neither an installed
296/// store name nor a resolvable ref.
297fn missing_parent(parent_ref: &str) -> Error {
298    Error::Failed(format!(
299        "template.extends '{parent_ref}' is not installed — store names must already be in the \
300         store; use an installable ref (path, git:, cf:, http:) or `tamias install {parent_ref}` first"
301    ))
302}

311    /// Write a manifest file with the given `template:` table body.
312    fn manifest(dir: &Path, body: &str) {
313        std::fs::write(
314            dir.join("tamias.yaml"),
315            format!("name: x\ntype: template\ntemplate:\n{body}"),
316        )
317        .unwrap();
318    }
319
320    #[test]
321    fn parents_reads_string_table_and_list() {
322        let dir = tempfile::tempdir().unwrap();
323        manifest(dir.path(), "  extends: git:org/base\n");
324        let p = parents(dir.path()).unwrap();
325        assert_eq!(p.len(), 1);
326        assert_eq!(p[0].source, "git:org/base");
327        assert_eq!(p[0].via, None);
328
329        manifest(
330            dir.path(),
331            "  extends:\n    source: git:org/base\n    via: tamias-ref\n",
332        );
333        let p = parents(dir.path()).unwrap();
334        assert_eq!(p.len(), 1);
335        assert_eq!(p[0].via.as_deref(), Some("tamias-ref"));
336
337        manifest(
338            dir.path(),
339            "  extends:\n    - git:org/a\n    - source: git:org/b\n      via: tamias\n",
340        );
341        let p = parents(dir.path()).unwrap();
342        assert_eq!(p.len(), 2);
343        assert_eq!(p[0].source, "git:org/a");
344        assert_eq!(p[1].source, "git:org/b");
345
346        // absent / null → empty
347        manifest(dir.path(), "  files: []\n");
348        assert!(parents(dir.path()).unwrap().is_empty());
349        manifest(dir.path(), "  extends: null\n");
350        assert!(parents(dir.path()).unwrap().is_empty());
351    }
352
353    #[test]
354    fn parents_rejects_malformed() {
355        let dir = tempfile::tempdir().unwrap();
356        for body in [
357            "  extends: 42\n",
358            "  extends: \"  \"\n",
359            "  extends: []\n",
360            "  extends:\n    - 42\n",
361            "  extends:\n    source: git:org/x\n    via: handl-kit\n",
362            "  extends:\n    source: git:org/x\n    via: 7\n",
363            "  extends:\n    via: tamias-ref\n",
364        ] {
365            manifest(dir.path(), body);
366            assert!(parents(dir.path()).is_err(), "body: {body}");
367        }
368    }

377        .unwrap();
378        let p = parents(dir.path()).unwrap();
379        assert_eq!(p[0].source, "git:org/t");
380    }
