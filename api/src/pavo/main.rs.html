2//! pavo — the CLI surface of the viz host.
3//!
4//! Verbs:
5//!
6//! - `drivers` — list discovered `tamias-driver-*` executables that
7//!   speak pavo (manifest version, view count, contributes summary).
8//! - `views [--driver <name>]` — every view as a `<driver>:<view>`
9//!   ref with title/route.
10//! - `chrome` — the merged contribution slots (navbar/dashboard/
11//!   statusbar/routes/sidebar/headerActions/notifications/jobs +
12//!   forward-compat extras).
13//! - `view <driver>:<view> [--param k=v …] [--format …]` — resolve a
14//!   deep link and render it: `text` (default) summarizes the
15//!   ViewNode tree + bound data, `json` dumps the raw
16//!   [`ViewResult`](pavo_protocol::wire::ViewResult), `dot`/`mermaid` emit the view's graph nodes.
17//!   `/path?query` routes (what `Action.route`/`item_route`
18//!   templates produce, e.g. `/myco/item?id=T-1`) resolve back to a
19//!   `<driver>:<view>` ref + params via declared view routes —
20//!   `view`, `open` (non-TTY), `actions`, `invoke`, and `submit`
21//!   all accept them.
22//! - `query <driver> <query> [--param k=v …]` — raw `query` op
23//!   passthrough.
24//! - `submit <driver>:<view> [--form <id>] --param k=v …` — the
25//!   `submit` wire op: send a form's field values to its driver and
26//!   print the `SubmitResult` (message/errors/data; `--json` dumps
27//!   the full payload including any updated view).
28//! - `actions <driver>:<view>|<driver>` — list the actions a view
29//!   (or a whole driver, manifest scope) offers: declared
30//!   `actions[]` (manifest/node/tree-item) plus ops-manifest derived
31//!   ones, with danger level and param schemas.
32//! - `invoke <driver>:<view> <action>` — dispatch an action through
33//!   the `invoke` wire op. `--param k=v` feeds `ParamSchema` params,
34//!   `--target`/`--target-id` carry the opaque action target,
35//!   `--interactive` prompts for missing params on a TTY, and
36//!   `--yes` bypasses `confirm`/`type-target` confirmations
37//!   (required off-TTY). Actions declaring `route` navigate instead
38//!   of dispatching: the template is interpolated over the invoke
39//!   context and the resulting route is printed (`{"route": …}`
40//!   with `--json`), exit 0. A real invoke whose `ActionResult`
41//!   carries `route` appends a `→ <route>` hint in text mode.
42//! - `jobs` / `notifications` — the aggregated platform feeds: every
43//!   provider driver's `jobs`/`notifications` wire op, tagged with
44//!   its driver (`--json` dumps the flattened rows).
45//! - `serve [--port N]` — the localhost HTTP bridge for the web host:
46//!   `/api/*` JSON endpoints over the same [`Host`] plus static
47//!   serving of the Nuxt build (`web/.output/public`). First stdout
48//!   line is `listening http://127.0.0.1:<port>`.
49//! - `open [<driver>:<view>]` — hand off to `pavo-tui` (the ratatui
50//!   surface) with the deep link, or the bare dashboard when no
51//!   target is given. On a non-TTY it falls back to `pavo view`
52//!   text output for the given target.
53//! - `manifest` / `schema` — the CLI's own capability manifest and
54//!   verb schema (sibling convention: tamias/kuda).
55//!
56//! `--json` is a per-subcommand flag (sibling convention) on the
57//! listing/query verbs; `view --format json` is its machine-readable
58//! mode. Errors go to stderr; exit codes are CI-meaningful: `0` ok,
59//! `1` view/driver error, `2` usage (bad `--param`, malformed view
60//! ref, bad format — clap reports the last itself).
61//!
62//! Discovery is environment-driven like pavo-core's
63//! [`Host::discover`]: drivers are found on `PATH` and under
64//! `$TAMIAS_HOME/drivers` — set `PATH`/`TAMIAS_HOME` to point the CLI
65//! at fixture drivers (integration tests do exactly this).

265/// A CLI failure: exit code + human message (stderr).
266struct Fail {
267    code: u8,
268    msg: String,
269}
270
271impl Fail {
272    /// View/driver/runtime error → exit 1.
273    fn err(msg: impl Into<String>) -> Self {
274        Self {
275            code: 1,
276            msg: msg.into(),
277        }
278    }
279
280    /// Usage error → exit 2.
281    fn usage(msg: impl Into<String>) -> Self {
282        Self {
283            code: 2,
284            msg: msg.into(),
285        }
286    }
287}
288
289impl From<PavoError> for Fail {
290    fn from(e: PavoError) -> Self {
291        // A malformed `<driver>:<view>` ref is a usage error; driver
292        // and resolution failures are runtime errors.
293        let code = match &e {
294            PavoError::BadViewRef(_) => 2,
295            _ => 1,
296        };
297        Self {
298            code,
299            msg: e.to_string(),
300        }
301    }
302}

315/// `Ok` carries the process exit code (only `open` can produce a
316/// non-zero one — it propagates `pavo-tui`'s status).
317fn run(cli: Cli) -> Result<u8, Fail> {
318    match cli.command {
319        None => {
320            println!("pavo {}", env!("CARGO_PKG_VERSION"));
321            Ok(0)
322        }
323        Some(Command::Drivers { json }) => cmd_drivers(json).map(|_| 0),
324        Some(Command::Views { driver, json }) => cmd_views(driver.as_deref(), json).map(|_| 0),
325        Some(Command::Chrome { json }) => cmd_chrome(json).map(|_| 0),
326        Some(Command::View {
327            target,
328            params,
329            format,
330        }) => cmd_view(&target, &params, &format).map(|_| 0),
331        Some(Command::Query {
332            driver,
333            query,
334            params,
335            json,
336        }) => cmd_query(&driver, &query, &params, json).map(|_| 0),
337        Some(Command::Submit {
338            target,
339            form,
340            params,
341            json,
342        }) => cmd_submit(&target, form.as_deref(), &params, json).map(|_| 0),
343        Some(Command::Actions {
344            target,
345            params,
346            json,
347        }) => cmd_actions(&target, &params, json).map(|_| 0),
348        Some(Command::Invoke {
349            target,
350            action,
351            params,
352            action_target,
353            target_id,
354            yes,
355            interactive,
356            json,
357        }) => cmd_invoke(
358            &target,
359            &action,
360            &params,
361            action_target.as_deref(),
362            target_id.as_deref(),
363            yes,
364            interactive,
365            json,
366        )
367        .map(|_| 0),
368        Some(Command::Jobs { json }) => cmd_jobs(json).map(|_| 0),
369        Some(Command::Notifications { json }) => cmd_notifications(json).map(|_| 0),
370        Some(Command::Serve { port }) => serve::run(port)
371            .map(|_| 0)
372            .map_err(|e| Fail::err(format!("serve: {e}"))),
373        Some(Command::Open { target }) => cmd_open(target.as_deref()),
374        Some(Command::Manifest) => {
375            println!(
376                "{}",
377                json!({
378                    "name": "pavo",
379                    "version": env!("CARGO_PKG_VERSION"),
380                    "verbs": [
381                        "drivers", "views", "chrome", "view", "query",
382                        "submit", "actions", "invoke", "jobs",
383                        "notifications", "serve", "open", "manifest",
384                        "schema", "gen-cli-docs"
385                    ],
386                })
387            );
388            Ok(0)
389        }
390        Some(Command::Schema) => {
391            println!(
392                "{}",
393                serde_json::to_string_pretty(&schema_doc())
394                    .map_err(|e| Fail::err(format!("schema serialization: {e}")))?
395            );
396            Ok(0)
397        }
398        Some(Command::GenCliDocs) => {
399            print!("{}", clap_markdown::help_markdown_command(&Cli::command()));
400            Ok(0)
401        }
402    }
403}

488/// `pavo open` — hand the deep link to `pavo-tui` on a TTY, else
489/// degrade to `view` text output. Returns `pavo-tui`'s exit status.
490fn cmd_open(target: Option<&str>) -> Result<u8, Fail> {
491    if std::io::stdout().is_terminal() {
492        return spawn_tui(target);
493    }
494    let Some(target) = target else {
495        return Err(Fail::usage(
496            "pavo open needs a terminal for the TUI — pass <driver>:<view> \
497             (or use `pavo view`) for text output",
498        ));
499    };
500    eprintln!("pavo: not a TTY — rendering text output instead of the TUI");
501    cmd_view(target, &[], "text").map(|_| 0)
502}

504/// Candidate `pavo-tui` binaries: the sibling of this executable
505/// (covers `cargo run`/target-dir installs) then `PATH`.
506fn tui_candidates() -> Vec<std::path::PathBuf> {
507    let mut out = Vec::new();
508    if let Ok(exe) = std::env::current_exe() {
509        if let Some(dir) = exe.parent() {
510            out.push(dir.join("pavo-tui"));
511        }
512    }
513    out.push(std::path::PathBuf::from("pavo-tui"));
514    out
515}

517/// Spawn `pavo-tui` (inheriting stdio) with the optional deep link
518/// and propagate its exit status.
519fn spawn_tui(target: Option<&str>) -> Result<u8, Fail> {
520    let mut last_err = String::new();
521    for cand in tui_candidates() {
522        let mut cmd = std::process::Command::new(&cand);
523        if let Some(t) = target {
524            cmd.arg(t);
525        }
526        match cmd.status() {
527            Ok(status) => {
528                // Signals yield no code; clamp exotic values to 1.
529                let code = status
530                    .code()
531                    .and_then(|c| u8::try_from(c).ok())
532                    .unwrap_or(1);
533                return Ok(code);
534            }
535            Err(e) => last_err = format!("{}: {e}", cand.display()),
536        }
537    }
538    Err(Fail::err(format!(
539        "pavo-tui not found on PATH or next to pavo ({last_err}) — \
540         install it (`cargo xtask install` or `cargo install --path crates/pavo-tui`) \
541         or run `cargo run -p pavo-tui`"
542    )))
543}

550/// [`cmd_drivers`] over an explicit host — split for tests.
551fn cmd_drivers_host(host: &mut Host, json: bool) -> Result<(), Fail> {
552    let kinds: Vec<String> = host.drivers().iter().map(|d| d.kind.clone()).collect();
553    let mut rows = Vec::new();
554    for kind in kinds {
555        match host.manifest(&kind) {
556            Ok(m) => rows.push(json!({
557                "driver": kind,
558                "title": m.title,
559                "version": m.version,
560                "views": m.views.len(),
561                "renderers": m.renderers.len(),
562                "contributes": contrib_counts(&m.contributes),
563            })),
564            Err(e) => rows.push(json!({
565                "driver": kind,
566                "manifest_error": e.to_string(),
567            })),
568        }
569    }
570    if json {
571        println!(
572            "{}",
573            serde_json::to_string(&rows).map_err(|e| Fail::err(e.to_string()))?
574        );
575        return Ok(());
576    }
577    if rows.is_empty() {
578        println!("no pavo-capable drivers found (PATH + $TAMIAS_HOME/drivers)");
579        return Ok(());
580    }
581    for row in &rows {
582        if let Some(err) = row.get("manifest_error").and_then(Value::as_str) {
583            println!(
584                "{}  —  manifest error: {err}",
585                row["driver"].as_str().unwrap_or("?")
586            );
587            continue;
588        }
589        let counts = &row["contributes"];
590        let contrib = [
591            "navbar",
592            "dashboard",
593            "statusbar",
594            "routes",
595            "sidebar",
596            "headerActions",
597            "notifications",
598            "jobs",
599        ]
600        .iter()
601        .filter(|slot| counts.get(**slot).and_then(Value::as_u64).unwrap_or(0) > 0)
602        .map(|slot| format!("{slot}:{}", counts[*slot].as_u64().unwrap_or(0)))
603        .collect::<Vec<_>>()
604        .join(" ");
605        println!(
606            "{}  {}  v{}  views:{}  renderers:{}  {}",
607            row["driver"].as_str().unwrap_or("?"),
608            row["title"].as_str().unwrap_or("?"),
609            row["version"].as_str().unwrap_or("?"),
610            row["views"].as_u64().unwrap_or(0),

815/// The `view` verb's format dispatch — returns the exact bytes to
816/// print (split from [`cmd_view_host`] so tests can assert on
817/// output).
818///
819/// Precedence: renderer plugins are consulted for **every** format
820/// before the built-in interpretation runs — `text`/`dot`/`mermaid`
821/// substitute claimed nodes' RenderModels per tree path and keep the
822/// built-in rendering for unclaimed nodes. `json` is exempt (the raw
823/// `ViewResult` dump is host-owned). Any *other* format string is a
824/// plugin-only format: every claimed node's RenderModel is emitted;
825/// no claims → usage error naming the built-ins.
826fn view_output(
827    host: &Host,
828    resolved: &ResolvedView,
829    target: &str,
830    format: &str,
831) -> Result<String, Fail> {
832    match format {
833        "json" => serde_json::to_string_pretty(&resolved.result)
834            .map(|s| format!("{s}\n"))
835            .map_err(|e| Fail::err(e.to_string())),
836        "text" => {
837            // Renderer plugins that claim `text` (or wildcard cli
838            // formats) substitute their RenderModel per node; the
839            // rest render built-in.
840            let renders = host.render_view_plugins(resolved, "cli", Some("text"));
841            let mut out = render::render_text_plugins(
842                &resolved.result.view.root,
843                &resolved.result.data,
844                &renders,
845            );
846            // Action affordance: one footer line naming the declared
847            // actions (danger-marked) so `pavo invoke` is discoverable.
848            if let Ok(manifest) = host.manifest(&resolved.driver) {
849                let mut seen = std::collections::BTreeSet::new();
850                let hints: Vec<String> = collect_actions(&manifest, &resolved.result)
851                    .iter()
852                    .filter(|s| seen.insert(s.action.id.clone()))
853                    .map(|s| {
854                        let suffix = match s.action.danger {
855                            DangerLevel::None => "",
856                            DangerLevel::Confirm => " (confirm)",
857                            DangerLevel::TypeTarget => " (type-target)",
858                        };
859                        format!("{}{}", s.action.id, suffix)
860                    })
861                    .collect();
862                if !hints.is_empty() {
863                    out.push_str(&format!("\nactions: {}\n", hints.join(", ")));
864                }
865            }
866            Ok(out)
867        }
868        "dot" => {
869            // Plugins claiming `dot` (or wildcard cli formats) take
870            // precedence per node; unclaimed graphs still emit
871            // digraph blocks.
872            let renders = host.render_view_plugins(resolved, "cli", Some("dot"));
873            render::render_dot_plugins(&resolved.result.view.root, &resolved.result.data, &renders)
874                .map_err(|e| Fail::err(format!("{target}: {e}")))
875        }
876        "mermaid" => {
877            let renders = host.render_view_plugins(resolved, "cli", Some("mermaid"));
878            render::render_mermaid_plugins(
879                &resolved.result.view.root,
880                &resolved.result.data,
881                &renders,
882            )
883            .map_err(|e| Fail::err(format!("{target}: {e}")))
884        }
885        other => {
886            // Plugin format: route every claimed node through its
887            // driver's `render` op and concatenate the models.
888            let renders = host.render_view_plugins(resolved, "cli", Some(other));
889            if renders.is_empty() {
890                return Err(Fail::err(format!(
891                    "{target}: no renderer plugin handles --format '{other}' \
892                     (built-ins: {})",
893                    BUILTIN_FORMATS.join(", ")
894                )));
895            }
896            // Deterministic order: sort by path ("" root first, then
897            // tree order).
898            let mut keyed: Vec<(&String, &pavo_protocol::RenderModel)> = renders.iter().collect();
899            keyed.sort_by(|a, b| a.0.cmp(b.0));
900            let mut out = String::new();
901            for (_, model) in keyed {
902                out.push_str(&render::render_model_cli(model));
903            }
904            Ok(out)
905        }
906    }
907}

1032// ---------- `/route?query` deep links ----------
1033//
1034// `Action.route`, `ViewSpec.item_route`, `MenuEntry.route`, and
1035// `ActionResult.route` all produce host-relative route strings like
1036// `/myco/item?id=T-1`. The CLI is non-interactive, so "navigate"
1037// means resolving the route back to a `<driver>:<view>` ref + view
1038// params and rendering/invoking there. Matching is local to this
1039// file (pavo-core stays transport-pure): every discovered
1040// manifest's `ViewSpec.route` and `item_route` is a candidate
1041// pattern.
1042//
1043// Limits (v1):
1044// - Only *view* routes resolve — `contributes.routes` pages are
1045//   driver-owned and have no `resolve_view` target.
1046// - `{name}` placeholders match exactly one whole path segment and
1047//   capture into a view param named `name` (`{item.id}` captures as
1048//   `"item.id"`); literals must match segment-for-segment. The
1049//   candidate's own `?query` template is ignored for matching —
1050//   the request's query pairs become the params.
1051// - Best match wins: fewest captures, then longest pattern.
1052
1053/// Match request path `segs` (decoded, split on `/`) against one
1054/// candidate route `pattern` (`/myco/item/{item.id}?id={item.id}` —
1055/// its query half is ignored). Returns the `{name}` captures on
1056/// success.
1057fn match_route_pattern(pattern: &str, segs: &[String]) -> Option<(Params, usize)> {
1058    let pat_path = pattern.split(['?', '#']).next().unwrap_or(pattern);
1059    let pat: Vec<&str> = pat_path.split('/').filter(|s| !s.is_empty()).collect();
1060    if pat.len() != segs.len() {
1061        return None;
1062    }
1063    let mut params = Params::new();
1064    let mut captures = 0;
1065    for (p, s) in pat.iter().zip(segs.iter()) {
1066        if let Some(name) = p.strip_prefix('{').and_then(|x| x.strip_suffix('}')) {
1067            let name = name.trim();
1068            if name.is_empty() {
1069                return None;
1070            }
1071            params.insert(name.to_string(), Value::String(s.clone()));
1072            captures += 1;
1073        } else if p != s {
1074            return None;
1075        }
1076    }
1077    Some((params, captures))
1078}

1080/// Segment count of a candidate pattern's path half (scoring).
1081fn route_pattern_len(pattern: &str) -> usize {
1082    pattern
1083        .split(['?', '#'])
1084        .next()
1085        .unwrap_or(pattern)
1086        .split('/')
1087        .filter(|s| !s.is_empty())
1088        .count()
1089}

1091/// Resolve a `/path?query` route to a `<driver>:<view>` ref + view
1092/// params by matching declared `ViewSpec.route`/`item_route`
1093/// patterns across all discovered manifests. `None` when nothing
1094/// matches — callers only invoke this for leading-`/` targets.
1095///
1096/// Query pairs become params via the same scalar rule as `--param`
1097/// ([`http::param_json`]: JSON when possible, else string) and
1098/// override `{name}` path captures on key conflicts.
1099#[must_use]
1100pub(crate) fn route_target(host: &Host, route: &str) -> Option<(String, Params)> {
1101    let route = route.trim();
1102    if !route.starts_with('/') {
1103        return None;
1104    }
1105    let (path, query) = route.split_once('?').unwrap_or((route, ""));
1106    let path = path.split('#').next().unwrap_or(path);
1107    let segs: Vec<String> = path
1108        .split('/')
1109        .filter(|s| !s.is_empty())
1110        .map(http::percent_decode)
1111        .collect();
1112    if segs.is_empty() {
1113        return None;
1114    }
1115    // Deterministic scan: driver kinds sorted, manifest view order.
1116    let mut kinds: Vec<String> = host.drivers().iter().map(|d| d.kind.clone()).collect();
1117    kinds.sort();
1118    // Best: (namespaced ref, captures, pattern len, is_item_route,
1119    // capture params). `ViewSpec.route` is canonical — it outranks
1120    // `item_route` templates on ties (an item_route like
1121    // `/item?id={item.id}` shares its path with the detail view's
1122    // own route and must not shadow it).
1123    let mut best: Option<(String, usize, usize, bool, Params)> = None;
1124    for kind in kinds {
1125        let Ok(manifest) = host.manifest(&kind) else {
1126            continue;
1127        };
1128        for spec in &manifest.views {
1129            for (pattern, is_item_route) in [
1130                (Some(spec.route.as_str()), false),
1131                (spec.item_route.as_deref(), true),
1132            ]
1133            .into_iter()
1134            .filter_map(|(p, i)| p.map(|p| (p, i)))
1135            {
1136                let Some((capture_params, captures)) = match_route_pattern(pattern, &segs) else {
1137                    continue;
1138                };
1139                if std::env::var_os("PAVO_DEBUG_ROUTE").is_some() {
1140                    eprintln!(
1141                        "  candidate {kind}:{} pattern={pattern:?} captures={captures}",
1142                        spec.id
1143                    );
1144                }
1145                let plen = route_pattern_len(pattern);
1146                let better = match &best {
1147                    None => true,
1148                    Some((_, bc, bl, bi, _)) => {
1149                        (captures, std::cmp::Reverse(plen), is_item_route)
1150                            < (*bc, std::cmp::Reverse(*bl), *bi)
1151                    }
1152                };
1153                if better {
1154                    best = Some((
1155                        format!("{kind}:{}", spec.id),
1156                        captures,
1157                        plen,
1158                        is_item_route,
1159                        capture_params,
1160                    ));
1161                }
1162            }
1163        }
1164    }
1165    if std::env::var_os("PAVO_DEBUG_ROUTE").is_some() {
1166        eprintln!("route_target({route:?}) -> {best:?}");
1167    }
1168    best.map(|(target, _, _, _, mut params)| {
1169        for (k, v) in http::parse_query(query) {
1170            if !k.is_empty() {
1171                params.insert(k, http::param_json(&v));
1172            }
1173        }
1174        (target, params)
1175    })
1176}

1178/// Normalize a CLI target that may be a `/route?query` deep link:
1179/// routes resolve to `<driver>:<view>` + params (explicit `params`
1180/// win over route-derived ones); anything else passes through
1181/// untouched. A leading-`/` target matching no declared route is a
1182/// hard error — the driver-ref grammar never produces leading
1183/// slashes, so it's unambiguously a route.
1184fn resolve_deep_link(host: &Host, target: &str, params: Params) -> Result<(String, Params), Fail> {
1185    if !target.trim_start().starts_with('/') {
1186        return Ok((target.to_string(), params));
1187    }
1188    match route_target(host, target) {
1189        Some((reference, mut route_params)) => {
1190            for (k, v) in params {
1191                route_params.insert(k, v);
1192            }
1193            Ok((reference, route_params))
1194        }
1195        None => Err(Fail::err(format!(
1196            "unknown route '{target}' — no view route or item_route \
1197             template matches (see `pavo views` for declared routes)"
1198        ))),
1199    }
1200}

1211    item_id: Option<&str>,
1212    action_target: Option<&Value>,
1213    params: &Params,
1214) -> Value {
1215    let driver = split_view_ref(target).map(|(d, _)| d).unwrap_or_default();
1216    let view = host.resolve_view(target, view_params.clone()).ok();
1217    let item = item_id.map(|id| json!({ "id": id }));
1218    interpolate_ctx_value(
1219        &driver,
1220        view.as_ref().map(|r| &r.result),
1221        action_target,
1222        params,
1223        item.as_ref(),
1224    )
1225}

1227/// Collect the action rows visible at `target`: declared actions
1228/// (manifest-global always; node/tree-item when the target names a
1229/// resolvable view) plus ops-manifest derived actions as a
1230/// zero-declaration fallback. `strict_view` propagates view
1231/// resolution errors (the `actions` verb); invoke passes `false`
1232/// and falls back to manifest scope so undeclared/param-gated views
1233/// still dispatch.
1234fn action_rows(
1235    host: &Host,
1236    target: &str,
1237    view_params: Params,
1238    strict_view: bool,
1239) -> Result<Vec<ActionRow>, Fail> {
1240    let (driver, view_id) = split_view_ref(target)?;
1241    let manifest = host.manifest(&driver)?;
1242    let mut rows: Vec<ActionRow> = Vec::new();
1243    let mut declared: std::collections::BTreeSet<String> = std::collections::BTreeSet::new();
1244
1245    // Node/item actions need the resolved view — only attempted when
1246    // the target names one (bare <driver> = manifest scope).
1247    let resolved = if view_id.is_some() {
1248        match host.resolve_view(target, view_params) {
1249            Ok(r) => Some(r),
1250            Err(e) if strict_view => return Err(e.into()),
1251            Err(_) => None,
1252        }
1253    } else {
1254        None
1255    };
1256    match &resolved {
1257        Some(r) => {
1258            for s in collect_actions(&manifest, &r.result) {
1259                declared.insert(s.action.id.clone());
1260                rows.push(ActionRow {
1261                    scope: match s.scope {
1262                        ActionScope::Manifest => "manifest",
1263                        ActionScope::Node => "node",
1264                        ActionScope::Item => "item",
1265                    },
1266                    view: s.view.clone(),
1267                    item_id: s.item_id.map(|id| id.into_owned()),
1268                    action: s.action.into_owned(),
1269                });
1270            }
1271        }
1272        None => {
1273            for a in manifest_actions(&manifest) {
1274                declared.insert(a.id.clone());
1275                rows.push(ActionRow {
1276                    scope: "manifest",
1277                    view: view_id.clone(),
1278                    item_id: None,
1279                    action: a.clone(),
1280                });
1281            }
1282        }
1283    }
1284
1285    // Ops-manifest fallback: `query "ops"` answers
1286    // `{ops:[{name,doc,params,mutates}]}` on surface drivers — each
1287    // op becomes an action when no declaration claimed the id.
1288    // Drivers without an ops manifest just error → skipped.
1289    if let Ok(q) = host.query(&driver, "ops", Params::new()) {
1290        for a in ops_actions(&q.data) {
1291            if declared.insert(a.id.clone()) {
1292                rows.push(ActionRow {
1293                    scope: "ops",
1294                    view: view_id.clone(),
1295                    item_id: None,
1296                    action: a,
1297                });
1298            }
1299        }
1300    }
1301    Ok(rows)
1302}

1425/// Read one line from stdin for `--interactive`/confirm prompts.
1426/// Prompts go to stderr so stdout stays machine-clean.
1427fn read_tty_line(prompt: &str) -> Result<String, Fail> {
1428    use std::io::Write;
1429    eprint!("{prompt}");
1430    let _ = std::io::stderr().flush();
1431    let mut line = String::new();
1432    std::io::stdin()
1433        .read_line(&mut line)
1434        .map_err(|e| Fail::err(format!("read stdin: {e}")))?;
1435    Ok(line.trim().to_string())
1436}

1438/// Enforce the action's [`DangerLevel`] before dispatch.
1439/// `action_target` is the already-parsed `--target`/`--target-id`
1440/// value. `--yes` short-circuits everything *except* the
1441/// type-target id requirement (the id is the confirmation).
1442fn check_danger(
1443    action: &str,
1444    danger: DangerLevel,
1445    action_target: Option<&Value>,
1446    yes: bool,
1447) -> Result<(), Fail> {
1448    match danger {
1449        DangerLevel::None => Ok(()),
1450        DangerLevel::Confirm => {
1451            if yes {
1452                return Ok(());
1453            }
1454            if !std::io::stdin().is_terminal() {
1455                return Err(Fail::usage(format!(
1456                    "action '{action}' needs confirmation — re-run with --yes \
1457                     (non-interactive) or on a TTY"
1458                )));
1459            }
1460            let ans = read_tty_line(&format!("invoke '{action}'? [y/N] "))?;
1461            if matches!(ans.to_ascii_lowercase().as_str(), "y" | "yes") {
1462                Ok(())
1463            } else {
1464                Err(Fail {
1465                    code: 1,
1466                    msg: format!("aborted — '{action}' not invoked"),
1467                })
1468            }
1469        }
1470        DangerLevel::TypeTarget => {
1471            // The typed id doubles as the action target — it must be
1472            // a plain string (`--target-id`, or `--target` carrying a
1473            // JSON string).
1474            let id = match action_target {
1475                Some(Value::String(s)) if !s.is_empty() => s.clone(),
1476                Some(_) => {
1477                    return Err(Fail::usage(format!(
1478                        "type-target action '{action}' needs a string id — \
1479                         pass --target-id <id>"
1480                    )));
1481                }
1482                None => {
1483                    return Err(Fail::usage(format!(
1484                        "type-target action '{action}' needs the target id — \
1485                         pass --target-id <id>"
1486                    )));
1487                }
1488            };
1489            if yes {
1490                return Ok(());
1491            }
1492            if !std::io::stdin().is_terminal() {
1493                return Err(Fail::usage(format!(
1494                    "type-target action '{action}' needs confirmation — \
1495                     re-run with --yes --target-id {id} (non-interactive) \
1496                     or on a TTY"
1497                )));
1498            }
1499            let ans = read_tty_line(&format!("type '{id}' to confirm '{action}': "))?;
1500            if ans == id {
1501                Ok(())
1502            } else {
1503                Err(Fail {
1504                    code: 1,
1505                    msg: format!("aborted — '{action}' not invoked"),
1506                })
1507            }
1508        }
1509    }
1510}

1512/// Prompt for one missing param on a TTY, guided by its
1513/// [`ParamSchema`]. Returns the coerced JSON value.
1514fn prompt_param(p: &ParamSchema) -> Result<Value, Fail> {
1515    let label = p.label.clone().unwrap_or_else(|| p.name.clone());
1516    let mut prompt = format!("{label}:");
1517    if let Some(h) = p.hint.as_deref().or(p.doc.as_deref()) {
1518        prompt.push_str(&format!(" ({h})"));
1519    }
1520    if !p.options.is_empty() {
1521        prompt.push_str(&format!(" [{}]", p.options.join("|")));
1522    }
1523    if let Some(d) = &p.default {
1524        prompt.push_str(&format!(" [default: {}]", d));
1525    }
1526    prompt.push(' ');
1527    let raw = read_tty_line(&prompt)?;
1528    if raw.is_empty() {
1529        if let Some(d) = &p.default {
1530            return Ok(d.clone());
1531        }
1532        return Ok(Value::Null);
1533    }
1534    let value = match p.kind {
1535        ParamKind::Bool => match raw.to_ascii_lowercase().as_str() {
1536            "y" | "yes" | "true" | "1" | "on" => Value::Bool(true),
1537            "n" | "no" | "false" | "0" | "off" => Value::Bool(false),
1538            _ => {
1539                return Err(Fail::usage(format!(
1540                    "param '{}': expected y/n, got '{raw}'",
1541                    p.name
1542                )))
1543            }
1544        },
1545        ParamKind::Integer => raw.parse::<i64>().map(Value::from).map_err(|_| {
1546            Fail::usage(format!("param '{}': expected integer, got '{raw}'", p.name))
1547        })?,
1548        ParamKind::Number => raw.parse::<f64>().map(Value::from).map_err(|_| {
1549            Fail::usage(format!("param '{}': expected number, got '{raw}'", p.name))
1550        })?,
1551        ParamKind::Enum => {
1552            // Exact option, or 1-based index into options.
1553            if p.options.iter().any(|o| o == &raw) {
1554                Value::String(raw)
1555            } else if let Ok(i) = raw.parse::<usize>() {
1556                match p.options.get(i.wrapping_sub(1)) {
1557                    Some(o) => Value::String(o.clone()),
1558                    None => {
1559                        return Err(Fail::usage(format!(
1560                            "param '{}': option {i} out of range [{}]",
1561                            p.name,
1562                            p.options.join("|")
1563                        )))
1564                    }
1565                }
1566            } else {
1567                return Err(Fail::usage(format!(
1568                    "param '{}': expected one of [{}], got '{raw}'",
1569                    p.name,
1570                    p.options.join("|")
1571                )));
1572            }
1573        }
1574        ParamKind::StringList => Value::Array(
1575            raw.split(',')
1576                .map(str::trim)
1577                .filter(|s| !s.is_empty())
1578                .map(|s| Value::String(s.to_string()))
1579                .collect(),
1580        ),
1581        _ => parse_scalar(&raw),
1582    };
1583    Ok(value)
1584}

1621/// [`cmd_invoke`] over an explicit host — split for tests.
1622#[allow(clippy::too_many_arguments)]
1623fn cmd_invoke_host(
1624    host: &mut Host,
1625    target: &str,
1626    action: &str,
1627    params: &mut Params,
1628    action_target: Option<Value>,
1629    yes: bool,
1630    interactive: bool,
1631    json: bool,
1632) -> Result<(), Fail> {
1633    // `/route?query` deep links → `<driver>:<view>`; route-derived
1634    // params are *view* params (they feed the view resolve inside
1635    // `action_rows`/`invoke_ctx`), never action params.
1636    let (target, view_params) = resolve_deep_link(host, target, Params::new())?;
1637    let target = target.as_str();
1638
1639    // Locate the declaration (danger + param metadata + route).
1640    // Best-effort: undeclared actions still dispatch — the driver is
1641    // authoritative (ops-derived rows cover zero-declaration
1642    // drivers). The row is kept whole: item scope feeds `{item.*}`
1643    // interpolation for route templates.
1644    let declared_row = action_rows(host, target, view_params.clone(), false)
1645        .ok()
1646        .and_then(|rows| rows.into_iter().find(|r| r.action.id == action));
1647    let declared = declared_row.as_ref().map(|r| r.action.clone());
1648    let item_id = declared_row.as_ref().and_then(|r| r.item_id.as_deref());
1649
1650    if let Some(a) = &declared {
1651        check_danger(action, a.danger, action_target.as_ref(), yes)?;
1652        // Fill declared defaults for absent params, then prompt for
1653        // the rest under --interactive.
1654        for p in &a.params {
1655            if !params.contains_key(&p.name) {
1656                if let Some(d) = &p.default {
1657                    params.insert(p.name.clone(), d.clone());
1658                }
1659            }
1660        }
1661        if interactive {
1662            if !std::io::stdin().is_terminal() {
1663                return Err(Fail::usage(
1664                    "--interactive needs a TTY — pass --param k=v instead",
1665                ));
1666            }
1667            for p in &a.params {
1668                let missing = match params.get(&p.name) {
1669                    None | Some(Value::Null) => true,
1670                    Some(Value::String(s)) => s.is_empty(),
1671                    _ => false,
1672                };
1673                if missing {
1674                    let v = prompt_param(p)?;
1675                    if !v.is_null() {
1676                        params.insert(p.name.clone(), v);
1677                    }
1678                }
1679            }
1680        }
1681        // Client-side validation (required/options/bounds) — drivers
1682        // stay authoritative via ActionResult.errors.
1683        let errors = validate_params(a, params);
1684        if !errors.is_empty() {
1685            let detail = errors
1686                .iter()
1687                .map(|(k, v)| format!("{k}: {}", v.as_str().unwrap_or("invalid")))
1688                .collect::<Vec<_>>()
1689                .join("; ");
1690            return Err(Fail {
1691                code: 1,
1692                msg: format!("invalid params for '{action}' — {detail}"),
1693            });
1694        }
1695    } else if interactive {
1696        return Err(Fail::usage(format!(
1697            "--interactive needs a declared action — '{action}' has no \
1698             ParamSchema (try `pavo actions {target}`)"
1699        )));
1700    }
1701
1702    // Route actions navigate instead of dispatching — danger and
1703    // params still applied host-side above (a form may collect
1704    // params the template consumes), but no `invoke` wire op is
1705    // sent. The interpolated route is the whole output.
1706    if let Some(template) = declared.as_ref().and_then(|a| a.route.clone()) {
1707        let ctx = invoke_ctx(
1708            host,
1709            target,
1710            &view_params,
1711            item_id,
1712            action_target.as_ref(),
1713            params,
1714        );
1715        let route = interpolate(&template, &ctx);
1716        if json {
1717            println!(
1718                "{}",
1719                serde_json::to_string_pretty(&json!({ "route": route }))
1720                    .map_err(|e| Fail::err(e.to_string()))?
1721            );
1722        } else {
1723            println!("{route}");
1724        }
1725        return Ok(());
1726    }
1727
1728    // `action_target` is cloned — `ActionResult.route` interpolation
1729    // below still needs it.
1730    let result = host.invoke(target, action, None, action_target.clone(), params.clone())?;
1731    if json {
1732        println!(
1733            "{}",
1734            serde_json::to_string_pretty(&result).map_err(|e| Fail::err(e.to_string()))?
1735        );
1736    } else {
1737        if let Some(message) = &result.message {
1738            println!("{message}");
1739        }
1740        match &result.data {
1741            Some(data) => println!(
1742                "{}",
1743                serde_json::to_string_pretty(data).map_err(|e| Fail::err(e.to_string()))?
1744            ),
1745            None if result.message.is_none() && result.errors.is_empty() => {
1746                println!("(empty result)");
1747            }
1748            None => {}
1749        }
1750        // Driver-requested redirect — interpolate the route template
1751        // over the invoke context and hint the destination. `--json`
1752        // carries the raw template in `result.route` already.
1753        if let Some(template) = &result.route {
1754            let ctx = invoke_ctx(
1755                host,
1756                target,
1757                &view_params,
1758                item_id,
1759                action_target.as_ref(),
1760                params,
1761            );
1762            println!("→ {}", interpolate(template, &ctx));
1763        }
1764    }
1765    // Per-param errors mean the driver *rejected* the invoke (the
1766    // wire still answered ok:true) — same convention as submit.
1767    if !result.errors.is_empty() {
1768        let detail = result
1769            .errors
1770            .iter()
1771            .map(|(k, v)| format!("{k}: {}", v.as_str().unwrap_or("invalid")))
1772            .collect::<Vec<_>>()
1773            .join("; ");
1774        return Err(Fail {
1775            code: 1,
1776            msg: format!("invoke rejected — {detail}"),
1777        });
1778    }
1779    Ok(())
1780}

1846/// Parse repeated `--param k=v` args into a [`Params`] map. Values
1847/// parse as JSON when possible (`k=42`, `k=true`, `k={…}`), else
1848/// become strings. Missing `=` or an empty key is a usage error.
1849fn parse_params(args: &[String]) -> Result<Params, Fail> {
1850    let mut params = Params::new();
1851    for arg in args {
1852        let Some((key, raw)) = arg.split_once('=') else {
1853            return Err(Fail::usage(format!(
1854                "--param expects key=value, got '{arg}'"
1855            )));
1856        };
1857        if key.is_empty() {
1858            return Err(Fail::usage(format!("--param has an empty key: '{arg}'")));
1859        }
1860        let value = serde_json::from_str(raw).unwrap_or_else(|_| Value::String(raw.to_string()));
1861        params.insert(key.to_string(), value);
1862    }
1863    Ok(params)
1864}
