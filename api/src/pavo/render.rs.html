1//! ViewNode → text / Graphviz-dot / Mermaid renderers.
2//!
3//! The CLI's `view` verb renders a resolved [`pavo_protocol::wire::ViewResult`]
4//! on a terminal: `--format text` walks the node tree and prints a
5//! human-readable summary, `--format dot`/`--format mermaid` emit the
6//! view's `graph` nodes in those graph description languages.
7//!
8//! Everything works on the node's **raw JSON** rather than the typed
9//! [`ViewNode`] enum so the protocol's forward-compat rule holds
10//! end-to-end: a node whose `kind` this host doesn't know degrades to
11//! [`kv_fallback_json`] (a generic key/value listing) instead of
12//! failing the render. Query-bound data comes from the
13//! [`ViewResult::data`](pavo_protocol::wire::ViewResult) map, keyed by
14//! the node's `query` field.
15//!
16//! Format choices (documented, deterministic):
17//!
18//! - **text** — containers (`panel`/`row`/`column`) print their title
19//!   then children indented two spaces per level; `row` does not
20//!   attempt side-by-side layout (children stack like `column`).
21//! - **dot** — one `digraph` block *per* graph node, concatenated with
22//!   blank lines, so graphs never merge ids across views.
23//! - **mermaid** — a *single* `graph TD` merging every graph node in
24//!   the view; vertex ids are rewritten to `n0`, `n1`, … in first-seen
25//!   order so ids from different graphs can't collide.
26//!
27//! Renderer plugins take precedence over every built-in format except
28//! `json` (the raw `ViewResult` dump — host-owned, never claimed):
29//! for `text`/`dot`/`mermaid` the `*_plugins` variants substitute a
30//! claimed node's [`RenderModel`] (keyed by tree path, see
31//! [`pavo_core::renderers::plugin_nodes`]) and keep the built-in
32//! interpretation for everything unclaimed — the same per-node
33//! fallback rule the TUI and web surfaces apply.
34

42/// Render the view tree + bound data as indented plain text.
43/// Convenience wrapper — no plugin renders. Used by tests; the CLI
44/// verb goes through [`render_text_plugins`].
45#[must_use]
46#[allow(dead_code)]
47pub fn render_text(root: &ViewNode, data: &Map<String, Value>) -> String {
48    render_text_plugins(root, data, &HashMap::new())
49}

51/// [`render_text`] with renderer-plugin substitution: nodes whose tree
52/// path (child indices joined by `.`, root = `""`) has a
53/// [`RenderModel`] in `renders` emit the plugin's output instead of
54/// the built-in interpretation. Paths come from
55/// [`pavo_core::renderers::plugin_nodes`].
56#[must_use]
57pub fn render_text_plugins(
58    root: &ViewNode,
59    data: &Map<String, Value>,
60    renders: &HashMap<String, RenderModel>,
61) -> String {
62    // ViewNode → JSON can't fail for a well-formed node; on the
63    // impossible error we render an empty doc rather than panic.
64    let value = serde_json::to_value(root).unwrap_or(Value::Null);
65    let mut out = String::new();
66    node_text_path(&value, data, 0, String::new(), renders, &mut out);
67    out
68}

70/// Render one [`RenderModel`] as terminal text — the CLI mapping for
71/// plugin output (`--format` values the plugin claims). `Svg` markup
72/// prints verbatim (that IS the requested format); other models use
73/// the text-substitution mapping.
74#[must_use]
75pub fn render_model_cli(model: &RenderModel) -> String {
76    match model {
77        RenderModel::Svg { markup } => format!("{markup}\n"),
78        other => {
79            let mut out = String::new();
80            render_model_text(other, "", &mut out);
81            out
82        }
83    }
84}

86/// Emit a [`RenderModel`] as indented plain text (shared by the
87/// `--format text` substitution path and [`render_model_cli`]).
88fn render_model_text(model: &RenderModel, ind: &str, out: &mut String) {
89    match model {
90        RenderModel::Text { lines } => {
91            for line in lines {
92                out.push_str(&format!("{ind}{line}\n"));
93            }
94        }
95        RenderModel::Rows {
96            headers,
97            aligns,
98            rows,
99        } => {
100            out.push_str(&rows_text(headers, aligns, rows, ind));
101        }
102        RenderModel::Svg { markup } => {
103            out.push_str(&format!("{ind}[svg: {} bytes]\n", markup.len()));
104        }
105        RenderModel::Placeholder { component, props } => {
106            out.push_str(&format!("{ind}[component: {component}]"));
107            if !props.is_empty() {
108                out.push_str(&format!(
109                    " {}",
110                    serde_json::to_string(props).unwrap_or_default()
111                ));
112            }
113            out.push('\n');
114        }
115    }
116}

118/// Aligned plain-text table for a [`RenderModel::Rows`] — same visual
119/// language as `table_text` (two-space gutters, `-` separator).
120fn rows_text(
121    headers: &[String],
122    aligns: &[pavo_protocol::Align],
123    rows: &[Vec<String>],
124    ind: &str,
125) -> String {
126    let mut out = String::new();
127    if headers.is_empty() && rows.is_empty() {
128        out.push_str(&format!("{ind}(empty)\n"));
129        return out;
130    }
131    let ncols = headers
132        .len()
133        .max(rows.iter().map(Vec::len).max().unwrap_or(0));
134    let align_at = |i: usize| -> &str {
135        aligns
136            .get(i)
137            .map(|a| match a {
138                pavo_protocol::Align::Left => "left",
139                pavo_protocol::Align::Center => "center",
140                pavo_protocol::Align::Right => "right",
141            })
142            .unwrap_or("left")
143    };
144    let mut widths: Vec<usize> = (0..ncols)
145        .map(|i| headers.get(i).map_or(0, |h| h.chars().count()))
146        .collect();
147    for row in rows {
148        for (i, cell) in row.iter().enumerate() {
149            if i < widths.len() {
150                widths[i] = widths[i].max(cell.chars().count());
151            }
152        }
153    }
154    let header_cells: Vec<String> = (0..ncols)
155        .map(|i| {
156            let h = headers.get(i).cloned().unwrap_or_default();
157            pad(&h, widths[i], align_at(i))
158        })
159        .collect();
160    out.push_str(&format!("{ind}{}\n", header_cells.join("  ")));
161    let sep: Vec<String> = widths.iter().map(|w| "-".repeat(*w)).collect();
162    out.push_str(&format!("{ind}{}\n", sep.join("  ")));
163    for row in rows {
164        let line: Vec<String> = (0..ncols)
165            .map(|i| {
166                let cell = row.get(i).cloned().unwrap_or_default();
167                pad(&cell, widths[i], align_at(i))
168            })
169            .collect();
170        out.push_str(&format!("{ind}{}\n", line.join("  ")));
171    }
172    out
173}

175/// Render every `graph` node in the view as Graphviz DOT — one
176/// `digraph` block per graph node, concatenated.
177///
178/// # Errors
179/// `Err` when the view contains no `graph` nodes.
180#[allow(dead_code)] // tests + the no-plugin path; the CLI goes through render_dot_plugins
181pub fn render_dot(root: &ViewNode, data: &Map<String, Value>) -> Result<String, String> {
182    render_dot_plugins(root, data, &HashMap::new())
183}

185/// [`render_dot`] with renderer-plugin substitution: a node claimed
186/// for `cli`+`dot` (a [`RenderModel`] in `renders`, keyed by tree
187/// path) emits the plugin's output in tree order instead of the
188/// built-in interpretation; unclaimed `graph` nodes still emit
189/// `digraph` blocks (numbered `pavo-0`, `pavo-1`, … over built-in
190/// graphs only). A claimed container suppresses its whole subtree,
191/// matching [`pavo_core::renderers::plugin_nodes`].
192///
193/// # Errors
194/// `Err` when nothing renders — no `graph` nodes and no plugin
195/// output.
196pub fn render_dot_plugins(
197    root: &ViewNode,
198    data: &Map<String, Value>,
199    renders: &HashMap<String, RenderModel>,
200) -> Result<String, String> {
201    let value = serde_json::to_value(root).unwrap_or(Value::Null);
202    let mut out = String::new();
203    let mut graph_index = 0usize;
204    node_dot_path(
205        &value,
206        data,
207        String::new(),
208        renders,
209        &mut graph_index,
210        &mut out,
211    );
212    if out.is_empty() {
213        return Err("view contains no graph nodes".to_string());
214    }
215    Ok(out)
216}

218/// Tree walk behind [`render_dot_plugins`] — see
219/// [`node_text_path`] for the shared path/plugin-substitution
220/// contract.
221fn node_dot_path(
222    v: &Value,
223    data: &Map<String, Value>,
224    path: String,
225    renders: &HashMap<String, RenderModel>,
226    graph_index: &mut usize,
227    out: &mut String,
228) {
229    if let Some(model) = renders.get(&path) {
230        out.push_str(&render_model_cli(model));
231        return;
232    }
233    if node_kind_json(v) == Some("graph") {
234        let (nodes, edges) = graph_parts(v, data);
235        // Blank line between digraph blocks (built-in output only —
236        // plugin models carry their own trailing newline).
237        if *graph_index > 0 {
238            out.push('\n');
239        }
240        out.push_str(&format!("digraph \"pavo-{graph_index}\" {{\n"));
241        *graph_index += 1;
242        for node in &nodes {
243            let id = node.get("id").and_then(Value::as_str).unwrap_or("");
244            let label = node.get("label").and_then(Value::as_str).unwrap_or(id);
245            out.push_str(&format!(
246                "  \"{}\" [label=\"{}\"];\n",
247                dot_str(id),
248                dot_str(label)
249            ));
250        }
251        for edge in &edges {
252            let from = edge.get("from").and_then(Value::as_str).unwrap_or("");
253            let to = edge.get("to").and_then(Value::as_str).unwrap_or("");
254            match edge.get("label").and_then(Value::as_str) {
255                Some(label) => out.push_str(&format!(
256                    "  \"{}\" -> \"{}\" [label=\"{}\"];\n",
257                    dot_str(from),
258                    dot_str(to),
259                    dot_str(label)
260                )),
261                None => out.push_str(&format!(
262                    "  \"{}\" -> \"{}\";\n",
263                    dot_str(from),
264                    dot_str(to)
265                )),
266            }
267        }
268        out.push_str("}\n");
269        return;
270    }
271    if let Some(children) = v.get("children").and_then(Value::as_array) {
272        for (i, child) in children.iter().enumerate() {
273            let child_path = if path.is_empty() {
274                i.to_string()
275            } else {
276                format!("{path}.{i}")
277            };
278            node_dot_path(child, data, child_path, renders, graph_index, out);
279        }
280    }
281}

283/// Render every `graph` node in the view as a single Mermaid
284/// `graph TD` — all graphs merged, vertex ids rewritten to `n0`, `n1`,
285/// … in first-seen order (nodes first, then edge endpoints).
286///
287/// # Errors
288/// `Err` when the view contains no `graph` nodes.
289#[allow(dead_code)] // tests + the no-plugin path; the CLI goes through render_mermaid_plugins
290pub fn render_mermaid(root: &ViewNode, data: &Map<String, Value>) -> Result<String, String> {
291    render_mermaid_plugins(root, data, &HashMap::new())
292}
293
294/// [`render_mermaid`] with renderer-plugin substitution: claimed
295/// nodes (a [`RenderModel`] in `renders`, keyed by tree path) emit
296/// the plugin's output *after* the merged `graph TD` block, in tree
297/// order; unclaimed `graph` nodes feed the mermaid block as usual.
298/// When every graph is claimed the `graph TD` header is omitted
299/// entirely — plugin output is the whole answer.
300///
301/// # Errors
302/// `Err` when nothing renders — no `graph` nodes and no plugin
303/// output.
304pub fn render_mermaid_plugins(
305    root: &ViewNode,
306    data: &Map<String, Value>,
307    renders: &HashMap<String, RenderModel>,
308) -> Result<String, String> {
309    let value = serde_json::to_value(root).unwrap_or(Value::Null);
310    let mut graphs: Vec<(Vec<Value>, Vec<Value>)> = Vec::new();
311    let mut claimed: Vec<RenderModel> = Vec::new();
312    collect_mermaid_parts(
313        &value,
314        data,
315        String::new(),
316        renders,
317        &mut graphs,
318        &mut claimed,
319    );
320    if graphs.is_empty() && claimed.is_empty() {
321        return Err("view contains no graph nodes".to_string());
322    }
323    let mut out = String::new();
324    if !graphs.is_empty() {
325        out.push_str(&mermaid_body(&graphs));
326    }
327    for model in &claimed {
328        out.push_str(&render_model_cli(model));
329    }
330    Ok(out)
331}
332
333/// Tree walk behind [`render_mermaid_plugins`]: claimed paths collect
334/// their [`RenderModel`] (subtree suppressed, like
335/// [`node_dot_path`]); unclaimed `graph` nodes collect `(nodes,
336/// edges)`; everything else recurses through `children`.
337fn collect_mermaid_parts(
338    v: &Value,
339    data: &Map<String, Value>,
340    path: String,
341    renders: &HashMap<String, RenderModel>,
342    graphs: &mut Vec<(Vec<Value>, Vec<Value>)>,
343    claimed: &mut Vec<RenderModel>,
344) {
345    if let Some(model) = renders.get(&path) {
346        claimed.push(model.clone());
347        return;
348    }
349    if node_kind_json(v) == Some("graph") {
350        graphs.push(graph_parts(v, data));
351        return;
352    }
353    if let Some(children) = v.get("children").and_then(Value::as_array) {
354        for (i, child) in children.iter().enumerate() {
355            let child_path = if path.is_empty() {
356                i.to_string()
357            } else {
358                format!("{path}.{i}")
359            };
360            collect_mermaid_parts(child, data, child_path, renders, graphs, claimed);
361        }
362    }
363}

365/// The merged `graph TD` block for `graphs` — the built-in mermaid
366/// body shared by [`render_mermaid`] and
367/// [`render_mermaid_plugins`].
368fn mermaid_body(graphs: &[(Vec<Value>, Vec<Value>)]) -> String {
369    let mut out = String::from("graph TD\n");
370    let mut ids: HashMap<String, String> = HashMap::new();
371    let mut emitted: std::collections::HashSet<String> = std::collections::HashSet::new();
372    let mut next = 0usize;
373    // Allocate ids for every vertex up front so node lines precede
374    // edges deterministically.
375    for (nodes, _) in graphs {
376        for node in nodes {
377            if let Some(id) = node.get("id").and_then(Value::as_str) {
378                if !ids.contains_key(id) {
379                    ids.insert(id.to_string(), format!("n{next}"));
380                    next += 1;
381                }
382            }
383        }
384    }
385    for (nodes, edges) in graphs {
386        for node in nodes {
387            let Some(id) = node.get("id").and_then(Value::as_str) else {
388                continue;
389            };
390            let Some(safe) = ids.get(id) else { continue };
391            // First-seen label wins — the same raw id across graphs is
392            // one merged vertex, declared once.
393            if !emitted.insert(id.to_string()) {
394                continue;
395            }
396            let label = node.get("label").and_then(Value::as_str).unwrap_or(id);
397            out.push_str(&format!("  {safe}[\"{}\"]\n", mermaid_str(label)));
398        }
399        for edge in edges {
400            let (Some(from), Some(to)) = (
401                edge.get("from").and_then(Value::as_str),
402                edge.get("to").and_then(Value::as_str),
403            ) else {
404                continue;
405            };
406            // Edge endpoints may reference ids never declared as
407            // nodes — allocate on demand.
408            for endpoint in [from, to] {
409                if !ids.contains_key(endpoint) {
410                    ids.insert(endpoint.to_string(), format!("n{next}"));
411                    next += 1;
412                }
413            }
414            let (Some(a), Some(b)) = (ids.get(from), ids.get(to)) else {
415                continue;
416            };
417            match edge.get("label").and_then(Value::as_str) {
418                Some(label) => out.push_str(&format!(
419                    "  {a}-->|{}|{b}\n",
420                    mermaid_str(label).replace('|', "/")
421                )),
422                None => out.push_str(&format!("  {a} --> {b}\n")),
423            }
424        }
425    }
426    out
427}

431/// [`node_text_path`] with a tree `path` (plugin-render lookup key) and
432/// the plugin `renders` map — a claimed node emits its
433/// [`RenderModel`] instead of the built-in dispatch.
434fn node_text_path(
435    v: &Value,
436    data: &Map<String, Value>,
437    depth: usize,
438    path: String,
439    renders: &HashMap<String, RenderModel>,
440    out: &mut String,
441) {
442    let ind = "  ".repeat(depth);
443    if let Some(model) = renders.get(&path) {
444        render_model_text(model, &ind, out);
445        return;
446    }
447    match node_kind_json(v).unwrap_or("") {
448        "panel" | "row" | "column" => {
449            if let Some(title) = scalar_str(v.get("title")) {
450                out.push_str(&format!("{ind}{title}\n"));
451            }
452            if let Some(children) = v.get("children").and_then(Value::as_array) {
453                for (i, child) in children.iter().enumerate() {
454                    let child_path = if path.is_empty() {
455                        i.to_string()
456                    } else {
457                        format!("{path}.{i}")
458                    };
459                    node_text_path(child, data, depth + 1, child_path, renders, out);
460                }
461            }
462        }
463        "table" => table_text(v, data, depth, out),
464        "tree" => tree_text(v, data, depth, out),
465        "graph" => graph_text(v, data, depth, out),
466        "form" => form_text(v, depth, out),
467        "kv" => kv_text(v, data, depth, out),
468        "markdown" => markdown_text(v, depth, out),
469        "embed" => embed_text(v, depth, out),
470        // Unknown or missing kind — degrade to KV per the protocol's
471        // forward-compat rule (kv_fallback_json preserves scalar
472        // fields as pairs and synthesizes a `kind` pair).
473        _ => {
474            let fallback = kv_fallback_json(v);
475            match serde_json::from_value::<KvNode>(fallback) {
476                Ok(kv) => kv_node_text(&kv, depth, out),
477                Err(_) => out.push_str(&format!("{ind}(unrenderable node)\n")),
478            }
479        }
480    }
481}

483/// The node's bound query payload, if `query` names a key in `data`.
484/// Returns `(query_name, payload)`.
485fn bound<'a>(v: &'a Value, data: &'a Map<String, Value>) -> Option<(&'a str, &'a Value)> {
486    let query = v.get("query").and_then(Value::as_str)?;
487    data.get(query).map(|d| (query, d))
488}
489
490/// Print the `{ind}[query 'q' error: …]` line when the bound payload is
491/// a degradation marker (`{"error": "…"}`). Returns true when printed.
492fn bound_error(v: &Value, data: &Map<String, Value>, ind: &str, out: &mut String) -> bool {
493    let Some((query, payload)) = bound(v, data) else {
494        return false;
495    };
496    if let Some(err) = payload.get("error").and_then(Value::as_str) {
497        out.push_str(&format!("{ind}[query '{query}' error: {err}]\n"));
498        return true;
499    }
500    false
501}

503fn title_line(v: &Value, ind: &str, out: &mut String) {
504    if let Some(title) = scalar_str(v.get("title")) {
505        out.push_str(&format!("{ind}{title}\n"));
506    }
507}

509fn table_text(v: &Value, data: &Map<String, Value>, depth: usize, out: &mut String) {
510    let ind = "  ".repeat(depth);
511    title_line(v, &ind, out);
512    if bound_error(v, data, &ind, out) {
513        return;
514    }
515    // Columns: declared `columns[].{key,title}`; fall back to
516    // positional `1..n` when rows exist but no columns are declared.
517    let mut columns: Vec<(String, String)> = v
518        .get("columns")
519        .and_then(Value::as_array)
520        .map(|cols| {
521            cols.iter()
522                .map(|c| {
523                    (
524                        c.get("key")
525                            .and_then(Value::as_str)
526                            .unwrap_or("")
527                            .to_string(),
528                        c.get("title")
529                            .and_then(Value::as_str)
530                            .unwrap_or_else(|| c.get("key").and_then(Value::as_str).unwrap_or(""))
531                            .to_string(),
532                    )
533                })
534                .collect()
535        })
536        .unwrap_or_default();
537    // Align hints, parallel to `columns`.
538    let aligns: Vec<String> = v
539        .get("columns")
540        .and_then(Value::as_array)
541        .map(|cols| {
542            cols.iter()
543                .map(|c| {
544                    c.get("align")
545                        .and_then(Value::as_str)
546                        .unwrap_or("left")
547                        .to_string()
548                })
549                .collect()
550        })
551        .unwrap_or_default();
552
553    // Rows: inline `rows` win; otherwise the bound query payload —
554    // an array of arrays, or an array of objects keyed by column key.
555    let mut rows: Vec<Vec<String>> = Vec::new();
556    if let Some(inline) = v.get("rows").and_then(Value::as_array) {
557        for row in inline {
558            if let Some(cells) = row.as_array() {
559                rows.push(cells.iter().map(cell_str).collect());
560            }
561        }
562    } else if let Some((_, payload)) = bound(v, data) {
563        match payload {
564            Value::Array(items) => {
565                for item in items {
566                    match item {
567                        Value::Array(cells) => {
568                            rows.push(cells.iter().map(cell_str).collect());
569                        }
570                        Value::Object(map) => {
571                            rows.push(
572                                columns
573                                    .iter()
574                                    .map(|(key, _)| cell_str(map.get(key).unwrap_or(&Value::Null)))
575                                    .collect(),
576                            );
577                        }
578                        other => rows.push(vec![cell_str(other)]),
579                    }
580                }
581            }
582            other => {
583                out.push_str(&format!("{ind}{}\n", cell_str(other)));
584                return;
585            }
586        }
587    }
588    if columns.is_empty() && !rows.is_empty() {
589        let n = rows.iter().map(Vec::len).max().unwrap_or(0);
590        columns = (1..=n).map(|i| (i.to_string(), i.to_string())).collect();
591    }
592    if columns.is_empty() {
593        out.push_str(&format!("{ind}(empty table)\n"));
594        return;
595    }
596    // Normalize row lengths to the column count.
597    for row in &mut rows {
598        row.resize(columns.len(), String::new());
599        row.truncate(columns.len());
600    }
601    // Column widths = max(header, cells) in chars.
602    let mut widths: Vec<usize> = columns
603        .iter()
604        .map(|(_, title)| title.chars().count())
605        .collect();
606    for row in &rows {
607        for (i, cell) in row.iter().enumerate() {
608            if i < widths.len() {
609                widths[i] = widths[i].max(cell.chars().count());
610            }
611        }
612    }
613    let header: Vec<String> = columns
614        .iter()
615        .enumerate()
616        .map(|(i, (_, title))| pad(title, widths[i], align_at(&aligns, i)))
617        .collect();
618    out.push_str(&format!("{ind}{}\n", header.join("  ")));
619    let sep: Vec<String> = widths.iter().map(|w| "-".repeat(*w)).collect();
620    out.push_str(&format!("{ind}{}\n", sep.join("  ")));
621    for row in &rows {
622        let line: Vec<String> = row
623            .iter()
624            .enumerate()
625            .map(|(i, cell)| pad(cell, widths[i], align_at(&aligns, i)))
626            .collect();
627        out.push_str(&format!("{ind}{}\n", line.join("  ")));
628    }
629}

631fn tree_text(v: &Value, data: &Map<String, Value>, depth: usize, out: &mut String) {
632    let ind = "  ".repeat(depth);
633    title_line(v, &ind, out);
634    if bound_error(v, data, &ind, out) {
635        return;
636    }
637    let mut items: Vec<TreeItem> = Vec::new();
638    let inline = v.get("items").and_then(Value::as_array);
639    let bound_items = bound(v, data).and_then(|(_, d)| {
640        d.get("items")
641            .and_then(Value::as_array)
642            .or_else(|| d.as_array())
643    });
644    let source = inline.or(bound_items);
645    if let Some(arr) = source {
646        for item in arr {
647            if let Ok(t) = serde_json::from_value::<TreeItem>(item.clone()) {
648                items.push(t);
649            }
650        }
651    }
652    if items.is_empty() {
653        // Bound payload that isn't an items array — show it raw.
654        if let Some((_, payload)) = bound(v, data) {
655            if inline.is_none() && !payload.is_array() {
656                out.push_str(&format!("{ind}{}\n", cell_str(payload)));
657                return;
658            }
659        }
660        out.push_str(&format!("{ind}(empty tree)\n"));
661        return;
662    }
663    for item in &items {
664        tree_item_text(item, depth, out);
665    }
666}
667
668fn tree_item_text(item: &TreeItem, depth: usize, out: &mut String) {
669    let ind = "  ".repeat(depth);
670    out.push_str(&format!("{ind}- {}", item.label));
671    if let Some(detail) = &item.detail {
672        out.push_str(&format!("  — {detail}"));
673    }
674    out.push('\n');
675    for child in &item.children {
676        tree_item_text(child, depth + 1, out);
677    }
678}

680fn graph_text(v: &Value, data: &Map<String, Value>, depth: usize, out: &mut String) {
681    let ind = "  ".repeat(depth);
682    title_line(v, &ind, out);
683    if bound_error(v, data, &ind, out) {
684        return;
685    }
686    let (nodes, edges) = graph_parts(v, data);
687    if edges.is_empty() && nodes.is_empty() {
688        out.push_str(&format!("{ind}(empty graph)\n"));
689        return;
690    }
691    // id → label lookup so edges read as names where possible.
692    let mut labels: HashMap<&str, &str> = HashMap::new();
693    for node in &nodes {
694        if let (Some(id), Some(label)) = (
695            node.get("id").and_then(Value::as_str),
696            node.get("label").and_then(Value::as_str),
697        ) {
698            labels.insert(id, label);
699        }
700    }
701    let disp = |id: &str| -> String {
702        labels
703            .get(id)
704            .map(|s| (*s).to_string())
705            .unwrap_or_else(|| id.to_string())
706    };
707    if edges.is_empty() {
708        for node in &nodes {
709            let id = node.get("id").and_then(Value::as_str).unwrap_or("");
710            out.push_str(&format!("{ind}- {} ({id})\n", node_label(node, id)));
711        }
712        return;
713    }
714    for edge in &edges {
715        let from = edge.get("from").and_then(Value::as_str).unwrap_or("");
716        let to = edge.get("to").and_then(Value::as_str).unwrap_or("");
717        out.push_str(&format!("{ind}{} -> {}", disp(from), disp(to)));
718        if let Some(label) = edge.get("label").and_then(Value::as_str) {
719            out.push_str(&format!("  ({label})"));
720        }
721        out.push('\n');
722    }
723}

725fn node_label<'a>(node: &'a Value, id: &'a str) -> &'a str {
726    node.get("label").and_then(Value::as_str).unwrap_or(id)
727}

729fn form_text(v: &Value, depth: usize, out: &mut String) {
730    let ind = "  ".repeat(depth);
731    title_line(v, &ind, out);
732    if let Some(fields) = v.get("fields").and_then(Value::as_array) {
733        for f in fields {
734            let key = f.get("key").and_then(Value::as_str).unwrap_or("");
735            let label = f.get("label").and_then(Value::as_str).unwrap_or(key);
736            let kind = f.get("kind").and_then(Value::as_str).unwrap_or("text");
737            out.push_str(&format!("{ind}- {key}: {label} [{kind}]"));
738            if f.get("required").and_then(Value::as_bool) == Some(true) {
739                out.push_str(" (required)");
740            }
741            // `editable:false` (or absent on very old drivers —
742            // treated editable per protocol default) marks fields a
743            // submit won't pick up; surfaces render them read-only.
744            if f.get("editable").and_then(Value::as_bool) == Some(false) {
745                out.push_str(" (read-only)");
746            }
747            if let Some(default) = f.get("default") {
748                out.push_str(&format!(" default={}", cell_str(default)));
749            }
750            if let Some(value) = f.get("value") {
751                out.push_str(&format!(" value={}", cell_str(value)));
752            }
753            if let Some(options) = f.get("options").and_then(Value::as_array) {
754                let opts: Vec<String> = options.iter().map(cell_str).collect();
755                out.push_str(&format!(" options={}", opts.join("|")));
756            }
757            if let Some(hint) = f.get("hint").and_then(Value::as_str) {
758                out.push_str(&format!(" — {hint}"));
759            }
760            if let Some(ph) = f.get("placeholder").and_then(Value::as_str) {
761                out.push_str(&format!(" placeholder=\"{ph}\""));
762            }
763            out.push('\n');
764        }
765    }
766    if let Some(op) = v.get("submit_op").and_then(Value::as_str) {
767        let label = v
768            .get("submit_label")
769            .and_then(Value::as_str)
770            .unwrap_or("Submit");
771        out.push_str(&format!("{ind}submit → {op} ({label})\n"));
772    }
773}

775fn kv_text(v: &Value, data: &Map<String, Value>, depth: usize, out: &mut String) {
776    let ind = "  ".repeat(depth);
777    title_line(v, &ind, out);
778    if bound_error(v, data, &ind, out) {
779        return;
780    }
781    let mut pairs: Vec<(String, Value)> = v
782        .get("pairs")
783        .and_then(Value::as_array)
784        .map(|arr| {
785            arr.iter()
786                .filter_map(|p| {
787                    Some((
788                        p.get("key").and_then(Value::as_str)?.to_string(),
789                        p.get("value").cloned().unwrap_or(Value::Null),
790                    ))
791                })
792                .collect()
793        })
794        .unwrap_or_default();
795    // Query-bound kv: an object payload becomes the pairs.
796    if pairs.is_empty() {
797        if let Some((_, payload)) = bound(v, data) {
798            if let Value::Object(map) = payload {
799                pairs = map
800                    .iter()
801                    .map(|(k, val)| (k.clone(), val.clone()))
802                    .collect();
803            } else {
804                out.push_str(&format!("{ind}{}\n", cell_str(payload)));
805                return;
806            }
807        }
808    }
809    if pairs.is_empty() {
810        out.push_str(&format!("{ind}(empty)\n"));
811        return;
812    }
813    kv_node_text(
814        &KvNode {
815            base: Default::default(),
816            pairs: pairs
817                .into_iter()
818                .map(|(key, value)| pavo_protocol::KvPair {
819                    key,
820                    value,
821                    kind: None,
822                })
823                .collect(),
824        },
825        depth,
826        out,
827    );
828}

830fn kv_node_text(kv: &KvNode, depth: usize, out: &mut String) {

832    for pair in &kv.pairs {
833        out.push_str(&format!("{ind}{}: {}\n", pair.key, cell_str(&pair.value)));
834    }
835}

865/// A graph node's effective `(nodes, edges)` — inline fields win,
866/// falling back to the bound query payload's `nodes`/`edges` keys.
867fn graph_parts(v: &Value, data: &Map<String, Value>) -> (Vec<Value>, Vec<Value>) {
868    let mut nodes = v
869        .get("nodes")
870        .and_then(Value::as_array)
871        .cloned()
872        .unwrap_or_default();
873    let mut edges = v
874        .get("edges")
875        .and_then(Value::as_array)
876        .cloned()
877        .unwrap_or_default();
878    if let Some((_, payload)) = bound(v, data) {
879        if payload.get("error").and_then(Value::as_str).is_none() {
880            if let Some(n) = payload.get("nodes").and_then(Value::as_array) {
881                if !n.is_empty() {
882                    nodes = n.clone();
883                }
884            }
885            if let Some(e) = payload.get("edges").and_then(Value::as_array) {
886                if !e.is_empty() {
887                    edges = e.clone();
888                }
889            }
890        }
891    }
892    (nodes, edges)
893}

895/// Stringify a JSON cell/pair value for terminal output: strings raw,
896/// numbers/bools via Display, null empty, anything else compact JSON.
897fn cell_str(v: &Value) -> String {
898    match v {
899        Value::String(s) => s.clone(),
900        Value::Null => String::new(),
901        Value::Number(n) => n.to_string(),
902        Value::Bool(b) => b.to_string(),
903        other => serde_json::to_string(other).unwrap_or_default(),
904    }
905}

907fn scalar_str(v: Option<&Value>) -> Option<String> {
908    v.and_then(Value::as_str).map(str::to_string)
909}

911/// Alignment for column `i`: declared `align`, default left.
912fn align_at(aligns: &[String], i: usize) -> &str {
913    aligns.get(i).map(String::as_str).unwrap_or("left")
914}

916/// Pad `text` to `width` chars per the alignment hint.
917fn pad(text: &str, width: usize, align: &str) -> String {
918    let len = text.chars().count();
919    if len >= width {
920        return text.to_string();
921    }
922    let fill = width - len;
923    match align {
924        "right" => format!("{}{text}", " ".repeat(fill)),
925        "center" => {
926            let left = fill / 2;
927            format!("{}{text}{}", " ".repeat(left), " ".repeat(fill - left))
928        }
929        _ => format!("{text}{}", " ".repeat(fill)),
930    }
931}
