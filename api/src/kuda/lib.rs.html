23/// stdout that survives `| head` — BrokenPipe exits 0 like a
24/// well-behaved unix filter instead of panicking.
25fn emit(args: std::fmt::Arguments) {
26    use std::io::Write;
27    let mut o = std::io::stdout().lock();
28    if let Err(e) = o.write_fmt(args).and_then(|_| o.flush()) {
29        if e.kind() == std::io::ErrorKind::BrokenPipe {
30            std::process::exit(0);
31        }
32    }
33}
34macro_rules! out {
35    ($($arg:tt)*) => { emit(format_args!($($arg)*)) };
36}
37macro_rules! outln {
38    ($($arg:tt)*) => { emit(format_args!("{}\n", format_args!($($arg)*))) };
39}

153/// One parsed command → exit code. The op table every driver
154/// shares; argv parsing is a caller concern.
155pub fn dispatch(cli: &Cli) -> i32 {
156    match &cli.cmd {
157        Cmd::Init => init(cli),
158        Cmd::Sync { full } => sync(cli, *full),
159        Cmd::Status => status(cli),
160        Cmd::Map => map(cli),
161        Cmd::Card { module } => card(cli, module),
162        Cmd::Find { symbol } => find(cli, symbol),
163        Cmd::Dependents { name } => dependents(cli, name),
164        Cmd::Refs { name, docs } => refs(cli, name, *docs),
165        Cmd::Doctor => doctor(cli),
166        // `watch` runs an unbounded loop and only stops on a signal —
167        // signals discard coverage data, so it's excluded below and
168        // exercised by the e2e test in normal builds.
169        #[cfg(not(coverage))]
170        Cmd::Watch { poll_refs } => watch(cli, *poll_refs),
171        #[cfg(coverage)]
172        Cmd::Watch { .. } => 0,
173        Cmd::Mcp => mcp(cli),
174        Cmd::Acp => acp(cli),
175        Cmd::GenCliDocs => {
176            outln!("{}", clap_markdown::help_markdown::<Cli>());
177            0
178        }
179    }
180}

234/// The effective domain set for a config: built-ins minus
235/// `[domains].disable` names, then `[[domain]]` entries overlaid —
236/// a same-named entry replaces the built-in (repo wins), the rest
237/// append. Deterministic; `domains()` stays the bare built-ins for
238/// tests/defaults.
239fn resolve_domains(cfg: &Config) -> Vec<DomainDef> {
240    let mut doms: Vec<DomainDef> = domains()
241        .into_iter()
242        .filter(|d| !cfg.domains.disable.iter().any(|n| n == &d.name))
243        .collect();
244    for d in cfg.domains.domain.iter().chain(&cfg.domain) {
245        match doms.iter_mut().find(|b| b.name == d.name) {
246            Some(slot) => *slot = d.clone(),
247            None => doms.push(d.clone()),
248        }
249    }
250    doms
251}

253/// git blob hash — provenance key. In-process sha1
254/// ("blob <len>\0" + bytes): identical values to
255/// `git hash-object`, no subprocess per file.
256fn hash_files(root: &Path, files: &[PathBuf]) -> Vec<String> {
257    files.iter().map(|f| input_hash(root, f)).collect()
258}

260/// Content hash for a provenance input: blob hash for real files;
261/// a non-tree input (a `refs/*` pseudo-path recorded by a domain
262/// source) resolves to the ref's current target sha — the same
263/// value sync stored, so cards don't report phantom drift.
264fn input_hash(root: &Path, rel: &Path) -> String {
265    let h = blob_hash(&root.join(rel));
266    if !h.is_empty() {
267        return h;
268    }
269    let Ok(o) = std::process::Command::new("git")
270        .arg("-C")
271        .arg(root)
272        .args(["rev-parse", "--verify", &rel.display().to_string()])
273        .output()
274    else {
275        return String::new();
276    };
277    if o.status.success() {
278        String::from_utf8_lossy(&o.stdout).trim().to_owned()
279    } else {
280        String::new()
281    }
282}

284fn blob_hash(path: &Path) -> String {
285    use sha1::Digest;
286    let Ok(bytes) = std::fs::read(path) else {
287        return String::new();
288    };
289    let mut h = sha1::Sha1::new();
290    h.update(format!("blob {}\0", bytes.len()).as_bytes());
291    h.update(&bytes);
292    h.finalize().iter().map(|b| format!("{b:02x}")).collect()
293}

302fn init(cli: &Cli) -> i32 {
303    let dir = root(cli);
304    let toml_path = dir.join("kuda.toml");
305    if toml_path.exists() {
306        eprintln!("kuda.toml already exists — not overwriting");
307        return 1;
308    }
309    if let Err(e) = std::fs::write(&toml_path, EXAMPLE_TOML) {
310        eprintln!("write {}: {e}", toml_path.display());
311        return 1;
312    }
313    for d in [".kuda", ".kuda/modules"] {
314        if let Err(e) = std::fs::create_dir_all(dir.join(d)) {
315            eprintln!("mkdir {d}: {e}");
316            return 1;
317        }
318    }
319    ensure_gitignore(&dir, &kuda_core::StoreCfg::default().path);
320    outln!("wrote {} + .kuda/ — run `kuda sync`", toml_path.display());
321    0
322}

324/// Append the rebuildable `.kuda` artifacts to `.gitignore` —
325/// `index.sqlite` + `meta.json` regen from the committed *.yaml
326/// cards, so they must never be checked in. Idempotent, and a no-op
327/// outside a git repo (`.git` may be a worktree *file*, so `exists`,
328/// not `is_dir`). `init` and `sync` both run it — `init` refuses
329/// when kuda.toml already exists, so `sync` is how existing installs
330/// pick the rules up.
331fn ensure_gitignore(dir: &Path, store: &Path) {
332    if !dir.join(".git").exists() {
333        return;
334    }
335    let store_line = if store.is_absolute() {
336        ".kuda/index.sqlite".to_owned()
337    } else {
338        store.to_string_lossy().replace('\\', "/")
339    };
340    let path = dir.join(".gitignore");
341    let mut text = std::fs::read_to_string(&path).unwrap_or_default();
342    let before = text.len();
343    for line in [store_line, ".kuda/meta.json".to_owned()] {
344        if !text.lines().any(|l| l.trim() == line) {
345            if !text.is_empty() && !text.ends_with('\n') {
346                text.push('\n');
347            }
348            text.push_str(&line);
349            text.push('\n');
350        }
351    }
352    if text.len() != before {
353        let _ = std::fs::write(&path, &text);
354    }
355}

357/// Workspace roots from `.kuda/workspace.yaml` (`roots: [...]`),
358/// resolved relative to the file's dir.
359fn workspace_roots(dir: &Path) -> Vec<PathBuf> {
360    let p = dir.join(".kuda/workspace.yaml");
361    let Ok(text) = std::fs::read_to_string(&p) else {
362        return vec![];
363    };
364    let Ok(v) = serde_yaml::from_str::<serde_yaml::Value>(&text) else {
365        return vec![];
366    };
367    v["roots"]
368        .as_sequence()
369        .map(|s| {
370            s.iter()
371                .filter_map(|r| r.as_str())
372                .map(|r| {
373                    let p = dir.join(r);
374                    std::fs::canonicalize(&p).unwrap_or(p)
375                })
376                .collect()
377        })
378        .unwrap_or_default()
379}

381/// The pipeline: extract -> per-module summarize -> cards + store.
382fn sync(cli: &Cli, full: bool) -> i32 {
383    let dir = root(cli);
384    let cfg = load_config(cli);
385    let roots = if cli.workspace {
386        workspace_roots(&dir)
387    } else {
388        vec![dir.clone()]
389    };
390    let roots = if roots.is_empty() {
391        vec![dir.clone()]
392    } else {
393        roots
394    };
395    // cross-root edge resolution: a dep naming a sibling repo's
396    // module must resolve to its canonical spelling (dash-named
397    // package, underscore import — module_key folds both), not fall
398    // through as external. Names come from each root's manifest
399    // pass + domain descriptors — cheap enough to pre-pass.
400    let union: std::collections::HashSet<String> = if roots.len() > 1 {
401        let ls = langs();
402        roots
403            .iter()
404            .flat_map(|r| {
405                kuda_extract_manifest::ManifestExtractor {
406                    excludes: cfg.extract.exclude.clone(),
407                    resolve_git_deps: cfg.workspace.resolve_git_deps,
408                }
409                .extract(r, &ls)
410                .map(|e| e.symbols.into_iter().map(|s| s.module).collect::<Vec<_>>())
411                .unwrap_or_default()
412            })
413            .chain(resolve_domains(&cfg).iter().map(|d| d.module.clone()))
414            .collect()
415    } else {
416        Default::default()
417    };
418    let mut code = 0;
419    for r in &roots {
420        outln!("sync {}", r.display());
421        ensure_gitignore(r, &cfg.store.path);
422        code |= sync_one(r, &cfg, cli, full, &union);
423    }
424    code
425}

427/// Walk inputs derived from the loaded langs: manifest basenames
428/// (ordered list + lookup set) and source extensions (dotless).
429fn walk_sets(
430    langs: &[LangDef],
431) -> (
432    Vec<String>,
433    std::collections::HashSet<String>,
434    std::collections::HashSet<String>,
435) {
436    let manifest_list: Vec<String> = langs
437        .iter()
438        .flat_map(|l| l.manifests.iter().cloned())
439        .collect();
440    let manifest_names = manifest_list.iter().cloned().collect();
441    let src_exts = langs
442        .iter()
443        .flat_map(|l| {

445                .iter()
446                .map(|e| e.trim_start_matches('.').to_owned())
447        })
448        .collect();
449    (manifest_list, manifest_names, src_exts)
450}

452/// Extractor registry — `[extract].kinds` names → adapter instances,
453/// tried in config order. New bundled adapter = one arm here + a dep
454/// in Cargo.toml; unknown kinds warn and skip (a config typo isn't a
455/// hard fail — the other extractors still run).
456fn extractors(
457    kinds: &[String],
458    cfg: &Config,
459    doms: &[DomainDef],
460    pkg_by_dir: &std::collections::HashMap<PathBuf, String>,

462    kinds
463        .iter()
464        .filter_map(|k| match k.as_str() {
465            "manifest" => Some(Box::new(kuda_extract_manifest::ManifestExtractor {
466                excludes: cfg.extract.exclude.clone(),
467                resolve_git_deps: cfg.workspace.resolve_git_deps,
468            }) as Box<dyn Extractor>),
469            "treesitter" => Some(Box::new(kuda_extract_treesitter::TreesitterExtractor {
470                excludes: cfg.extract.exclude.clone(),
471                pkg_names: pkg_by_dir.clone(),
472            })),
473            "lsp" => Some(Box::new(kuda_extract_lsp::LspExtractor {
474                server_cmd: cfg.extract.lsp_server.clone(),
475            })),
476            "records" => Some(Box::new(kuda_records::RecordsExtractor {
477                excludes: cfg.extract.exclude.clone(),
478                domains: doms.to_vec(),
479            })),
480            other => {
481                eprintln!("extract: unknown kind {other} (adapter crate?)");
482                None
483            }
484        })
485        .collect()
486}

502/// Which domain corpus a file belongs to — declared root prefix +
503/// extension. Returns the descriptor's module name on match.
504fn domain_module<'a>(rel: &Path, domains: &'a [DomainDef]) -> Option<&'a str> {
505    domains
506        .iter()
507        .find(|d| {
508            d.roots.iter().any(|r| rel.starts_with(r))
509                && rel
510                    .extension()
511                    .and_then(|x| x.to_str())
512                    .map(|x| {
513                        d.extensions
514                            .iter()
515                            .any(|de| de.trim_start_matches('.') == x)
516                    })
517                    .unwrap_or(false)
518        })
519        .map(|d| d.module.as_str())
520}

522/// Shared repo walk for `sync` + `status`: collect manifests and
523/// source files (same `dir_excluded` + worktree `.git`-file skip),
524/// group them into modules by manifest dir -> package name, and
525/// blob-hash every file. `pkg_by_dir` only affects module naming —
526/// pass an empty map when just the file set + hashes matter (the
527/// tree digest). `domains` buckets record-corpus files under their
528/// descriptor's module (cards + provenance ride the same pipeline).
529fn scan_repo(
530    dir: &Path,
531    cfg: &Config,
532    manifest_list: &[String],
533    manifest_names: &std::collections::HashSet<String>,
534    src_exts: &std::collections::HashSet<String>,
535    pkg_by_dir: &std::collections::HashMap<PathBuf, String>,
536    domains: &[DomainDef],
537) -> (
538    BTreeMap<String, Vec<PathBuf>>,
539    std::collections::HashMap<PathBuf, String>,
540) {
541    // Declared corpus roots — a `.git`-file worktree marker does not
542    // stop the descent here (myco's project/ is such a worktree).
543    let domain_roots: std::collections::HashSet<PathBuf> = domains
544        .iter()
545        .flat_map(|d| d.roots.iter().cloned())
546        .collect();
547    // Cheap (no parse): drives both the incremental diff and the
548    // per-module file lists in sync, and the tree digest in status.
549    let mut mod_files: BTreeMap<String, Vec<PathBuf>> = BTreeMap::new();
550    let mut stack = vec![dir.to_path_buf()];
551    while let Some(d) = stack.pop() {
552        let Ok(rd) = std::fs::read_dir(&d) else {
553            continue;
554        };
555        for e in rd.flatten() {
556            let p = e.path();
557            if p.is_dir() {
558                let rel_dir = p.strip_prefix(dir).unwrap_or(&p);
559                let worktree = p.join(".git").is_file() && !domain_roots.contains(rel_dir);
560                if !e
561                    .file_name()
562                    .to_str()
563                    .map(|n| kuda_core::dir_excluded(n, &cfg.extract.exclude))
564                    .unwrap_or(true)
565                    && !worktree
566                {
567                    stack.push(p);
568                }
569                continue;
570            }
571            let name = e.file_name().to_string_lossy().into_owned();
572            let rel = p.strip_prefix(dir).unwrap_or(&p).to_path_buf();
573            if let Some(m) = domain_module(&rel, domains) {
574                mod_files.entry(m.to_owned()).or_default().push(rel);
575                continue;
576            }
577            let is_manifest = manifest_names.contains(name.as_str());
578            let is_src = p
579                .extension()
580                .and_then(|x| x.to_str())
581                .map(|x| src_exts.contains(x))
582                .unwrap_or(false);
583            if !is_manifest && !is_src {
584                continue;
585            }
586            let module = kuda_extract_treesitter::manifest_dir_of(dir, &p, manifest_list)
587                .map(|d| {
588                    let rel_dir = d.strip_prefix(dir).unwrap_or(&d).to_path_buf();
589                    pkg_by_dir.get(&rel_dir).cloned().unwrap_or_else(|| {
590                        d.file_name()
591                            .map(|f| f.to_string_lossy().into_owned())
592                            .unwrap_or_else(|| "_root".into())
593                    })
594                })
595                .unwrap_or_else(|| "_root".into());
596            mod_files.entry(module).or_default().push(rel);
597        }
598    }
599    // Non-tree domain sources contribute pseudo-inputs before the
600    // sort: each source "file" (a `refs/sturn/*` ref for sturn-wal)
601    // rides the module set with its head sha as the content hash —
602    // a moved or deleted ref diffs exactly like a file write, and
603    // the module survives the vanished-modules sweep while refs
604    // exist. No refs = no inputs = silent skip.
605    let mut ref_hashes: Vec<(PathBuf, String)> = Vec::new();
606    for d in domains {
607        for (f, h) in kuda_records::source_inputs(dir, d) {
608            ref_hashes.push((f.clone(), h));
609            mod_files.entry(d.module.clone()).or_default().push(f);
610        }
611    }
612    for v in mod_files.values_mut() {
613        v.sort();
614    }
615    // in-process blob hashes for every walked file; ref inputs
616    // carry their own hashes (a rev-parse, not a blob read).
617    let mut filehash: std::collections::HashMap<PathBuf, String> = mod_files
618        .values()
619        .flatten()
620        .map(|f| (f.clone(), blob_hash(&dir.join(f))))
621        .collect();
622    filehash.extend(ref_hashes);
623    (mod_files, filehash)
624}
625
626fn sync_one(
627    dir: &Path,
628    cfg: &Config,
629    _cli: &Cli,
630    full: bool,
631    union: &std::collections::HashSet<String>,
632) -> i32 {
633    let langs = langs();
634    let doms = resolve_domains(cfg);
635    let (manifest_list, manifest_names, src_exts) = walk_sets(&langs);
636
637    // --- manifest pass first (cheap): package names decide module
638    // identity — walk groups by manifest dir -> pkg name, matching
639    // the Module symbols the manifest extractor emits.
640    let manifest_ext = kuda_extract_manifest::ManifestExtractor {
641        excludes: cfg.extract.exclude.clone(),
642        resolve_git_deps: cfg.workspace.resolve_git_deps,
643    }
644    .extract(dir, &langs)
645    .unwrap_or_default();
646    // manifest parent dir (repo-relative) -> package name
647    let mut pkg_by_dir: std::collections::HashMap<PathBuf, String> =
648        std::collections::HashMap::new();
649    for s in &manifest_ext.symbols {
650        if let Some(d) = s.file.parent() {
651            pkg_by_dir.insert(d.to_path_buf(), s.module.clone());
652        }
653    }
654
655    // --- walk once: group manifests + source files by module ---
656    let (mod_files, filehash) = scan_repo(
657        dir,
658        cfg,
659        &manifest_list,
660        &manifest_names,
661        &src_exts,
662        &pkg_by_dir,
663        &doms,
664    );
665
666    let cards_dir = dir.join(&cfg.cards.dir);
667    if let Err(e) = std::fs::create_dir_all(cards_dir.join("modules")) {
668        eprintln!("mkdir {}: {e}", cards_dir.display());
669        return 1;
670    }
671    let mut store = match kuda_store_sqlite::SqliteStore::open(&dir.join(&cfg.store.path)) {
672        Ok(s) => s,
673        Err(e) => {
674            eprintln!("store {}: {e}", cfg.store.path.display());
675            return 1;
676        }
677    };
678    // upgrade path: no file-hash rows yet -> seed them from the
679    // committed cards' provenance so nothing re-parses (or
680    // re-prompts an agent) on the first post-upgrade sync.
681    // `sync --full` skips all of it: an empty diff baseline marks
682    // every module changed.
683    let mut old_hashes = if full {
684        std::collections::HashMap::new()
685    } else {
686        store.file_hashes().unwrap_or_default()
687    };
688    if old_hashes.is_empty() && !full {
689        // read_dir can't fail: modules/ was just create_dir_all'd —
690        // flattened instead of `if let` so no dead error arm exists.
691        for e in std::fs::read_dir(cards_dir.join("modules"))
692            .into_iter()
693            .flatten()
694            .flatten()
695        {
696            let Ok(text) = std::fs::read_to_string(e.path()) else {
697                continue;
698            };
699            let Ok(card) = kuda_cards::parse_module_card(&text) else {
700                continue;
701            };
702            let rows: Vec<(String, String)> = card
703                .provenance
704                .files
705                .iter()
706                .zip(&card.provenance.hashes)
707                .map(|(f, h)| (f.display().to_string(), h.clone()))
708                .collect();
709            let _ = store.record_files(&card.name, &rows);
710        }
711        old_hashes = store.file_hashes().unwrap_or_default();
712    }
713
714    // diff: which modules changed?
715    let mut changed: std::collections::HashSet<String> = std::collections::HashSet::new();
716    for (module, files) in &mod_files {
717        for f in files {
718            match old_hashes.get(&f.display().to_string()) {
719                Some((h, _)) if h == &filehash[f] => {}
720                _ => {
721                    changed.insert(module.clone());
722                    break;
723                }
724            }
725        }
726        // no card on disk -> force (store may outlive .kuda/modules)
727        let card_path = cards_dir
728            .join("modules")
729            .join(format!("{}.yaml", safe(module)));
730        if !card_path.exists() {
731            changed.insert(module.clone());
732        }
733    }
734    for (path, (_, module)) in &old_hashes {
735        if !filehash.contains_key(&PathBuf::from(path)) {
736            changed.insert(module.clone()); // deleted file
737        }
738    }
739    // sturn WAL ops feed the diff *additively* — non-sturn
740    // writes still surface via the hash walk; sturn-managed
741    // writes (write/rename/delete/claim/release) mark their
742    // modules changed even when content happened to match.
743    let wal_touched = sturn_wal_changes(dir, &mut store);
744    if !wal_touched.is_empty() {
745        eprintln!(
746            "sturn wal: {}",
747            wal_touched.iter().cloned().collect::<Vec<_>>().join(", ")
748        );
749    }
750    for t in wal_touched {
751        for (module, mfiles) in &mod_files {
752            let hit = mfiles.iter().any(|f| {
753                let fs = f.display().to_string();
754                fs == t || fs.starts_with(&format!("{t}/"))
755            });
756            if hit {
757                changed.insert(module.clone());
758            }
759        }
760    }
761    // summarizer adapter: acp spawns an agent; anything else is template
762    let acp = acp_summarizer(cfg, dir);
763    let use_acp = acp.is_some();
764    // Provenance names the adapter that writes the prose — an
765    // acp card stamps `kuda-acp`, never `kuda-template`. The
766    // same stamp drives mechanism-version invalidation: a card
767    // written by a different generator (older mechanism, or a
768    // different summarizer) is re-summarized even on
769    // byte-identical inputs.
770    // 0.1.3: acp replies that fail card-prose parsing (monologue,
771    // embedded-field blobs) now fall back to template instead of
772    // writing garbage into `purpose` — restamp so degraded cards
773    // re-summarize on next sync (I-002).
774    // 0.1.4: empty agent turns (end_turn with zero text — cline on
775    // a dead model id) are rejected instead of silently stamping
776    // `purpose: <module name>`; fallback cards carry an `acp:`
777    // warnings entry explaining the degradation (I-002).
778    const GENERATOR_VERSION: &str = "0.1.4";
779    let stamp = |kind: &str| format!("kuda-{kind}/{GENERATOR_VERSION}");
780    // manifest-only pseudo-modules (root Cargo.toml, docs/
781    // package.json) have no source to ground on — the agent
782    // would return empty prose. Skip the prompt entirely.
783    let manifest_only = |mfiles: &[PathBuf]| {
784        !mfiles.is_empty()
785            && mfiles.iter().all(|f| {
786                f.file_name()
787                    .and_then(|n| n.to_str())
788                    .map(|n| manifest_names.contains(n))
789                    .unwrap_or(false)
790            })
791    };
792    // Generator staleness must join `changed` *before* the parse
793    // filter: a fresh-but-stale module still needs its files
794    // re-extracted, else card and store rows would rebuild with
795    // zero symbols (treesitter honors parse_set).
796    for (module, mfiles) in &mod_files {
797        if changed.contains(module) {
798            continue;
799        }
800        let generator = stamp(if use_acp && !manifest_only(mfiles) {
801            "acp"
802        } else {
803            "template"
804        });
805        let card_path = cards_dir
806            .join("modules")
807            .join(format!("{}.yaml", safe(module)));
808        let stale_generator = std::fs::read_to_string(&card_path)
809            .ok()
810            .and_then(|t| kuda_cards::parse_module_card(&t).ok())
811            .map(|old| old.provenance.generator != generator)
812            .unwrap_or(true);
813        if stale_generator {
814            changed.insert(module.clone());
815        }
816    }
817    let parse_set: std::collections::HashSet<PathBuf> = changed
818        .iter()
819        .flat_map(|m| mod_files.get(m).cloned().unwrap_or_default())
820        .collect();
821
822    let extractors = extractors(&cfg.extract.kinds, cfg, &doms, &pkg_by_dir);
823    let mut ext = kuda_extract::Extraction::default();
824    for x in &extractors {
825        // manifest is cheap (few files) — full pass keeps module
826        // symbols + edges complete; treesitter honors the filter.
827        let filtered = x.kind() == "treesitter" && !parse_set.is_empty();
828        let res = if filtered {
829            x.extract_filtered(dir, &langs, &parse_set)
830        } else {
831            x.extract(dir, &langs)
832        };
833        #[cfg(not(coverage))]
834        match res {
835            Ok(e) => {
836                ext.symbols.extend(e.symbols);
837                ext.edges.extend(e.edges);
838                ext.files.extend(e.files);
839                ext.warnings.extend(e.warnings);
840            }
841            // bundled extractors are infallible by construction
842            Err(e) => eprintln!("extract {}: {e}", x.kind()),
843        }
844        #[cfg(coverage)]
845        if let Ok(e) = res {
846            ext.symbols.extend(e.symbols);
847            ext.edges.extend(e.edges);
848            ext.files.extend(e.files);
849            ext.warnings.extend(e.warnings);
850        }
851    }
852    // resolve raw dep/import specifiers to indexed module names —
853    // `import neuralmind_shared` lands an edge on `neuralmind-shared`;
854    // unmatched names (serde, os) stay external and queryable. The
855    // workspace union lets sibling-repo modules resolve cross-root.
856    let module_set: Vec<String> = mod_files
857        .keys()
858        .cloned()
859        .chain(union.iter().cloned())
860        .collect();
861    {
862        let mut seen = std::collections::HashSet::new();
863        for e in &mut ext.edges {
864            if let Some(m) = kuda_core::resolve_module(&e.to, &module_set) {
865                e.to = m;
866            }
867        }
868        ext.edges.retain(|e| {
869            // self-imports (`import my_pkg` inside my-pkg's own
870            // sources) resolve to the module itself — a self-edge
871            // answers nothing in `dependents`.
872            e.from != e.to && seen.insert((e.from.clone(), e.to.clone(), e.kind))
873        });
874    }
875    let sha = lasius_vcs_git::GitVcs
876        .head(dir)
877        .unwrap_or_else(|| "unknown".into());
878    let now = unix_now();
879
880    // group symbols + edges by module (filtered pass: changed only)
881    let mut by_mod: BTreeMap<String, Vec<kuda_core::Symbol>> = BTreeMap::new();
882    // symbol name -> module, so sub-module edges (record -> record)
883    // still store under the module that owns their source symbol.
884    let mut sym_module: std::collections::HashMap<String, String> =
885        std::collections::HashMap::new();
886    for s in &ext.symbols {
887        sym_module.insert(s.name.clone(), s.module.clone());
888        by_mod.entry(s.module.clone()).or_default().push(s.clone());
889    }
890
891    let summ = kuda_template::TemplateSummarizer;
892    let mut module_names = Vec::new();
893    // acp replies that fail card-prose parsing (monologue, garbage)
894    // fall back to the template per-module — counted for the
895    // summary line so degraded syncs are visible, not silent.
896    let mut acp_fallbacks = 0usize;
897    // modules that vanished from the walk entirely -> drop card +
898    // store rows
899    let present: std::collections::HashSet<&String> = mod_files.keys().collect();
900    for m in store.modules().unwrap_or_default() {
901        if !present.contains(&m) {
902            let _ = store.replace_module(&m, &[], &[]);
903            let _ = store.record_files(&m, &[]);
904            let stale = cards_dir.join("modules").join(format!("{}.yaml", safe(&m)));
905            let _ = std::fs::remove_file(stale);
906        }
907    }
908    // L1 file-card stale sweep — housekeeping, not summarization, so
909    // it runs whenever .kuda/files exists even with file_cards/acp
910    // off (stale cards are leftovers from a previous config).
911    // Mirrors the module sweep above: drop dirs of vanished
912    // modules, then per live module drop yamls whose slug no
913    // longer maps to a walked file (deleted/renamed sources).
914    let files_dir = cards_dir.join("files");
915    if files_dir.is_dir() {
916        let live: std::collections::HashSet<String> = mod_files.keys().map(|m| safe(m)).collect();
917        // a files/ dir that exists but can't be read just yields no
918        // entries — flattened so no untriggerable error arm exists.
919        for e in std::fs::read_dir(&files_dir)
920            .into_iter()
921            .flatten()
922            .flatten()
923        {
924            let p = e.path();
925            if p.is_dir() && !live.contains(e.file_name().to_string_lossy().as_ref()) {
926                let _ = std::fs::remove_dir_all(&p);
927            }
928        }
929        for (module, mfiles) in &mod_files {
930            let mdir = files_dir.join(safe(module));
931            let Ok(rd) = std::fs::read_dir(&mdir) else {
932                continue;
933            };
934            // expected card names — same slug scheme as
935            // sync_file_cards; manifests never get file cards.
936            let expected: std::collections::HashSet<String> = mfiles
937                .iter()
938                .filter(|f| {
939                    !f.file_name()
940                        .and_then(|n| n.to_str())
941                        .map(|n| manifest_names.contains(n))
942                        .unwrap_or(false)
943                })
944                .map(|f| format!("{}.yaml", f.display().to_string().replace('/', "__")))
945                .collect();
946            for e in rd.flatten() {
947                let name = e.file_name().to_string_lossy().into_owned();
948                if name.ends_with(".yaml") && !expected.contains(&name) {
949                    let _ = std::fs::remove_file(e.path());
950                }
951            }
952        }
953    }
954    for (module, mfiles) in &mod_files {
955        let want_acp = use_acp && !manifest_only(mfiles);
956        let generator = stamp(if want_acp { "acp" } else { "template" });
957        let card_path = cards_dir
958            .join("modules")
959            .join(format!("{}.yaml", safe(module)));
960        if !changed.contains(module) {
961            // store + card already agree with the bytes — generator
962            // staleness was folded into `changed` pre-extraction.
963            module_names.push(module.clone());
964            continue;
965        }
966        let prov = Provenance {
967            hashes: mfiles
968                .iter()
969                .map(|f| filehash.get(f).cloned().unwrap_or_default())
970                .collect(),
971            files: mfiles.clone(),
972            commit_sha: sha.clone(),
973            generated_at: now.clone(),
974            generator: generator.clone(),
975        };
976        let empty: Vec<kuda_core::Symbol> = Vec::new();
977        let syms: &[kuda_core::Symbol] = by_mod.get(module).map(|v| v.as_slice()).unwrap_or(&empty);
978        let mod_edges: Vec<_> = ext
979            .edges
980            .iter()
981            .filter(|e| {
982                e.from == *module
983                    || sym_module
984                        .get(&e.from)
985                        .map(|m| m == module)
986                        .unwrap_or(false)
987            })
988            .cloned()
989            .collect();
990        // L1 file cards — per-source-file chunked summaries.
991        // Per-file hash-skip makes the steady state free; only
992        // changed files spend an agent prompt.
993        if use_acp && cfg.summarize.file_cards {
994            if let Some(acp) = acp.as_ref() {
995                sync_file_cards(FileCardJob {
996                    dir,
997                    module,
998                    syms,
999                    mfiles,
1000                    hashes: &prov.hashes,
1001                    manifest_names: &manifest_names,
1002                    acp,
1003                    files_dir: &cards_dir.join("files"),
1004                    sha: &sha,
1005                    now: &now,
1006                    full,
1007                });
1008            }
1009        }
1010        // module `path` for the card: its manifest if present,
1011        // else first input file
1012        let path = mfiles
1013            .iter()
1014            .find(|f| {
1015                f.file_name()
1016                    .and_then(|n| n.to_str())
1017                    .map(|n| manifest_names.contains(n))
1018                    .unwrap_or(false)
1019            })
1020            .cloned()
1021            .or_else(|| mfiles.first().cloned())
1022            .unwrap_or_else(|| PathBuf::from(module));
1023        // set when the acp reply failed and the template wrote the
1024        // prose — lands on the committed card's `warnings:` so the
1025        // degradation is visible in the artifact, not just stderr.
1026        let mut acp_warn: Option<String> = None;
1027        let card_res = if want_acp {
1028            acp.as_ref()
1029                .unwrap()
1030                .summarize(
1031                    module,
1032                    &path,
1033                    &kuda_extract::Extraction {
1034                        symbols: syms.to_vec(),
1035                        edges: mod_edges.clone(),
1036                        files: mfiles.clone(),
1037                        warnings: Vec::new(),
1038                    },
1039                    prov.clone(),
1040                )
1041                .or_else(|e| {
1042                    acp_fallbacks += 1;
1043                    eprintln!("acp {module}: {e} — template fallback");
1044                    acp_warn = Some(format!("acp: {e} — template fallback"));
1045                    // the fallback wrote the prose — stamp it
1046                    // `kuda-template` so the mismatch re-prompts
1047                    // the agent next sync.
1048                    let mut p = prov.clone();
1049                    p.generator = stamp("template");
1050                    summ.summarize(
1051                        module,
1052                        &path,
1053                        &kuda_extract::Extraction {
1054                            symbols: syms.to_vec(),
1055                            edges: mod_edges.clone(),
1056                            files: mfiles.clone(),
1057                            warnings: Vec::new(),
1058                        },
1059                        p,
1060                    )
1061                })
1062        } else {
1063            summ.summarize(
1064                module,
1065                &path,
1066                &kuda_extract::Extraction {
1067                    symbols: syms.to_vec(),
1068                    edges: mod_edges.clone(),
1069                    files: mfiles.clone(),
1070                    warnings: Vec::new(),
1071                },
1072                prov.clone(),
1073            )
1074        };
1075        #[cfg(not(coverage))]
1076        let mut card = match card_res {
1077            Ok(c) => c,
1078            Err(e) => {
1079                eprintln!("summarize {module}: {e}");
1080                continue;
1081            }
1082        };
1083        // acp errors already fell back to the infallible template
1084        #[cfg(coverage)]
1085        let mut card = card_res.unwrap();
1086        // extraction findings on this module's files land on the
1087        // card — `card`/`status`/MCP surface them, sync prints them
1088        for w in &ext.warnings {
1089            if mfiles.contains(&w.file) {
1090                eprintln!("warn {}: {}", w.file.display(), w.message);
1091            }
1092        }
1093        card.warnings = ext
1094            .warnings
1095            .iter()
1096            .filter(|w| mfiles.contains(&w.file))
1097            .map(|w| format!("{}: {}", w.file.display(), w.message))
1098            .collect();
1099        // the acp failure that produced this card rides the card —
1100        // `warnings:` records *why* the prose is template-stamped
1101        if let Some(w) = acp_warn {
1102            card.warnings.push(w);
1103        }
1104        sanitize_card(&mut card, &manifest_names);
1105        // last-resort hygiene on agent-written purposes: the acp
1106        // summarizer already rejects monologue-shaped prose, but
1107        // nothing that smells like agent reasoning may be committed
1108        // under a kuda-acp stamp — reset to the module name so the
1109        // fallback chain below re-grounds it, and say so on the
1110        // card. Template/repo-authored purposes are trusted as-is —
1111        // a README may legitimately start "First,".
1112        if card.provenance.generator.starts_with("kuda-acp")
1113            && card.purpose != card.name
1114            && kuda_acp::looks_like_monologue(&card.purpose)
1115        {
1116            card.warnings
1117                .push("acp: purpose looked like agent reasoning — reset".into());
1118            card.purpose = card.name.clone();
1119        }
1120        // purpose: domain descriptor -> manifest description ->
1121        // crate docs -> README. Domain modules have no manifest —
1122        // skipping straight to the chain would steal the repo
1123        // README's first line as their purpose.
1124        if card.purpose == card.name {
1125            if let Some(p) = doms
1126                .iter()
1127                .find(|d| d.module == *module)
1128                .and_then(|d| d.purpose.as_ref())
1129            {
1130                card.purpose = p.clone();
1131            } else if let Some(d) = module_purpose(dir, &dir.join(&path)) {
1132                card.purpose = d;
1133            }
1134        }
1135        // `notes:` is the human-owned region (T-008): the committed
1136        // card's notes carry forward verbatim — regen rewrites the
1137        // machine fields only, never clobbers human prose.
1138        if let Ok(prev) = std::fs::read_to_string(&card_path)
1139            .map_err(|e| e.to_string())
1140            .and_then(|t| kuda_cards::parse_module_card(&t))
1141        {
1142            kuda_cards::carry_notes(&mut card, &prev);
1143        }
1144        let out = cards_dir
1145            .join("modules")
1146            .join(format!("{}.yaml", safe(&card.name)));
1147        #[cfg(not(coverage))]
1148        match kuda_cards::render_module_card(&card) {
1149            Ok(y) => {
1150                if let Err(e) = std::fs::write(&out, y) {
1151                    eprintln!("write {}: {e}", out.display());
1152                }
1153            }
1154            // serde_yaml can't fail on plain card data
1155            Err(e) => eprintln!("render {module}: {e}"),
1156        }
1157        #[cfg(coverage)]
1158        if let Ok(y) = kuda_cards::render_module_card(&card) {
1159            let _ = std::fs::write(&out, y);
1160        }
1161        #[cfg(not(coverage))]
1162        if let Err(e) = store.replace_module(module, syms, &mod_edges) {
1163            eprintln!("store {module}: {e}");
1164        }
1165        #[cfg(coverage)]
1166        store.replace_module(module, syms, &mod_edges).unwrap();
1167        let rows: Vec<(String, String)> = mfiles
1168            .iter()
1169            .zip(&prov.hashes)
1170            .map(|(f, h)| (f.display().to_string(), h.clone()))
1171            .collect();
1172        #[cfg(not(coverage))]
1173        if let Err(e) = store.record_files(module, &rows) {
1174            eprintln!("record_files {module}: {e}");
1175        }
1176        #[cfg(coverage)]
1177        store.record_files(module, &rows).unwrap();
1178        module_names.push(card.name);
1179    }
1180    module_names.sort();
1181    if acp_fallbacks > 0 {
1182        eprintln!("acp: {acp_fallbacks} card(s) fell back to template — unparseable agent replies");
1183    }
1184
1185    // L0 repo card
1186    let name = cfg
1187        .repo
1188        .name
1189        .clone()
1190        .or_else(|| dir.file_name().map(|f| f.to_string_lossy().into_owned()))
1191        .unwrap_or_else(|| "repo".into());
1192    // L0 provenance covers only the card's real inputs — manifests
1193    // (module identity) and the README (purpose). The full walked set
1194    // is content-addressed by `tree_digest`; embedding ~10k path:hash
1195    // pairs made index.yaml an always-AVOID blob on real repos.
1196    let prov_files: Vec<PathBuf> = {
1197        let mut v: Vec<PathBuf> = ext
1198            .files
1199            .iter()
1200            .filter(|f| {
1201                f.file_name()
1202                    .and_then(|n| n.to_str())
1203                    .map(|n| manifest_names.contains(n))
1204                    .unwrap_or(false)
1205            })
1206            .cloned()
1207            .collect();
1208        let readme = PathBuf::from("README.md");
1209        if dir.join(&readme).exists() && !v.contains(&readme) {
1210            v.push(readme);
1211        }
1212        v.sort();
1213        v
1214    };
1215    let prov = Provenance {
1216        files: prov_files.clone(),
1217        hashes: hash_files(dir, &prov_files),
1218        commit_sha: sha.clone(),
1219        generated_at: now.clone(),
1220        generator: "kuda/0.1.0".into(),
1221    };
1222    let repo_card = RepoCard {
1223        name,
1224        root: dir.to_path_buf(),
1225        purpose: readme_purpose(&dir.join("README.md")).unwrap_or_default(),
1226        modules: module_names.clone(),
1227        commands: detect_commands(dir, &manifest_ext.files),
1228        tree_digest: tree_digest(&filehash),
1229        files_total: filehash.len(),
1230        provenance: prov,
1231    };
1232    #[cfg(not(coverage))]
1233    match kuda_cards::render_repo_card(&repo_card) {
1234        Ok(y) => {
1235            if let Err(e) = std::fs::write(cards_dir.join("index.yaml"), y) {
1236                eprintln!("write index.yaml: {e}");
1237            }
1238        }
1239        // serde_yaml can't fail on plain card data
1240        Err(e) => eprintln!("render index: {e}"),
1241    }
1242    #[cfg(coverage)]
1243    if let Ok(y) = kuda_cards::render_repo_card(&repo_card) {
1244        let _ = std::fs::write(cards_dir.join("index.yaml"), y);
1245    }
1246    let _ = store.set_meta("commit_sha", &sha);
1247    let _ = store.set_meta("generated_at", &now);
1248    let _ = std::fs::write(
1249        cards_dir.join("meta.json"),
1250        format!("{{\"commit_sha\":\"{sha}\",\"generated_at\":{now}}}\n"),
1251    );
1252    outln!("indexed {} module(s) at {sha:.7}", module_names.len());
1253    0
1254}

1271/// L1 file-card sync — one card per source file. Chunk with text-splitter
1272/// (code-aware via tree-sitter where a grammar exists), prompt the
1273/// shared ACP session, write `.kuda/files/<module>/<slug>.yaml`.
1274/// Per-file provenance hash gates: unchanged files cost nothing.
1275fn sync_file_cards(job: FileCardJob<'_>) {
1276    let FileCardJob {
1277        dir,
1278        module,
1279        syms,
1280        mfiles,
1281        hashes,
1282        manifest_names,
1283        acp,
1284        files_dir,
1285        sha,
1286        now,
1287        full,
1288    } = job;
1289    use text_splitter::{ChunkConfig, CodeSplitter, MarkdownSplitter, TextSplitter};
1290    const CHUNK: usize = 1_500; // chars per chunk
1291    const BUDGET: usize = 14_000; // total source chars per prompt
1292    const SKIP_BYTES: u64 = 200_000; // files bigger than this: not prose-material
1293                                     // provenance stamp + mechanism gate — cards from older writers
1294                                     // (pre empty-reply rejection) re-prompt even on unchanged bytes
1295    const FILE_CARD_GENERATOR: &str = "kuda-acp/0.1.1";
1296
1297    // Slug namespace is lossy: `a/b.py` and `a__b.py` map to the
1298    // same card name. mfiles is sorted, so the first file wins
1299    // deterministically; the loser is skipped loudly instead of
1300    // silently overwriting the other's card.
1301    let mut emitted: std::collections::HashSet<String> = std::collections::HashSet::new();
1302    for (rel, hash) in mfiles.iter().zip(hashes) {
1303        if rel
1304            .file_name()
1305            .and_then(|n| n.to_str())
1306            .map(|n| manifest_names.contains(n))
1307            .unwrap_or(false)
1308        {
1309            continue; // manifests are card inputs, not file cards
1310        }
1311        let slug = rel.display().to_string().replace('/', "__");
1312        if !emitted.insert(slug.clone()) {
1313            eprintln!(
1314                "file card: slug collision {slug} — skipping {}",
1315                rel.display()
1316            );
1317            continue;
1318        }
1319        let out = files_dir.join(safe(module)).join(format!("{slug}.yaml"));
1320        if out.exists() && !full {
1321            if let Ok(old) = std::fs::read_to_string(&out)
1322                .map_err(|e| e.to_string())
1323                .and_then(|t| kuda_cards::parse_file_card(&t))
1324            {
1325                // hash gate + generator gate: cards written by an
1326                // older mechanism (e.g. before empty-reply
1327                // rejection, purpose = bare filename) re-prompt
1328                // even on unchanged bytes.
1329                if old.provenance.hashes.first() == Some(hash)
1330                    && old.provenance.generator == FILE_CARD_GENERATOR
1331                {
1332                    continue;
1333                }
1334            }
1335        }
1336        let abs = dir.join(rel);
1337        let Ok(text) = std::fs::read_to_string(&abs) else {
1338            continue;
1339        };
1340        if abs.metadata().map(|m| m.len()).unwrap_or(0) > SKIP_BYTES {
1341            continue;
1342        }
1343        let cfg = ChunkConfig::new(CHUNK);
1344        let chunks: Vec<String> = match rel.extension().and_then(|e| e.to_str()) {
1345            Some("rs") => CodeSplitter::new(tree_sitter_rust::LANGUAGE, cfg)
1346                .map(|s| s.chunk_indices(&text).map(|(_, c)| c.to_string()).collect())
1347                .unwrap_or_else(|_| vec![text.clone()]),
1348            Some("ts") | Some("tsx") => {
1349                CodeSplitter::new(tree_sitter_typescript::LANGUAGE_TYPESCRIPT, cfg)
1350                    .map(|s| s.chunk_indices(&text).map(|(_, c)| c.to_string()).collect())
1351                    .unwrap_or_else(|_| vec![text.clone()])
1352            }
1353            Some("py") => CodeSplitter::new(tree_sitter_python::LANGUAGE, cfg)
1354                .map(|s| s.chunk_indices(&text).map(|(_, c)| c.to_string()).collect())
1355                .unwrap_or_else(|_| vec![text.clone()]),
1356            #[cfg(not(coverage))]
1357            Some("go") => CodeSplitter::new(tree_sitter_go::LANGUAGE, cfg)
1358                .map(|s| s.chunk_indices(&text).map(|(_, c)| c.to_string()).collect())
1359                .unwrap_or_else(|_| vec![text.clone()]),
1360            // walked extensions are only rs/ts/tsx/py/go — these arms
1361            // exist for future data/langs entries
1362            #[cfg(not(coverage))]
1363            Some("md") | Some("mdx") => MarkdownSplitter::new(cfg)
1364                .chunk_indices(&text)
1365                .map(|(_, c)| c.to_string())
1366                .collect(),
1367            #[cfg(not(coverage))]
1368            _ => TextSplitter::new(cfg)
1369                .chunk_indices(&text)
1370                .map(|(_, c)| c.to_string())
1371                .collect(),
1372            #[cfg(coverage)]
1373            _ => CodeSplitter::new(tree_sitter_go::LANGUAGE, cfg)
1374                .map(|s| s.chunk_indices(&text).map(|(_, c)| c.to_string()).collect())
1375                .unwrap_or_else(|_| vec![text.clone()]),
1376        };
1377        // budget the prompt: take chunks until the char cap
1378        let mut body = String::new();
1379        let mut used = 0usize;
1380        for c in &chunks {
1381            if body.len() + c.len() > BUDGET {
1382                break;
1383            }
1384            used += 1;
1385            body.push_str("\n---\n");
1386            body.push_str(c);
1387        }
1388        let sym_names: Vec<String> = syms
1389            .iter()
1390            .filter(|s| &s.file == rel)
1391            .map(|s| s.name.clone())
1392            .take(30)
1393            .collect();
1394        let prompt = format!(
1395            "Summarize this source file for the repo memory index.\n\
1396             File: {}\nModule: {module}\nSymbols defined: {}\n\n\
1397             Contents ({used} of {} semantic chunks{}):\n{body}\n\n\
1398             Reply with ONLY `purpose:` followed by 1-3 grounded \
1399             sentences — this text is re-verified against the file.",
1400            rel.display(),
1401            sym_names.join(", "),
1402            chunks.len(),
1403            if used < chunks.len() {
1404                ", truncated"
1405            } else {
1406                ""
1407            },
1408        );
1409        let prose = match acp.complete(&prompt) {
1410            Ok(p) => p,
1411            Err(e) => {
1412                eprintln!("acp file {}: {e}", rel.display());
1413                continue;
1414            }
1415        };
1416        // an empty turn — `end_turn` with zero message text (cline
1417        // on a dead model id) — is a failed prompt: skip the card
1418        // so next sync retries rather than stamping purpose = the
1419        // bare filename.
1420        if prose.trim().is_empty() {
1421            eprintln!("acp file {}: empty reply — card skipped", rel.display());
1422            continue;
1423        }
1424        // purpose = the reply's `purpose:` line if present — a
1425        // reasoning preamble can precede it — else the first
1426        // non-empty line
1427        let purpose = prose
1428            .lines()
1429            .rev()
1430            .find_map(|l| {
1431                let t = l.trim();
1432                t.strip_prefix("purpose:")
1433                    .or_else(|| t.strip_prefix("Purpose:"))
1434                    .map(|p| p.trim().to_owned())
1435                    .filter(|p| !p.is_empty())
1436            })
1437            .or_else(|| {
1438                prose
1439                    .lines()
1440                    .map(str::trim)
1441                    .find(|t| !t.is_empty())
1442                    .map(|t| t.to_owned())
1443            })
1444            .unwrap_or_else(|| rel.display().to_string());
1445        // same hygiene as module cards: a monologue/garbage reply
1446        // must not land in `purpose` — skip the card, retry next
1447        // sync (no card written -> hash gate re-prompts).
1448        if kuda_acp::looks_like_monologue(&purpose) {
1449            eprintln!(
1450                "acp file {}: unparseable reply (monologue) — card skipped",
1451                rel.display()
1452            );
1453            continue;
1454        }
1455        let card = kuda_core::FileCard {
1456            path: rel.clone(),
1457            module: module.to_string(),
1458            purpose,
1459            symbols: sym_names,
1460            provenance: Provenance {
1461                files: vec![rel.clone()],
1462                hashes: vec![hash.clone()],
1463                commit_sha: sha.to_string(),
1464                generated_at: now.to_string(),
1465                generator: FILE_CARD_GENERATOR.into(),
1466            },
1467        };
1468        // serde_yaml can't fail on plain card data
1469        #[cfg(not(coverage))]
1470        let Ok(y) = kuda_cards::render_file_card(&card) else {
1471            continue;
1472        };
1473        #[cfg(coverage)]
1474        let y = kuda_cards::render_file_card(&card).unwrap();
1475        if let Some(parent) = out.parent() {
1476            let _ = std::fs::create_dir_all(parent);
1477        }
1478        #[cfg(not(coverage))]
1479        if let Err(e) = std::fs::write(&out, y) {
1480            eprintln!("write {}: {e}", out.display());
1481        }
1482        #[cfg(coverage)]
1483        let _ = std::fs::write(&out, y);
1484    }
1485}

1497/// First `//!` doc line — the crate's self-description.
1498fn crate_doc_purpose(lib: &Path) -> Option<String> {

1500    for line in text.lines() {

1503            let d = d.trim();
1504            if !d.is_empty() {
1505                return Some(d.to_owned());

1507        } else if !l.is_empty() && !l.starts_with("//") {
1508            return None; // first real code line — no doc comment
1509        }
1510    }

1554/// sha1 over sorted "path:hash" lines — content-addresses the
1555/// walked file set without embedding the list in the L0 card.
1556fn tree_digest(filehash: &std::collections::HashMap<PathBuf, String>) -> String {
1557    use sha1::Digest;
1558    let mut lines: Vec<String> = filehash
1559        .iter()
1560        .map(|(p, h)| format!("{}:{h}", p.display()))
1561        .collect();
1562    lines.sort();
1563    let mut h = sha1::Sha1::new();
1564    for l in &lines {
1565        h.update(l.as_bytes());
1566        h.update(b"\n");
1567    }
1568    h.finalize().iter().map(|b| format!("{b:02x}")).collect()
1569}

1571/// Bounded cards, whatever the summarizer produced: drop
1572/// private-by-convention symbols, dedupe, cap `api` (remainder
1573/// counted in `api_omitted`) and `verify` to entry points.
1574fn sanitize_card(
1575    card: &mut kuda_core::ModuleCard,
1576    manifest_names: &std::collections::HashSet<String>,
1577) {
1578    const API_CAP: usize = 40;
1579    const VERIFY_CAP: usize = 10;
1580    card.api.retain(|s| !s.name.starts_with('_'));
1581    card.api.sort_by(|a, b| {
1582        a.name
1583            .cmp(&b.name)
1584            .then(a.file.cmp(&b.file))
1585            .then(a.line.cmp(&b.line))
1586    });
1587    card.api
1588        .dedup_by(|a, b| a.name == b.name && a.file == b.file && a.line == b.line);
1589    if card.api.len() > API_CAP {
1590        card.api_omitted = card.api.len() - API_CAP;
1591        card.api.truncate(API_CAP);
1592    }
1593    if card.verify.len() > VERIFY_CAP {
1594        let entry = |f: &PathBuf| {
1595            f.file_name()
1596                .and_then(|n| n.to_str())
1597                .map(|n| {
1598                    manifest_names.contains(n)
1599                        || matches!(
1600                            n,
1601                            "lib.rs"
1602                                | "main.rs"
1603                                | "mod.rs"
1604                                | "__init__.py"
1605                                | "__main__.py"
1606                                | "index.ts"
1607                                | "index.tsx"
1608                                | "index.js"
1609                                | "main.go"
1610                                | "go.mod"
1611                        )
1612                })
1613                .unwrap_or(false)
1614        };
1615        card.verify.sort_by_key(|f| !entry(f));
1616        card.verify.truncate(VERIFY_CAP);
1617    }
1618}

1620/// Build/test entry points for the L0 card — advisory hints, not a
1621/// contract. Sources: Makefile targets (root + package dirs),
1622/// justfile/Taskfile, package.json scripts, ecosystem manifests.
1623fn detect_commands(dir: &Path, manifests: &[PathBuf]) -> Vec<String> {
1624    fn fname(m: &Path) -> &str {
1625        m.file_name().and_then(|n| n.to_str()).unwrap_or("")
1626    }
1627    let has = |name: &str| manifests.iter().any(|m| fname(m) == name);
1628    let mut cmds: Vec<String> = Vec::new();
1629
1630    // Makefile targets — root first, then package dirs (make -C).
1631    let mut dirs: Vec<PathBuf> = vec![PathBuf::new()];
1632    dirs.extend(
1633        manifests
1634            .iter()
1635            .filter_map(|m| m.parent().map(|p| p.to_path_buf())),
1636    );
1637    let mut seen_dirs = std::collections::HashSet::new();
1638    for rel in dirs {
1639        if !seen_dirs.insert(rel.clone()) {
1640            continue;
1641        }
1642        let Ok(text) = std::fs::read_to_string(dir.join(&rel).join("Makefile")) else {
1643            continue;
1644        };
1645        for t in makefile_targets(&text).into_iter().take(3) {
1646            cmds.push(if rel.as_os_str().is_empty() {
1647                format!("make {t}")
1648            } else {
1649                format!("make -C {} {t}", rel.display())
1650            });
1651        }
1652    }
1653    if dir.join("justfile").exists() || dir.join("Justfile").exists() {
1654        cmds.push("just --list".into());
1655    }
1656    if dir.join("Taskfile.yml").exists() || dir.join("Taskfile.yaml").exists() {
1657        cmds.push("task --list".into());
1658    }
1659    if has("Cargo.toml") {
1660        cmds.extend(["cargo check", "cargo test --workspace"].map(str::to_owned));
1661    }
1662    if has("go.mod") {
1663        cmds.push("go test ./...".into());
1664    }
1665    for m in manifests
1666        .iter()
1667        .filter(|m| fname(m) == "package.json")
1668        .take(2)
1669    {
1670        let Ok(text) = std::fs::read_to_string(dir.join(m)) else {
1671            continue;
1672        };
1673        let Ok(v) = serde_json::from_str::<serde_json::Value>(&text) else {
1674            continue;
1675        };
1676        let Some(scripts) = v.get("scripts").and_then(|s| s.as_object()) else {
1677            continue;
1678        };
1679        // prefer the verbs agents actually need, then fill to 3
1680        let mut keys: Vec<&str> = [
1681            "test",
1682            "build",
1683            "dev",
1684            "lint",
1685            "start",
1686            "typecheck",
1687            "check",
1688        ]
1689        .into_iter()
1690        .filter(|k| scripts.contains_key(*k))
1691        .collect();
1692        for k in scripts.keys().map(|k| k.as_str()) {
1693            if !keys.contains(&k) {
1694                keys.push(k);
1695            }
1696        }
1697        let rel = m.parent().map(|p| p.to_path_buf()).unwrap_or_default();
1698        for k in keys.into_iter().take(3) {
1699            cmds.push(if rel.as_os_str().is_empty() {
1700                format!("npm run {k}")
1701            } else {
1702                format!("(cd {} && npm run {k})", rel.display())
1703            });
1704        }
1705    }
1706    if has("pyproject.toml")
1707        && manifests
1708            .iter()
1709            .filter(|m| fname(m) == "pyproject.toml")
1710            .any(|m| {
1711                dir.join("tests").is_dir()
1712                    || m.parent()
1713                        .map(|p| dir.join(p).join("tests").is_dir())
1714                        .unwrap_or(false)
1715            })
1716    {
1717        cmds.push("pytest".into());
1718    }
1719    let mut seen = std::collections::HashSet::new();
1720    cmds.retain(|c| seen.insert(c.clone()));
1721    cmds.truncate(8);
1722    cmds
1723}

1725/// `^[A-Za-z0-9_.-]+:` recipe names — skips assignments (`:=`),
1726/// dot-directives (.PHONY), pattern rules (%) and comments.
1727fn makefile_targets(text: &str) -> Vec<String> {
1728    let mut out: Vec<String> = Vec::new();
1729    for line in text.lines() {
1730        if line.starts_with(['\t', ' ', '#']) {
1731            continue;
1732        }
1733        let Some(colon) = line.find(':') else {
1734            continue;
1735        };
1736        let name = line[..colon].trim();
1737        if name.is_empty()
1738            || name.starts_with('.')
1739            || name.contains('%')
1740            || !name
1741                .chars()
1742                .all(|c| c.is_ascii_alphanumeric() || matches!(c, '_' | '-' | '.'))
1743            || line[colon..].starts_with(":=")
1744        {
1745            continue;
1746        }
1747        if !out.iter().any(|t| t == name) {
1748            out.push(name.to_owned());
1749        }
1750    }
1751    out
1752}

1754/// Drift report: per-card hashes vs current content + sha vs HEAD,
1755/// plus a repo-level check of the L0 card's `tree_digest`.
1756fn status(cli: &Cli) -> i32 {
1757    let dir = root(cli);
1758    let cfg = load_config(cli);
1759    let moddir = dir.join(".kuda/modules");
1760    let Ok(entries) = std::fs::read_dir(&moddir) else {
1761        eprintln!("no .kuda/modules — run `kuda sync` first");
1762        return 2;
1763    };
1764    let head = lasius_vcs_git::GitVcs.head(&dir).unwrap_or_default();
1765    let (mut fresh, mut drifted) = (0, 0);
1766    for e in entries.flatten() {
1767        let p = e.path();
1768        if p.extension().map(|x| x == "yaml").unwrap_or(false) {
1769            let Ok(card) =
1770                kuda_cards::parse_module_card(&std::fs::read_to_string(&p).unwrap_or_default())
1771            else {
1772                continue;
1773            };
1774            let hashes = hash_files(&dir, &card.provenance.files);
1775            let stale = card
1776                .provenance
1777                .hashes
1778                .iter()
1779                .zip(&hashes)
1780                .any(|(a, b)| a != b)
1781                || card.provenance.files.len() != hashes.len();
1782            if stale {
1783                drifted += 1;
1784                outln!("drifted  {}", card.name);
1785            } else {
1786                fresh += 1;
1787            }
1788        }
1789    }
1790    // repo-level drift: re-walk the tree and compare against the
1791    // digest the last sync wrote into index.yaml. Cards from older
1792    // versions lack the field — skip silently.
1793    if let Ok(text) = std::fs::read_to_string(dir.join(".kuda/index.yaml")) {
1794        if let Ok(l0) = kuda_cards::parse_repo_card(&text) {
1795            if !l0.tree_digest.is_empty() {
1796                let (manifest_list, manifest_names, src_exts) = walk_sets(&langs());
1797                // module naming is irrelevant to the file set, so an
1798                // empty pkg map skips the manifest pass entirely.
1799                let (_, filehash) = scan_repo(
1800                    &dir,
1801                    &cfg,
1802                    &manifest_list,
1803                    &manifest_names,
1804                    &src_exts,
1805                    &std::collections::HashMap::new(),
1806                    &resolve_domains(&cfg),
1807                );
1808                if tree_digest(&filehash) != l0.tree_digest {
1809                    outln!(
1810                        "tree drifted (tree_digest mismatch — {} files walked, {} at sync)",
1811                        filehash.len(),
1812                        l0.files_total
1813                    );
1814                }
1815            }
1816        }
1817    }
1818    // domain corpus validity — schema'd records vs descriptors.
1819    // Warnings name the file; absent corpus/schema = silent skip.
1820    if cfg.extract.kinds.iter().any(|k| k == "records") {
1821        if let Ok(ws) = kuda_records::validate(&dir, &resolve_domains(&cfg), &cfg.extract.exclude) {
1822            for w in &ws {
1823                outln!("invalid  {}: {}", w.file.display(), w.message);
1824            }
1825            if !ws.is_empty() {
1826                outln!("{} invalid record(s)", ws.len());
1827            }
1828        }
1829    }
1830    outln!(
1831        "{fresh} fresh, {drifted} drifted (HEAD {})",
1832        &head[..head.len().min(7)]
1833    );
1834    0
1835}

2021fn refs_in(dir: &Path, name: &str, docs_only: bool, tag: bool) -> usize {
2022    let exts: &[&str] = if docs_only {
2023        &["md", "mdx", "rst", "txt"]
2024    } else {
2025        &[
2026            "md", "mdx", "rs", "py", "ts", "tsx", "go", "toml", "yaml", "yml", "sh", "txt",
2027        ]
2028    };
2029    let mut hits = 0;
2030    let mut stack = vec![dir.to_path_buf()];
2031    while let Some(d) = stack.pop() {
2032        let Ok(rd) = std::fs::read_dir(&d) else {
2033            continue;
2034        };

2036            let p = e.path();

2042                    .file_name()

2058            }

2069                        outln!("{}:{}: {}{t}", rel.display(), i + 1, line.trim());

2072            }
2073        }
2074    }

2076}

2322/// sturn WAL → repo-relative paths touched since last sync.
2323/// Walks `refs/sturn/wal/*` commits newer than each ref's stored
2324/// head (`meta` key `sturn_wal:<ref>`), pulling `path`/`from`/`to`
2325/// and static glob prefixes out of `event.json` ops. No sturn refs
2326/// = no work — the hash diff still covers everything else.
2327fn sturn_wal_changes(dir: &Path, store: &mut dyn IndexStore) -> std::collections::HashSet<String> {
2328    use std::process::Command;
2329    let mut touched = std::collections::HashSet::new();
2330    let Ok(out) = Command::new("git")
2331        .args(["for-each-ref", "--format=%(refname)", "refs/sturn/wal"])
2332        .current_dir(dir)
2333        .output()
2334    else {
2335        return touched;
2336    };
2337    if !out.status.success() {
2338        return touched;
2339    }
2340    for line in String::from_utf8_lossy(&out.stdout).lines() {
2341        let refname = line.trim();
2342        // for-each-ref never emits blank lines
2343        #[cfg(not(coverage))]
2344        if refname.is_empty() {
2345            continue;
2346        }
2347        let key = format!("sturn_wal:{refname}");
2348        let since = store.meta(&key).ok().flatten().unwrap_or_default();
2349        let range = if since.is_empty() {
2350            refname.to_owned()
2351        } else {
2352            format!("{since}..{refname}")
2353        };
2354        let log_out = Command::new("git")
2355            .args(["log", "--format=%H", &range])
2356            .current_dir(dir)
2357            .output();
2358        #[cfg(not(coverage))]
2359        let Ok(log) = log_out
2360        else {
2361            continue;
2362        };
2363        // git spawned fine a moment ago (for-each-ref) — can't vanish
2364        #[cfg(coverage)]
2365        let log = log_out.unwrap();
2366        for sha in String::from_utf8_lossy(&log.stdout).lines() {
2367            let ev_out = Command::new("git")
2368                .args(["show", &format!("{sha}:event.json")])
2369                .current_dir(dir)
2370                .output();
2371            #[cfg(not(coverage))]
2372            let Ok(ev) = ev_out
2373            else {
2374                continue;
2375            };
2376            #[cfg(coverage)]
2377            let ev = ev_out.unwrap();
2378            let Ok(v) = serde_json::from_slice::<serde_json::Value>(&ev.stdout) else {
2379                continue;
2380            };
2381            let Some(op) = v.get("op") else { continue };
2382            for k in ["path", "from", "to"] {
2383                if let Some(p) = op.get(k).and_then(|x| x.as_str()) {
2384                    touched.insert(p.to_owned());
2385                }
2386            }
2387            // claim/release globs (docs/**) → static prefix dir
2388            if let Some(g) = op.get("glob").and_then(|x| x.as_str()) {
2389                let prefix = g.split('*').next().unwrap_or("").trim_end_matches('/');
2390                if !prefix.is_empty() {
2391                    touched.insert(prefix.to_owned());
2392                }
2393            }
2394        }
2395        let head_out = Command::new("git")
2396            .args(["rev-parse", refname])
2397            .current_dir(dir)
2398            .output();
2399        #[cfg(not(coverage))]
2400        if let Ok(head) = head_out {
2401            let h = String::from_utf8_lossy(&head.stdout).trim().to_owned();
2402            if !h.is_empty() {
2403                let _ = store.set_meta(&key, &h);
2404            }
2405        }
2406        #[cfg(coverage)]
2407        {
2408            let head = head_out.unwrap();
2409            let h = String::from_utf8_lossy(&head.stdout).trim().to_owned();
2410            if !h.is_empty() {
2411                let _ = store.set_meta(&key, &h);
2412            }
2413        }
2414    }
2415    touched
2416}

2418/// Watch wakeups — fs events from notify plus synthetic kicks from
2419/// the ref poller, funneled through one channel so a burst of either
2420/// debounces into a single sync.
2421#[cfg(not(coverage))]
2422enum Wake {
2423    Fs(Result<notify::Event, notify::Error>),
2424    Refs,
2425}
2426
2427/// Fingerprint of the non-tree domain inputs (ref -> head sha
2428/// pairs): a moved, new, or deleted ref changes it. Tree domains
2429/// yield no inputs — fs events already cover them.
2430#[cfg(not(coverage))]
2431fn ref_fingerprint(dir: &Path, doms: &[DomainDef]) -> String {
2432    use sha1::Digest;
2433    let mut h = sha1::Sha1::new();
2434    for d in doms {
2435        for (p, sha) in kuda_records::source_inputs(dir, d) {
2436            h.update(p.display().to_string().as_bytes());
2437            h.update(sha.as_bytes());
2438        }
2439    }
2440    h.finalize().iter().map(|b| format!("{b:02x}")).collect()
2441}

2443#[cfg(not(coverage))]
2444fn watch(cli: &Cli, poll_refs: Option<u64>) -> i32 {
2445    use notify::{recommended_watcher, RecursiveMode, Watcher};
2446    // Same root sync resolves — a `--config` elsewhere must not
2447    // watch the cwd while indexing the config's repo.
2448    let dir = root(cli);
2449    let cfg = load_config(cli);
2450    let (tx, rx) = std::sync::mpsc::channel();
2451    let mut w = match recommended_watcher({
2452        let tx = tx.clone();
2453        move |r| {
2454            let _ = tx.send(Wake::Fs(r));
2455        }
2456    }) {
2457        Ok(w) => w,
2458        Err(e) => {
2459            eprintln!("watch: {e}");
2460            return 1;
2461        }
2462    };
2463    if let Err(e) = w.watch(&dir, RecursiveMode::Recursive) {
2464        eprintln!("watch {e}");
2465        return 1;
2466    }
2467    // Domain corpus roots get an explicit watch on top of the
2468    // repo-tree one: a `roots` entry that's a symlink or reaches
2469    // outside the watched tree (`../shared`) escapes the recursive
2470    // descent but is still corpus the sync walk indexes. Missing
2471    // roots skip silently — the corpus appearing later still lands
2472    // in repo-tree events.
2473    let doms = resolve_domains(&cfg);
2474    for d in doms.iter().filter(|d| d.source.is_none()) {
2475        for r in &d.roots {
2476            let p = dir.join(r);
2477            if p.exists() {
2478                let _ = w.watch(&p, RecursiveMode::Recursive);
2479            }
2480        }
2481    }
2482    // WAL refs (refs/sturn/*) aren't filesystem-watchable: a
2483    // `git update-ref` writes inside .git, which the event filter
2484    // drops on purpose. `--poll-refs <secs>` is the seam — the
2485    // proper fix is a sturn-side post-op hook, not a deeper watch.
2486    if let Some(secs) = poll_refs {
2487        let dir = dir.clone();
2488        let doms = doms.clone();
2489        std::thread::spawn(move || {
2490            let mut last = ref_fingerprint(&dir, &doms);
2491            loop {
2492                std::thread::sleep(std::time::Duration::from_secs(secs.max(1)));
2493                let now = ref_fingerprint(&dir, &doms);
2494                if now != last {
2495                    last = now;
2496                    if tx.send(Wake::Refs).is_err() {
2497                        return;
2498                    }
2499                }
2500            }
2501        });
2502    }
2503    outln!(
2504        "watching {} — sync on change (ctrl-c to stop)",
2505        dir.display()
2506    );
2507    loop {
2508        let fire = match rx.recv() {
2509            Ok(Wake::Fs(Ok(evt))) => {
2510                if std::env::var_os("KUDA_WATCH_DEBUG").is_some() {
2511                    eprintln!("watch-dbg fs {:?} {:?}", evt.kind, evt.paths);
2512                }
2513                // Access events (open/read/close-nowrite) can't
2514                // change content — without this guard the sync's own
2515                // repo walk and the poller's git reads re-trigger
2516                // the watcher forever.
2517                !matches!(evt.kind, notify::EventKind::Access(_))
2518                    && evt.paths.iter().any(|p| {
2519                        // same noise set as the sync walk — writes
2520                        // under excluded dirs (.git, .kuda, target, …)
2521                        // can't change the index either.
2522                        !p.components().any(|c| {
2523                            c.as_os_str()
2524                                .to_str()
2525                                .map(|n| kuda_core::dir_excluded(n, &cfg.extract.exclude))
2526                                .unwrap_or(true)
2527                        })
2528                    })
2529            }
2530            Ok(Wake::Refs) => {
2531                if std::env::var_os("KUDA_WATCH_DEBUG").is_some() {
2532                    eprintln!("watch-dbg refs wake");
2533                }
2534                true
2535            }
2536            Ok(Wake::Fs(Err(e))) => {
2537                eprintln!("watch: {e}");
2538                continue;
2539            }
2540            Err(_) => return 0,
2541        };
2542        if !fire {
2543            continue;
2544        }
2545        // debounce: drain the burst, then one sync
2546        std::thread::sleep(std::time::Duration::from_millis(300));
2547        while let Ok(w) = rx.try_recv() {
2548            if std::env::var_os("KUDA_WATCH_DEBUG").is_some() {
2549                match w {
2550                    Wake::Fs(Ok(e)) => eprintln!("watch-dbg fs {:?}", e.paths),
2551                    Wake::Fs(Err(e)) => eprintln!("watch-dbg fs err {e}"),
2552                    Wake::Refs => eprintln!("watch-dbg refs"),
2553                }
2554            }
2555        }
2556        let _ = sync_one(&dir, &cfg, cli, false, &Default::default());
2557    }
2558}
