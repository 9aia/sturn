2//! tamias-serve — the ndjson JSON-RPC surface for embedding: read one JSON
3//! request per line on stdin, write one JSON response per line on stdout.
4//! Same wire as `sepia serve` / `cuora serve` / `myco serve`.
5//!
6//! Ops: `install`, `plan`, `sync`, `list`, `info`, `verify`, `remove`,
7//! `search`, `publish`, `registry.list`, `registry.get`, `registry.add`,
8//! `registry.remove`, `driver.list`, `driver.manifest`, `driver.query`,
9//! `driver.exec`. Requests are `{"op":"…", …}`; responses are
10//! `{"ok":true,"result":…}` or `{"ok":false,"error":"…"}`.
11//!
12//! `plan` is `tamias install --plan` — `{"op":"plan","ref":…,
13//! "allow_drivers":…}` fetches/plans and reports what a real install
14//! would do (includes, drivers, provider services, trust tier,
15//! would-be problems) without committing or dispatching.
16//!
17//! `sync` reconciles the *server's* cwd project dep set against the
18//! store + `tamias.lock` — the same `tamias sync` semantics
19//! (`{"op":"sync","locked":…,"from_lock":…,"allow_drivers":…,
20//! "dir":…}`; `dir` defaults to `.`). `from_lock: true` treats
21//! `tamias.lock` itself as the spec (no manifest consult, no lock
22//! rewrite). Per-dep outcomes come back in `result.deps`; a
23//! bundled-driver consent refusal is an `ok:false` error like
24//! `install`'s.
25//!
26//! `search`/`publish`/`registry.*` resolve named registry sources and
27//! config writes through `tamias-registry-open` — the same code the
28//! `tamias` CLI verbs use, so the wire surface can't diverge.
29//!
30//! `driver.*` ops (P-096) proxy installed `tamias-driver-<name>`
31//! binaries so orchestrators can drive them through this wire:
32//! `driver.list` reports every discovered driver (`{"drivers":[{name,
33//! path, manifest|error}]}` — a per-driver `capabilities` failure
34//! degrades to an `error` field, never fails the op);
35//! `driver.manifest` fetches one driver's capabilities JSON;
36//! `driver.query` spawns `<bin> serve`, writes ONE
37//! `{"op":"query","query":…,"params":{…}}` line and returns the
38//! driver's single response line verbatim — a driver-side `ok:false`
39//! propagates as an op error; `driver.exec` runs `<bin> <argv…>`
40//! verbatim and reports `{code, stdout, stderr, json}` (`json` is the
41//! parsed stdout or null). `driver` names are content-type kinds —
42//! the same `[a-z0-9._-]` charset [`tamias_core::is_valid_kind`]
43//! enforces, so hostile names never reach argv or path joins.
44//! `timeout_ms` (default 120s, clamped to 30min) bounds `driver.query`
45//! and `driver.exec`. Driver sessions are spawn-per-call — pooled
46//! sessions are a follow-up.
47//!
48//! **Trust boundary.** Stdin is fully trusted: `install` accepts
49//! `allow_drivers:true` (bundled-driver consent), and
50//! `publish`/`registry.add`/`registry.remove` mutate config and the
51//! registry. Embedders must treat the pipe as a privileged channel —
52//! never expose it to untrusted input.

106/// The real blocking stdin loop — compiled out under `cargo-llvm-cov`
107/// (`cfg(coverage)`) so the coverage build doesn't hang on stdin.
108#[cfg(not(coverage))]
109fn serve_loop(env: Env) -> Result<(), Error> {
110    let stdin = std::io::stdin();
111    let mut stdout = std::io::stdout();
112    for line in stdin.lock().lines() {
113        let line = line.map_err(Error::io("read stdin"))?;
114        if line.trim().is_empty() {
115            continue;
116        }
117        let response = handle(&env, &line);
118        writeln!(stdout, "{response}").map_err(Error::io("write stdout"))?;
119        stdout.flush().map_err(Error::io("flush stdout"))?;
120    }
121    Ok(())
122}

124/// The coverage probe — a fast exit so the loop's lines aren't measured.
125#[cfg(coverage)]
126fn serve_loop(_env: Env) -> Result<(), Error> {
127    Err(Error::Failed(
128        "serve loop is excluded under coverage".to_string(),
129    ))
130}

132fn handle(env: &Env, line: &str) -> String {
133    match dispatch(env, line) {
134        Ok(result) => json!({"ok": true, "result": result}).to_string(),
135        Err(e) => json!({"ok": false, "error": e.to_string()}).to_string(),
136    }
137}

145fn dispatch(env: &Env, line: &str) -> Result<Value, Error> {
146    let req: Value =
147        serde_json::from_str(line).map_err(|e| Error::Failed(format!("bad request: {e}")))?;
148    let op = str_field(&req, "op")?;
149    match op {
150        "install" => {
151            let reference = str_field(&req, "ref")?;
152            // Optional consent for payload-shipped `drivers/` — same rule as
153            // `tamias install --allow-drivers` / `TAMIAS_ALLOW_DRIVERS=1`.
154            let allow_drivers = req
155                .get("allow_drivers")
156                .and_then(Value::as_bool)
157                .unwrap_or(false);
158            // The shared install pipeline — silent progress, `"allow_drivers"`
159            // consent hints (see `tamias_install::Surface::Rpc`).
160            let config = env_config(env);
161            let entry = tamias_install::install(
162                &tamias_install::Env {
163                    store_root: env.store_root.clone(),
164                    cache_root: env.cache_root.clone(),
165                    // Global `limits.max_download` caps http(s)/`cf:`
166                    // fetches — same resolution as the CLI's install.
167                    max_download: Some(config.effective_limits(None, None).max_download),
168                    // `signing.*` — package-signature verification policy.
169                    signing: tamias_sign::SigningPolicy::resolve(Some(&config)),
170                    bin_dir: None,
171                },
172                reference,
173                allow_drivers,
174            )?;
175            Ok(json!({
176                "label": entry.label(),
177                "resolved": entry.record.resolved,
178                "digest": entry.record.digest,
179                "signed": entry.record.signature.is_some(),
180            }))
181        }
182        "plan" => {
183            // `tamias install --plan` — the dry-run over the same
184            // store/cache roots + config resolution. Reports what a
185            // real install would do; commits and dispatches nothing.
186            let reference = str_field(&req, "ref")?;
187            let allow_drivers = req
188                .get("allow_drivers")
189                .and_then(Value::as_bool)
190                .unwrap_or(false);
191            let plan = tamias_install::plan_install(
192                &tamias_install::Env::for_config(
193                    env.store_root.clone(),
194                    env.cache_root.clone(),
195                    Some(&env_config(env)),
196                ),
197                reference,
198                allow_drivers,
199            )?;
200            Ok(serde_json::to_value(&plan).map_err(|e| Error::Failed(e.to_string()))?)
201        }
202        "sync" => {
203            // `tamias sync` — reconcile the project dep set against
204            // the store + `tamias.lock`. `dir` (default `.`) is the
205            // project dir; fs refs inside `deps:` still resolve
206            // against the server's cwd, same as `install`.
207            // `from_lock: true` swaps the spec to `tamias.lock`
208            // itself (the manifest's `deps:` is not consulted).
209            let allow_drivers = req
210                .get("allow_drivers")
211                .and_then(Value::as_bool)
212                .unwrap_or(false);
213            let locked = req.get("locked").and_then(Value::as_bool).unwrap_or(false);
214            let from_lock = req
215                .get("from_lock")
216                .and_then(Value::as_bool)
217                .unwrap_or(false);
218            let dir = req
219                .get("dir")
220                .and_then(Value::as_str)
221                .map_or_else(|| Path::new("."), Path::new);
222            let config = env_config(env);
223            let report = tamias_install::sync_project(
224                &tamias_install::Env {
225                    store_root: env.store_root.clone(),
226                    cache_root: env.cache_root.clone(),
227                    max_download: Some(config.effective_limits(None, None).max_download),
228                    signing: tamias_sign::SigningPolicy::resolve(Some(&config)),
229                    bin_dir: None,
230                },
231                dir,
232                tamias_install::Surface::Rpc,
233                tamias_install::SyncOptions {
234                    allow_drivers,
235                    locked,
236                    source: if from_lock {
237                        tamias_install::SyncSource::Lock
238                    } else {
239                        tamias_install::SyncSource::Deps
240                    },
241                },
242                &mut |_| {},
243            )?;
244            let deps: Vec<Value> = report
245                .outcomes
246                .iter()
247                .map(|o| {
248                    json!({
249                        "dep": o.dep,
250                        "status": o.status.as_str(),
251                        "detail": o.detail,
252                    })
253                })
254                .collect();
255            Ok(json!({"deps": deps, "failed": report.failed()}))
256        }
257        "list" => {
258            let entries = Store::new(&env.store_root).entries()?;
259            let items: Vec<Value> = entries
260                .iter()
261                .map(|e| {
262                    json!({
263                        "label": e.label(),
264                        "source_ref": e.record.source_ref,
265                        "digest": e.record.digest,
266                        "description": e.record.description,
267                    })
268                })
269                .collect();
270            Ok(json!(items))
271        }
272        "info" => {
273            let name = str_field(&req, "name")?;
274            let entry = Store::new(&env.store_root).find(name)?;
275            // `sealed:` format when the payload manifest declares a
276            // ciphertext-only pack — advisory badge, never a gate.
277            let sealed = entry
278                .manifest()
279                .ok()
280                .and_then(|m| m.sealed)
281                .map(|s| s.format);
282            Ok(json!({
283                "label": entry.label(),
284                "resolved": entry.record.resolved,
285                "digest": entry.record.digest,
286                "description": entry.record.description,
287                "keywords": entry.record.keywords,
288                "homepage": entry.record.homepage,
289                "repository": entry.record.repository,
290                "license": entry.record.license,
291                "authors": entry.record.authors,
292                "sealed": sealed,
293                "size": entry.size()?,
294            }))
295        }
296        "verify" => {
297            let name = str_field(&req, "name")?;
298            let store = Store::new(&env.store_root);
299            let report = store.verify(name)?;
300            // Signature status alongside the digest check — the record's
301            // `signature` re-verified against `signing.allowed_signers`.
302            let entry = store.find(name)?;
303            let config = env_config(env);
304            let policy = tamias_sign::SigningPolicy::resolve(Some(&config));
305            let signature = tamias_sign::record_signature_status(
306                &entry.record,
307                policy.allowed_signers.as_deref(),
308                &policy.namespace,
309            )?
310            .as_str();
311            // Trust tier + signer re-derived live (same rule as the
312            // CLI's `verify`) — recorded `signer` wins when present.
313            let (trust, live_signer) = tamias_sign::trust_and_signer(
314                &entry.record.source_ref,
315                entry.record.signature.as_deref(),
316                &entry.record.name,
317                entry.record.scope.as_deref(),
318                entry.record.version.as_deref(),
319                &entry.record.digest,
320                &policy,
321            )?;
322            let signer = entry.record.signer.clone().or(live_signer);
323            Ok(json!({
324                "name": report.name,
325                "ok": report.ok,
326                "signature": signature,
327                "trust": trust.as_str(),
328                "signer": signer,
329            }))
330        }
331        "remove" => {
332            let name = str_field(&req, "name")?;
333            // The shared lightweight uninstall (`tamias_install::remove`)
334            // — same teardown rule as `tamias ui`'s Remove job.
335            let record = tamias_install::remove(&env.store_root, name, &mut |m| {
336                eprintln!("tamias-serve: warning — {m}");
337            })?;
338            Ok(json!({"name": record.name}))
339        }
340        // --- registry sources (shared with the CLI verbs) ---
341        "registry.list" => {
342            let (_path, config) = tamias_registry_open::config_target(env.config_file.as_deref())?;
343            let items: Vec<Value> = config
344                .registries
345                .iter()
346                .map(|(name, reg)| tamias_registry_open::registry_json(name, reg))
347                .collect();
348            Ok(json!({"registries": items}))
349        }
350        "registry.get" => {
351            let name = str_field(&req, "name")?;
352            let reg = tamias_registry_open::get_registry(env.config_file.as_deref(), name)?;
353            Ok(tamias_registry_open::registry_json(name, &reg))
354        }
355        "registry.add" => {
356            let name = str_field(&req, "name")?;
357            let kind = str_field(&req, "kind")?;
358            let url = str_field(&req, "url")?;
359            let token_env = req
360                .get("token_env")
361                .and_then(Value::as_str)
362                .map(str::to_string);
363            let id = req.get("id").and_then(Value::as_str).map(str::to_string);
364            let path = tamias_registry_open::add_registry(
365                env.config_file.as_deref(),
366                name,
367                kind,
368                url,
369                token_env,
370                id,
371            )?;
372            Ok(json!({"name": name, "config": path.display().to_string()}))
373        }
374        "registry.remove" => {
375            let name = str_field(&req, "name")?;
376            let path = tamias_registry_open::remove_registry(env.config_file.as_deref(), name)?;
377            Ok(json!({"name": name, "config": path.display().to_string()}))
378        }
379        "search" => {
380            let query = str_field(&req, "query")?;
381            let flag = req.get("registry").and_then(Value::as_str);
382            let config = env_config(env);
383            // Same routing as `tamias search`: flag → TAMIAS_REGISTRY →
384            // search.default → publish.default → local default root —
385            // the shared `search` (`tamias-registry-open`).
386            let (found, _target) =
387                tamias_registry_open::search(&env.registry_root, Some(&config), flag, query)?;
388            Ok(json!({"results": found.iter().map(registry_entry_json).collect::<Vec<_>>()}))
389        }
390        "publish" => {
391            let dir = str_field(&req, "dir")?;
392            let flag = req.get("registry").and_then(Value::as_str);
393            let config = env_config(env);
394            // Same routing as `tamias publish`: flag → TAMIAS_REGISTRY →
395            // publish.default → local default root — the shared
396            // `publish_plan` resolution (`tamias-registry-open`).
397            // `"keep_includes":true` ships the raw payload — the
398            // default flattens `include:` bases read-only.
399            let keep_includes = req
400                .get("keep_includes")
401                .and_then(Value::as_bool)
402                .unwrap_or(false);
403            // `"keep":<n>` — per-invocation retention override (P-024);
404            // absent → `retention.keep` config.
405            let keep = req.get("keep").and_then(Value::as_u64);
406            let plan = tamias_registry_open::publish_plan(
407                Path::new(dir),
408                &env.registry_root,
409                Some(&config),
410                flag,
411                &[config.publish.default.as_deref()],
412                &env.store_root,
413                keep_includes,
414                keep,
415                &mut |_, _| {},
416            )?;
417            // `"sign":true` — ssh-keygen -Y sign the provenance anchors;
418            // the key resolves like the CLI's `publish --sign`.
419            let signature = if req.get("sign").and_then(Value::as_bool).unwrap_or(false) {
420                let key = tamias_sign::resolve_signing_key(Some(&config), None)?;
421                let policy = tamias_sign::SigningPolicy::resolve(Some(&config));
422                Some(tamias_sign::sign_package(
423                    &plan.root,
424                    &plan.manifest,
425                    &key,
426                    &policy.namespace,
427                )?)
428            } else {
429                None
430            };
431            let outcome = plan.publish(signature.as_deref())?;
432            // `pruned` — index entries the retention policy dropped
433            // (P-024); payloads stay.
434            let mut out = registry_entry_json(&outcome.entry);
435            out["pruned"] = json!(outcome.pruned);
436            Ok(out)
437        }
438        // --- driver proxy ops (P-096) ---
439        "driver.list" => driver_list(),
440        "driver.manifest" => {
441            let (_name, path) = resolve_driver(&req)?;
442            Ok(tamias_driver::capabilities(&path)?)
443        }
444        "driver.query" => {
445            let (name, path) = resolve_driver(&req)?;
446            let query = str_field(&req, "query")?;
447            let params = req.get("params").cloned().unwrap_or_else(|| json!({}));
448            if !params.is_object() {
449                return Err(Error::Failed("'params' must be an object".to_string()));
450            }
451            let request = json!({"op": "query", "query": query, "params": params});
452            let response = driver_serve_once(&path, &request, driver_timeout(&req))
453                .map_err(|e| Error::Failed(format!("driver '{name}': {e}")))?;
454            // The driver's response line comes back verbatim on
455            // `ok:true` — `{"ok":true,"result":{…}}` including the
456            // pavo `QueryResult` envelope. A driver-side `ok:false`
457            // is this op's error, with the driver's message.
458            match response.get("ok").and_then(Value::as_bool) {
459                Some(true) => Ok(response),
460                Some(false) => Err(Error::Failed(format!(
461                    "driver '{name}' query '{query}' failed: {}",
462                    response
463                        .get("error")
464                        .and_then(Value::as_str)
465                        .unwrap_or("unknown error")
466                ))),
467                _ => Err(Error::Failed(format!(
468                    "driver '{name}' returned a malformed response: {response}"
469                ))),
470            }
471        }
472        "driver.exec" => {
473            let (name, path) = resolve_driver(&req)?;
474            let argv = req
475                .get("argv")
476                .and_then(Value::as_array)
477                .ok_or_else(|| Error::Failed("missing 'argv'".to_string()))?;
478            let mut args = Vec::with_capacity(argv.len());
479            for v in argv {
480                args.push(
481                    v.as_str()
482                        .ok_or_else(|| Error::Failed("'argv' must be strings".to_string()))?
483                        .to_string(),
484                );
485            }
486            let mut cmd = std::process::Command::new(&path);
487            // stdin null — the serve loop owns stdin; an exec'd
488            // driver must never eat wire lines.
489            cmd.args(&args).stdin(Stdio::null());
490            let out = tamias_core::output_with_timeout(&mut cmd, driver_timeout(&req))
491                .map_err(|e| Error::Failed(format!("driver '{name}': {e}")))?;
492            let stdout = String::from_utf8_lossy(&out.stdout).into_owned();
493            Ok(json!({
494                "code": out.status.code().unwrap_or(-1),
495                "stdout": stdout,
496                "stderr": String::from_utf8_lossy(&out.stderr),
497                "json": serde_json::from_str::<Value>(stdout.trim()).ok(),
498            }))
499        }
500        other => Err(Error::Failed(format!("unknown op '{other}'"))),
501    }
502}

528/// Default bound for `driver.query`/`driver.exec` — the same 120s the
529/// driver's own lifecycle dispatch uses (`tamias-driver`'s
530/// `DISPATCH_TIMEOUT` is private to that crate).
531const DRIVER_OP_TIMEOUT: Duration = Duration::from_secs(120);
532
533/// `timeout_ms` ceiling — a wire client can't park the serve loop
534/// forever (or overflow the deadline arithmetic) with an absurd
535/// value.
536const DRIVER_OP_TIMEOUT_MAX_MS: u64 = 30 * 60 * 1000;
537
538/// Every discovered `tamias-driver-*` binary with its capabilities
539/// manifest — `{"drivers":[{name, path, manifest}]}`. A driver whose
540/// `capabilities` probe fails reports `"error"` instead of `manifest`
541/// and never fails the op (same tolerance `tamias drivers` has for
542/// unreachable drivers).
543fn driver_list() -> Result<Value, Error> {
544    let drivers: Vec<Value> = tamias_driver::discovered_drivers()
545        .into_iter()
546        .map(|(name, path)| match tamias_driver::capabilities(&path) {
547            Ok(manifest) => json!({
548                "name": name,
549                "path": path.display().to_string(),
550                "manifest": manifest,
551            }),
552            Err(e) => json!({
553                "name": name,
554                "path": path.display().to_string(),
555                "error": e.to_string(),
556            }),
557        })
558        .collect();
559    Ok(json!({"drivers": drivers}))
560}

562/// Resolve `"driver":"<name>"` to its binary. The name is a
563/// content-type kind — [`tamias_core::is_valid_kind`] keeps hostile
564/// values (`../`, separators, uppercase) away from argv/path joins;
565/// [`tamias_driver::find_driver`] then does the PATH/`<tamias-home>`
566/// lookup plus its capabilities-fallback scan.
567fn resolve_driver(req: &Value) -> Result<(String, PathBuf), Error> {
568    let name = str_field(req, "driver")?;
569    if !tamias_core::is_valid_kind(name) {
570        return Err(Error::Failed(format!("invalid driver name {name:?}")));
571    }

577/// `timeout_ms` param → bounded [`Duration`] (default
578/// [`DRIVER_OP_TIMEOUT`], clamped so a wire client can't park the
579/// loop or overflow the deadline arithmetic).
580fn driver_timeout(req: &Value) -> Duration {
581    req.get("timeout_ms")
582        .and_then(Value::as_u64)
583        .map(|ms| Duration::from_millis(ms.clamp(1, DRIVER_OP_TIMEOUT_MAX_MS)))
584        .unwrap_or(DRIVER_OP_TIMEOUT)
585}

587/// The `driver.query` transport: spawn `<bin> serve`, write ONE
588/// request line, read ONE response line. Spawn-per-call is the v1
589/// contract — pooled driver sessions are a follow-up. stdin closes
590/// after the request so a well-behaved serve loop answers then exits
591/// on EOF; the child is killed once the response line lands (or at
592/// the deadline) so a driver that ignores EOF can't leak.
593fn driver_serve_once(bin: &Path, request: &Value, timeout: Duration) -> Result<Value, Error> {
594    let line = serde_json::to_string(request)
595        .map_err(|e| Error::Failed(format!("encode driver request: {e}")))?;
596    let mut child = std::process::Command::new(bin)
597        .arg("serve")
598        .stdin(Stdio::piped())
599        .stdout(Stdio::piped())
600        .stderr(Stdio::piped())
601        .spawn()
602        .map_err(|e| Error::Failed(format!("spawn {} serve: {e}", bin.display())))?;
603    if let Some(mut stdin) = child.stdin.take() {
604        // A dead-on-arrival child makes the write fail — that surfaces
605        // as "exited without a response" below, no need to fail early.
606        let _ = writeln!(stdin, "{line}");
607    }
608    // stderr drains on a thread so a chatty driver can't wedge the
609    // wait loop on a full pipe (same rule as `output_with_timeout`).
610    let stderr = child.stderr.take();
611    let stderr_thr = thread::spawn(move || {
612        let mut buf = Vec::new();
613        if let Some(mut p) = stderr {
614            let _ = p.read_to_end(&mut buf);
615        }
616        buf
617    });
618    // stdout streams lines over a channel — the FIRST line is the
619    // response; we don't wait for process exit (a serve loop may keep
620    // running after answering).
621    let stdout = child.stdout.take();
622    let (tx, rx) = mpsc::channel::<String>();
623    let stdout_thr = thread::spawn(move || {
624        let Some(stdout) = stdout else { return };
625        let mut reader = BufReader::new(stdout);
626        let mut line = String::new();
627        loop {
628            line.clear();
629            match reader.read_line(&mut line) {
630                Ok(0) | Err(_) => break,
631                Ok(_) => {
632                    if tx.send(line.trim_end().to_string()).is_err() {
633                        break;
634                    }
635                }
636            }
637        }
638    });
639    let deadline = Instant::now() + timeout;
640    let outcome = loop {
641        let now = Instant::now();
642        if now >= deadline {
643            break Err(Error::Failed(format!(
644                "{} serve timed out after {timeout:?}",
645                bin.display()
646            )));
647        }
648        match rx.recv_timeout(deadline - now) {
649            Ok(line) => {
650                break serde_json::from_str::<Value>(&line).map_err(|e| {
651                    Error::Failed(format!("{} serve: bad response line: {e}", bin.display()))
652                })
653            }
654            Err(mpsc::RecvTimeoutError::Timeout) => continue,
655            Err(mpsc::RecvTimeoutError::Disconnected) => {
656                break Err(Error::Failed(format!(
657                    "{} serve: exited without a response",
658                    bin.display()
659                )));
660            }
661        }
662    };
663    // Reap: kill a still-running child (the response is already in
664    // hand, or the deadline hit), then collect stderr for error
665    // context.
666    let _ = child.kill();
667    let _ = child.wait();
668    let _ = stdout_thr.join();
669    let stderr_bytes = stderr_thr.join().unwrap_or_default();
670    match outcome {
671        Ok(v) => Ok(v),
672        Err(e) => {
673            let tail = String::from_utf8_lossy(&stderr_bytes);
674            let tail = tail.trim();
675            if tail.is_empty() {
676                Err(e)
677            } else {
678                Err(Error::Failed(format!("{e} — stderr: {tail}")))
679            }
680        }
681    }
682}

717            let env = std::env::vars_os().collect();
718            let cwd = std::env::current_dir().unwrap();
719            Self {
720                _lock: lock,
721                env,
722                cwd,
723            }
724        }

728        fn drop(&mut self) {
729            for (k, _) in std::env::vars_os() {
730                if !self.env.iter().any(|(sk, _)| *sk == k) {
731                    std::env::remove_var(&k);
732                }
733            }
734            for (k, v) in &self.env {
735                std::env::set_var(k, v);
736            }
737            let _ = std::env::set_current_dir(&self.cwd);
738        }
739    }
