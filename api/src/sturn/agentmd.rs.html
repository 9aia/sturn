1//! The managed `AGENTS.md` agent-awareness block — `init` writes
2//! it so every sturn-wired repo carries the same operating rules.
3//!
4//! The block is marker-delimited so human prose elsewhere in the
5//! file survives syncs; a stale or hand-written `Sturn awareness`
6//! section is upgraded in place, everything else is untouched.
7
8use std::path::Path;
9
10/// Marker pair delimiting the sturn-managed block.
11pub const BEGIN: &str = "<!-- sturn:agent-awareness -->";
12/// Closing marker.
13pub const END: &str = "<!-- /sturn:agent-awareness -->";
14
15/// Canonical managed section, between the markers — the text
16/// deployed org-wide in sturn-wired repos.
17const SECTION: &str = "\
18## Sturn awareness
19
20- **No PRs.** The pushed branch/commit is the deliverable; review
21  streams continuously (sturn WAL), merge is a human/sturn call.
22- sturn-wired repo (`sturn.toml` present): `sturn claim \"<uri>\"
23  --ws <ws>` *before* edits — claims surface conflicts in ms.
24- Ops via `sturn write|rename|delete`; `sturn snap` folds the ws
25  into a real commit; `sturn push` publishes heads + refs/sturn/*.
26- `sturn leave` on exit — drops the actor's claims.
27";
28
29/// The canonical emitted block, markers included.
30fn canonical() -> String {
31    format!("{BEGIN}\n\n{SECTION}\n{END}\n")
32}
33
34/// What a `sync` did.
35#[derive(Debug, Clone, Copy, PartialEq, Eq)]
36pub enum Change {
37    /// `AGENTS.md` was created.
38    Created,
39    /// The block was inserted or rewritten in an existing file.
40    Updated,
41    /// Already canonical — nothing written.
42    Unchanged,
43}
44
45/// ATX heading level for a line (`## foo` → 2), else 0.
46fn heading_level(line: &str) -> usize {
47    let t = line.trim_start();
48    let n = t.bytes().take_while(|b| *b == b'#').count();
49    if n > 0 && t[n..].starts_with(char::is_whitespace) {
50        n
51    } else {
52        0
53    }
54}
55
56/// Heading text for a line, if it's an ATX heading.
57fn heading_text(line: &str) -> Option<&str> {
58    let n = heading_level(line);
59    (n > 0).then(|| line.trim_start()[n..].trim())
60}
61
62/// Index of the `Sturn awareness` heading line (any level).
63fn section_start(lines: &[String]) -> Option<usize> {
64    lines
65        .iter()
66        .position(|l| heading_text(l).is_some_and(|t| t.eq_ignore_ascii_case("sturn awareness")))
67}

69/// Merge the canonical block into `src`:
70/// - a marked block is rewritten in place;
71/// - an unmarked `Sturn awareness` section is replaced (upgrade);
72/// - stray markers are dropped before either rule runs;
73/// - otherwise the block is appended.
74///
75/// Emission is deterministic: exactly one blank line on either
76/// side of the block, file ends with a single newline.
77fn merged(src: &str) -> String {
78    let canon: Vec<String> = canonical()
79        .trim_end_matches('\n')
80        .split('\n')
81        .map(str::to_string)
82        .collect();
83    let mut lines: Vec<String> = src.lines().map(str::to_string).collect();
84
85    let begin = lines.iter().position(|l| l.trim() == BEGIN);
86    let end = begin.and_then(|b| {
87        lines
88            .iter()
89            .enumerate()
90            .skip(b + 1)
91            .find(|(_, l)| l.trim() == END)
92            .map(|(i, _)| i)
93    });
94
95    // [..h) head prose, [h..e) replaced span, [e..) tail prose
96    let (h, e) = match (begin, end) {
97        (Some(b), Some(e)) => (b, e + 1),
98        _ => {
99            // malformed or absent markers: drop strays, then try a
100            // hand-written section
101            lines.retain(|l| {
102                let t = l.trim();
103                t != BEGIN && t != END
104            });
105            match section_start(&lines) {
106                Some(s) => {
107                    let lvl = heading_level(&lines[s]);
108                    let e = lines[s + 1..]
109                        .iter()
110                        .position(|l| {
111                            let n = heading_level(l);
112                            n > 0 && n <= lvl
113                        })
114                        .map_or(lines.len(), |p| s + 1 + p);
115                    (s, e)
116                }
117                None => (lines.len(), lines.len()),
118            }
119        }
120    };
121
122    let mut out: Vec<String> = lines[..h].to_vec();
123    while out.last().is_some_and(|l| l.trim().is_empty()) {
124        out.pop();
125    }
126    if !out.is_empty() {
127        out.push(String::new());
128    }
129    out.extend(canon);
130    let tail = lines[e..]
131        .iter()
132        .skip_while(|l| l.trim().is_empty())
133        .cloned()
134        .collect::<Vec<_>>();
135    if !tail.is_empty() {
136        out.push(String::new());
137        out.extend(tail);
138    }
139    // Canonical emission ends with exactly one newline — trailing
140    // blank lines are dropped so a repeat `sync` is a no-op.
141    while out.last().is_some_and(|l| l.trim().is_empty()) {
142        out.pop();
143    }
144    format!("{}\n", out.join("\n"))
145}

147/// Write the canonical block into `<root>/AGENTS.md`.
148pub fn sync(root: &Path) -> Result<Change, String> {
149    let path = root.join("AGENTS.md");
150    let existed = path.exists();
151    let src = std::fs::read_to_string(&path).unwrap_or_default();
152    let next = merged(&src);
153    if existed && next == src {
154        return Ok(Change::Unchanged);
155    }
156    std::fs::write(&path, &next).map_err(|e| format!("write {}: {e}", path.display()))?;
157    Ok(if existed {
158        Change::Updated
159    } else {
160        Change::Created
161    })
162}

164#[cfg(test)]
165#[allow(clippy::unwrap_used)]
166mod tests {
167    use super::*;
168
169    fn scratch(name: &str) -> std::path::PathBuf {
170        let d = std::env::temp_dir().join(format!("sturn-agentmd-{}-{name}", std::process::id()));
171        let _ = std::fs::remove_dir_all(&d);
172        std::fs::create_dir_all(&d).unwrap();
173        d
174    }
175
176    #[test]
177    fn missing_file_created_then_unchanged() {
178        let d = scratch("missing");
179        assert_eq!(sync(&d).unwrap(), Change::Created);
180        assert_eq!(
181            std::fs::read_to_string(d.join("AGENTS.md")).unwrap(),
182            canonical()
183        );
184        assert_eq!(sync(&d).unwrap(), Change::Unchanged);
185    }
186
187    #[test]
188    fn appends_to_existing_prose() {
189        let d = scratch("append");
190        std::fs::write(d.join("AGENTS.md"), "# Repo\n\nRules here.\n").unwrap();
191        assert_eq!(sync(&d).unwrap(), Change::Updated);
192        let out = std::fs::read_to_string(d.join("AGENTS.md")).unwrap();
193        assert!(out.starts_with("# Repo\n\nRules here.\n\n"));
194        assert!(out.contains(BEGIN));
195        assert!(out.ends_with(&format!("{END}\n")));
196        assert_eq!(sync(&d).unwrap(), Change::Unchanged);
197    }
198
199    #[test]
200    fn rewrites_drifted_marked_block() {
201        let d = scratch("drift");
202        let bad = canonical().replace("sturn leave", "sturn yeet");
203        std::fs::write(d.join("AGENTS.md"), &bad).unwrap();
204        assert_eq!(sync(&d).unwrap(), Change::Updated);
205        assert_eq!(
206            std::fs::read_to_string(d.join("AGENTS.md")).unwrap(),
207            canonical()

209    }

211    #[test]
212    fn keeps_tail_prose_after_marked_block() {
213        let d = scratch("tail");

215            d.join("AGENTS.md"),
216            format!("# R\n\n{BEGIN}\n\nold\n\n{END}\n\n\n## Layout\n\nstuff\n\n\n"),
217        )
218        .unwrap();
219        sync(&d).unwrap();
220        let out = std::fs::read_to_string(d.join("AGENTS.md")).unwrap();
221        assert!(out.starts_with("# R\n"));
222        assert!(out.ends_with("## Layout\n\nstuff\n"));
223        assert!(!out.contains("\nold\n"));
224        assert_eq!(sync(&d).unwrap(), Change::Unchanged);
225    }
