1//! Driver conformance — [`check_driver`] runs the pavo wire contract
2//! suite against a `tamias-driver-*` executable and returns structured
3//! [`Violation`]s rather than panicking.
4//!
5//! The suite (all over `<driver> serve`, one persistent session):
6//!
7//! 1. `{"op":"manifest"}` → parses as [`DriverManifest`] and
8//!    advertises pavo (`"serve"` in `verbs` or `"pavo"` in
9//!    `capabilities`).
10//! 2. `{"op":"views"}` → parses as `Vec<ViewSummary>` whose id set
11//!    matches the manifest's declared views.
12//! 3. `{"op":"view","view":<id>"}` for every declared view → parses
13//!    as [`ViewResult`], echoes the requested view id, and its `data`
14//!    map carries every declared query binding.
15//! 4. `{"op":"query","query":<name>"}` for every declared binding →
16//!    parses as [`QueryResult`].
17//! 5. `{"op":"invoke","action":<id>"}` for every declared action —
18//!    manifest-global, node-scoped (`NodeBase.actions[]`), and
19//!    tree-item-scoped (`TreeItem.actions[]`, inline or query-bound)
20//!    — with the view/target context hosts would send → parses as
21//!    [`ActionResult`]. `route`/`driver`-scoped actions are skipped
22//!    (they don't dispatch over this driver's `invoke`). An
23//!    undeclared action id must answer `ok:false`.
24//! 6. Malformed requests — an unknown op and a non-JSON line — must
25//!    both get an `ok:false` [`Response`].
26//! 7. The process must still answer after all of the above (session
27//!    liveness across multiple requests).
28//!
29//! Transport failures (timeout, EOF, unparseable response lines)
30//! become violations on the check that hit them; the suite keeps
31//! going so one broken op doesn't hide the rest.
32

44/// Which contract check produced a [`Violation`].
45#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash, PartialOrd, Ord, Serialize)]
46#[serde(rename_all = "snake_case")]
47pub enum Check {
48    /// The `serve` process couldn't be spawned.
49    Spawn,
50    /// `manifest` op: transport failure, `ok:false`, or a result that
51    /// doesn't parse as [`DriverManifest`].
52    Manifest,
53    /// The manifest parses but doesn't advertise pavo service.
54    AdvertisesPavo,
55    /// `views` op: transport failure, `ok:false`, or a result that
56    /// doesn't parse as `Vec<ViewSummary>`.
57    Views,
58    /// `views` result ids don't match the manifest's view ids.
59    ViewsConsistent,
60    /// `view` op for a declared view: transport failure, `ok:false`,
61    /// or a result that doesn't parse as [`ViewResult`].
62    View,
63    /// `view` result shape: echoed id mismatch or `data` missing a
64    /// declared query binding.
65    ViewShape,
66    /// `query` op for a declared binding: transport failure,
67    /// `ok:false`, or a result that doesn't parse as [`QueryResult`].
68    Query,
69    /// `invoke` op for a declared action: transport failure,
70    /// `ok:false`, or a result that doesn't parse as [`ActionResult`].
71    Invoke,
72    /// `invoke` of an undeclared action id didn't get `ok:false`.
73    InvokeUnknown,
74    /// Malformed requests (unknown op, non-JSON) didn't get `ok:false`.
75    MalformedRejected,
76    /// The session died or stopped answering partway through.
77    SessionAlive,
78}
79
80impl std::fmt::Display for Check {
81    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
82        let s = match self {
83            Self::Spawn => "spawn",
84            Self::Manifest => "manifest",
85            Self::AdvertisesPavo => "advertises_pavo",
86            Self::Views => "views",
87            Self::ViewsConsistent => "views_consistent",
88            Self::View => "view",
89            Self::ViewShape => "view_shape",
90            Self::Query => "query",
91            Self::Invoke => "invoke",
92            Self::InvokeUnknown => "invoke_unknown",
93            Self::MalformedRejected => "malformed_rejected",
94            Self::SessionAlive => "session_alive",
95        };
96        f.write_str(s)
97    }
98}

100/// One contract violation — which check failed and why.
101#[derive(Clone, Debug, PartialEq, Eq, Serialize)]
102pub struct Violation {
103    /// The check that failed.
104    pub check: Check,
105    /// Human/machine-readable detail (driver errors, parse errors,
106    /// mismatched ids, …).
107    pub detail: String,
108}
109
110impl std::fmt::Display for Violation {
111    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
112        write!(f, "[{}] {}", self.check, self.detail)
113    }
114}
115
116impl std::error::Error for Violation {}
117
118fn violation(check: Check, detail: impl Into<String>) -> Violation {
119    Violation {
120        check,
121        detail: detail.into(),
122    }
123}

125/// Knobs for [`check_driver_with`].
126#[derive(Clone, Debug)]
127pub struct CheckOptions {
128    /// Per-request response timeout. Default 10s.
129    pub timeout: Duration,
130}
131
132impl Default for CheckOptions {
133    fn default() -> Self {
134        Self {
135            timeout: Duration::from_secs(10),
136        }
137    }

140/// Run the conformance suite against the `tamias-driver-*` executable
141/// at `driver` with default options. Returns every violation found —
142/// an empty vec means the driver is conformant.
143#[must_use]
144pub fn check_driver(driver: &Path) -> Vec<Violation> {
145    check_driver_with(driver, CheckOptions::default())
146}
147
148/// [`check_driver`] with explicit [`CheckOptions`] (short timeouts for
149/// delay-injection tests, etc.).
150#[must_use]
151pub fn check_driver_with(driver: &Path, options: CheckOptions) -> Vec<Violation> {
152    let mut out = Vec::new();
153    let timeout = options.timeout;
154    let mut session = match Session::spawn(driver) {
155        Ok(s) => s,
156        Err(e) => {
157            out.push(violation(Check::Spawn, format!("{driver:?}: {e}")));
158            return out;
159        }
160    };
161
162    // -- 1. manifest -------------------------------------------------
163    let manifest = match typed_call(
164        &mut session,
165        json!({"op": "manifest"}),
166        timeout,
167        Check::Manifest,
168        &mut out,
169    ) {
170        Some(value) => match serde_json::from_value::<DriverManifest>(value) {
171            Ok(m) => {
172                let advertises = m.verbs.iter().any(|v| v == "serve")
173                    || m.capabilities.iter().any(|c| c == "pavo");
174                if !advertises {
175                    out.push(violation(
176                        Check::AdvertisesPavo,
177                        format!(
178                            "manifest doesn't advertise pavo service (verbs={:?}, capabilities={:?})",
179                            m.verbs, m.capabilities
180                        ),
181                    ));
182                }
183                Some(m)
184            }
185            Err(e) => {
186                out.push(violation(
187                    Check::Manifest,
188                    format!("manifest result isn't a DriverManifest: {e}"),
189                ));
190                None
191            }
192        },
193        None => None,
194    };
195
196    // -- 2. views ------------------------------------------------------
197    let summaries = typed_call(
198        &mut session,
199        json!({"op": "views"}),
200        timeout,
201        Check::Views,
202        &mut out,
203    )
204    .and_then(
205        |value| match serde_json::from_value::<Vec<ViewSummary>>(value) {
206            Ok(v) => Some(v),
207            Err(e) => {
208                out.push(violation(
209                    Check::Views,
210                    format!("views result isn't Vec<ViewSummary>: {e}"),
211                ));
212                None
213            }
214        },
215    );
216
217    // views ↔ manifest consistency
218    if let (Some(m), Some(s)) = (&manifest, &summaries) {
219        let declared: BTreeSet<&str> = m.views.iter().map(|v| v.id.as_str()).collect();
220        let served: BTreeSet<&str> = s.iter().map(|v| v.id.as_str()).collect();
221        let missing: Vec<&str> = declared.difference(&served).copied().collect();
222        let extra: Vec<&str> = served.difference(&declared).copied().collect();
223        if !missing.is_empty() || !extra.is_empty() {
224            out.push(violation(
225                Check::ViewsConsistent,
226                format!(
227                    "views/manifest mismatch — in manifest not views: {missing:?}; in views not manifest: {extra:?}"
228                ),
229            ));
230        }
231    }
232
233    // -- 3. view per declared view -------------------------------------
234    // View ids to exercise: the manifest's when we have it, else the
235    // views op's.
236    let view_ids: Vec<String> = manifest
237        .as_ref()
238        .map(|m| m.views.iter().map(|v| v.id.clone()).collect())
239        .or_else(|| {
240            summaries
241                .as_ref()
242                .map(|s| s.iter().map(|v| v.id.clone()).collect())
243        })
244        .unwrap_or_default();
245    let mut binding_names: BTreeSet<String> = BTreeSet::new();
246    if let Some(m) = &manifest {
247        for spec in &m.views {
248            for b in &spec.queries {
249                binding_names.insert(b.name.clone());
250            }
251        }
252    }
253
254    // Action probes collected while serving views — node/item-scoped
255    // declarations live inside ViewResults; manifest-global ones are
256    // added once `view_ids` is known (they still get a view context,
257    // like `Host::invoke` always sends).
258    let mut probes: Vec<InvokeProbe> = Vec::new();
259
260    for id in &view_ids {
261        let request = json!({"op": "view", "view": id});
262        let Some(value) = typed_call(&mut session, request, timeout, Check::View, &mut out) else {
263            continue;
264        };
265        match serde_json::from_value::<ViewResult>(value) {
266            Ok(result) => {
267                if result.view.id != *id {
268                    out.push(violation(
269                        Check::ViewShape,
270                        format!("view '{id}' echoed id '{}'", result.view.id),
271                    ));
272                }
273                for b in &result.view.queries {
274                    if !result.data.contains_key(&b.name) {
275                        out.push(violation(
276                            Check::ViewShape,
277                            format!("view '{id}' data missing binding '{}'", b.name),
278                        ));
279                    }
280                }
281                for name in result.view.queries.iter().map(|b| b.name.clone()) {
282                    binding_names.insert(name);
283                }
284                probes.extend(view_action_probes(&result));
285            }
286            Err(e) => out.push(violation(
287                Check::View,
288                format!("view '{id}' result isn't a ViewResult: {e}"),
289            )),
290        }
291    }
292
293    // -- 4. query per declared binding ---------------------------------
294    for name in &binding_names {
295        let request = json!({"op": "query", "query": name});
296        if let Some(value) = typed_call(&mut session, request, timeout, Check::Query, &mut out) {
297            if let Err(e) = serde_json::from_value::<QueryResult>(value) {
298                out.push(violation(
299                    Check::Query,
300                    format!("query '{name}' result isn't a QueryResult: {e}"),
301                ));
302            }
303        }
304    }
305
306    // -- 5. invoke per declared action ----------------------------------
307    // Manifest-global actions dispatch with the driver's first view as
308    // context (what `Host::invoke` falls back to); no view context at
309    // all when the driver declares none.
310    if let Some(m) = &manifest {
311        for a in &m.actions {
312            if dispatchable(a) {
313                probes.push(InvokeProbe {
314                    view: view_ids.first().cloned(),
315                    action: a.id.clone(),
316                    target: None,
317                });
318            }
319        }
320    }
321
322    // Dedupe identical wire requests — the same action id declared at
323    // several scopes collapses to one probe per distinct context.
324    let mut seen_requests = BTreeSet::new();
325    for probe in &probes {
326        let mut request = json!({"op": "invoke", "action": probe.action, "params": {}});
327        if let Some(v) = &probe.view {
328            request["view"] = json!(v);
329        }
330        if let Some(t) = &probe.target {
331            request["target"] = t.clone();
332        }
333        if !seen_requests.insert(serde_json::to_string(&request).unwrap_or_default()) {
334            continue;
335        }
336        if let Some(value) = typed_call(&mut session, request, timeout, Check::Invoke, &mut out) {
337            if let Err(e) = serde_json::from_value::<ActionResult>(value) {
338                out.push(violation(
339                    Check::Invoke,
340                    format!(
341                        "invoke '{}' result isn't an ActionResult: {e}",
342                        probe.action
343                    ),
344                ));
345            }
346        }
347    }
348
349    // An undeclared action id must be rejected — `ok:false` is the
350    // contract for undispatchable actions (unknown op, unknown
351    // action). Runs even for action-less drivers: `ok:false` is the
352    // right answer there too.
353    let bogus = json!({"op": "invoke", "action": "pavo.conformance.no-such-action"});
354    match session.call(&bogus, timeout) {
355        Ok(resp) if !resp.ok => {}
356        Ok(resp) => out.push(violation(
357            Check::InvokeUnknown,
358            format!(
359                "undeclared action got ok:true: {}",
360                serde_json::to_string(&resp.result.unwrap_or(Value::Null)).unwrap_or_default()
361            ),
362        )),
363        Err(e) => out.push(violation(Check::InvokeUnknown, wire_detail(&bogus, &e))),
364    }
365
366    // -- 6. malformed requests must get ok:false ------------------------
367    for raw in ["{\"op\":\"bogus\"}", "this is not json"] {
368        match session.request_raw(raw, timeout) {
369            Ok(line) => match serde_json::from_str::<Response>(&line) {
370                Ok(resp) if !resp.ok => {}
371                Ok(resp) => out.push(violation(
372                    Check::MalformedRejected,
373                    format!("malformed request {raw:?} got ok:{}", resp.ok),
374                )),
375                Err(_) => out.push(violation(
376                    Check::MalformedRejected,
377                    format!("malformed request {raw:?} got non-Response line: {line}"),
378                )),
379            },
380            Err(e) => out.push(violation(
381                Check::MalformedRejected,
382                format!("malformed request {raw:?}: {e}"),
383            )),
384        }
385    }
386
387    // -- 7. session still alive -----------------------------------------
388    match session.call(&json!({"op": "views"}), timeout) {
389        Ok(resp) if resp.ok => {}
390        Ok(resp) => out.push(violation(
391            Check::SessionAlive,
392            format!(
393                "final views probe got ok:false: {}",
394                resp.error.unwrap_or_default()
395            ),
396        )),
397        Err(e) => out.push(violation(Check::SessionAlive, format!("{e}"))),
398    }
399
400    out
401}

403/// One `invoke` probe — the wire fields a host would send for a
404/// declared action.
405struct InvokeProbe {
406    /// View-id context (`None` only for manifest-global actions on
407    /// view-less drivers).
408    view: Option<String>,
409    /// The action id.
410    action: String,
411    /// Opaque target — `{"id": …}` for item-scoped actions.
412    target: Option<Value>,
413}
414
415/// Whether an [`Action`] dispatches over *this* driver's `invoke`.
416/// `route` actions navigate host-locally (no wire round-trip, per
417/// protocol); `driver`-scoped actions name another driver's serve —
418/// v1 hosts may treat them as local, but the declaring driver isn't
419/// obligated to resolve the id, so probing would risk false
420/// violations.
421fn dispatchable(a: &Action) -> bool {
422    a.route.is_none() && a.driver.is_none()
423}
424
425/// Node- and tree-item-scoped action probes declared inside one
426/// served [`ViewResult`] — node actions first (pre-order), then
427/// tree-item actions (inline items, else query-bound `data` items —
428/// same resolution rule renderers/hosts use). Item probes are
429/// deduped to one per action id per view (first item wins) so a
430/// 1000-row tree doesn't turn into 1000 invokes.
431fn view_action_probes(result: &ViewResult) -> Vec<InvokeProbe> {
432    let view = result.view.id.clone();
433    let mut probes = Vec::new();
434    let mut seen_node = BTreeSet::new();
435    let mut seen_item = BTreeSet::new();
436    walk_view_node(&result.view.root, &mut |node| {
437        for a in &node.base().actions {
438            if dispatchable(a) && seen_node.insert(a.id.clone()) {
439                probes.push(InvokeProbe {
440                    view: Some(view.clone()),
441                    action: a.id.clone(),
442                    target: None,
443                });
444            }
445        }
446        if let ViewNode::Tree(tree) = node {
447            if !tree.items.is_empty() {
448                collect_item_probes(&tree.items, &view, &mut seen_item, &mut probes);
449            } else if let Some(items) = tree
450                .base
451                .query
452                .as_ref()
453                .and_then(|q| result.data.get(q))
454                .and_then(payload_tree_items)
455            {
456                collect_item_probes(&items, &view, &mut seen_item, &mut probes);
457            }
458        }
459    });
460    probes
461}

463/// `TreeItem[]` or `{"items": [...]}` — the two payload shapes tree
464/// query bindings carry (mirrors `pavo_core::actions`).
465fn payload_tree_items(payload: &Value) -> Option<Vec<TreeItem>> {
466    let value = if payload.is_array() {
467        payload.clone()
468    } else {
469        payload.get("items")?.clone()
470    };
471    serde_json::from_value(value).ok()
472}

474/// Item-scoped probes for one item list, recursing into `children`.
475fn collect_item_probes(
476    items: &[TreeItem],
477    view: &str,
478    seen: &mut BTreeSet<String>,
479    out: &mut Vec<InvokeProbe>,
480) {
481    for item in items {
482        for a in &item.actions {
483            if dispatchable(a) && seen.insert(a.id.clone()) {
484                out.push(InvokeProbe {
485                    view: Some(view.to_string()),
486                    action: a.id.clone(),
487                    // The canonical item target hosts send —
488                    // `{"id": <item-id>}` (see
489                    // `pavo_core::commands` dispatch).
490                    target: item.id.as_ref().map(|id| json!({"id": id})),
491                });
492            }
493        }
494        collect_item_probes(&item.children, view, seen, out);
495    }
496}

498/// A node's children — `panel`/`row`/`column` nest `Vec<ViewNode>`;
499/// every other kind is a leaf (mirrors `pavo_core::viewnode`).
500fn view_node_children(node: &ViewNode) -> &[ViewNode] {
501    match node {
502        ViewNode::Panel(n) => &n.children,
503        ViewNode::Row(n) => &n.children,
504        ViewNode::Column(n) => &n.children,
505        _ => &[],
506    }
507}
508
509/// Depth-first pre-order walk over a [`ViewNode`] tree.
510fn walk_view_node<'a>(node: &'a ViewNode, f: &mut impl FnMut(&'a ViewNode)) {
511    f(node);
512    for child in view_node_children(node) {
513        walk_view_node(child, f);
514    }
515}

517/// One wire round-trip → the `result` value on `ok:true`.
518///
519/// Pushes a [`Violation`] under `check` and returns `None` on
520/// transport failure, `ok:false`, or an unparseable response line.
521fn typed_call(
522    session: &mut Session,
523    request: Value,
524    timeout: Duration,
525    check: Check,
526    out: &mut Vec<Violation>,
527) -> Option<Value> {
528    match session.call(&request, timeout) {
529        Ok(resp) if resp.ok => Some(resp.result.unwrap_or(Value::Null)),
530        Ok(resp) => {
531            out.push(violation(
532                check,
533                resp.error
534                    .unwrap_or_else(|| "ok:false without error".to_string()),
535            ));
536            None
537        }
538        Err(e) => {
539            out.push(violation(check, wire_detail(&request, &e)));
540            None
541        }
542    }
543}

545/// Render a transport error with the request that triggered it.
546fn wire_detail(request: &Value, e: &WireError) -> String {
547    let op = request.get("op").and_then(Value::as_str).unwrap_or("?");
548    let extra = request
549        .get("query")
550        .or_else(|| request.get("action"))
551        .or_else(|| request.get("view"))
552        .and_then(Value::as_str)
553        .map(|s| format!(" '{s}'"))
554        .unwrap_or_default();
555    format!("op '{op}'{extra}: {e}")
556}
