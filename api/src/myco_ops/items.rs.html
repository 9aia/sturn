54/// `done:` archive stamp — write-once: set when the item sits at
55/// `Done` with no stamp; a stamp already present is never
56/// rewritten (undo/redo restores, re-archives). Callers run it
57/// after status mutations that can land on `Done`; the reopen
58/// direction clears the field instead (see `walk_to` /
59/// `item set`'s apply path).
60pub fn stamp_done(item: &mut Item) {
61    if item.status == Status::Done && item.done.is_none() {
62        item.done = Some(today());
63    }
64}
65
66/// Stamp every `Done` node in `item`'s subtree — the archive
67/// funnel stamps the whole subtree at once (children die with
68/// their parent), write-once per node.
69fn stamp_subtree_done(item: &mut Item) {
70    let mut stack = vec![item];
71    while let Some(i) = stack.pop() {
72        stamp_done(i);
73        stack.extend(i.children.iter_mut());
74    }
75}

77/// Append `item` (subtree included — every node must already be
78/// terminal, `Done` or `Cancelled`; emit rejects anything else)
79/// to the done doc's flat unsectioned list. Duplicate ids — the
80/// item's whole
81/// subtree — are refused: an id lives in exactly one place.
82pub fn done_push(doc: &mut myco_core::DoneDoc, mut item: Item) -> Result<(), String> {
83    stamp_subtree_done(&mut item);
84    let live: std::collections::BTreeSet<_> = doc
85        .items()
86        .flat_map(|i| i.subtree())
87        .map(|i| i.id)
88        .collect();
89    if item.subtree().any(|i| live.contains(&i.id)) {
90        return Err(format!("{}: already in DONE.md", item.id));
91    }
92    // the flat list lives in the unsectioned run — it must stay
93    // at index 0 (hand-edited `##` sections may precede nothing)
94    match doc.sections.first_mut() {
95        Some(s) if s.name.is_none() => s.items.push(item),
96        _ => doc.sections.insert(
97            0,
98            myco_core::Section {
99                name: None,
100                items: vec![item],
101            },
102        ),
103    }
104    Ok(())
105}

107/// Archive a done task subtree into `DONE.md` — load, push, save.
108/// Callers that must validate *before* any write stage the doc
109/// themselves (`load_done` → [`done_push`] → `save_ledger` →
110/// `save_done`) — this convenience form loads fresh, so a dup id
111/// surfaces only after earlier writes landed.
112pub fn archive_done(store: &impl Store, item: Item) -> Result<(), String> {
113    let mut doc = load_done(store)?;
114    done_push(&mut doc, item)?;
115    save_done(store, &doc)
116}

118/// Load a file-backed item for mutation. Archived items are
119/// refused — `save_item` writes to the live dir, so editing one
120/// would silently un-archive it.
121pub fn item_for_edit(store: &impl Store, id: ItemId) -> Result<ItemFile, String> {
122    if store.is_archived(id).map_err(|e| e.to_string())? {
123        return Err(format!("{id} is archived — mutating it is refused"));
124    }
125    store.item(id).map_err(|e| e.to_string())
126}

133/// A store handle holding the worktree lock — exclusive from
134/// [`open_with`] (mutating verbs), shared from [`open_read_with`].
135/// The lock releases when this drops. Derefs to the inner store,
136/// so `&locked` coerces to `&S` at every call site. Generic over
137/// the store — the front-end (composition root) names the impl.
138pub struct LockedStore<S> {
139    store: S,
140    _lock: Option<ops::WorktreeLock>,
141}
142
143impl<S> LockedStore<S> {
144    /// A store holding no lock — for tests and callers that
145    /// guarantee a single writer by construction.
146    pub fn unlocked(store: S) -> Self {
147        LockedStore { store, _lock: None }
148    }
149}

152    type Target = S;
153
154    fn deref(&self) -> &S {
155        &self.store
156    }

242/// Resolve the project + open the store `mk` builds under the
243/// exclusive mutation lock — every verb that writes the ledger
244/// takes this path. `mk` gets the discovered worktree and names
245/// the concrete impl (`myco`'s `ops::store_at` → `MdStore`), so
246/// this crate stays generic over [`Store`]; `cfg` is the same
247/// story for config (discovery reads `project_dir`). The lock
248/// spans discovery → mutation → the per-verb autocommit;
249/// contention fails fast (`another myco process is writing —
250/// retry`) instead of corrupting the ledger.
251pub fn open_with<S: Store>(
252    dir: Option<&Path>,
253    root: Option<&Path>,
254    cfg: crate::config::ConfigLoader,
255    mk: impl FnOnce(&Path, &myco_core::KindTable) -> Result<S, String>,
256) -> Result<(discover::Project, LockedStore<S>), String> {
257    open_with_index(dir, root, cfg, None, mk)
258}

260/// [`open_with`] with the repo index at discovery's disposal —
261/// the org-ledger fallback: a *registered* repo with no ledger
262/// of its own resolves to the `ledger = "org"` entry's worktree.
263/// `None` keeps discovery strict (unregistered repos fail).
264pub fn open_with_index<S: Store>(
265    dir: Option<&Path>,
266    root: Option<&Path>,
267    cfg: crate::config::ConfigLoader,
268    index: Option<&dyn myco_index::Index>,
269    mk: impl FnOnce(&Path, &myco_core::KindTable) -> Result<S, String>,
270) -> Result<(discover::Project, LockedStore<S>), String> {
271    let proj = discover::discover_with(dir, root, cfg, index)?;
272    // a detached worktree would swallow the write — refuse before
273    // the store opens (`check --fix` repairs the worktree itself)
274    ops::ensure_committable(&proj.worktree)?;
275    // a ledger behind its remote must sync before mutating —
276    // every verb commit would otherwise deepen the divergence.
277    // `sync.auto_fetch` (default on) refreshes the remote-tracking
278    // ref first — best-effort, a failed fetch never blocks.
279    ops::ensure_synced(
280        &proj.worktree,
281        crate::config::sync_auto_fetch_for(&proj.root, cfg),
282    )?;
283    let kinds = crate::config::kinds_for(&proj.root, cfg);
284    let lock = ops::lock_worktree(&proj.worktree)?;
285    let store = mk(&proj.worktree, &kinds)?;
286    Ok((
287        proj,
288        LockedStore {
289            store,
290            _lock: Some(lock),
291        },
292    ))
293}

341/// Standard mutation output: `--json` gets `payload` plus a
342/// `committed` key; human output is one line.
343pub fn report(
344    mut payload: serde_json::Value,
345    committed: bool,
346    json_out: bool,
347    human: &str,
348) -> Result<i32, String> {
349    if json_out {

358/// `--status/--owner/--label/--pri/--epic/--milestone/--due/--est/
359/// --cycle/--query` list filter + `--limit` — the parsed form of
360/// the CLI's `ListFlags`, usable by any front-end.
361#[derive(Debug, Clone, Default)]
362pub struct Filter {
363    /// Keep only items with this status.
364    pub status: Option<Status>,
365    /// Keep only items owned by this actor string.
366    pub owner: Option<String>,
367    /// Keep only items carrying all of these labels.
368    pub labels: Vec<String>,
369    /// Keep only items at this priority.
370    pub pri: Option<Priority>,
371    /// Keep only items in this epic.
372    pub epic: Option<String>,
373    /// Keep only items in this milestone.
374    pub milestone: Option<String>,
375    /// Keep only items due on this exact date.
376    pub due: Option<time::Date>,
377    /// Keep only items with this point estimate.
378    pub estimate: Option<u32>,
379    /// Keep only items in this cycle.
380    pub cycle: Option<String>,
381    /// Keep only items whose id, title, or a label contains this
382    /// text (case-insensitive).
383    pub query: Option<String>,
384    /// Keep only the first N items after filtering.
385    pub limit: Option<usize>,
386}
387
388impl Filter {
389    /// Does `item` pass the filters?
390    pub fn matches(&self, item: &Item) -> bool {
391        self.status.is_none_or(|s| item.status == s)
392            && self
393                .owner
394                .as_deref()
395                .is_none_or(|o| item.owner.as_ref().map(myco_core::Actor::as_str) == Some(o))
396            && self.labels.iter().all(|l| item.labels.contains(l))
397            && self.pri.is_none_or(|p| item.priority == p)
398            && self
399                .epic
400                .as_deref()
401                .is_none_or(|e| item.epic.as_deref() == Some(e))
402            && self
403                .milestone
404                .as_deref()
405                .is_none_or(|m| item.milestone.as_deref() == Some(m))
406            && self.due.is_none_or(|d| item.due == Some(d))
407            && self.estimate.is_none_or(|e| item.estimate == Some(e))
408            && self
409                .cycle
410                .as_deref()
411                .is_none_or(|c| item.cycle.as_deref() == Some(c))
412            && self.query.as_deref().is_none_or(|q| {
413                let q = q.to_lowercase();
414                item.id.to_string().to_lowercase().contains(&q)
415                    || item.title.to_lowercase().contains(&q)
416                    || item.labels.iter().any(|l| l.to_lowercase().contains(&q))
417            })
418    }
419
420    /// `limit` as a `.take()` bound.
421    pub fn take(&self) -> usize {
422        self.limit.unwrap_or(usize::MAX)
423    }
424}

426/// `--json` → a bare array of item objects; else compact lines.
427pub fn print_list(items: &[Item], json_out: bool) -> Result<i32, String> {
428    if json_out {
429        let v: Vec<serde_json::Value> = items
430            .iter()
431            .map(|i| serde_json::to_value(i).unwrap_or(json!(null)))
432            .collect();
433        ops::print_json(&json!(v));
434    } else {
435        for i in items {
436            println!("{}", compact(i));
437        }
438    }
439    Ok(0)
440}

442/// Like [`print_list`] for depth-tagged items — human output
443/// indents children 2 spaces per level under their parent.
444pub fn print_tree(pairs: &[(usize, &Item)], json_out: bool) -> Result<i32, String> {
445    if json_out {
446        let v: Vec<serde_json::Value> = pairs
447            .iter()
448            .map(|(_, i)| serde_json::to_value(i).unwrap_or(json!(null)))
449            .collect();
450        ops::print_json(&json!(v));
451    } else {
452        for (depth, i) in pairs {
453            for _ in 0..*depth {
454                print!("  ");
455            }
456            println!("{}", compact(i));
457        }
458    }
459    Ok(0)
460}

462/// Input bundle for [`new_file`] — one per `<kind> new` call.
463/// `word` is the commit-message noun (`chore: new issue I-001`).
464pub struct NewItem<'a> {
465    /// The kind of item to create.
466    pub kind: ItemKind,
467    /// One-line title (drives the filename slug).
468    pub title: String,
469    /// Body markdown — `None` → `template` fills in.
470    pub body: Option<&'a str>,
471    /// Optional metadata (`pri:`/`owner:`/`#label`/…).
472    pub meta: &'a NewItemMeta,
473    /// The kind's default body template.
474    pub template: fn(&Item) -> String,
475    /// Commit-message noun — `chore: new <word> <id>`.
476    pub word: &'a str,
477    /// `--json` output.
478    pub json_out: bool,
479    /// `--no-commit`.
480    pub no_commit: bool,
481}
482
483/// The shared `<kind> new` body — alloc id → apply meta → body or
484/// `template` → `save_item` → commit + report.
485pub fn new_file(
486    proj: &discover::Project,
487    store: &impl Store,
488    spec: &NewItem<'_>,
489) -> Result<i32, String> {
490    let id = store.alloc_id(spec.kind).map_err(|e| e.to_string())?;
491    let mut item = Item::new(id, &spec.title).created(today());
492    spec.meta.apply(&mut item)?;
493    let body = spec
494        .body
495        .map(str::to_string)
496        .unwrap_or_else(|| (spec.template)(&item));
497    let file = ItemFile::new(item.clone()).body(body);
498    store.save_item(&file).map_err(|e| e.to_string())?;
499    let rel = item_rel_path(store, &item)?;
500    let committed = commit(
501        &proj.worktree,
502        &format!("chore: new {} {id}", spec.word),
503        spec.meta.owner.as_ref(),
504        spec.no_commit,
505    )?;
506    report(
507        json!({"item": item, "path": rel}),
508        committed,
509        spec.json_out,
510        &format!("created {id} {rel}"),
511    )
512}

514/// The read behind [`list_files`] — live item files (plus
515/// `archive/` entries when `include_archived`), filtered, as
516/// `(item, archived)` pairs. Non-CLI front-ends (`myco serve`)
517/// shape it themselves — [`marked_json`] is the list shape.
518pub fn query_files(
519    store: &impl Store,
520    kind: ItemKind,
521    include_archived: bool,
522    filter: &Filter,
523) -> Result<Vec<(Item, bool)>, String> {
524    let live = store.list_items(kind).map_err(|e| e.to_string())?;
525    let mut found: Vec<(Item, bool)> = live.into_iter().map(|f| (f.meta, false)).collect();
526    if include_archived {
527        found.extend(
528            store
529                .list_archived(kind)
530                .map_err(|e| e.to_string())?
531                .into_iter()
532                .map(|f| (f.meta, true)),
533        );
534    }
535    Ok(found
536        .into_iter()
537        .filter(|(i, _)| filter.matches(i))
538        .take(filter.take())
539        .collect())
540}

542/// `(item, archived)` pairs → a bare array of item objects, each
543/// tagged `"archived": bool` — the `--json` list shape. Live and
544/// archived items share one array; the flag is the marker.
545pub fn marked_json(items: &[(Item, bool)]) -> serde_json::Value {
546    json!(items
547        .iter()
548        .map(|(i, archived)| {
549            let mut v = serde_json::to_value(i).unwrap_or(json!(null));
550            if let Some(obj) = v.as_object_mut() {
551                obj.insert("archived".into(), json!(archived));
552            }
553            v
554        })
555        .collect::<Vec<_>>())
556}

558/// The shared `<kind> list` — live item files, plus `archive/`
559/// entries when `include_archived`.
560pub fn list_files(
561    store: &impl Store,
562    kind: ItemKind,
563    include_archived: bool,
564    filter: &Filter,
565    json_out: bool,
566) -> Result<i32, String> {
567    let found = query_files(store, kind, include_archived, filter)?;
568    if json_out {
569        ops::print_json(&marked_json(&found));
570        return Ok(0);
571    }
572    let items: Vec<Item> = found.into_iter().map(|(i, _)| i).collect();
573    print_list(&items, false)
574}

587/// Locate an item anywhere in the ledger — TODO.md, BACKLOG.md, or
588/// an item file (live or `archive/`). Returns the item, its
589/// location label, and the body for file-backed items.
590pub fn find_item(
591    store: &impl Store,
592    id: ItemId,
593) -> Result<Option<(Item, String, Option<String>)>, String> {
594    let ledger = load_ledger(store)?;
595    if let Some(loc) = ledger.locate(id) {
596        let file = if loc.backlog { "BACKLOG.md" } else { "TODO.md" };
597        if let Some(item) = ledger.item(&loc) {
598            return Ok(Some((item.clone(), file.to_string(), None)));
599        }
600    }
601    // DONE.md — finished tasks stay resolvable (`item show`,
602    // dependents, links) after they leave the live ledger
603    let done = load_done(store)?;
604    if let Some(item) = done.items().flat_map(|i| i.subtree()).find(|i| i.id == id) {
605        return Ok(Some((item.clone(), "DONE.md".to_string(), None)));
606    }
607    match store.item(id) {
608        Ok(f) => {
609            let loc = item_rel_path(store, &f.meta)?;
610            Ok(Some((f.meta.clone(), loc, Some(f.body))))
611        }
612        Err(myco_store::Error::NotFound { .. }) => Ok(None),
613        Err(e) => Err(e.to_string()),
614    }
615}

701    fn concurrent_open_fails_fast_no_dup_ids() {
702        let (_d, wt) = project_dir();
703        let (_p, s1) = open_with(
704            Some(wt.as_path()),
705            None,
706            crate::config::test_loader(),
707            md_store,
708        )
709        .unwrap();
710        let id1 = s1.alloc_id(ItemKind::Task).unwrap();
711        // a racing verb can't even open — prove the failure is
712        // the lock, not the store, by retrying after the drop
713        let wt2 = wt.clone();
714        let (tx, rx) = std::sync::mpsc::channel();
715        let racing = std::thread::spawn(move || {
716            tx.send(
717                open_with(
718                    Some(wt2.as_path()),
719                    None,
720                    crate::config::test_loader(),
721                    md_store,
722                )
723                .map(|_| ()),
724            )
725            .unwrap();
726        });
727        let err = rx.recv().unwrap().unwrap_err();
728        assert!(err.contains("another myco process"), "{err}");
729        drop(s1);
730        // after release, a fresh verb allocates the *next* id —
731        // the lock is what kept two alloc_id RMWs from both
732        // reading "next = id1" and handing out a duplicate
733        let (_p, s2) = open_with(
734            Some(wt.as_path()),
735            None,
736            crate::config::test_loader(),
737            md_store,
738        )
739        .unwrap();
740        let id2 = s2.alloc_id(ItemKind::Task).unwrap();
741        assert_ne!(id1, id2);
742        racing.join().unwrap();

745    #[test]
746    fn done_push_stamps_done_write_once() {
747        use myco_core::{date, DoneDoc, Item, ItemId, Status};
748        let mut doc = DoneDoc::default();
749        let items_of = |d: &DoneDoc| d.items().cloned().collect::<Vec<_>>();
750        // a done item gets today's stamp
751        let mut item = Item::new(ItemId::new(ItemKind::Task, 1), "x");
752        item.status = Status::Done;
753        done_push(&mut doc, item).unwrap();
754        let stamped = items_of(&doc)[0].done.expect("done: stamp");
755        assert_eq!(stamped, today());
756        // an item carrying an older stamp keeps it — write-once
757        let mut old = Item::new(ItemId::new(ItemKind::Task, 2), "y");
758        old.status = Status::Done;
759        old.done = Some(date::parse("2020-01-01").unwrap());
760        done_push(&mut doc, old).unwrap();
761        assert_eq!(
762            items_of(&doc)[1].done,
763            Some(date::parse("2020-01-01").unwrap())
764        );
765        // cancelled records why it left — never a done: stamp
766        let mut dead = Item::new(ItemId::new(ItemKind::Task, 3), "z");
767        dead.status = Status::Cancelled;
768        done_push(&mut doc, dead).unwrap();
769        assert!(items_of(&doc)[2].done.is_none());
770        // and the stamps survive the emit round-trip
771        let src = myco_store_md::emit_done(&doc).unwrap();
772        assert!(src.contains("done:"), "{src}");
773        let back = myco_store_md::parse_done(&src).unwrap();
774        assert_eq!(items_of(&back)[0].done, Some(stamped));
775    }
