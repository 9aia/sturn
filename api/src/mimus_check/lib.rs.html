1//! Check port — verify a claim against reality.
2//! Impls: `mimus-check-fs` (paths), `mimus-check-cmd`
3//! (commands), `mimus-check-symbol` (kuda index), `check-llm`
4//! (behavior prose — semantic tier, advisory). New checker =
5//! new adapter crate. Set-level classes (claim↔index, ADR-0002)
6//! use [`GraphChecker`] instead — impl: `mimus-check-graph`.
7
8use mimus_core::{Claim, ClaimKind, Graph, Violation};
9
10/// A verifier for some subset of `ClaimKind`s.
11pub trait Checker: Send + Sync {
12    /// The claim kinds this checker can prove or refute.
13    fn handles(&self) -> &[ClaimKind];
14    /// `Ok(Some(v))` = claim refuted with evidence.
15    /// `Ok(None)` = verified — OR unverifiable. Evidence-or-silence:
16    /// a checker that can't prove the claim false emits nothing.
17    fn check(&self, claim: &Claim) -> Result<Option<Violation>, String>;
18
19    /// Batch path — default is per-claim. Checkers that talk to a
20    /// rate-limited backend (ACP agents) override this to fold many
21    /// claims into one prompt.
22    fn check_batch(&self, claims: &[Claim]) -> Vec<Result<Option<Violation>, String>> {
23        claims.iter().map(|c| self.check(c)).collect()
24    }
25}
26
27/// Set-level port — compare the *whole claim set* against a Graph
28/// snapshot (ADR-0002). `Checker` is per-claim; contradiction
29/// classes are claim↔claim / claim↔index and can't fold into it.
30/// Impl: `mimus-check-graph` (index-edge classes).
31pub trait GraphChecker: Send + Sync {
32    /// Evidence-or-silence over the full set — a class that can't
33    /// prove a conflict emits nothing.
34    fn run_set(&self, claims: &[Claim], graph: &Graph) -> Vec<Violation>;
35}
36
37/// Route claims to the checkers that handle their kind.
38pub fn run<'a>(
39    checkers: &'a [Box<dyn Checker>],
40    claims: &'a [Claim],
41) -> impl Iterator<Item = Result<Violation, String>> + 'a {
42    // group claims by the first checker that handles each kind,
43    // then batch through it — one claim still works
44    let mut grouped: Vec<(usize, Vec<Claim>)> = Vec::new();
45    for c in claims {
46        if let Some(i) = checkers
47            .iter()
48            .position(|ch| ch.handles().contains(&c.kind))
49        {
50            match grouped.iter_mut().find(|(g, _)| *g == i) {
51                Some((_, v)) => v.push(c.clone()),
52                None => grouped.push((i, vec![c.clone()])),
53            }
54        }
55    }
56    grouped.into_iter().flat_map(move |(i, cs)| {
57        checkers[i]
58            .check_batch(&cs)
59            .into_iter()
60            .filter_map(|r| r.transpose())
61    })
62}
