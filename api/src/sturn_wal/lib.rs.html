1//! Wal port — the append-only op log that *is* the state.
2//! Git impl in `sturn-wal-git`; libgrite = a future impl.
3//!
4//! The ports speak git's vocabulary (refs, sha, commit) as the
5//! contract — a non-git impl maps equivalent concepts (bookmark,
6//! change-id, object id) onto the same semantics.
7
8use std::io;
9use std::path::Path;
10use sturn_core::Event;
11
12/// Append/read the per-workspace op log — one log per workspace,
13/// serialized per workspace, parallel across them (git impl:
14/// commits on `refs/sturn/wal/<ws>`). Replicates with ordinary
15/// push/fetch — the bus on top is a cache, never the authority.
16pub trait Wal: Send + Sync {
17    /// Append `ev` to the ws WAL; returns the assigned event id.
18    /// Must be linearizable under concurrent appends — mechanism
19    /// is impl-defined (git: CAS retry on the ref).
20    fn append(&self, repo: &Path, ws: &str, ev: &Event) -> io::Result<String>;
21    /// All events for `ws`, oldest first.
22    fn events(&self, repo: &Path, ws: &str) -> io::Result<Vec<Event>> {
23        self.events_since(repo, ws, None)
24    }
25    /// Events appended after `from` (an opaque cursor returned by
26    /// `head`, exclusive) — `None` reads the whole log. Delta reads
27    /// are what the daemon and catch-up need; `events` is the full
28    /// replay.
29    fn events_since(
30        &self,
31        repo: &Path,
32        ws: &str,
33        from: Option<&str>,
34    ) -> io::Result<Vec<Event>>;
35    /// Opaque tip cursor for the ws log — cheap change detection.
36    /// Callers must not interpret the value; feed it back to
37    /// `events_since`.
38    fn head(&self, repo: &Path, ws: &str) -> io::Result<Option<String>>;
39    /// Workspace ids that have a log.
40    fn workspaces(&self, repo: &Path) -> io::Result<Vec<String>>;
41    /// Drop pre-snapshot history: events before the last `Snapshot`
42    /// are already in the tip — replaying them is waste. Returns
43    /// events dropped. Old history moves to an archive namespace
44    /// (git impl: `refs/sturn/archive/<ws>`) — compaction loses
45    /// nothing.
46    fn compact(&self, repo: &Path, ws: &str) -> io::Result<usize>;
47}
