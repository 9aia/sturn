1//! The process side of surface drivers — discovery and one-shot wire
2//! calls against a spawned `tamias-driver-<type>`. Standalone surface
3//! binaries (`tamias-driver-cli`, `tamias-driver-mcp`) are thin
4//! `main`s over [`DriverClient`]; an engine driver embedding a
5//! surface implements [`Surface`] itself and never spawns.
6//!
7//! Discovery mirrors the tamias driver contract: `tamias-driver-<name>`
8//! on `PATH`, then `<tamias-home>/drivers/` (`$TAMIAS_HOME` →
9//! `$XDG_DATA_HOME/tamias` → `~/.local/share/tamias`). A name that is
10//! (or resolves to) an existing file — any arg containing a path
11//! separator — is used as-is, which keeps tests and developer builds
12//! (`target/debug/tamias-driver-kuda`) trivially usable.
13
14use std::io::{BufRead, BufReader, Write};
15use std::path::{Path, PathBuf};
16use std::process::{Command, Stdio};

18use serde_json::{json, Map, Value};

20use super::{OpOutcome, OpSpec, OpsManifest, Surface};

22/// `<tamias-home>/drivers/` — the second discovery root after `PATH`.
23#[must_use]
24pub fn drivers_dir() -> PathBuf {
25    std::env::var_os("TAMIAS_HOME")
26        .map(PathBuf::from)
27        .or_else(|| std::env::var_os("XDG_DATA_HOME").map(|d| PathBuf::from(d).join("tamias")))
28        .or_else(|| std::env::var_os("HOME").map(|h| PathBuf::from(h).join(".local/share/tamias")))
29        .unwrap_or_else(|| PathBuf::from("./.tamias"))
30        .join("drivers")
31}

33/// Resolve `driver` — an engine name (`kuda` → `tamias-driver-kuda`),
34/// a full driver name, or a path — to an executable file.
35#[must_use]
36pub fn find_driver(driver: &str) -> Option<PathBuf> {
37    let path = Path::new(driver);
38    if path.components().count() > 1 {
39        return path.is_file().then(|| path.to_path_buf());
40    }
41    let name = if let Some(stripped) = driver.strip_prefix("tamias-driver-") {
42        stripped.to_string()
43    } else {
44        driver.to_string()
45    };
46    let name = format!("tamias-driver-{name}");
47    std::env::var_os("PATH")
48        .and_then(|p| {
49            std::env::split_paths(&p)
50                .map(|dir| dir.join(&name))
51                .find(|c| c.is_file())
52        })
53        .or_else(|| {
54            let candidate = drivers_dir().join(&name);
55            candidate.is_file().then_some(candidate)
56        })
57}

59/// Fetch a driver's ops manifest — the `ops` argv verb, one bounded
60/// spawn. `Err` text is caller-renderable (spawn/exit/parse detail).
61///
62/// # Errors
63/// Fails on spawn errors, non-zero exits, or an unparseable manifest.
64pub fn ops_manifest(driver: &Path) -> Result<OpsManifest, String> {
65    let out = Command::new(driver)
66        .arg("ops")
67        .output()
68        .map_err(|e| format!("{} ops: {e}", driver.display()))?;
69    if !out.status.success() {
70        return Err(format!(
71            "{} ops exited {}: {}",
72            driver.display(),
73            out.status,
74            String::from_utf8_lossy(&out.stderr).trim()
75        ));
76    }
77    let value: Value = serde_json::from_slice(&out.stdout)
78        .map_err(|e| format!("{} ops: bad JSON: {e}", driver.display()))?;
79    OpsManifest::from_value(&value).map_err(|e| format!("{} ops: {e}", driver.display()))
80}

82/// A [`Surface`] backed by a spawned `tamias-driver-<type>` binary.
83pub struct DriverClient {
84    path: PathBuf,
85}
86
87impl DriverClient {
88    /// Wrap one resolved driver binary — see [`find_driver`].
89    #[must_use]
90    pub fn new(path: PathBuf) -> Self {
91        Self { path }
92    }
93
94    /// One request line → one response line: spawn `<driver> serve`,
95    /// write the request, read the first response, close stdin so the
96    /// loop's EOF rule ends the child. `Err` = transport failure
97    /// (spawn/write/read/parse), never a driver `ok:false`.
98    fn request(&self, request: &Value) -> Result<Value, String> {
99        let mut child = Command::new(&self.path)
100            .arg("serve")
101            .stdin(Stdio::piped())
102            .stdout(Stdio::piped())
103            .stderr(Stdio::null())
104            .spawn()
105            .map_err(|e| format!("{} serve: {e}", self.path.display()))?;
106        let result = (|| {
107            let mut stdin = child
108                .stdin
109                .take()
110                .ok_or_else(|| "serve: no stdin".to_string())?;
111            let stdout = child
112                .stdout
113                .take()
114                .ok_or_else(|| "serve: no stdout".to_string())?;
115            let line =
116                serde_json::to_string(request).map_err(|e| format!("serve: request JSON: {e}"))?;
117            stdin
118                .write_all(line.as_bytes())
119                .and_then(|()| stdin.write_all(b"\n"))
120                .map_err(|e| format!("serve: write: {e}"))?;
121            drop(stdin); // EOF lets serve exit after answering
122            let mut reply = String::new();
123            BufReader::new(stdout)
124                .read_line(&mut reply)
125                .map_err(|e| format!("serve: read: {e}"))?;
126            serde_json::from_str(&reply).map_err(|e| format!("serve: bad reply: {e}"))
127        })();
128        let _ = child.wait();
129        result
130    }
131}
132
133impl Surface for DriverClient {
134    /// `cli.<op>` as a one-shot wire query; the `{code,stdout,stderr}`
135    /// convention in the response data becomes the [`OpOutcome`]. A
136    /// transport failure reports as `code: 2` on stderr.
137    fn op(&mut self, op: &OpSpec, params: &Map<String, Value>) -> OpOutcome {
138        let request = json!({
139            "op": "query",
140            "query": format!("cli.{}", op.name),
141            "params": params,
142        });
143        match self.request(&request) {
144            Ok(resp) => OpOutcome::from_response(
145                &serde_json::from_value(resp)
146                    .unwrap_or_else(|_| pavo_protocol::wire::Response::err("bad response shape")),
147            ),
148            Err(e) => OpOutcome {
149                code: 2,
150                stdout: String::new(),
151                stderr: e,
152            },
153        }
154    }
155
156    /// `<driver> run <name> <args>` with inherited stdio — unbounded
157    /// surfaces stream through.
158    fn surface(&mut self, name: &str, args: &[String]) -> i32 {
159        match Command::new(&self.path)
160            .arg("run")
161            .arg(name)
162            .args(args)
163            .stdin(Stdio::inherit())
164            .stdout(Stdio::inherit())
165            .stderr(Stdio::inherit())
166            .status()
167        {
168            Ok(status) => status.code().unwrap_or(1),
169            Err(e) => {
170                eprintln!("{} run {name}: {e}", self.path.display());
171                2
172            }
173        }
174    }
175}

178#[cfg_attr(coverage, coverage(off))]
179mod tests {

181
182    use super::*;
183
184    /// A path argument bypasses discovery; a bare name looks up
185    /// `tamias-driver-<name>` (miss here — nothing is installed).
186    #[test]
187    fn find_driver_paths_and_misses() {
188        assert!(find_driver("definitely-no-such-engine-9aia").is_none());
189        let cwd = std::env::current_dir().unwrap();
190        let fake = cwd.join("Cargo.toml"); // any existing file
191        assert_eq!(
192            find_driver(fake.to_str().unwrap()).as_deref(),
193            Some(fake.as_path())
194        );
195        assert!(find_driver("Cargo.toml").is_none()); // no prefix, not on PATH
196    }
