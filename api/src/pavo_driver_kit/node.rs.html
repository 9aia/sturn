1//! [`ViewNode`] constructors — the declarative tree as function
2//! calls instead of nested struct literals (or JSON).
3//!
4//! Every kind function takes the shared [`NodeBase`] fields first
5//! (`impl Into<NodeBase>` — pass [`base()`]'s builder or
6//! `NodeBase::default()`), then the kind-specific payload. Part
7//! helpers ([`table_column`], [`pair`], [`field`], [`tree_item`],
8//! [`vertex`]) cover the sub-structs; anything not covered is a
9//! public-field mutation away.
10//!
11//! ```
12//! # use pavo_driver_kit as kit;
13//! let tree = kit::panel(
14//!     kit::base().title("Board").query("items"),
15//!     vec![
16//!         kit::table(
17//!             kit::base(),
18//!             vec![kit::table_column("id", "ID"), kit::table_column("title", "Title")],
19//!             vec![vec!["T-1".into(), "ship it".into()]],
20//!         ),
21//!         kit::kv(kit::base().title("Counts"), vec![kit::pair("open", 3)]),
22//!     ],
23//! );
24//! assert_eq!(tree.base().title.as_deref(), Some("Board"));
25//! ```
26

36/// Builder for the [`NodeBase`] fields every node kind shares —
37/// `id`, `title`, `query`, `description`, `props`.
38///
39/// ```
40/// # use pavo_driver_kit as kit;
41/// let b = kit::base().id("n1").title("Board").query("items")
42///     .description("the board").prop("density", "compact").build();
43/// assert_eq!(b.query.as_deref(), Some("items"));
44/// assert_eq!(b.props["density"], "compact");
45/// ```
46pub struct Base {
47    base: NodeBase,
48}
49
50/// Start a [`NodeBase`] builder — chain setters, pass the result (or
51/// `NodeBase::default()`) to any [`crate::node`] kind function.
52#[must_use]
53pub fn base() -> Base {
54    Base {
55        base: NodeBase::default(),
56    }
57}
58
59impl Base {
60    /// Stable id — deep-link target, diffing, renderer expansion state.
61    #[must_use]
62    pub fn id(mut self, id: impl Into<String>) -> Self {
63        self.base.id = Some(id.into());
64        self
65    }
66
67    /// Heading the renderer may display above the node.
68    #[must_use]
69    pub fn title(mut self, title: impl Into<String>) -> Self {
70        self.base.title = Some(title.into());
71        self
72    }
73
74    /// Binding name — references a [`crate::manifest::View`]'s
75    /// [`pavo_protocol::QueryBinding`] (via [`crate::manifest::View::query`]).
76    #[must_use]
77    pub fn query(mut self, query: impl Into<String>) -> Self {
78        self.base.query = Some(query.into());
79        self
80    }
81
82    /// Longer description (tooltips, help panes).
83    #[must_use]
84    pub fn description(mut self, description: impl Into<String>) -> Self {
85        self.base.description = Some(description.into());
86        self
87    }
88
89    /// Renderer hint — kind-specific knob (`density`, `direction`,
90    /// `collapsed`, …). Renderers ignore keys they don't know.
91    #[must_use]
92    pub fn prop(mut self, key: &str, value: impl Serialize) -> Self {
93        self.base.props.insert(key.to_string(), to_value(value));
94        self
95    }
96
97    /// Add an action scoped to this node — dispatched via the
98    /// `invoke` wire op. See [`mod@crate::action`].
99    #[must_use]
100    pub fn action(mut self, action: pavo_protocol::Action) -> Self {
101        self.base.actions.push(action);
102        self
103    }
104
105    /// Set all node actions at once.
106    #[must_use]
107    pub fn actions(mut self, actions: Vec<pavo_protocol::Action>) -> Self {
108        self.base.actions = actions;
109        self
110    }
111
112    /// Finish — the [`NodeBase`] the kind functions take.
113    #[must_use]
114    pub fn build(self) -> NodeBase {
115        self.base
116    }
117}
118
119impl From<Base> for NodeBase {
120    fn from(base: Base) -> Self {
121        base.base
122    }
123}

125/// `panel` — a titled container grouping children; the universal
126/// wrapper.
127#[must_use]
128pub fn panel(base: impl Into<NodeBase>, children: Vec<ViewNode>) -> ViewNode {

313/// One [`FormField`] — text input; `required`/`options`/`default`/

357/// One [`KvPair`] — `key` + arbitrary JSON `value` (serialized via

387/// Serialize a prop/pair value — `null` on failure (a driver must
388/// never panic on the wire).
389fn to_value(value: impl Serialize) -> Value {
390    serde_json::to_value(value).unwrap_or(Value::Null)
391}

395mod tests {
396    #![allow(clippy::unwrap_used)]

399    use serde_json::json;

401    /// Constructors emit the right `kind` wire shapes.
402    #[test]
403    fn kinds_serialize() {
404        let nodes = vec![
405            panel(base().title("P"), vec![]),
406            row(base(), vec![]),
407            row_gap(base(), 2, vec![]),
408            column(base(), vec![]),
409            column_gap(base(), 3, vec![]),
410            table(base(), vec![table_column("k", "K")], vec![]),
411            sortable_table(base(), vec![], vec![]),
412            tree(base(), vec![tree_item("root")]),
413            expandable_tree(base(), vec![]),
414            graph(base(), vec![vertex("a", "A")], vec![]),
415            form(base(), vec![field("k", "K")]),
416            form_submit(
417                base().id("f"),
418                vec![select_field("s", "S", vec!["x".into()])],
419                "go",
420            ),
421            kv(base(), vec![pair("k", 1)]),
422            markdown(base(), "hi"),
423            embed(base().prop("p", 1), "comp"),
424        ];
425        let kinds: Vec<&str> = nodes
426            .iter()
427            .map(|n| match n {
428                ViewNode::Panel(_) => "panel",
429                ViewNode::Row(_) => "row",
430                ViewNode::Column(_) => "column",
431                ViewNode::Table(_) => "table",
432                ViewNode::Tree(_) => "tree",
433                ViewNode::Graph(_) => "graph",
434                ViewNode::Form(_) => "form",
435                ViewNode::Kv(_) => "kv",
436                ViewNode::Markdown(_) => "markdown",
437                ViewNode::Embed(_) => "embed",
438            })
439            .collect();
440        assert_eq!(
441            kinds,
442            [
443                "panel", "row", "row", "column", "column", "table", "table", "tree", "tree",
444                "graph", "form", "form", "kv", "markdown", "embed"
445            ]
446        );
447
448        // Spot-check wire details: gap, sortable, submit_op, props.
449        let row_gap = serde_json::to_value(&nodes[2]).unwrap();
450        assert_eq!(row_gap["gap"], json!(2));
451        let st = serde_json::to_value(&nodes[6]).unwrap();
452        assert_eq!(st["sortable"], json!(true));
453        let fs = serde_json::to_value(&nodes[11]).unwrap();
454        assert_eq!(fs["submit_op"], json!("go"));
455        assert_eq!(fs["id"], json!("f"));
456        assert_eq!(fs["fields"][0]["kind"], json!("select"));
457        let em = serde_json::to_value(&nodes[14]).unwrap();
458        assert_eq!(em["component"], json!("comp"));
459        assert_eq!(em["props"]["p"], json!(1));
460    }

462    /// `Base` covers the shared NodeBase fields.
463    #[test]
464    fn base_builder() {
465        let b = base()
466            .id("i")
467            .title("t")
468            .query("q")
469            .description("d")
470            .prop("k", "v")
471            .build();
472        assert_eq!(b.id.as_deref(), Some("i"));
473        assert_eq!(b.title.as_deref(), Some("t"));
474        assert_eq!(b.query.as_deref(), Some("q"));
475        assert_eq!(b.description.as_deref(), Some("d"));
476        assert_eq!(b.props["k"], json!("v"));
477        // NodeBase itself satisfies Into<NodeBase> (identity).
478        let n = markdown(NodeBase::default(), "x");
479        assert!(n.base().title.is_none());
480    }
