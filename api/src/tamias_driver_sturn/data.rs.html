1//! data — one read of a repo's sturn state per wire op.
2//!
3//! The driver is a *reader*: every `view`/`query` resolves the
4//! target repo, folds every `refs/sturn/wal/*` log through the same
5//! pure functions the `sturn` CLI uses (`sturn_claim::active`,
6//! `sturn_ask::all`), and drops the result. No mutations — `sturn
7//! claim`/`write`/`snap` stay the write surface; this crate only
8//! needs what `sturn claims`/`sturn log`/`sturn asks` already show.
9//!
10//! Repo resolution order (first hit wins) — mirrors `sturn --repo`
11//! plus driver conveniences:
12//!
13//! 1. Request params — `dir`, `root`, or `repo` (all name the repo
14//!    dir), e.g. `pavo view sturn:claims --param dir=/path/to/repo`.
15//! 2. `$STURN_REPO` — driver-level convenience; the `sturn` CLI
16//!    itself has no repo env var.
17//! 3. `sturn.toml`'s `repo.dir` — config search identical to the
18//!    CLI: `--config`-equivalent param `config` → `$STURN_CONFIG` →
19//!    `./sturn.toml` → `~/.config/sturn/sturn.toml`. A relative
20//!    `repo.dir` resolves against the process cwd, same as the CLI.
21//! 4. `.` — the driver's cwd (pavo inherits the user's cwd, so
22//!    `pavo view sturn:claims` inside a wired repo reads it).
23//!
24//! When nothing resolves to a git repo, [`Snapshot::error`] carries
25//! the reason and every query degrades to an `{"error": …}` payload
26//! — the wire stays `ok:true` so hosts render the failure inline.
27

39/// One workspace row — union of `refs/sturn/ws/*` namespaces and
40/// `refs/sturn/wal/*` logs (same union `sturn ws ls` prints).
41#[derive(Debug, Clone)]
42pub struct WsRow {
43    /// Workspace name.
44    pub name: String,
45    /// `namespace` = `refs/sturn/ws/<name>` exists,
46    /// `wal` = `refs/sturn/wal/<name>` exists; both when both do.
47    pub kinds: Vec<&'static str>,
48    /// WAL events in the log (0 for namespace-only ws).
49    pub events: usize,
50    /// Events since the last `Snapshot` marker — ops still in
51    /// flight, not yet folded into a commit.
52    pub pending: usize,
53    /// WAL tip short sha, when the log exists.
54    pub tip: Option<String>,
55}
56
57/// Everything the views need, loaded once per op.
58pub struct Snapshot {
59    /// Resolved repo dir (may not be a git repo).
60    pub repo: PathBuf,
61    /// Repo id — `repo.id` from config, else dir basename.
62    pub repo_id: String,
63    /// Where `sturn.toml` came from (`<path>`), None = flag/env only.
64    pub config: Option<String>,
65    /// WAL ref prefix in effect (default `refs/sturn/wal`).
66    pub wal_prefix: String,
67    /// Union of ws namespaces + WAL logs.
68    pub workspaces: Vec<WsRow>,
69    /// Every event across every WAL, fold order per ws.
70    pub events: Vec<Event>,
71    /// Live claims at `now` — `sturn claims` output, structured.
72    pub claims: Vec<claims::Active>,
73    /// Every ask and its terminal state — `sturn asks --all`.
74    pub asks: Vec<(sturn_ask::Open, AskStatus)>,
75    /// Last `Presence` status per `(ws, actor)` — `joined` actors
76    /// are the live swarm; `left`/absent are gone.
77    pub presence: BTreeMap<(String, String), String>,
78    /// Unix seconds the snapshot was taken at.
79    pub now: u64,
80    /// Per-source load problems — surfaced, never fatal.
81    pub warnings: Vec<String>,
82    /// Why there is no usable repo state at all.
83    pub error: Option<String>,
84}
85
86impl Default for Snapshot {
87    fn default() -> Self {
88        Snapshot {
89            repo: PathBuf::from("."),
90            repo_id: String::new(),
91            config: None,
92            wal_prefix: "refs/sturn/wal".into(),
93            workspaces: Vec::new(),
94            events: Vec::new(),
95            claims: Vec::new(),
96            asks: Vec::new(),
97            presence: BTreeMap::new(),
98            now: 0,
99            warnings: Vec::new(),
100            error: None,
101        }
102    }
103}

105impl Snapshot {
106    /// The empty/degraded snapshot — `error` says why.
107    fn failed(error: String) -> Self {
108        Snapshot {
109            error: Some(error),
110            ..Snapshot::default()
111        }
112    }
113
114    /// The `{"error": …}` payload every query emits when the
115    /// snapshot couldn't load — the degradation marker pavo
116    /// renderers already understand.
117    pub fn error_payload(&self) -> serde_json::Value {
118        serde_json::json!({"error": self.error.clone().unwrap_or_else(|| "no sturn repo".into())})
119    }
120
121    /// Is there usable repo state?
122    pub fn ok(&self) -> bool {
123        self.error.is_none()
124    }
125}

127fn now() -> u64 {
128    std::time::SystemTime::now()
129        .duration_since(std::time::UNIX_EPOCH)
130        .map(|d| d.as_secs())
131        .unwrap_or(0)
132}

134/// `params` → `(repo_dir, config_path)` overrides. `dir`/`root`/
135/// `repo` all name the repo; `dir` wins when several are given
136/// (mirrors the CLI globals' precedence).
137fn overrides(
138    params: &serde_json::Map<String, serde_json::Value>,
139) -> (Option<PathBuf>, Option<String>) {
140    let repo = ["dir", "root", "repo"]
141        .iter()
142        .find_map(|k| params.get(*k).and_then(|v| v.as_str()))
143        .map(PathBuf::from);
144    let config = params
145        .get("config")
146        .and_then(|v| v.as_str())
147        .map(str::to_string);
148    (repo, config)
149}

151/// Which config file the source chain would read — same candidate
152/// order as `sturn_config::FileSource`, reported for the status view.
153fn config_path_hint(explicit: Option<&str>) -> Option<String> {
154    sturn_config::FileSource {
155        path: explicit.map(str::to_string),
156    }
157    .candidates()
158    .into_iter()
159    .find(|p| p.is_file())
160    .map(|p| p.display().to_string())
161}
162
163/// Load config via the same source chain the CLI uses. Failure is
164/// a warning, not fatal — sturn works flag-driven with no config.
165fn load_config(path: Option<String>) -> (Option<Config>, Vec<String>) {
166    let mut warnings = Vec::new();
167    match (sturn_config::FileSource { path }).load() {
168        Ok(cfg) => (Some(cfg), warnings),
169        Err(e) => {
170            // "no sturn.* found" is the normal case outside a
171            // wired repo — only surface real parse/read errors.
172            if !e.contains("no sturn.") {
173                warnings.push(format!("config: {e}"));
174            }
175            (None, warnings)
176        }
177    }
178}

180/// Where a `cli.<verb>` re-exec should run — the repo dir plus the
181/// config file the same chain `load` uses would read, so engine ops
182/// land exactly where the views see state. Both come back absolute:
183/// the child's cwd is the repo dir, so a relative candidate like
184/// `sturn.toml` would re-resolve against the wrong root.
185pub fn resolve(params: &serde_json::Map<String, serde_json::Value>) -> (PathBuf, Option<String>) {
186    let (param_repo, param_config) = overrides(params);
187    let (cfg, _) = load_config(param_config.clone());
188    let repo = param_repo
189        .or_else(|| {
190            std::env::var("STURN_REPO")
191                .ok()
192                .filter(|s| !s.is_empty())
193                .map(PathBuf::from)
194        })
195        .or_else(|| cfg.as_ref().map(|c| c.repo.dir.clone()))
196        .unwrap_or_else(|| PathBuf::from("."));
197    let repo = std::fs::canonicalize(&repo).unwrap_or(repo);
198    let config = config_path_hint(param_config.as_deref())
199        .map(|p| {
200            std::fs::canonicalize(&p)
201                .map(|a| a.to_string_lossy().into_owned())
202                .unwrap_or(p)
203        })
204        .or(param_config);
205    (repo, config)
206}

208/// Resolve + load. `params` are the wire op's caller params —
209/// `dir`/`root`/`repo`/`config` steer resolution, the rest pass
210/// through to the query resolvers untouched.
211pub fn load(params: &serde_json::Map<String, serde_json::Value>) -> Snapshot {
212    let (param_repo, param_config) = overrides(params);
213    let config = config_path_hint(param_config.as_deref());
214    let (cfg, warnings) = load_config(param_config);
215
216    // Repo resolution — params → $STURN_REPO → cfg.repo.dir → cwd.
217    let repo = param_repo
218        .or_else(|| {
219            std::env::var("STURN_REPO")
220                .ok()
221                .filter(|s| !s.is_empty())
222                .map(PathBuf::from)
223        })
224        .or_else(|| cfg.as_ref().map(|c| c.repo.dir.clone()))
225        .unwrap_or_else(|| PathBuf::from("."));
226
227    // Not a git repo → honest degradation, no WAL reads attempted.
228    if sturn_odb::git_dir(&repo).is_err() {
229        return Snapshot::failed(format!(
230            "{}: not a git repo — sturn state lives in refs/sturn/* under a repo",
231            repo.display()
232        ));
233    }
234
235    let repo_id = cfg
236        .as_ref()
237        .map(|c| c.repo.id())
238        .or_else(|| {
239            repo.canonicalize()
240                .ok()
241                .and_then(|p| p.file_name().map(|s| s.to_string_lossy().into_owned()))
242        })
243        .unwrap_or_else(|| "repo".into());
244    let wal_prefix = cfg
245        .as_ref()
246        .map(|c| c.wal.ref_prefix.clone())
247        .unwrap_or_else(|| "refs/sturn/wal".into());
248
249    let wal = GitWal {
250        ref_prefix: wal_prefix.clone(),
251        // Read-only driver — never signs.
252        sign_key: None,
253    };
254    let ws_impl = sturn_ws_git::GitWs;
255    let now = now();
256    let mut snap = Snapshot {
257        repo,
258        repo_id,
259        config,
260        wal_prefix,
261        now,
262        warnings,
263        ..Snapshot::default()
264    };
265
266    // Workspace union — namespaces (`refs/sturn/ws/*`) plus WAL
267    // logs (`refs/sturn/wal/*`), same as `sturn ws ls`.
268    let mut names: Vec<String> = ws_impl
269        .workspaces(&snap.repo)
270        .map_err(|e| e.to_string())
271        .unwrap_or_else(|e| {
272            snap.warnings.push(format!("ws namespaces: {e}"));
273            Vec::new()
274        });
275    names.extend(wal.workspaces(&snap.repo).unwrap_or_else(|e| {
276        snap.warnings.push(format!("wal workspaces: {e}"));
277        Vec::new()
278    }));
279    names.sort();
280    names.dedup();
281
282    let mut events_by_ws: BTreeMap<String, Vec<Event>> = BTreeMap::new();
283    for name in &names {
284        let mut row = WsRow {
285            name: name.clone(),
286            kinds: Vec::new(),
287            events: 0,
288            pending: 0,
289            tip: None,
290        };
291        if ws_impl.tip(&snap.repo, name).is_ok() {
292            row.kinds.push("namespace");
293        }
294        // `wal.events` answers `Ok(vec![])` for a missing log — the
295        // head check is what distinguishes "wal-only/absent".
296        let head = wal.head(&snap.repo, name).unwrap_or(None);
297        if head.is_some() {
298            row.kinds.push("wal");
299            row.tip = head.map(|h| if h.len() > 8 { h[..8].to_string() } else { h });
300            match wal.events(&snap.repo, name) {
301                Ok(evs) => {
302                    row.events = evs.len();
303                    row.pending = evs
304                        .len()
305                        .saturating_sub(sturn_core::since_last_snapshot(&evs));
306                    events_by_ws.insert(name.clone(), evs);
307                }
308                Err(e) => snap.warnings.push(format!("wal {name}: {e}")),
309            }
310        }
311        snap.workspaces.push(row);
312    }
313
314    // Fold order: per-ws WAL order, concatenated — the same order
315    // `all_events` in the CLI feeds `claims::active`.
316    for evs in events_by_ws.values() {
317        snap.events.extend(evs.iter().cloned());
318    }
319
320    snap.claims = claims::active(&snap.events, now);
321    snap.asks = sturn_ask::all(&snap.events, now);
322    for ev in &snap.events {
323        if let Op::Presence { status } = &ev.op {
324            snap.presence
325                .insert((ev.ws.clone(), ev.actor.clone()), status.clone());
326        }
327    }
328    snap
329}

331/// The five bus topic suffixes under `repo.<id>.` — derived names,
332/// not stored state (the bus is a cache; topics are vocabulary).
333pub fn topic_rows(repo_id: &str) -> Vec<serde_json::Value> {
334    let descs = [
335        (topic::OPS, "every op append — the firehose"),
336        (topic::CLAIMS, "claim/release ops only — collision warnings"),
337        (topic::REVIEW, "approve/reject verdicts on events"),
338        (topic::PRESENCE, "actors joining/leaving workspaces"),
339        (topic::ASKS, "pending-response ops — asks, answers, cancels"),
340    ];
341    descs
342        .iter()
343        .map(|(suffix, desc)| {
344            serde_json::json!({
345                "topic": format!("repo.{repo_id}.{suffix}"),
346                "suffix": suffix,
347                "description": desc,
348            })
349        })
350        .collect()
351}
