1//! hypha-hub — the node-API gateway. Bridges client traffic to the
2//! node daemons and runs the multi-node sync engine:
3//!
4//! - `/api/events` — SSE bridge to `HYPHA_NODE_URL` (EventSource
5//!   can't carry cross-origin auth cleanly; proxying keeps the client
6//!   same-origin).
7//! - `/api/entities/{id}/stream` — per-entity live stream, routed to
8//!   the owning node and shaped by the kind's
9//!   [`hypha_sync::KindPolicy::stream_path`].
10//! - `/api/push/*` — the hub-owned push subscription store (one
11//!   subscription fans out to every paired node).
12//! - `/api/*` — generic passthrough for the rest of the node API,
13//!   entity-scoped calls routed to the owning node on multi-node
14//!   setups (`HYPHA_NODES`).
15//!
16//! [`sync_api::SyncEntityApi`] is the hub's [`EntityApi`]: entity
17//! lists read the merged projection and writes queue in the durable
18//! outbox when a node is down.
19//!
20//! The hub is **policy-injected**: [`kind_policies::default_kind_policies`]
21//! is an empty registry — integrations (peregrin's session plane,
22//! campaign planes, …) supply their [`hypha_sync::KindPolicies`] via
23//! [`hub_state_with`]/[`hub_state_from_with`]. A bare hub still
24//! gateways the whole substrate contract.

48/// Everything the router needs — the node-API port (`EntityApi` is a
49/// trait so tests can stub it) and the upstream address for the
50/// HTTP/SSE bridges.
51/// Entity id → `(url, token, upstream path+query)` for the entity's
52/// live event stream — the kind policy's `stream_path` +
53/// `call_params` resolved against the owning node.
54pub type StreamRouter = Arc<dyn Fn(&str) -> Option<(String, Option<String>, String)> + Send + Sync>;

56#[derive(Clone)]
57pub struct HubState {
58    pub node: Arc<dyn EntityApi>,
59    pub node_url: Arc<str>,
60    pub node_token: Option<Arc<str>>,
61    /// Live-stream routing — `(url, token, path)` per entity id.
62    /// `None` → `/api/entities/{id}/stream` 404s.
63    pub stream_router: Option<StreamRouter>,
64    /// Hub-owned push subscription store — `None` → `/api/push/*` 501s.
65    pub push: Option<Arc<hypha_push::PushStore>>,
66    /// `HYPHA_HUB_TOKEN` — the client→hub gate. `None` → every route
67    /// is open (loopback-only deployments).
68    pub hub_token: Option<Arc<str>>,
69    /// Long-lived streams: no global deadline and no body budget —
70    /// ureq's `recv_body` is a total deadline that fails every read
71    /// instantly once spent; a half-dead client is instead detected
72    /// when the next frame's `blocking_send` fails.
73    sse_agent: ureq::Agent,
74    /// Bounded request/response passthroughs.
75    http_agent: ureq::Agent,
76}
77
78impl HubState {
79    /// Build the state from parts. `node_url` is the node's origin
80    /// (`http://127.0.0.1:18787`); `node_token` is forwarded as a bearer
81    /// token on hub → node calls, falling back to the caller's own
82    /// `Authorization` header when unset.
83    pub fn new(
84        node: Arc<dyn EntityApi>,
85        node_url: impl Into<Arc<str>>,
86        node_token: Option<Arc<str>>,
87    ) -> Self {
88        let sse_agent = ureq::Agent::new_with_config(
89            ureq::Agent::config_builder()
90                .http_status_as_error(false)
91                .timeout_global(None)
92                .timeout_connect(Some(std::time::Duration::from_secs(10)))
93                .timeout_recv_body(None)
94                .build(),
95        );
96        let http_agent = ureq::Agent::new_with_config(
97            ureq::Agent::config_builder()
98                .http_status_as_error(false)
99                .timeout_global(Some(std::time::Duration::from_secs(60)))
100                .build(),
101        );
102        Self {
103            node,
104            node_url: node_url.into(),
105            node_token,
106            stream_router: None,
107            push: None,
108            hub_token: None,
109            sse_agent,
110            http_agent,
111        }
112    }
113
114    /// Agent configured for open-ended SSE reads.
115    pub(crate) fn sse_agent(&self) -> ureq::Agent {
116        self.sse_agent.clone()
117    }
118
119    /// Agent configured for bounded JSON calls.
120    pub(crate) fn json_agent(&self) -> ureq::Agent {
121        self.http_agent.clone()
122    }
123}

125/// Runtime configuration, all `HYPHA_*` env.
126#[derive(Clone, Debug)]
127pub struct HubConfig {
128    /// `HYPHA_HUB_HOST` — default `127.0.0.1`.
129    pub host: String,
130    /// `HYPHA_HUB_PORT` — default `13000`.
131    pub port: u16,
132    /// `HYPHA_NODE_URL` — default `http://127.0.0.1:18787`.
133    pub node_url: String,
134    /// `HYPHA_NODE_TOKEN` — bearer token forwarded hub → node.
135    pub node_token: Option<String>,
136    /// `HYPHA_HUB_TOKEN` — client→hub auth; required on non-loopback
137    /// binds, honored as bearer header or `?token=` query.
138    pub hub_token: Option<String>,
139}

141impl Default for HubConfig {
142    fn default() -> Self {
143        Self {
144            host: "127.0.0.1".into(),
145            port: 13000,
146            node_url: "http://127.0.0.1:18787".into(),
147            node_token: None,
148            hub_token: None,
149        }
150    }

153/// Loopback binds may serve tokenless; anything else requires
154/// `HYPHA_HUB_TOKEN` (mirrors `hypha_http::env::is_loopback_host`).
155fn is_loopback_host(host: &str) -> bool {
156    hypha_http::env::is_loopback_host(host)
157}
158
159impl HubConfig {
160    /// Parse from env. `Err` on a malformed port/address or a
161    /// non-loopback bind without `HYPHA_HUB_TOKEN` — fail fast.
162    pub fn from_env() -> Result<Self, String> {
163        Self::from_lookup(|k| std::env::var(k).ok())
164    }
165
166    /// Test seam — parse from a key/value map instead of `std::env`.
167    pub fn from_map(vars: &HashMap<String, String>) -> Result<Self, String> {
168        Self::from_lookup(|key| vars.get(key).cloned())
169    }
170
171    /// Shared parser behind [`from_env`](Self::from_env).
172    pub fn from_lookup(get: impl Fn(&str) -> Option<String>) -> Result<Self, String> {
173        let mut cfg = Self::default();
174        if let Some(v) = get("HYPHA_HUB_HOST") {
175            cfg.host = v;
176        }
177        if let Some(v) = get("HYPHA_HUB_PORT").or_else(|| get("HYPHA_PORT")) {
178            cfg.port = v.parse::<u16>().map_err(|_| {
179                format!("HYPHA_HUB_PORT/HYPHA_PORT must be a port number, got {v:?}")
180            })?;
181        }
182        if let Some(v) = get("HYPHA_NODE_URL") {
183            cfg.node_url = v.trim_end_matches('/').to_string();
184        }
185        cfg.node_token = get("HYPHA_NODE_TOKEN").filter(|t| !t.is_empty());
186        cfg.hub_token = get("HYPHA_HUB_TOKEN").filter(|t| !t.is_empty());
187        // The node's bind guard, applied to the client surface: a
188        // wildcard/LAN bind with no hub token would expose the API.
189        if !is_loopback_host(&cfg.host) && cfg.hub_token.is_none() {
190            return Err(format!(
191                "hypha-hub refuses to bind {} without HYPHA_HUB_TOKEN. Set HYPHA_HUB_TOKEN or bind a loopback address (HYPHA_HUB_HOST=127.0.0.1).",
192                cfg.host
193            ));
194        }
195        Ok(cfg)
196    }
197
198    pub fn socket_addr(&self) -> Result<SocketAddr, String> {
199        format!("{}:{}", self.host, self.port)
200            .parse()
201            .map_err(|_| format!("invalid hub address {}:{}", self.host, self.port))
202    }
203}

205/// Build the axum router. `HubConfig` → `HubState` happens in
206/// [`hub_state`]; tests construct `HubState` directly with a stub
207/// `EntityApi`.
208pub fn router(state: HubState) -> Router {
209    Router::new()
210        // The merged entity surface — substrate reads hit the
211        // projection, ops ride the outbox. Must be registered before
212        // the `/api/*` passthrough so they win over single-node routing.
213        // Canonical spellings only — `hypha-node-api`'s `HttpEntityApi`
214        // and `hypha-http`'s node routes speak the same paths.
215        .route("/api/entities", any(entity_routes::collection))
216        .route(
217            "/api/entities/snapshot",
218            any(entity_routes::snapshot_collection),
219        )
220        .route("/api/entities/{id}", any(entity_routes::item))
221        .route("/api/entities/{id}/ops", any(entity_routes::ops))
222        .route(
223            "/api/entities/{id}/snapshot",
224            any(entity_routes::snapshot_item),
225        )
226        .route("/api/entity-index", get(entity_routes::entity_index))
227        .route("/api/node-status", get(entity_routes::node_status))
228        .route("/api/pending-ops", get(entity_routes::pending_ops))
229        // SSE bridges — must be registered before the `/api/*` wildcard.
230        .route("/api/events", get(proxy::sse_events))
231        .route("/api/entities/{id}/stream", get(proxy::sse_entity_stream))
232        // Hub-owned push — intercepted before the passthrough so every
233        // node notifies through one subscription store.
234        .route("/api/push/vapid", any(push_routes::vapid))
235        .route("/api/push/subscribe", any(push_routes::subscribe))
236        .route("/api/{*rest}", any(proxy::passthrough))
237        // Hard ceiling for *every* request body. Sized at the
238        // passthrough cap so legit node-bound bodies never hit it.
239        .layer(tower_http::limit::RequestBodyLimitLayer::new(
240            proxy::MAX_PROXY_BODY,
241        ))
242        // Client gate — a no-op until `hub_token` is set.
243        .layer(middleware::from_fn_with_state(
244            state.clone(),
245            auth::auth_gate,
246        ))
247        // OTEL server span — outermost: client→hub requests get an
248        // `http_request` span the proxy's `traceparent` injection links
249        // hub→node calls into. Inert when export is off.
250        .layer(middleware::from_fn(trace_layer))
251        .with_state(state)
252}

254/// OTEL server span — extracts any inbound `traceparent` (federated
255/// callers), records the response status onto the span.
256async fn trace_layer(req: Request<Body>, next: Next) -> Response {
257    let span = hypha_telemetry::server_span(req.method(), req.uri().path(), req.headers());
258    let for_status = span.clone();
259    async move {
260        let res = next.run(req).await;
261        hypha_telemetry::record_response(&for_status, res.status().as_u16());
262        res
263    }
264    .instrument(span)
265    .await
266}

268/// HubState for a running hub: the sync engine + [`SyncEntityApi`](crate::sync_api::SyncEntityApi).
269/// `HYPHA_NODES` (`id=url[@token];…`) wins; `HYPHA_NODE_URL`/`HYPHA_NODE_TOKEN`
270/// is the single-node shorthand. Nodes listed without a token resolve
271/// one from the [`credentials`] store (OS keyring → file fallback —
272/// that store is also where the pair flow persists minted tokens), and
273/// an env-provided token seeds an empty store slot so the plaintext
274/// can leave the env. The projection + outbox live under
275/// `$HYPHA_HOME/hub/` so they survive restarts.
276///
277/// Boots with [`kind_policies::default_kind_policies`] — the empty
278/// registry. Integrations inject theirs via [`hub_state_with`].
279///
280/// # Errors
281/// `String` on a missing/empty node list or a projection/outbox open failure.
282pub fn hub_state(config: &HubConfig) -> Result<HubState, String> {
283    hub_state_with(config, kind_policies::default_kind_policies())
284}

308/// [`hub_state_with`] with the environment injected — the full test
309/// seam (env map + policy registry).
310#[doc(hidden)]
311pub fn hub_state_from_with(
312    get: impl Fn(&str) -> Option<std::ffi::OsString>,
313    config: &HubConfig,
314    policies: KindPolicies,
315) -> Result<HubState, String> {
316    let home = get("HYPHA_HOME")
317        .map(std::path::PathBuf::from)
318        .or_else(|| get("HOME").map(|h| std::path::PathBuf::from(h).join(".local/share/hypha")))
319        .unwrap_or_else(|| std::path::PathBuf::from(".hypha"))
320        .join("hub");
321    std::fs::create_dir_all(&home).map_err(|e| format!("create {}: {e}", home.display()))?;
322    // Credential-store selection (`HYPHA_CREDENTIAL_STORE`) rides the
323    // same injected env as the node list — integrations mapping their
324    // own env names (peregrin's `PEREGRIN_*`) get it for free.
325    let creds = credentials::from_lookup(&home, |k| get(k).and_then(|v| v.into_string().ok()));
326    let mut nodes = get("HYPHA_NODES")
327        .and_then(|v| v.into_string().ok())
328        .filter(|v| !v.trim().is_empty())
329        .map(|raw| sync_api::parse_nodes(&raw))
330        .filter(|v| !v.is_empty())
331        .unwrap_or_else(|| {
332            vec![sync_api::HubNode {
333                id: "node".to_string(),
334                url: config.node_url.clone(),
335                token: config.node_token.clone(),
336            }]
337        });
338    // Env tokens are explicit config — they win, and seed an empty
339    // store slot so the plaintext can leave the env. Tokenless nodes
340    // resolve a stored credential.
341    let (stored, seeded) = sync_api::resolve_node_tokens(&mut nodes, creds.as_ref());
342    // Hoisted — `tracing` fields evaluate only when a subscriber is
343    // attached, which coverage runs don't guarantee.
344    let backend = creds.backend();
345    tracing::info!(backend, stored, seeded, "node credential store");
346    let projection = hypha_sync::ProjectionStore::open(&home.join("projection.db"))
347        .map_err(|e| format!("projection open: {e}"))?
348        .with_kind_policies(policies.clone());
349    // Rows from node ids no longer configured (renamed, removed, or the
350    // legacy `node` shorthand) have no sync loop — purge them once at
351    // boot so dead entities stop surfacing in the index.
352    let keep: Vec<String> = nodes.iter().map(|n| n.id.clone()).collect();
353    if let Err(e) = projection.purge_unregistered(&keep) {
354        tracing::warn!("purge_unregistered failed: {e}");
355    }
356    let outbox = hypha_outbox::Outbox::open(&home.join("outbox.db")).map_err(|e| e.to_string())?;
357    let targets = nodes.iter().map(|n| {
358        let node = hypha_sync::NodeRef {
359            id: n.id.clone(),
360            url: n.url.clone(),
361            label: n.id.clone(),
362        };
363        match &n.token {
364            Some(t) => hypha_sync::engine::NodeTarget::with_token(node, t.clone()),
365            None => hypha_sync::engine::NodeTarget::new(node),
366        }
367    });
368    let push = Arc::new(hypha_push::PushStore::open(hypha_push::JsonDocStore::open(
369        &home.join("push.json"),
370    )));
371    // One registry, three consumers: the sync engine (diffs, op
372    // params), the feed hook (push rules), and the entity API (gating,
373    // creates, actors, stream paths).
374    // `HYPHA_EPHEMERAL_DEAD_AFTER` (seconds) tunes how long a
375    // self-declared ephemeral node may stay down before its entities
376    // are reaped — CI runners that never come back shouldn't pin
377    // projections forever. Unset/invalid → `SyncOptions` default.
378    let mut options = hypha_sync::engine::SyncOptions {
379        on_event: Some(feed_hook(projection.clone(), Arc::clone(&push), &policies)),
380        policies: policies.clone(),
381        ..hypha_sync::engine::SyncOptions::default()
382    };
383    if let Some(secs) = get("HYPHA_EPHEMERAL_DEAD_AFTER")
384        .and_then(|v| v.into_string().ok())
385        .and_then(|v| v.trim().parse::<u64>().ok())
386    {
387        options.ephemeral_dead_after = std::time::Duration::from_secs(secs);
388    }
389    let engine = hypha_sync::SyncEngine::spawn(projection, outbox, targets, options);
390    // `unwrap_or_else` above guarantees at least the single-node row.
391    let upstream = nodes.first().cloned().unwrap_or_default();
392    let api = Arc::new(sync_api::SyncEntityApi::new(engine, nodes, creds).with_policies(policies));
393    let mut state = HubState::new(
394        Arc::clone(&api) as Arc<dyn EntityApi>,
395        upstream.url.as_str(),
396        upstream.token.clone().map(Into::into),
397    );
398    state.push = Some(push);
399    state.hub_token = config.hub_token.clone().map(Into::into);
400    let stream_api = Arc::clone(&api);
401    state.stream_router = Some(Arc::new(move |id| stream_api.stream_upstream(id)));
402    Ok(state)
403}

405/// The hub's push-notification feed hook — a diff event a kind
406/// policy claims produces that kind's [`hypha_sync::Notice`]s,
407/// delivered as Web Push. Which events notify and what they say is
408/// the policy's call; this body is the generic dispatch. Extracted so
409/// tests can drive it directly (the live hook only fires on real
410/// node feeds).
411#[doc(hidden)]
412pub fn feed_hook(
413    projection: hypha_sync::ProjectionStore,
414    push: Arc<hypha_push::PushStore>,
415    policies: &KindPolicies,
416) -> hypha_sync::engine::FeedHook {
417    let policies = policies.clone();
418    Arc::new(
419        move |node_id: &str, event: &hypha_sync::client::FeedEvent| {
420            let hypha_sync::client::FeedEvent::Diff {
421                kind, id, patch, ..
422            } = event
423            else {
424                return;
425            };
426            let Some(policy) = policies.for_diff_event(kind) else {
427                return;
428            };
429            let row = projection.entity(node_id, id).ok().flatten();
430            for notice in policy.notices(kind, id, row.as_ref(), patch) {
431                let kind = match notice.kind {
432                    hypha_sync::NoticeKind::Done => hypha_push::Kind::Done,
433                    hypha_sync::NoticeKind::Permission => hypha_push::Kind::Permission,
434                };
435                push.send(kind, &notice.title, &notice.body, &notice.url);
436            }
437        },
438    )
