1//! hypha-kv-file — the JSON-file [`KvStore`] impl, porting
2//! `sepia_meta::MetaStore`'s persistence mechanics: one document,
3//! rewritten atomically (tmp + rename) after every mutation, and a
4//! corrupt or half-written file degrades to empty rather than
5//! failing.
6//!
7//! On-disk shape:
8//!
9//! ```json
10//! {
11//!   "namespaces": {
12//!     "config": { "visibility": "public", "entries": { "theme": "dark" } },
13//!     "push":   { "visibility": "internal", "entries": { "vapid": {} } }
14//!   }
15//! }
16//! ```
17//!
18//! Reads are tolerant in both directions: a top-level object without
19//! a `namespaces` key is read as the flat `{ns: {key: value}}` shape
20//! (every top-level object value becomes a public namespace's
21//! entries), which is the shape a migrated `meta.json` `config`/
22//! `projects` section already has. Writes always emit the canonical
23//! `namespaces` form.
24
25use std::collections::BTreeMap;
26use std::path::{Path, PathBuf};
27use std::sync::{Arc, Mutex, MutexGuard};
28
29use hypha_kv::{KvStore, NamespaceInfo, Visibility};
30use serde::{Deserialize, Serialize};
31use serde_json::Value;
32
33fn lock_inner(mutex: &Mutex<Inner>) -> MutexGuard<'_, Inner> {
34    mutex
35        .lock()
36        .unwrap_or_else(std::sync::PoisonError::into_inner)
37}
38
39/// One namespace on disk: its wire visibility plus the entry map.
40#[derive(Clone, Debug, Default, Serialize, Deserialize)]
41struct Namespace {
42    #[serde(default)]
43    visibility: Visibility,
44    #[serde(default)]
45    entries: BTreeMap<String, Value>,
46}
47
48/// The canonical document — `namespaces` keyed by namespace name.
49#[derive(Default, Serialize)]
50struct KvFile {
51    namespaces: BTreeMap<String, Namespace>,
52}
53
54struct Inner {
55    path: PathBuf,
56    namespaces: BTreeMap<String, Namespace>,
57}
58
59/// A [`KvStore`] over one JSON file. `Clone` shares the same
60/// in-memory map and path.
61#[derive(Clone)]
62pub struct JsonKvStore {
63    inner: Arc<Mutex<Inner>>,
64}
65
66impl JsonKvStore {
67    /// Open the store at `path` (created on first write). A missing,
68    /// unreadable, or corrupt file degrades to empty.
69    #[must_use]
70    pub fn open(path: &Path) -> Self {
71        let namespaces = read_namespaces(path);
72        Self {
73            inner: Arc::new(Mutex::new(Inner {
74                path: path.to_path_buf(),
75                namespaces,
76            })),
77        }
78    }
79
80    /// The file this store flushes to.
81    #[must_use]
82    pub fn path(&self) -> PathBuf {
83        self.inner
84            .lock()
85            .unwrap_or_else(std::sync::PoisonError::into_inner)
86            .path
87            .clone()
88    }
89
90    fn flush(inner: &Inner) {
91        if let Some(dir) = inner.path.parent() {
92            let _ = std::fs::create_dir_all(dir);
93        }
94        let tmp = inner
95            .path
96            .with_extension(format!("tmp-{}", std::process::id()));
97        let doc = KvFile {
98            namespaces: inner.namespaces.clone(),
99        };
100        let json = serde_json::to_vec(&doc).unwrap_or_default();
101        if std::fs::write(&tmp, &json).is_ok() {
102            let _ = std::fs::rename(&tmp, &inner.path);
103        }
104    }
105
106    /// Mutable access to one namespace's entries, creating it public
107    /// when undeclared — callers flush after mutating.
108    fn entries_mut<'a>(
109        namespaces: &'a mut BTreeMap<String, Namespace>,
110        ns: &str,
111    ) -> &'a mut BTreeMap<String, Value> {
112        &mut namespaces.entry(ns.to_string()).or_default().entries
113    }
114}
115
116/// Parse the file into the namespace map — canonical `namespaces`
117/// shape first, flat `{ns: {entries}}` as the tolerant fallback.
118fn read_namespaces(path: &Path) -> BTreeMap<String, Namespace> {
119    let Ok(raw) = std::fs::read_to_string(path) else {
120        return BTreeMap::new();
121    };
122    let Ok(Value::Object(record)) = serde_json::from_str::<Value>(&raw) else {
123        return BTreeMap::new();
124    };
125    if let Some(Value::Object(ns_map)) = record.get("namespaces") {
126        return ns_map
127            .iter()
128            .filter_map(|(name, value)| {
129                let ns: Namespace = serde_json::from_value(value.clone()).ok()?;
130                Some((name.clone(), ns))
131            })
132            .collect();
133    }
134    // Flat shape: every top-level object value is one namespace's
135    // entry map (public by default).
136    record
137        .iter()
138        .filter_map(|(name, value)| {
139            value.as_object().map(|entries| {
140                (
141                    name.clone(),
142                    Namespace {
143                        visibility: Visibility::Public,
144                        entries: entries
145                            .iter()
146                            .map(|(k, v)| (k.clone(), v.clone()))
147                            .collect(),
148                    },
149                )
150            })
151        })
152        .collect()
153}
154
155impl KvStore for JsonKvStore {
156    fn get(&self, ns: &str, key: &str) -> Option<Value> {
157        lock_inner(&self.inner)
158            .namespaces
159            .get(ns)
160            .and_then(|n| n.entries.get(key))
161            .cloned()
162    }
163
164    fn set(&self, ns: &str, key: &str, value: Value) {
165        let mut inner = lock_inner(&self.inner);
166        Self::entries_mut(&mut inner.namespaces, ns).insert(key.to_string(), value);
167        Self::flush(&inner);
168    }
169
170    fn delete(&self, ns: &str, key: &str) -> bool {
171        let mut inner = lock_inner(&self.inner);
172        let removed = inner
173            .namespaces
174            .get_mut(ns)
175            .is_some_and(|n| n.entries.remove(key).is_some());
176        if removed {
177            Self::flush(&inner);
178        }
179        removed
180    }
181
182    fn list(&self, ns: &str) -> BTreeMap<String, Value> {
183        lock_inner(&self.inner)
184            .namespaces
185            .get(ns)
186            .map(|n| n.entries.clone())
187            .unwrap_or_default()
188    }
189
190    fn namespaces(&self) -> Vec<NamespaceInfo> {
191        lock_inner(&self.inner)
192            .namespaces
193            .iter()
194            .map(|(name, ns)| NamespaceInfo {
195                name: name.clone(),
196                visibility: ns.visibility,
197            })
198            .collect()
199    }
200
201    fn visibility(&self, ns: &str) -> Visibility {
202        lock_inner(&self.inner)
203            .namespaces
204            .get(ns)
205            .map_or(Visibility::Public, |ns| ns.visibility)
206    }
207
208    fn declare(&self, ns: &str, visibility: Visibility) {
209        let mut inner = lock_inner(&self.inner);
210        inner
211            .namespaces
212            .entry(ns.to_string())
213            .or_default()
214            .visibility = visibility;
215        Self::flush(&inner);
216    }
217
218    fn import(&self, ns: &str, entries: BTreeMap<String, Value>, visibility: Visibility) {
219        let mut inner = lock_inner(&self.inner);
220        let namespace = inner.namespaces.entry(ns.to_string()).or_default();
221        namespace.visibility = visibility;
222        namespace.entries.extend(entries);
223        Self::flush(&inner);
224    }
225}

227#[cfg(test)]
228mod tests {
229    #![allow(clippy::unwrap_used, clippy::assert_is_empty)]
230    use super::*;

232    fn temp_store() -> (tempfile::TempDir, JsonKvStore) {
233        let dir = tempfile::tempdir().unwrap();
234        let store = JsonKvStore::open(&dir.path().join("kv.json"));
235        (dir, store)
236    }
237
238    #[test]
239    fn crud_roundtrip() {
240        let (_dir, store) = temp_store();
241        assert_eq!(store.get("config", "theme"), None);
242        store.set("config", "theme", Value::String("dark".into()));
243        store.set("config", "scale", Value::from(2));
244        assert_eq!(
245            store.get("config", "theme"),
246            Some(Value::String("dark".into()))
247        );
248        assert_eq!(store.list("config").len(), 2);
249        assert!(store.delete("config", "theme"));
250        assert!(!store.delete("config", "theme"));
251        assert_eq!(store.get("config", "theme"), None);
252        assert_eq!(store.list("missing").len(), 0);
253    }
254
255    #[test]
256    fn namespaces_are_independent() {
257        let (_dir, store) = temp_store();
258        store.set("config", "k", Value::from(1));
259        store.set("project", "p1", serde_json::json!({ "name": "Web" }));
260        assert_eq!(store.list("config").len(), 1);
261        assert_eq!(store.list("project").len(), 1);
262        let names: Vec<String> = store.namespaces().iter().map(|n| n.name.clone()).collect();
263        assert_eq!(names, vec!["config".to_string(), "project".to_string()]);
264        assert!(
265            store
266                .namespaces()
267                .iter()
268                .all(|n| n.visibility == Visibility::Public)
269        );
270    }
271
272    #[test]
273    fn visibility_declared_and_preserved() {
274        let (_dir, store) = temp_store();
275        store.declare("push", Visibility::Internal);
276        assert_eq!(store.visibility("push"), Visibility::Internal);
277        assert_eq!(store.visibility("undeclared"), Visibility::Public);
278        // Store API ignores visibility — internal entries still read.
279        store.set("push", "vapid", serde_json::json!({ "k": "v" }));
280        assert!(store.get("push", "vapid").is_some());
281        // Deleting the last entry keeps the namespace + visibility.
282        assert!(store.delete("push", "vapid"));
283        assert_eq!(store.visibility("push"), Visibility::Internal);
284        let info = store
285            .namespaces()
286            .into_iter()
287            .find(|n| n.name == "push")
288            .unwrap();
289        assert_eq!(info.visibility, Visibility::Internal);
290    }
291
292    #[test]
293    fn import_merges_entries_and_sets_visibility() {
294        let (_dir, store) = temp_store();
295        store.set("config", "keep", Value::from(true));
296        let mut entries = BTreeMap::new();
297        entries.insert("theme".to_string(), Value::String("dark".into()));
298        store.import("config", entries, Visibility::Internal);
299        assert_eq!(store.get("config", "keep"), Some(Value::from(true)));
300        assert_eq!(store.get("config", "theme"), Some("dark".into()));
301        assert_eq!(store.visibility("config"), Visibility::Internal);
302    }

304    #[test]
305    fn persists_across_reopen() {
306        let dir = tempfile::tempdir().unwrap();
307        let path = dir.path().join("kv.json");
308        {
309            let store = JsonKvStore::open(&path);
310            store.set("config", "theme", Value::String("dark".into()));
311            store.declare("push", Visibility::Internal);
312            store.set("push", "vapid", serde_json::json!({ "pub": "x" }));
313        }
314        let store = JsonKvStore::open(&path);
315        assert_eq!(
316            store.get("config", "theme"),
317            Some(Value::String("dark".into()))
318        );
319        assert_eq!(store.visibility("push"), Visibility::Internal);
320        assert!(store.get("push", "vapid").is_some());
321    }

323    #[test]
324    fn corrupt_file_degrades_to_empty() {
325        let dir = tempfile::tempdir().unwrap();
326        let path = dir.path().join("kv.json");
327        std::fs::write(&path, b"{ not json").unwrap();
328        let store = JsonKvStore::open(&path);
329        assert!(store.namespaces().is_empty());
330        // And the store still works — first write rewrites the file.
331        store.set("config", "k", Value::Null);
332        assert!(path.exists());
333    }
