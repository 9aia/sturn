1//! myco-testkit — golden fixtures + conformance for the `project/`
2//! file spec. `fixtures/` holds canonical and malformed files;
3//! [`strat`] holds proptest strategies; later crates reuse both
4//! (conformance is shared, not per-crate).
5//!
6//! Tests here freeze the spec:
7//!
8//! - insta snapshots — `emit(parse(fixture))` byte-for-byte
9//! - proptest — `parse(emit(doc)) == doc` and emit idempotence
10
11use std::path::{Path, PathBuf};
12
13/// Absolute path to `fixtures/` (works from any test cwd).
14pub fn fixtures_dir() -> PathBuf {
15    Path::new(env!("CARGO_MANIFEST_DIR")).join("fixtures")
16}
17
18/// Read a fixture file, e.g. `fixture("todo/basic.md")`.
19pub fn fixture(rel: &str) -> String {
20    let path = fixtures_dir().join(rel);
21    std::fs::read_to_string(&path).unwrap_or_else(|e| panic!("fixture {rel} unreadable: {e}"))
22}
23
24/// Fixture paths (relative to `fixtures/`) under `dir`
25/// (e.g. `"todo/invalid"`), sorted, `*.md`/`*.toml` only.
26pub fn fixtures_under(dir: &str) -> Vec<String> {
27    let root = fixtures_dir();
28    let mut out = Vec::new();
29    collect(&root, &root.join(dir), &mut out);
30    out.sort();
31    out
32}
33
34fn collect(root: &Path, dir: &Path, out: &mut Vec<String>) {
35    let Ok(read) = std::fs::read_dir(dir) else {
36        return;
37    };
38    for entry in read.flatten() {
39        let path = entry.path();
40        if path.is_dir() {
41            collect(root, &path, out);
42        } else if matches!(
43            path.extension().and_then(|e| e.to_str()),
44            Some("md") | Some("toml")
45        ) {
46            let rel = path
47                .strip_prefix(root)
48                .unwrap_or(&path)
49                .to_string_lossy()
50                .replace('\\', "/");
51            out.push(rel);
52        }
53    }
54}

56/// proptest strategies for `myco-core` models — constrained to what
57/// the canonical grammar can express (see each fn's docs).
58pub mod strat {
59    use myco_core::{
60        Actor, BacklogDoc, DoneDoc, HandoffEntry, Item, ItemFile, ItemId, ItemKind, NextCounters,
61        Priority, ProjectMeta, Section, Status, TodoDoc,
62    };
63    use proptest::prelude::*;
64    use time::Date;
65
66    /// Any kind.
67    pub fn item_kind() -> impl Strategy<Value = ItemKind> {
68        prop::sample::select(ItemKind::ALL.to_vec())
69    }
70
71    /// A file-backed kind (no `Task` — tasks live in TODO/BACKLOG).
72    pub fn file_kind() -> impl Strategy<Value = ItemKind> {
73        prop::sample::select(vec![
74            ItemKind::Issue,
75            ItemKind::Proposal,
76            ItemKind::Adr,
77            ItemKind::Plan,
78        ])
79    }
80
81    /// `1..=9999` — padded ids stay in spec range.
82    pub fn item_id(kind: ItemKind) -> impl Strategy<Value = ItemId> {
83        (1u32..=9999).prop_map(move |n| ItemId::new(kind, n))
84    }
85
86    /// Any status.
87    pub fn status() -> impl Strategy<Value = Status> {
88        prop::sample::select(Status::ALL.to_vec())
89    }
90
91    /// Any priority.
92    pub fn priority() -> impl Strategy<Value = Priority> {
93        prop::sample::select(Priority::ALL.to_vec())
94    }
95
96    /// `[a-z0-9._-]{1,12}` optionally `human:`/`agent:` prefixed.
97    pub fn actor() -> impl Strategy<Value = Actor> {
98        (
99            prop::sample::select(vec!["", "human:", "agent:"]),
100            "[a-z0-9._-]{1,12}",
101        )
102            .prop_map(|(p, s)| Actor::new(&format!("{p}{s}")).expect("valid actor"))
103    }
104
105    /// `[a-z0-9_-]{1,10}` — a valid `#label`.
106    pub fn label() -> impl Strategy<Value = String> {
107        "[a-z0-9_-]{1,10}"
108    }
109
110    /// A `YYYY-MM-DD` date in the 2000s.
111    pub fn date() -> impl Strategy<Value = Date> {
112        (2000i32..=2099, 1u8..=12, 1u8..=28).prop_map(|(y, m, d)| {
113            Date::from_calendar_date(y, time::Month::try_from(m).expect("1..=12"), d)
114                .expect("valid date")
115        })
116    }
117
118    /// Trimmed, nonempty, no `·`, no newlines — a legal title.
119    /// Includes markdown specials to exercise inline escaping.
120    pub fn title() -> impl Strategy<Value = String> {
121        "[a-zA-Z0-9*#_\\[\\](),.:' -]{1,60}".prop_map(|s| s.trim().to_string())
122    }
123
124    /// Optional trimmed section name — same charset as titles,
125    /// including `#` so emit's ATX-heading escaping is exercised
126    /// (`x ###` must survive a `## ` line byte-for-byte).
127    pub fn section_name() -> impl Strategy<Value = String> {
128        "[a-zA-Z0-9*#_\\[\\](),.:' -]{1,20}"
129            .prop_map(|s| s.trim().to_string())
130            .prop_filter("nonempty after trim", |s| !s.is_empty())
131    }
132
133    /// `[a-z0-9._-]{1,12}` — a valid `epic:`/`ms:` group name.
134    pub fn group_name() -> impl Strategy<Value = String> {
135        "[a-z0-9._-]{1,12}"
136    }
137
138    /// A line-safe `note:` value — trimmed text without `·` or
139    /// newlines (`src/x.rs:12`-style provenance strings).
140    pub fn line_note() -> impl Strategy<Value = String> {
141        "[a-z0-9 .:/_-]{1,40}".prop_map(|s| s.trim().to_string())
142    }
143
144    /// The scheduling fields as one strategy — `(due, est, cycle,
145    /// done)` — keeping the item tuples under proptest's arity
146    /// cap.
147    fn sched_fields(
148    ) -> impl Strategy<Value = (Option<Date>, Option<u32>, Option<String>, Option<Date>)> {
149        (
150            prop::option::of(date()),
151            prop::option::of(0u32..=myco_core::MAX_ESTIMATE),
152            prop::option::of(group_name()),
153            prop::option::of(date()),
154        )
155    }
156
157    /// A leaf line item — full metadata, no children.
158    fn leaf_item(forbid: &'static [Status]) -> impl Strategy<Value = Item> {
159        (
160            item_id(ItemKind::Task),
161            title(),
162            prop::sample::select(
163                Status::ALL
164                    .into_iter()
165                    .filter(|s| !forbid.contains(s))
166                    .collect::<Vec<_>>(),
167            ),
168            priority(),
169            prop::option::of(actor()),
170            prop::collection::btree_set(label(), 0..3),
171            prop::collection::btree_set(item_id(ItemKind::Task), 0..3),
172            prop::option::of(date()),
173            prop::option::of(group_name()),
174            prop::option::of(group_name()),
175            sched_fields(),
176            prop::option::of(line_note()),
177        )
178            .prop_map(
179                |(
180                    id,
181                    title,
182                    status,
183                    priority,
184                    owner,
185                    labels,
186                    blocks,
187                    created,
188                    epic,
189                    ms,
190                    (due, est, cycle, done),
191                    note,
192                )| {
193                    let mut item = Item::new(id, title).status(status).priority(priority);
194                    item.owner = owner;
195                    item.labels = labels;
196                    item.blocks = blocks;
197                    // an item can't block itself — emit rejects it
198                    item.blocks.remove(&item.id);
199                    item.created = created;
200                    item.epic = epic;
201                    item.milestone = ms;
202                    item.due = due;
203                    item.estimate = est;
204                    item.cycle = cycle;
205                    item.done = done;
206                    item.note = note.filter(|n| !n.is_empty());
207                    item
208                },
209            )
210    }
211
212    /// A task item valid for a line grammar — notes are the
213    /// trimmed `note:` token kind only, and status avoids `forbid`
214    /// at every level. Children are full items too; generated
215    /// depth is ≤ 2 (one nesting level — the recommended shape,
216    /// deeper nesting is legal but rare).
217    pub fn line_item(forbid: &'static [Status]) -> impl Strategy<Value = Item> {
218        (
219            leaf_item(forbid),
220            prop::collection::vec(leaf_item(forbid), 0..3),
221        )
222            .prop_map(|(mut item, children)| {
223                item.children = children;
224                item
225            })
226    }
227
228    /// Set `status` on a whole subtree (deferred propagates to
229    /// children — BACKLOG.md is deferred-only).
230    fn status_tree(item: &mut Item, status: Status) {
231        item.status = status;
232        for c in &mut item.children {
233            status_tree(c, status);
234        }
235    }
236
237    fn sections(items: impl Strategy<Value = Item>) -> impl Strategy<Value = Vec<Section>> {
238        prop::collection::vec(
239            (
240                prop::option::of(section_name()),
241                prop::collection::vec(items, 0..4),
242            ),
243            0..4,
244        )
245        .prop_map(|v| {
246            let mut out: Vec<Section> = Vec::new();
247            let mut first_unsectioned = true;
248            for (name, items) in v {
249                match name {
250                    Some(n) => out.push(Section {
251                        name: Some(n),
252                        items,
253                    }),
254                    None => {
255                        // unsectioned only legal at index 0, and
256                        // needs items to be representable
257                        if first_unsectioned && out.is_empty() && !items.is_empty() {
258                            out.push(Section { name: None, items });
259                            first_unsectioned = false;
260                        }
261                    }
262                }
263            }
264            out
265        })
266    }
267
268    /// A `TodoDoc` — statuses exclude the terminals (`Done`,
269    /// `Cancelled` — both archive into DONE.md) and `Deferred`
270    /// (BACKLOG.md's status), task ids only. The repo
271    /// name shares the section-name charset — it fills
272    /// `# TODO — <repo>` and must survive the heading too.
273    pub fn todo_doc() -> impl Strategy<Value = TodoDoc> {
274        (
275            section_name(),
276            sections(line_item(&[
277                Status::Done,
278                Status::Deferred,
279                Status::Cancelled,
280            ])),
281        )
282            .prop_map(|(repo, sections)| TodoDoc { repo, sections })
283    }
284
285    /// A `BacklogDoc` — every item (children included) `Deferred`.
286    pub fn backlog_doc() -> impl Strategy<Value = BacklogDoc> {
287        sections(line_item(&[]).prop_map(|mut i| {
288            status_tree(&mut i, Status::Deferred);
289            i
290        }))
291        .prop_map(|sections| BacklogDoc { sections })
292    }
293
294    /// A `DoneDoc` — every item (children included) terminal: one
295    /// of `Done`/`Cancelled` per tree, so the `status:cancelled`
296    /// token round-trips through emit/parse.
297    pub fn done_doc() -> impl Strategy<Value = DoneDoc> {
298        sections(
299            (
300                line_item(&[]),
301                prop::sample::select(vec![Status::Done, Status::Cancelled]),
302            )
303                .prop_map(|(mut i, status)| {
304                    status_tree(&mut i, status);
305                    i
306                }),
307        )
308        .prop_map(|sections| DoneDoc { sections })
309    }
310
311    /// A `ProjectMeta` — arbitrary counters.
312    pub fn project_meta() -> impl Strategy<Value = ProjectMeta> {
313        (
314            "[a-z0-9-]{1,16}",
315            (1u32..=999, 1u32..=999, 1u32..=999, 1u32..=999, 1u32..=999),
316        )
317            .prop_map(|(name, (t, i, p, adr, pl))| ProjectMeta {
318                name,
319                next: NextCounters {
320                    t,
321                    i,
322                    p,
323                    adr,
324                    pl,
325                    ..Default::default()
326                },
327            })
328    }
329
330    /// A session id — `[A-Za-z0-9._:-]{1,12}`, the full charset
331    /// `is_valid_session` allows (anything else — space, `#`, `\r`,
332    /// `·` — can't survive the `##` heading).
333    pub fn session() -> impl Strategy<Value = String> {
334        "[a-zA-Z0-9._:-]{1,12}"
335    }
336
337    /// A `HandoffEntry` — notes are plain text lines (never a `##`
338    /// heading, which the grammar can't carry inside a note).
339    pub fn handoff_entry() -> impl Strategy<Value = HandoffEntry> {
340        (
341            date(),
342            actor(),
343            prop::option::of(session()),
344            prop::collection::vec("[a-z0-9.:]([a-z0-9 .:]{0,38}[a-z0-9.:])?", 0..4),
345        )
346            .prop_map(|(date, actor, session, lines)| HandoffEntry {
347                date,
348                actor,
349                session,
350                note: lines.join("\n"),
351            })
352    }
353
354    /// An `ItemFile` — any status, a non-empty line-safe note,
355    /// trimmed body of simple markdown lines.
356    pub fn item_file() -> impl Strategy<Value = ItemFile> {
357        (
358            file_kind().prop_flat_map(item_id),
359            title(),
360            status(),
361            priority(),
362            prop::option::of(actor()),
363            prop::collection::btree_set(label(), 0..3),
364            prop::collection::btree_set(item_id(ItemKind::Task), 0..2),
365            prop::option::of(date()),
366            prop::option::of(line_note()).prop_map(|o| o.filter(|n| !n.is_empty())),
367            (
368                prop::option::of(group_name()),
369                prop::option::of(group_name()),
370            ),
371            sched_fields(),
372            prop::collection::vec("(#+ |[a-z0-9 .-]+|- \\[[ x]\\] )[a-z0-9 .-]{0,30}", 0..6),
373        )
374            .prop_map(
375                |(
376                    id,
377                    title,
378                    status,
379                    priority,
380                    owner,
381                    labels,
382                    blocks,
383                    created,
384                    note,
385                    (epic, ms),
386                    (due, est, cycle, done),
387                    lines,
388                )| {
389                    let mut item = Item::new(id, title).status(status).priority(priority);
390                    item.owner = owner;
391                    item.labels = labels;
392                    item.blocks = blocks;
393                    // an item can't block itself — emit rejects it
394                    item.blocks.remove(&item.id);
395                    item.created = created;
396                    item.note = note;
397                    item.epic = epic;
398                    item.milestone = ms;
399                    item.due = due;
400                    item.estimate = est;
401                    item.cycle = cycle;
402                    item.done = done;
403                    let mut file = ItemFile::new(item);
404                    // canonical body: no leading blank lines, no
405                    // trailing whitespace — emit rejects less
406                    file.body = myco_core::canonical_body(&lines.join("\n\n")).to_string();
407                    file
408                },
409            )
410    }
