1//! hypha-http — the axum driving adapter for the hypha entity
2//! substrate. Serves the kind-agnostic `/api/*` surface: entity
3//! planes (`/api/entities*`), the node event feed (`/api/events`),
4//! node identity, pairing + bearer auth, directory/user info, and web
5//! push subscriptions. Integrations (peregrin's session plane, future
6//! campaign planes) compose [`app`] with their own routes and register
7//! [`EntityPlane`]s through [`HyphaState::with_kind`]; the
8//! `EntityContext::host` downcast is how planes reach host state.

43/// Async agent-inventory provider behind `GET /api/node`'s `agents`
44/// field — integrations whose roster is a live query (e.g. a session
45/// plane's `list_agents`) install one via
46/// [`HyphaState::with_agent_lister`]. When unset the static
47/// [`HyphaState::node_agents`] list is reported.
48pub type AgentLister =
49    Arc<dyn Fn() -> futures::future::BoxFuture<'static, Vec<String>> + Send + Sync>;

51/// Shared state threaded through every substrate handler — the
52/// `createApp` options object minus everything an integration owns
53/// (session planes, meta stores, campaign state). `Clone` is cheap
54/// (Arc/Mutex/registry clones) because axum clones state per request.
55#[derive(Clone)]
56pub struct HyphaState {
57    /// `kind` → [`EntityPlane`] registry behind `/api/entities*`.
58    /// Empty on a bare substrate — integrations register their kinds
59    /// at composition ([`Self::with_kind`]).
60    pub kinds: KindRegistry,
61    /// Process-wide event feed behind `GET /api/events`.
62    pub feed: EventFeed,
63    /// Identity reported by `GET /api/node`; `PATCH /api/node` renames
64    /// it in place (a mutex because `HyphaState` clones per request).
65    pub node: Arc<Mutex<NodeIdentity>>,
66    /// Where the identity persists (`HYPHA_NODE`); `None` keeps a
67    /// `PATCH /api/node` rename in-memory only.
68    pub node_path: Option<PathBuf>,
69    /// Agent ids advertised by `GET /api/node` — the integration's
70    /// runtime inventory (empty on a bare substrate; peregrin fills it
71    /// from its session plane). Static fallback: a live
72    /// [`Self::agent_lister`] wins when installed.
73    pub node_agents: Vec<String>,
74    /// Live agent-inventory provider for `GET /api/node` — see
75    /// [`AgentLister`]. `None` → `node_agents` is reported verbatim.
76    pub agent_lister: Option<AgentLister>,
77    /// Full `capabilities` override for `GET /api/node` — integrations
78    /// whose wire contract names its own capability tags (peregrin's
79    /// `sessions`/`projects`/`import`) install the exact list; `None`
80    /// → the substrate computes `entities`/`events`/`export` plus
81    /// `push`/`pairing` when those backends are wired.
82    pub node_capabilities: Option<Vec<String>>,
83    /// Opaque payload the [`EntityContext`] handed to every
84    /// [`EntityPlane`] carries as `host` — integrations whose planes
85    /// wrap host handlers (peregrin's session plane downcasts it back
86    /// to `AppState`) install their own state via
87    /// [`Self::with_entity_host`]. `None` → the `HyphaState` itself.
88    pub entity_host: Option<Arc<dyn Any + Send + Sync>>,
89    /// Auth-cookie name the login/logout exchange sets and the gate
90    /// reads — `hypha_token` on a bare substrate; integrations keep
91    /// their historical name (peregrin serves `sepia_token`).
92    pub auth_cookie: Arc<str>,
93    /// Pairing backend for `POST /api/pair` + issued-credential auth.
94    pub pairing: Option<Pairing>,
95    /// Per-client-IP throttle on `POST /api/pair` — the one route that
96    /// mints a bearer credential with no prior auth, so it's the brute-
97    /// force surface. Keyed by `ConnectInfo` socket IP (never
98    /// `X-Forwarded-For` — the node doesn't run behind a trusted proxy).
99    pub pair_limit: RateLimiter,
100    /// `HYPHA_TOKEN` — `None`/empty means bearer auth is off entirely.
101    pub token: Option<String>,
102    /// CORS allowlist — `*` echoes any Origin back (bearer auth, not
103    /// CORS, is the gate).
104    pub origins: Vec<String>,
105    /// SSE keep-alive cadence; `0` disables.
106    pub keep_alive: Duration,
107    /// Web Push service — `None` → `/api/push/*` answers 501.
108    pub push: Option<Arc<hypha_push::PushStore>>,
109    /// Deep health probe for `GET /api/health` — integrations with a
110    /// real liveness check (peregrin's session-plane probe) install a
111    /// handler returning the full response; `None` → the static
112    /// `{"ok":true}` liveness answer.
113    pub health_probe: Option<HealthProbe>,
114    /// Node-local key/value store behind `/api/kv*` (P-006) —
115    /// `None` → those routes answer 501. [`Self::from_env`] opens the
116    /// JSON-file impl at `HYPHA_KV` (`<home>/kv.json`); integrations
117    /// wanting a different backend or the meta.json migration install
118    /// their own via [`Self::with_kv`].
119    pub kv: Option<Arc<dyn hypha_kv::KvStore>>,
120    /// Post-write hook for `/api/kv*` mutations — see [`KvHook`].
121    pub kv_hook: Option<KvHook>,
122}

138/// the write plus the shared [`EventFeed`] so it can emit its own

369fn hostname_or_unknown() -> String {
370    nix::unistd::gethostname()
371        .ok()
372        .map(|h| h.to_string_lossy().into_owned())
373        .filter(|h| !h.is_empty())
374        .unwrap_or_else(|| "unknown".to_string())
375}

377/// CORS — port of `corsHeaders` + the OPTIONS preflight: `vary: Origin`
378/// on every response, and the allow-* triple only when the request's
379/// Origin is in the allowlist (or the allowlist carries `*`).
380async fn cors_layer(State(state): State<HyphaState>, req: Request<Body>, next: Next) -> Response {
381    let origin = req
382        .headers()
383        .get(header::ORIGIN)
384        .and_then(|v| v.to_str().ok())
385        .map(str::to_string);
386    let allowed: HashSet<&str> = state.origins.iter().map(String::as_str).collect();
387    let mut res = if req.method() == Method::OPTIONS {
388        StatusCode::NO_CONTENT.into_response()
389    } else {
390        next.run(req).await
391    };
392    let headers = res.headers_mut();
393    headers.insert(header::VARY, HeaderValue::from_static("Origin"));
394    if let Some(origin) = origin {
395        if allowed.contains(origin.as_str()) || allowed.contains("*") {
396            if let Ok(v) = HeaderValue::from_str(&origin) {
397                headers.insert(header::ACCESS_CONTROL_ALLOW_ORIGIN, v);
398            }
399            headers.insert(
400                header::ACCESS_CONTROL_ALLOW_METHODS,
401                HeaderValue::from_static("GET,POST,PATCH,DELETE,OPTIONS"),
402            );
403            headers.insert(
404                header::ACCESS_CONTROL_ALLOW_HEADERS,
405                HeaderValue::from_static("content-type,authorization"),
406            );
407        }
408    }
409    res
410}

412/// OTEL server span — the outermost layer. Extracts the W3C remote
413/// parent (`traceparent`/`tracestate`) so proxied hub→node calls join
414/// the hub's trace; requests without context get a local-root
415/// `http_request` span. Inert (plain span) when export is off.
416async fn trace_layer(req: Request<Body>, next: Next) -> Response {
417    let span = hypha_telemetry::server_span(req.method(), req.uri().path(), req.headers());
418    let for_status = span.clone();
419    async move {
420        let res = next.run(req).await;
421        hypha_telemetry::record_response(&for_status, res.status().as_u16());
422        res
423    }
424    .instrument(span)
425    .await
426}

428/// One line per request — `/api/health` excluded, like the TS logger.
429async fn log_layer(req: Request<Body>, next: Next) -> Response {
430    let method = req.method().clone();
431    let path = req.uri().path().to_string();
432    let started = Instant::now();
433    let res = next.run(req).await;
434    if path != "/api/health" {
435        // Hoisted — `tracing` fields evaluate only when a subscriber
436        // is attached, which coverage runs don't guarantee.
437        let status = res.status().as_u16();
438        let elapsed_ms = started.elapsed().as_millis();
439        tracing::info!("{method} {path} {status} {elapsed_ms}ms");
440    }
441    res
442}

444/// Build the substrate route table + middleware WITHOUT the terminal
445/// fallback — returns `Router<HyphaState>` so an integration can
446/// `.with_state(...)` it and `merge` it with its own router under one
447/// listener. axum panics merging two routers that both carry a custom
448/// fallback, so the composer installs exactly one (see
449/// [`app_routes`] vs [`app`]).
450///
451/// Only the kind-agnostic surface: health, pairing + auth bootstrap,
452/// node, fs, user, the event feed, `/api/entities*` plus the
453/// `/api/entity-index`/`/api/pending-ops` reads, and `/api/push/*`.
454/// Every route is `any()` + an internal method guard so
455/// wrong-method requests fall through to the 404 (rather than axum's
456/// bare 405). Integrations nest their own routes around or inside
457/// this router and register entity planes on the shared
458/// [`HyphaState`].
459pub fn app_routes(state: &HyphaState) -> Router<HyphaState> {
460    Router::new()
461        // Pre-auth routes — the auth gate itself passes these through
462        // (health for probes; pair/login are the credential bootstrap).
463        .route("/api/health", any(routes::misc::health))
464        .route("/api/pair", any(auth::pair))
465        .route("/api/auth/login", any(auth::handle))
466        .route("/api/auth/logout", any(auth::handle))
467        .route("/api/user", any(routes::misc::user))
468        .route("/api/node", any(routes::misc::node))
469        .route("/api/fs", any(routes::misc::fs))
470        .route("/api/events", any(routes::events::events))
471        .route("/api/entities", any(routes::entities::collection))
472        .route(
473            "/api/entities/snapshot",
474            any(routes::entities::snapshot_collection),
475        )
476        .route("/api/entities/{id}", any(routes::entities::item))
477        .route("/api/entities/{id}/ops", any(routes::entities::ops))
478        .route(
479            "/api/entities/{id}/snapshot",
480            any(routes::entities::snapshot_item),
481        )
482        .route("/api/entity-index", any(routes::entities::entity_index))
483        .route("/api/pending-ops", any(routes::entities::pending_ops))
484        .route("/api/push/vapid", any(routes::push::vapid))
485        .route("/api/push/subscribe", any(routes::push::subscribe))
486        .route("/api/kv", any(routes::kv::namespaces))
487        .route("/api/kv/{ns}", any(routes::kv::namespace))
488        .route("/api/kv/{ns}/{key}", any(routes::kv::entry))
489        .route("/api/client/keypair", any(routes::kv::keypair))
490        // Every body read goes through `read_json_body` with an
491        // explicit per-route cap (SMALL/MAX_JSON_BODY); axum's
492        // 2MiB `DefaultBodyLimit` stays on as the backstop for any
493        // future extractor-based read.
494        // Innermost → outermost: auth gate, request log, CORS, trace.
495        .layer(middleware::from_fn_with_state(
496            state.clone(),
497            auth::auth_gate,
498        ))
499        .layer(middleware::from_fn(log_layer))
500        .layer(middleware::from_fn_with_state(state.clone(), cors_layer))
501        .layer(middleware::from_fn(trace_layer))
502}

504/// The standalone substrate router — [`app_routes`] plus the 404
505/// fallback and the state applied. Integrations composing several
506/// routers under one listener use [`app_routes`] instead and install
507/// their own single fallback.
508pub fn app(state: HyphaState) -> Router {
509    app_routes(&state)
510        .fallback(async || routes::not_found())
511        .with_state(state)
512}

514/// Bind `env.host:env.port` and serve the app. The bind guard already
515/// ran in [`Env::parse`]; callers wanting graceful shutdown own the
516/// listener themselves.
517///
518/// # Errors
519/// `std::io::Error` on bind/serve failure.
520pub async fn serve(env: &Env, state: HyphaState) -> std::io::Result<()> {
521    let listener = tokio::net::TcpListener::bind((env.host.as_str(), env.port)).await?;
522    tracing::info!("hypha-http listening on {}:{}", env.host, env.port);
523    // `ConnectInfo` feeds the /api/pair limiter its client-IP key.
524    let serving = axum_serve(
525        listener,
526        app(state).into_make_service_with_connect_info::<std::net::SocketAddr>(),
527    );
528    #[cfg(not(coverage))]
529    {
530        serving.await
531    }
532    #[cfg(coverage)]
533    {
534        // The accept loop never returns; under coverage a ready arm
535        // lets `serve` complete so the function tail counts.
536        tokio::select! {
537            r = serving => r,
538            () = std::future::ready(()) => Ok(()),
539        }
540    }
