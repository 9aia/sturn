1//! Pairing — port of `apps/server/src/pair.ts`: the one-time short code
2//! `hypha pair` writes to `$HYPHA_HOME/pair-code` redeems for a long-lived
3//! `hypha_…` bearer credential. Issued credentials persist as sha256
4//! hashes in `$HYPHA_HOME/tokens.json` — the file alone can't
5//! authenticate.
6
7use std::collections::{HashMap, HashSet};
8use std::path::PathBuf;
9use std::sync::Mutex;
10use std::time::{SystemTime, UNIX_EPOCH};
11
12use base64::Engine;
13use base64::engine::general_purpose::URL_SAFE_NO_PAD;
14use sha2::{Digest, Sha256};

16/// How long a minted code stays redeemable (docs/protocol.md: ~60s).
17pub const PAIR_CODE_TTL_MS: u64 = 60_000;
18
19// Crockford base32 — no I/L/O/U, so codes survive being read aloud.
20const CROCKFORD: &[u8] = b"0123456789ABCDEFGHJKMNPQRSTVWXYZ";
21
22fn now_ms() -> u64 {
23    SystemTime::now()
24        .duration_since(UNIX_EPOCH)
25        .map_or(0, |d| d.as_millis() as u64)
26}
27
28fn random_bytes(n: usize) -> Vec<u8> {
29    // uuid v4 is already backed by the OS CSPRNG — 16 bytes per call.
30    let mut out = Vec::with_capacity(n);
31    while out.len() < n {
32        out.extend_from_slice(uuid::Uuid::new_v4().as_bytes());
33    }
34    out.truncate(n);
35    out
36}
37
38/// 8 Crockford chars in a 4-4 group, e.g. `7K2M-9PQX` (~40 bits).
39pub fn mint_pair_code() -> String {
40    let bytes = random_bytes(8);
41    let chars: String = bytes
42        .iter()
43        .map(|b| CROCKFORD[usize::from(b % u8::try_from(CROCKFORD.len()).unwrap_or(32))] as char)
44        .collect();
45    format!("{}-{}", &chars[..4], &chars[4..])
46}
47
48/// User input → lookup key: case-insensitive, dashes/spaces optional.
49pub fn normalize_pair_code(input: &str) -> String {
50    input
51        .to_uppercase()
52        .chars()
53        .filter(char::is_ascii_alphanumeric)
54        .collect()
55}

57/// The long-lived credential handed out on redeem — `hypha_`-prefixed
58/// on a bare substrate; integrations keep their historical prefix via
59/// [`Pairing::with_token_prefix`].
60pub fn mint_pair_token() -> String {
61    mint_pair_token_prefixed("hypha_")
62}
63
64/// `mint_pair_token` with an explicit credential prefix — the token
65/// is `<prefix><base64url(32 random bytes)>`.
66pub fn mint_pair_token_prefixed(prefix: &str) -> String {
67    format!("{prefix}{}", URL_SAFE_NO_PAD.encode(random_bytes(32)))
68}
69
70/// Tokens persist as hashes only — the file alone can't authenticate.
71pub fn hash_pair_token(token: &str) -> String {
72    use std::fmt::Write;
73    let digest = Sha256::digest(token.as_bytes());
74    digest.iter().fold(String::with_capacity(64), |mut out, b| {
75        let _ = write!(out, "{b:02x}");
76        out
77    })
78}

80/// In-memory one-time codes; expired entries are swept on every access
81/// so the map can't grow via abandoned codes.
82#[derive(Default)]
83pub struct PairingStore {
84    codes: Mutex<HashMap<String, u64>>,
85}
86
87impl PairingStore {
88    /// Register an externally minted code (the `pair-code` file).
89    pub fn register(&self, code: &str, expires_at: u64) {
90        let normalized = normalize_pair_code(code);
91        if normalized.is_empty() {
92            return;
93        }
94        self.codes
95            .lock()
96            .unwrap_or_else(std::sync::PoisonError::into_inner)
97            .insert(normalized, expires_at);
98    }
99
100    /// Mint + register a fresh code; returns what the operator sees.
101    pub fn mint(&self, ttl_ms: u64) -> (String, u64) {
102        let code = mint_pair_code();
103        let expires_at = now_ms() + ttl_ms;
104        self.register(&code, expires_at);
105        (code, expires_at)
106    }
107
108    /// Single-use redeem: consumes the code on success; unknown, malformed
109    /// or expired codes all return false — never distinguish the case.
110    pub fn redeem(&self, code: &str) -> bool {
111        let now = now_ms();
112        let mut codes = self
113            .codes
114            .lock()
115            .unwrap_or_else(std::sync::PoisonError::into_inner);
116        codes.retain(|_, expires_at| *expires_at > now);
117        codes.remove(&normalize_pair_code(code)).is_some()
118    }
119}
120
121/// The on-disk mint channel: `{code, expiresAt}` JSON read on every
122/// `POST /api/pair` and deleted on first read.
123fn read_pair_code_file(path: &std::path::Path) -> Option<(String, u64)> {
124    let raw = std::fs::read_to_string(path).ok()?;
125    let parsed: serde_json::Value = serde_json::from_str(&raw).ok()?;
126    let record = parsed.as_object()?;
127    let code = record.get("code")?.as_str()?;
128    // serde_json rejects non-finite numbers at parse — `as_f64` is
129    // always finite here.
130    let expires_at = record.get("expiresAt")?.as_f64()?;
131    Some((code.to_string(), expires_at as u64))
132}

134/// The seam the app consumes: redeem a code, check a credential.
135/// Filesystem-backed — `code_file` is the mint gate (whoever could write
136/// it is the machine owner); `tokens_file` persists issued-credential
137/// hashes across restarts.
138#[derive(Clone)]
139pub struct Pairing {
140    inner: std::sync::Arc<PairingInner>,
141}
142
143struct PairingInner {
144    code_file: PathBuf,
145    tokens_file: PathBuf,
146    store: PairingStore,
147    hashes: Mutex<HashSet<String>>,
148    /// Credential prefix minted on redeem — `hypha_` by default;
149    /// integrations keep their historical prefix (peregrin's
150    /// `sepia_`) so existing tooling keeps matching. `Mutex` so
151    /// [`Pairing::with_token_prefix`] also works on an already-shared
152    /// (cloned) instance — composition roots install the prefix after
153    /// the pairing has been handed to app state.
154    token_prefix: Mutex<std::sync::Arc<str>>,
155}
156
157impl Pairing {
158    /// Open the pairing backend; a corrupt tokens file degrades to empty.
159    pub fn open(code_file: PathBuf, tokens_file: PathBuf) -> Self {
160        let mut hashes = HashSet::new();
161        if let Ok(raw) = std::fs::read_to_string(&tokens_file) {
162            if let Ok(serde_json::Value::Object(record)) = serde_json::from_str(&raw) {
163                if let Some(serde_json::Value::Array(tokens)) = record.get("tokens") {
164                    for token in tokens {
165                        if let Some(hash) = token.as_str() {
166                            hashes.insert(hash.to_string());
167                        }
168                    }
169                }
170            }
171        }
172        Self {
173            inner: std::sync::Arc::new(PairingInner {
174                code_file,
175                tokens_file,
176                store: PairingStore::default(),
177                hashes: Mutex::new(hashes),
178                token_prefix: Mutex::new(std::sync::Arc::from("hypha_")),
179            }),
180        }
181    }
182
183    /// In-memory pairing for tests — no file channel.
184    pub fn in_memory(store: PairingStore) -> Self {
185        Self {
186            inner: std::sync::Arc::new(PairingInner {
187                code_file: PathBuf::new(),
188                tokens_file: PathBuf::new(),
189                store,
190                hashes: Mutex::new(HashSet::new()),
191                token_prefix: Mutex::new(std::sync::Arc::from("hypha_")),
192            }),
193        }
194    }
195
196    /// Rename the credential prefix `redeem` mints (default
197    /// `hypha_`) — integrations whose wire contract names its own
198    /// prefix install it here, even after the `Pairing` has been
199    /// cloned into shared state. Issued credentials are hash-persisted,
200    /// so already-issued tokens keep working either way.
201    #[must_use]
202    pub fn with_token_prefix(self, prefix: impl Into<std::sync::Arc<str>>) -> Self {
203        *self
204            .inner
205            .token_prefix
206            .lock()
207            .unwrap_or_else(std::sync::PoisonError::into_inner) = prefix.into();
208        self
209    }
210
211    fn flush_tokens(&self) {
212        let tmp = self
213            .inner
214            .tokens_file
215            .with_extension(format!("tmp-{}", std::process::id()));
216        let hashes = self
217            .inner
218            .hashes
219            .lock()
220            .unwrap_or_else(std::sync::PoisonError::into_inner);
221        let body = serde_json::json!({ "tokens": hashes.iter().collect::<Vec<_>>() });
222        if let Some(dir) = self.inner.tokens_file.parent() {
223            let _ = std::fs::create_dir_all(dir);
224        }
225        // `to_vec` is infallible for this shape — `ok().inspect`
226        // keeps the write best-effort without an untestable else arm.
227        serde_json::to_vec(&body).ok().inspect(|json| {
228            if std::fs::write(&tmp, json).is_ok() {
229                #[cfg(unix)]
230                {
231                    use std::os::unix::fs::PermissionsExt;
232                    let _ = std::fs::set_permissions(&tmp, std::fs::Permissions::from_mode(0o600));
233                }
234                let _ = std::fs::rename(&tmp, &self.inner.tokens_file);
235            }
236        });
237    }
238
239    /// Absorb a pending `pair-code` file — consume it regardless of
240    /// contents so a stale code can't linger past its TTL.
241    fn absorb_code_file(&self) {
242        if !self.inner.code_file.exists() {
243            return;
244        }
245        let file = read_pair_code_file(&self.inner.code_file);
246        let _ = std::fs::remove_file(&self.inner.code_file);
247        if let Some((code, expires_at)) = file {
248            if expires_at > now_ms() {
249                self.inner.store.register(&code, expires_at);
250            }
251        }
252    }
253
254    /// `POST /api/pair`: absorb any pending code file, then redeem. On
255    /// success returns a fresh long-lived credential (already persisted).
256    pub fn redeem(&self, code: &str) -> Option<String> {
257        self.absorb_code_file();
258        if !self.inner.store.redeem(code) {
259            return None;
260        }
261        let token = mint_pair_token_prefixed(
262            &self
263                .inner
264                .token_prefix
265                .lock()
266                .unwrap_or_else(std::sync::PoisonError::into_inner),
267        );
268        {
269            let mut hashes = self
270                .inner
271                .hashes
272                .lock()
273                .unwrap_or_else(std::sync::PoisonError::into_inner);
274            hashes.insert(hash_pair_token(&token));
275        }
276        if !self.inner.tokens_file.as_os_str().is_empty() {
277            self.flush_tokens();
278        }
279        Some(token)
280    }
281
282    /// Bearer-auth check for credentials this node has issued.
283    pub fn accepts(&self, token: &str) -> bool {
284        self.inner
285            .hashes
286            .lock()
287            .unwrap_or_else(std::sync::PoisonError::into_inner)
288            .contains(&hash_pair_token(token))
289    }
290}

292#[cfg(test)]
293mod cov_tests {
294    #![allow(clippy::unwrap_used)]
295    use super::*;

297    #[test]
298    fn code_file_and_open_arms() {
299        let dir = tempfile::tempdir().unwrap();
300        let code_file = dir.path().join("pair-code");
301        // Non-finite expiry can't even parse → None via `.ok()?`.
302        std::fs::write(&code_file, r#"{"code":"c","expiresAt":1e999}"#).unwrap();
303        assert_eq!(read_pair_code_file(&code_file), None);
304        // A tokens file with non-string entries is skipped.
305        let tokens_file = dir.path().join("tokens.json");
306        std::fs::write(&tokens_file, r#"{"tokens":[1,"h",null]}"#).unwrap();
307        let pairing = Pairing::open(code_file.clone(), tokens_file);
308        // An already-expired code file is absorbed but not registered.
309        std::fs::write(&code_file, r#"{"code":"c","expiresAt":1.0}"#).unwrap();
310        pairing.absorb_code_file();
311        // An unparseable code file → `file` is None (outer else).
312        std::fs::write(&code_file, "not json").unwrap();
313        pairing.absorb_code_file();
314    }
315
316    #[test]
317    fn open_and_flush_failure_arms() {
318        let dir = tempfile::tempdir().unwrap();
319        // `tokens` present but not an array / file not a JSON object.
320        for body in [r#"{"tokens":"x"}"#, "[1,2]"] {
321            let tokens_file = dir.path().join("t.json");
322            std::fs::write(&tokens_file, body).unwrap();
323            let _ = Pairing::open(dir.path().join("c"), tokens_file);
324        }
325        // flush_tokens: tokens_file lives under a *file* — write fails.
326        let blocker = dir.path().join("blocker");
327        std::fs::write(&blocker, "x").unwrap();
328        let pairing = Pairing::open(PathBuf::new(), blocker.join("deep/tokens.json"));
329        pairing.inner.store.register("ABCD-1234", now_ms() + 60_000);
330        assert!(pairing.redeem("abcd-1234").is_some());
331    }
