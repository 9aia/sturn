1//! Persistent `cuora serve` client — the ndjson JSON-RPC transport
2//! alternative to shelling the `cuora` CLI per call.
3//!
4//! `cuora serve` is one JSON request per stdin line → one JSON
5//! response per stdout line (protocol `cuora-serve/1`); audit ndjson
6//! stays on stderr, which this client discards (`Stdio::null`) so it
7//! never pollutes the parent's stream. The child spawns lazily on
8//! the first request and is reused for the client's lifetime; a
9//! dead child, a closed stdout (EOF), a malformed reply line, or a
10//! request timeout drops the session and the next call respawns.
11//!
12//! Cold-start deaths get a wider retry budget than warm ones: at
13//! service boot a `cuora serve` child can race vault/keyring
14//! readiness or a SIGKILL'd predecessor's lock, and the millisecond
15//! flakes the per-request retry covers clear long before the child
16//! is spawnable. Until the first reply ever lands, `call` keeps
17//! respawning with backoff for [`CuoraServe::dead_budget`]; after
18//! that a crash-looping child is a broken `cuora` and gets the
19//! historical three fast attempts. A genuinely missing binary
20//! (`ENOENT` at spawn) never enters either budget — it fails fast
21//! on every call.
22//!
23//! Requests serialize on a mutex — the wire is one-request-at-a-time
24//! by design, and the timeout applies to the whole
25//! write+read roundtrip while the lock is held.
26//!
27//! Value handling mirrors the CLI surface deliberately: [`CuoraServe::get`]
28//! answers `Option<String>` (`None` on *any* failure — missing binary,
29//! transport error, `not_found`, `denied` — so callers' fallthrough
30//! chains behave identically), [`CuoraServe::put`] answers
31//! `io::Result<()>`. `put` carries the value in the request's
32//! `value` field — never stdin of a side channel, never argv.
33//!
34//! The audit `actor` is a constructor param ([`CuoraServe::new`]) —
35//! each consumer passes its own name (`peregrin`, `bupha`, …); it
36//! seeds both the child's `--actor` flag and every request's
37//! `actor` field.
38

51/// Default per-request deadline — generous because a `[serve.pdp]`
52/// consult alone may take its 5s default before the vault is even
53/// touched. Override with [`CuoraServe::with_timeout`].
54const DEFAULT_TIMEOUT: Duration = Duration::from_secs(30);
55
56/// How long [`Drop`] waits for a clean `shutdown`-less exit (stdin
57/// EOF) before killing the child.
58const DROP_GRACE: Duration = Duration::from_millis(500);
59
60/// Cold-start dead-child budget — how long `call` keeps respawning
61/// while *no reply has ever landed* on this client. Boot-time
62/// readiness races (vault/keyring not up yet, a predecessor's lock
63/// still held) clear in seconds; the per-request backoff in
64/// [`crate::GET_RETRY_BACKOFF`] is sized for millisecond flakes and
65/// can't see them. Bounded — a `cuora` that never answers within
66/// the budget is broken, not warming up.
67const COLD_DEAD_BUDGET: Duration = Duration::from_secs(10);
68/// First cold-retry sleep — doubles each dead attempt up to
69/// [`COLD_DEAD_BACKOFF_MAX`]. Small enough that a child dying on
70/// the first write doesn't stall the common path.
71const COLD_DEAD_BACKOFF: Duration = Duration::from_millis(50);
72const COLD_DEAD_BACKOFF_MAX: Duration = Duration::from_secs(1);
73/// Warm-path dead attempts — once any reply has landed, a child
74/// that keeps dying is a broken `cuora`; the historical three fast
75/// respawns, no sleep (the caller's own retry — `get`'s
76/// [`crate::GET_ATTEMPTS`] — still applies on top).
77const WARM_DEAD_ATTEMPTS: u32 = 3;

79/// Which cuora transport [`crate::CuoraSecrets`] uses.
80///
81/// Consumers select it via their own env knob — the convention is
82/// `<PRODUCT>_CUORA_TRANSPORT` (e.g. `PEREGRIN_CUORA_TRANSPORT`,
83/// `BUPHA_CUORA_TRANSPORT`); [`CuoraTransport::from_env`] takes the
84/// variable *name* so the shared crate stays product-agnostic. Only
85/// `serve` (any case) selects the serve transport; unset, `cli`, or
86/// anything else stays on the CLI so a typo can't silently change
87/// the wire.
88#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
89pub enum CuoraTransport {
90    /// Shell `cuora get`/`put` per call — the historical behavior.
91    #[default]
92    Cli,
93    /// Persistent `cuora serve` child speaking ndjson on stdio.
94    Serve,
95}
96
97impl CuoraTransport {
98    /// Read the transport knob from the env var `name` — see the
99    /// type docs for the `<PRODUCT>_CUORA_TRANSPORT` convention.
100    #[must_use]
101    pub fn from_env(name: &str) -> Self {
102        Self::parse(std::env::var(name).ok().as_deref())
103    }
104
105    /// The parse rule, split out so tests pin it without touching
106    /// the process environment: only `serve` (any case) flips.
107    #[must_use]
108    fn parse(value: Option<&str>) -> Self {
109        match value {
110            Some(v) if v.eq_ignore_ascii_case("serve") => Self::Serve,
111            _ => Self::Cli,
112        }
113    }
114}

116/// A live `cuora serve` child plus the plumbing to talk to it.
117struct Session {
118    child: Child,
119    stdin: ChildStdin,
120    /// Reply lines from the stdout reader thread — `Err` carries the
121    /// reader's io failure; channel close means stdout hit EOF.
122    lines: Receiver<io::Result<String>>,
123}
124
125/// Why a roundtrip failed at the transport layer (as opposed to a
126/// well-formed `ok:false` reply, which is a cuora answer, not a
127/// transport failure).
128enum Fail {
129    /// Write/read/EOF/protocol desync — the session is dead.
130    Dead,
131    /// Spawn failed `ENOENT` — the binary (or its interpreter) is
132    /// absent. Permanent for the process lifetime: no retry budget
133    /// applies, `call` fails fast so a missing `cuora` under the
134    /// serve transport misses as quickly as under the CLI.
135    NoBinary,
136    /// The deadline passed — session dropped, caller sees the error.
137    Timeout,
138}

140/// Persistent `cuora serve` client. Spawn is lazy — construction is
141/// free; the child appears on the first request. Cloning shares the
142/// child via [`Arc`](std::sync::Arc) at the [`crate::CuoraSecrets`] layer; `CuoraServe`
143/// itself is not `Clone`.
144pub struct CuoraServe {
145    /// The `cuora` binary — the consumer's `*_CUORA_BIN` override or
146    /// `cuora` on PATH.
147    bin: PathBuf,
148    /// Audit actor — seeds the child's `--actor` and each request's
149    /// `actor` field (a claimed label, like every local actor claim).
150    actor: String,
151    timeout: Duration,
152    /// Cold-start dead-child budget — see [`COLD_DEAD_BUDGET`].
153    dead_budget: Duration,
154    /// Set on the first reply of any kind (`ok` or `ok:false` —
155    /// either proves the wire works). Gates the cold/warm dead
156    /// retry regimes in [`Self::call`].
157    ever_connected: AtomicBool,
158    next_id: AtomicU64,
159    session: Mutex<Option<Session>>,
160}

162impl CuoraServe {
163    /// A client over `bin` attributing requests to `actor` with
164    /// `DEFAULT_TIMEOUT`.
165    #[must_use]

168            bin,
169            actor: actor.into(),
170            timeout: DEFAULT_TIMEOUT,

173            next_id: AtomicU64::new(1),
174            session: Mutex::new(None),
175        }

181        self.timeout = timeout;
182        self
183    }

185    /// Override the cold-start dead-child budget (tests use tiny
186    /// values — the default is sized for real boot-time readiness
187    /// races, which no test wants to wait out).
188    #[must_use]
189    pub fn with_dead_budget(mut self, dead_budget: Duration) -> Self {
190        self.dead_budget = dead_budget;

194    /// `get` — the secret value, or `None` on *any* failure.
195    ///
196    /// `Ok` replies yield `result.value` (a JSON string); empty
197    /// values answer `None`, matching the CLI transport's
198    /// empty-stdout rule. `not_found`, `denied`, transport errors,
199    /// and a missing binary all collapse to `None` — "not here" —
200    /// so secret-chain fallthrough is identical across transports.
201    ///
202    /// Transient answers retry with the same bounded budget as the
203    /// CLI leg (T-059): a dead child already retries inside
204    /// `Self::call`, but an `ok:false` `op_error`/`pdp_unreachable`
205    /// is a *cuora* answer — a vault transport flake wearing a
206    /// well-formed reply — which used to collapse straight to
207    /// `None`. Definitive codes (`not_found`, `denied`,
208    /// `bad_request`, `unknown_op`) still miss on the first reply.
209    pub fn get(&self, rref: &str) -> Option<String> {
210        for attempt in 0..crate::GET_ATTEMPTS {
211            if attempt > 0 {
212                thread::sleep(crate::GET_RETRY_BACKOFF * attempt);
213            }
214            let Ok(reply) = self.call(&json!({
215                "op": "get",
216                "ref": rref,
217                "actor": self.actor,
218            })) else {
219                continue;
220            };
221            if reply.get("ok").and_then(Value::as_bool) != Some(true) {
222                if transient_reply(&reply) {
223                    continue;
224                }
225                return None;
226            }
227            let value = reply.get("result")?.get("value")?.as_str()?.to_string();
228            return (!value.is_empty()).then_some(value);
229        }
230        None
231    }
232
233    /// `put` — store `value` under `rref`. The value rides the
234    /// request's `value` field (the wire's designed channel — the
235    /// parent owns the pipe), capped at 64 KiB by cuora itself.
236    pub fn put(&self, rref: &str, value: &str) -> io::Result<()> {
237        let reply = self
238            .call(&json!({
239                "op": "put",
240                "ref": rref,
241                "value": value,
242                "actor": self.actor,
243            }))
244            .map_err(io::Error::other)?;
245        if reply.get("ok").and_then(Value::as_bool) == Some(true) {
246            Ok(())
247        } else {
248            Err(io::Error::other(format!(
249                "cuora serve put {rref}: {}",
250                reply_error(&reply)
251            )))
252        }
253    }
254
255    /// `check` — the config + declared-surface report, `result`
256    /// verbatim. Useful for diagnostics; not on the secret path.
257    pub fn check(&self) -> io::Result<Value> {
258        let reply = self
259            .call(&json!({"op": "check", "actor": self.actor}))
260            .map_err(io::Error::other)?;
261        if reply.get("ok").and_then(Value::as_bool) == Some(true) {
262            Ok(reply.get("result").cloned().unwrap_or(Value::Null))
263        } else {
264            Err(io::Error::other(format!(
265                "cuora serve check: {}",
266                reply_error(&reply)
267            )))
268        }
269    }

271    /// One request → one reply. Transport failures retry on a
272    /// freshly spawned child (a request that never got answered is
273    /// safe to replay — `get` is idempotent and `put`/`rotate`
274    /// re-putting the same value is a no-op semantically); a parsed
275    /// reply — `ok` or `ok:false` — returns as-is.
276    ///
277    /// Dead-child retries come in two regimes (see the module docs):
278    /// *cold* — no reply has ever landed on this client, so the
279    /// spawn/readiness race gets [`Self::dead_budget`] of respawns
280    /// with backoff; *warm* — a session proved itself once, so a
281    /// crash-looping child is a broken `cuora` and gets
282    /// [`WARM_DEAD_ATTEMPTS`] fast attempts. `Fail::NoBinary` is
283    /// neither — `ENOENT` at spawn is permanent and fails fast.
284    fn call(&self, request: &Value) -> Result<Value, String> {
285        let mut req = request.clone();
286        // `id` is echoed verbatim for correlation; absent → null is
287        // legal, but a real id makes a desynced wire diagnosable.
288        req["id"] = json!(self.next_id.fetch_add(1, Ordering::Relaxed));
289        let line = serde_json::to_string(&req).map_err(|e| e.to_string())?;
290        // The cold deadline is fixed at call entry; `ever_connected`
291        // flipping mid-call (a concurrent request landed a reply)
292        // demotes this call to the warm regime on the next dead
293        // attempt — readiness is a property of the transport, not
294        // the request.
295        let cold_deadline = (!self.ever_connected.load(Ordering::Relaxed))
296            .then(|| Instant::now() + self.dead_budget);
297        let mut backoff = COLD_DEAD_BACKOFF;
298        let mut dead_attempts = 0u32;
299        loop {
300            match self.roundtrip(&line) {
301                Ok(reply) => {
302                    self.ever_connected.store(true, Ordering::Relaxed);
303                    return Ok(reply);
304                }
305                Err(Fail::NoBinary) => {
306                    return Err("cuora serve binary unavailable".to_string());
307                }
308                Err(Fail::Dead) => {
309                    self.drop_session();
310                    dead_attempts += 1;
311                    let cold =
312                        cold_deadline.filter(|_| !self.ever_connected.load(Ordering::Relaxed));
313                    match cold {
314                        Some(deadline) => {
315                            let now = Instant::now();
316                            if now >= deadline {
317                                break;
318                            }
319                            thread::sleep(backoff.min(deadline - now));
320                            backoff = (backoff * 2).min(COLD_DEAD_BACKOFF_MAX);
321                        }
322                        None => {
323                            if dead_attempts >= WARM_DEAD_ATTEMPTS {
324                                break;
325                            }
326                        }
327                    }
328                }
329                Err(Fail::Timeout) => {
330                    self.drop_session();
331                    return Err(format!(
332                        "cuora serve {} timed out after {:?}",
333                        request.get("op").and_then(Value::as_str).unwrap_or("?"),
334                        self.timeout
335                    ));
336                }
337            }
338        }
339        // The only `break` is the dead-child budget — every other
340        // outcome returned above.
341        Err("cuora serve child died".to_string())
342    }

344    /// Write `line` + `\n`, flush, read one reply line, parse it.
345    /// Any transport-level failure marks the session dead.
346    fn roundtrip(&self, line: &str) -> Result<Value, Fail> {
347        let mut guard = self
348            .session
349            .lock()
350            .unwrap_or_else(std::sync::PoisonError::into_inner);
351        if guard.is_none() {
352            match self.spawn() {
353                Ok(session) => *guard = Some(session),
354                // `ENOENT` — binary or interpreter absent. Permanent:
355                // surface it so `call` fails fast instead of burning
356                // the cold budget on a spawn that can never succeed.
357                Err(e) if e.kind() == io::ErrorKind::NotFound => {
358                    return Err(Fail::NoBinary);
359                }
360                // Any other spawn/io failure (fork pressure, EAGAIN)
361                // is transient-shaped — the dead-child retry path.
362                Err(_) => return Err(Fail::Dead),
363            }
364        }
365        let Some(session) = guard.as_mut() else {
366            return Err(Fail::Dead);
367        };
368        session.exchange(line, self.timeout)
369    }
370
371    /// Spawn `cuora serve` — stdin/stdout piped, stderr discarded
372    /// (audit ndjson is cuora's own stream; inheriting it would
373    /// interleave with the parent's). A stdout reader thread feeds
374    /// reply lines into a channel so [`Session::exchange`] can apply
375    /// the timeout without blocking the read itself.
376    fn spawn(&self) -> io::Result<Session> {
377        let mut child = Command::new(&self.bin)
378            // `--actor` seeds the audit default; each request also
379            // carries `actor` explicitly (the serve wire's field).
380            .arg("--actor")
381            .arg(&self.actor)
382            .arg("serve")
383            .stdin(Stdio::piped())
384            .stdout(Stdio::piped())
385            .stderr(Stdio::null())
386            .spawn()?;
387        let stdin = child
388            .stdin
389            .take()
390            .ok_or_else(|| io::Error::other("cuora serve spawned without stdin"))?;
391        let stdout = child
392            .stdout
393            .take()
394            .ok_or_else(|| io::Error::other("cuora serve spawned without stdout"))?;
395        let (tx, lines) = std::sync::mpsc::channel();
396        thread::spawn(move || {
397            let mut reader = BufReader::new(stdout);
398            let mut buf = String::new();
399            loop {
400                buf.clear();
401                match reader.read_line(&mut buf) {
402                    // EOF or a read error — the child's stdout is
403                    // gone; channel close tells the client.
404                    Ok(0) | Err(_) => break,
405                    Ok(_) => {
406                        if tx.send(Ok(buf.clone())).is_err() {
407                            break;
408                        }
409                    }
410                }
411            }
412        });
413        Ok(Session {
414            child,
415            stdin,
416            lines,
417        })
418    }
419
420    /// Kill + reap the current session, if any. Dropping the
421    /// `ChildStdin` first gives cuora its EOF → clean exit; we then
422    /// wait briefly before killing so `Drop` stays bounded.
423    fn drop_session(&self) {
424        let mut guard = self
425            .session
426            .lock()
427            .unwrap_or_else(std::sync::PoisonError::into_inner);
428        if let Some(mut session) = guard.take() {
429            reap(&mut session.child);
430        }
431    }
432}

434impl Session {
435    /// One request/response on the live child. `try_wait` catches an
436    /// already-exited child before we write into a dead pipe; the
437    /// channel states (timeout / disconnect / reader error) catch
438    /// everything else.
439    fn exchange(&mut self, line: &str, timeout: Duration) -> Result<Value, Fail> {
440        // Already reaped-exited — dead before we even write.
441        if let Ok(Some(_)) = self.child.try_wait() {
442            return Err(Fail::Dead);
443        }
444        if self
445            .stdin
446            .write_all(line.as_bytes())
447            .and_then(|()| self.stdin.write_all(b"\n"))
448            .and_then(|()| self.stdin.flush())
449            .is_err()
450        {
451            return Err(Fail::Dead);
452        }
453        match self.lines.recv_timeout(timeout) {
454            Ok(Ok(reply)) => serde_json::from_str(&reply).map_err(|_| Fail::Dead),
455            // Reader thread hit an io error or stdout EOF.
456            Ok(Err(_)) | Err(RecvTimeoutError::Disconnected) => Err(Fail::Dead),
457            Err(RecvTimeoutError::Timeout) => Err(Fail::Timeout),
458        }
459    }
460}

462/// Close stdin (EOF → cuora exits cleanly), wait up to [`DROP_GRACE`],
463/// then kill + reap. Bounded — a wedged child can't hang drop.
464fn reap(child: &mut Child) {
465    // `stdin` was moved into the Session; dropping the Session
466    // closes it. Here we only own `child` — close is implicit when
467    // the Session dropped. Wait briefly, then kill.
468    let deadline = Instant::now() + DROP_GRACE;
469    loop {
470        match child.try_wait() {
471            Ok(Some(_)) => return,
472            Ok(None) if Instant::now() < deadline => thread::sleep(Duration::from_millis(10)),
473            _ => {
474                let _ = child.kill();
475                let _ = child.wait();
476                return;
477            }
478        }
479    }
480}

482impl Drop for CuoraServe {
483    fn drop(&mut self) {
484        // Take the session out so its `ChildStdin` closes (EOF) —
485        // `cuora serve` exits cleanly on stdin EOF — then reap.
486        if let Ok(mut guard) = self.session.lock() {
487            if let Some(mut session) = guard.take() {
488                drop(session.stdin);
489                reap(&mut session.child);
490            }
491        }
492    }
493}

506/// Is an `ok:false` reply a *transient* answer worth retrying?
507/// `op_error` wraps vault/ops failures (HTTP flakes, decrypt
508/// transport errors) and `pdp_unreachable` is a policy engine
509/// that may come back; `not_found`/`denied`/`bad_request`/
510/// `unknown_op` are definitive — a retry can't change them.
511/// Unknown future codes default to definitive: retrying is the
512/// risky direction (it can mask a real denial behind latency),
513/// missing is the safe one.
514fn transient_reply(reply: &Value) -> bool {
515    matches!(
516        reply
517            .get("error")
518            .and_then(|e| e.get("code"))
519            .and_then(Value::as_str),
520        Some("op_error" | "pdp_unreachable")
521    )
522}
523
524/// `"code: message"` for an `ok:false` reply — diagnostics only.
525fn reply_error(reply: &Value) -> String {
526    reply.get("error").map_or_else(
527        || "no error detail".to_string(),
528        |e| {
529            format!(
530                "{}: {}",
531                e.get("code").and_then(Value::as_str).unwrap_or("?"),
532                e.get("message").and_then(Value::as_str).unwrap_or("?")
533            )
534        },
535    )
536}

544    /// A fake `cuora` speaking the ndjson serve wire. Requests are
545    /// compact single-line JSON we generate, so `grep -o` field
546    /// extraction is exact. State lives in `$DIR/db-<key>` files;
547    /// `$DIR/spawns` counts child invocations (spawn-on-demand and
548    /// respawn tests read it). `$DIR/die-once` makes the *next*
549    /// spawn consume one line then exit without replying — EOF
550    /// mid-request. `$DIR/die-after` makes the *running* child exit
551    /// right after its next reply — death between requests.
552    /// `$DIR/hang` makes every spawn sleep forever — the timeout
553    /// trigger. `$DIR/operr-<key>` makes the next `get` for that
554    /// key answer a transient `op_error` once (consumed);
555    /// `$DIR/operr-always` makes every `get` answer it — the
556    /// T-059 retry-budget fixtures. `$DIR/flaky` holds a count N:
557    /// the next N spawns exit 1 immediately without reading —
558    /// the cold-start readiness-race fixture.
559    fn stub(dir: &std::path::Path, extra: &str) -> PathBuf {
560        let script = format!(
561            "#!/bin/sh\n\
562             dir='{dir}'\n\
563             echo x >> \"$dir/spawns\"\n\
564             {extra}\n\
565             if [ -f \"$dir/flaky\" ]; then\n\
566               n=$(cat \"$dir/flaky\")\n\
567               if [ \"$n\" -gt 0 ]; then\n\
568                 echo $((n - 1)) > \"$dir/flaky\"\n\
569                 exit 1\n\
570               fi\n\
571             fi\n\
572             if [ -f \"$dir/die-once\" ]; then\n\
573               rm \"$dir/die-once\"\n\
574               IFS= read -r _line\n\
575               exit 0\n\
576             fi\n\
577             if [ -f \"$dir/hang\" ]; then exec sleep 60; fi\n\
578             while IFS= read -r line; do\n\
579               op=$(printf '%s' \"$line\" | grep -o '\"op\":\"[a-z]*\"' | cut -d'\"' -f4)\n\
580               ref=$(printf '%s' \"$line\" | grep -o '\"ref\":\"[^\"]*\"' | cut -d'\"' -f4)\n\
581               val=$(printf '%s' \"$line\" | grep -o '\"value\":\"[^\"]*\"' | cut -d'\"' -f4)\n\
582               key=${{ref##*/}}\n\
583               case \"$op\" in\n\

741    #[test]
742    fn warm_deaths_keep_the_fast_budget() {
743        // Once a reply has landed the transport is proven — a
744        // crash-looping child is a broken `cuora`, so dead retries
745        // go back to the historical three fast attempts per call
746        // (`get`'s own GET_ATTEMPTS on top), not the cold budget.
747        let dir = tempfile::tempdir().unwrap();
748        let bin = stub(dir.path(), "");
749        std::fs::write(dir.path().join("db-A"), "one").unwrap();
750        // A generous budget proves the warm path ignores it.
751        let serve = CuoraServe::new(bin, "test-actor").with_dead_budget(Duration::from_secs(30));
752        assert_eq!(serve.get("cuora://test/A").as_deref(), Some("one"));
753        assert_eq!(spawns(dir.path()), 1);
754        // `die-after` retires the live session (it answers once
755        // more, then exits); `flaky` then kills every respawn —
756        // warm regime: each `call` gets WARM_DEAD_ATTEMPTS
757        // roundtrips, `get` retries GET_ATTEMPTS calls.
758        std::fs::write(dir.path().join("die-after"), "").unwrap();
759        std::fs::write(dir.path().join("flaky"), "100").unwrap();
760        assert_eq!(serve.get("cuora://test/A").as_deref(), Some("one"));
761        assert_eq!(serve.get("cuora://test/A"), None);
762        // 1 serving spawn + the warm retries: the first `call`'s
763        // opening roundtrip dies on the *stale* session (no spawn),
764        // leaving WARM_DEAD_ATTEMPTS-1 respawns; each later `get`
765        // attempt spawns WARM_DEAD_ATTEMPTS children.
766        assert_eq!(
767            spawns(dir.path()),
768            1 + (WARM_DEAD_ATTEMPTS - 1) as usize
769                + WARM_DEAD_ATTEMPTS as usize * (crate::GET_ATTEMPTS - 1) as usize,
770            "warm calls give up after WARM_DEAD_ATTEMPTS each"
771        );
772    }
