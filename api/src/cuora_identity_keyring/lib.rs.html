1//! cuora-identity-keyring — the shared age identity in the
2//! platform's session secret store.
3//!
4//! The org has ONE age identity per human/machine, owned by the
5//! `tardigrada` tool; cuora resolves that same identity. The
6//! `AGE-SECRET-KEY-…` string lives as a keyed secret in the
7//! user-session credential store — it unlocks with the login
8//! session, so reads need no prompt:
9//!
10//! - Linux/*nix — org.freedesktop.secrets (secret-service: GNOME
11//!   Keyring, KeePassXC, KDE Wallet's wrapper) via the synchronous
12//!   libdbus client — no async runtime.
13//! - macOS — the login keychain.
14//! - Windows — Credential Manager (DPAPI).
15//!
16//! [`KeyringIdentity::platform`] is the availability probe: it
17//! fails with [`IdentityError::Unavailable`] on a headless box
18//! (no session bus, no daemon) and callers fall back to
19//! `cuora-identity-file` — never a crash. Tests drive any
20//! `keyring-core` store through [`KeyringIdentity::with_store`]
21//! (the mock store needs no daemon).
22//!
23//! Store calls run under a bounded wait (`bounded`): a locked
24//! collection's unlock prompt has no client-side deadline on the
25//! dbus store, so an unanswered prompt would hang the caller
26//! forever. `CUORA_KEYRING_TIMEOUT_MS` tunes the bound (`0`
27//! waits forever).
28

37/// The canonical secret-store service name — the shared
38/// `tardigrada` identity entry cuora reads and writes.
39pub const CANONICAL_SERVICE: &str = "tardigrada";
40/// The entry/user name the age identity is stored under.
41pub const USER: &str = "age-identity";
42/// Legacy service names `load` falls back to, in order — identities
43/// stored by `cuora`/`arca` keep resolving; `store` always writes
44/// [`CANONICAL_SERVICE`] only.
45pub const LEGACY_SERVICES: &[&str] = &["cuora", "arca"];
46
47/// Upper bound on one blocking credential-store call. The dbus
48/// secret-service store connects via `SecretService::connect`,
49/// which waits *indefinitely* for a user to answer an unlock
50/// prompt — after a keyring-daemon restart left the login
51/// collection locked, every `cuora` verb sat on the prompt and
52/// looked hung while long-lived `cuora serve` legs (identity
53/// already in memory) kept working. On timeout the op fails with
54/// a typed error naming the escape hatches; the still-blocked
55/// worker thread is left to die with the process (CLI verbs are
56/// one-shot; `serve` just sees an op error per request).
57const OP_TIMEOUT: std::time::Duration = std::time::Duration::from_secs(120);
58
59/// The effective op bound — `CUORA_KEYRING_TIMEOUT_MS` overrides
60/// [`OP_TIMEOUT`]; `0` disables the bound entirely (the old
61/// block-forever behavior, for a human willing to wait out a
62/// prompt). An unparseable value falls back to the default.
63fn op_timeout() -> Option<std::time::Duration> {
64    match std::env::var("CUORA_KEYRING_TIMEOUT_MS") {
65        Ok(v) => match v.parse::<u64>() {
66            Ok(0) => None,
67            Ok(ms) => Some(std::time::Duration::from_millis(ms)),
68            Err(_) => Some(OP_TIMEOUT),
69        },
70        Err(_) => Some(OP_TIMEOUT),
71    }
72}

74/// Run `f` on a worker thread and bound the wait — credential
75/// stores are synchronous and a locked collection's unlock prompt
76/// has no client-side deadline, so the bound lives here, above
77/// every backend (dbus prompt, Keychain dialog, Credential
78/// Manager). `what` names the op in the timeout error.
79fn bounded<T, F>(what: &str, f: F) -> Result<T, IdentityError>
80where
81    T: Send + 'static,
82    F: FnOnce() -> Result<T, IdentityError> + Send + 'static,
83{
84    let Some(timeout) = op_timeout() else {
85        return f();
86    };
87    let (tx, rx) = std::sync::mpsc::channel();
88    std::thread::Builder::new()
89        .name(format!("cuora-keyring-{what}"))
90        .spawn(move || {
91            // A timed-out caller is gone; a late result has
92            // nowhere to go — dropping the send is fine.
93            let _ = tx.send(f());
94        })
95        .map_err(|e| IdentityError::Store(format!("keyring {what} thread: {e}")))?;
96    match rx.recv_timeout(timeout) {
97        Ok(res) => res,
98        Err(_) => Err(IdentityError::Store(format!(
99            "keyring {what} timed out after {}s — the session store is \
100             waiting on an unlock prompt; unlock it, or configure your \
101             tool to use its identity-file backend instead \
102             (CUORA_KEYRING_TIMEOUT_MS adjusts the bound, 0 waits \
103             forever)",
104            timeout.as_secs()
105        ))),
106    }
107}

109/// The `<service>`/`age-identity` entry in `store` — the
110/// store-level twin of [`KeyringIdentity::entry_as`], for calls
111/// that run on the bounded worker thread.
112fn entry_in(store: &CredentialStore, service: &str) -> Result<Entry, IdentityError> {
113    store
114        .build(service, USER, None)
115        .map_err(|e| IdentityError::Store(format!("keyring entry: {e}")))
116}
117
118/// The `load` body, run inside [`bounded`] — `store` is the
119/// `Arc`'d credential store moved onto the worker thread. Reads
120/// probe [`CANONICAL_SERVICE`] first, then each of
121/// [`LEGACY_SERVICES`] in order — the first entry that exists
122/// wins, so a tardigrada-canonical identity shadows stale
123/// cuora/arca copies.
124fn load_from(store: &CredentialStore) -> Result<Option<x25519::Identity>, IdentityError> {
125    for service in std::iter::once(CANONICAL_SERVICE).chain(LEGACY_SERVICES.iter().copied()) {
126        let entry = entry_in(store, service)?;
127        let text = match entry.get_password() {
128            Ok(t) => t,
129            Err(keyring_core::Error::NoEntry) => continue,
130            Err(e) => return Err(IdentityError::Store(format!("keyring get: {e}"))),
131        };
132        // get_password returns a plain String holding raw key
133        // material — zeroize on drop (same hygiene as the
134        // keyfile body in cuora-identity-file).
135        let text = zeroize::Zeroizing::new(text);
136        let line = text.trim();
137        // Deliberate: a present-but-corrupt entry is a hard error,
138        // never a skip to the next service — silently falling back
139        // to an older identity could decrypt with the WRONG key and
140        // look like data loss. Surface the corruption.
141        return x25519::Identity::from_str(line)
142            .map(Some)
143            .map_err(|e| IdentityError::Parse(format!("keyring identity {service}/{USER}: {e}")));
144    }
145    Ok(None)
146}

148/// An age identity held in a `keyring-core` credential store.
149pub struct KeyringIdentity {
150    store: Arc<CredentialStore>,
151    /// Whether `store()` must claim the `default` collection alias
152    /// first — true only for the platform secret-service store
153    /// (the I-006 fixup). `with_store` callers own their store's
154    /// semantics, so a test/embedded store never triggers a real
155    /// dbus mutation.
156    ensure_collection: bool,
157}
158
159/// Does the platform store need the `default`-alias ensure before
160/// writes? Only the unix secret-service backend — Keychain and
161/// Credential Manager have no collection-alias concept.
162#[cfg(all(
163    unix,
164    not(any(target_os = "macos", target_os = "ios", target_os = "android")),
165    not(coverage)
166))]
167const PLATFORM_NEEDS_DEFAULT_COLLECTION: bool = true;
168
169#[cfg(not(all(
170    unix,
171    not(any(target_os = "macos", target_os = "ios", target_os = "android")),
172    not(coverage)
173)))]
174const PLATFORM_NEEDS_DEFAULT_COLLECTION: bool = false;
175
176impl KeyringIdentity {
177    /// The platform session store — the availability probe.
178    ///
179    /// Read-only: probing (and every `load`) never creates
180    /// collections or claims aliases — `tardigrada show`-style
181    /// consumers that only *load* the identity must not mutate the
182    /// keyring. The default-collection fixup runs lazily inside
183    /// `store`, the only mutating op.
184    ///
185    /// `Err(Unavailable)` means no session secret store is
186    /// reachable right now (headless, no dbus session, no
187    /// daemon); `Ok` means ops are possible but can still fail
188    /// per-call (daemon dies, collection locked).
189    pub fn platform() -> Result<Self, IdentityError> {
190        Ok(Self {
191            store: platform_store()?,
192            ensure_collection: PLATFORM_NEEDS_DEFAULT_COLLECTION,
193        })
194    }
195
196    /// Any `keyring-core` credential store — tests (the mock),
197    /// alternative backends, embeddings.
198    pub fn with_store(store: Arc<CredentialStore>) -> Self {
199        KeyringIdentity {
200            store,
201            ensure_collection: false,
202        }
203    }
204
205    /// The `tardigrada`/`age-identity` entry in this store — test
206    /// surface only; ops go through [`entry_in`] on the bounded
207    /// worker thread.
208    #[cfg(test)]
209    fn entry(&self) -> Result<Entry, IdentityError> {
210        self.store
211            .build(CANONICAL_SERVICE, USER, None)
212            .map_err(|e| IdentityError::Store(format!("keyring entry: {e}")))
213    }
214}

216impl IdentitySource for KeyringIdentity {
217    /// `Ok(None)` when no identity entry exists in the store;
218    /// store failures and unparseable material are `Err`.
219    fn load(&self) -> Result<Option<x25519::Identity>, IdentityError> {
220        let store = Arc::clone(&self.store);
221        bounded("load", move || load_from(&*store))
222    }
223
224    /// Write the identity into the store (upsert — the platform
225    /// store owns unlock/prompt policy, bounded by `bounded`).
226    fn store(&self, identity: &x25519::Identity) -> Result<(), IdentityError> {
227        let store = Arc::clone(&self.store);
228        let ensure = self.ensure_collection;
229        let key = identity.to_string();
230        bounded("store", move || {
231            // The only mutation point — claim the `default`
232            // collection alias here, never on the load/probe
233            // path (a read-only consumer must not create a
234            // `tardigrada` collection).
235            if ensure {
236                ensure_default_collection();
237            }
238            entry_in(&*store, CANONICAL_SERVICE)?
239                .set_password(key.expose_secret())
240                .map_err(|e| IdentityError::Store(format!("keyring set: {e}")))
241        })
242    }
243
244    fn kind(&self) -> &'static str {
245        "keyring"
246    }
247
248    fn describe(&self) -> String {
249        format!("system keyring (entry {CANONICAL_SERVICE}/{USER})")
250    }
251}

269/// Claim the `default` collection alias when the service has none.
270///
271/// Called only from the `store` path on platform-built identities —
272/// writes are the sole mutation this crate performs on the session
273/// store; `load`/probe stay read-only.
274///
275/// `dbus-secret-service-keyring-store`'s create-item path resolves
276/// the literal `default` alias; on a fresh Secret Service (daemon
277/// just spawned, no keyring unlocked yet) `ReadAlias` returns `/`
278/// → `NoResult`, surfaced as `NoStorageAccess` — the first
279/// `set_password` fails even though the service is reachable.
280/// gnome-keyring only honors the `default` alias (others are
281/// `NotSupported`), so the fix is always claiming *that* alias:
282/// re-alias an existing collection (the login keyring keeps the
283/// secret inside the user's real keyring) or create one when the
284/// service is empty. Best-effort: any failure here surfaces later
285/// as the store's own error — this never makes things worse.
286#[cfg(all(
287    unix,
288    not(any(target_os = "macos", target_os = "ios", target_os = "android")),
289    not(coverage)
290))]
291fn ensure_default_collection() {
292    use dbus_secret_service::{EncryptionType, SecretService};
293    // Prompt timeout 0 — a create that would block on a GUI
294    // prompter cancels instead; ops stay non-interactive.
295    let Ok(ss) = SecretService::connect_with_max_prompt_timeout(EncryptionType::Plain, 0) else {
296        return;
297    };
298    if ss.get_default_collection().is_ok() {
299        return;
300    }
301    // Collections exist but none owns `default` — claim the alias
302    // for the login keyring (else the first) via SetAlias, which
303    // dbus-secret-service doesn't expose publicly.
304    if let Ok(cols) = ss.get_all_collections() {
305        let pick = cols
306            .iter()
307            .find(|c| c.get_label().ok().as_deref() == Some("login"))
308            .or_else(|| cols.first());
309        if let Some(col) = pick {
310            if set_default_alias(&col.path) && ss.get_default_collection().is_ok() {
311                return;
312            }
313        }
314    }
315    // No usable collection — create one, then claim `default`
316    // explicitly: CreateCollection's alias arg is only a hint.
317    if ss.create_collection("tardigrada", "default").is_ok() {
318        if let Ok(cols) = ss.get_all_collections() {
319            if let Some(col) = cols
320                .iter()
321                .find(|c| c.get_label().ok().as_deref() == Some("tardigrada"))
322            {
323                set_default_alias(&col.path);
324            }
325        }
326    }
327}

329/// `org.freedesktop.Secret.Service.SetAlias` — gnome-keyring
330/// implements it but dbus-secret-service keeps the proxy private,
331/// so this is one raw call on a short timeout. `true` = the method
332/// returned without a dbus error.
333#[cfg(all(
334    unix,
335    not(any(target_os = "macos", target_os = "ios", target_os = "android")),
336    not(coverage)
337))]
338fn set_default_alias(collection: &dbus_secret_service::Path) -> bool {
339    let Ok(conn) = dbus::blocking::Connection::new_session() else {
340        return false;
341    };
342    let proxy = conn.with_proxy(
343        "org.freedesktop.secrets",
344        "/org/freedesktop/secrets",
345        std::time::Duration::from_secs(5),
346    );
347    proxy
348        .method_call::<(), _, _, _>(
349            "org.freedesktop.Secret.Service",
350            "SetAlias",
351            ("default", collection.clone()),
352        )
353        .is_ok()
354}

585        // Unavailable-vs-Ok branch is environment-dependent).
586        let _ = KeyringIdentity::platform();
587    }

598    /// A store whose `build` always fails — no daemon, no entry.

604        }

606            "dead".into()
607        }

609            Err(Error::PlatformFailure("dead bus".into()))
610        }
611        fn as_any(&self) -> &dyn Any {
612            self
613        }
614    }

616    #[test]
617    fn entry_build_failures_surface_as_store_errors() {
618        // entry's map_err arm — a typed error, never a panic
619        let store = Arc::new(DeadStore);
620        let k = KeyringIdentity::with_store(store.clone() as Arc<CredentialStore>);
621        assert!(matches!(k.load(), Err(IdentityError::Store(_))));
622        // the trait surface DeadStore implements, exercised
623        assert_eq!(
624            keyring_core::api::CredentialStoreApi::vendor(store.as_ref()),
625            "dead"
626        );
627        assert_eq!(
628            keyring_core::api::CredentialStoreApi::id(store.as_ref()),
629            "dead"
630        );
631        let _ = keyring_core::api::CredentialStoreApi::as_any(store.as_ref());
632    }

634    /// A store whose `build` never answers — the shape of a
635    /// secret-service unlock prompt: reachable bus, no deadline.
636    struct StuckStore;

647            Err(Error::PlatformFailure("unreachable".into()))

649        fn as_any(&self) -> &dyn Any {

651        }

654    #[test]
655    fn a_blocked_store_call_times_out_instead_of_hanging() {
656        // CUORA_KEYRING_TIMEOUT_MS shrinks the bound for the test;
657        // sibling tests in this binary only drive the in-memory
658        // mock (microseconds), so the tiny window can't flake them.
659        std::env::set_var("CUORA_KEYRING_TIMEOUT_MS", "200");
660        let k = KeyringIdentity::with_store(Arc::new(StuckStore) as Arc<CredentialStore>);
661        let e = k.load().err().unwrap().to_string();
662        std::env::remove_var("CUORA_KEYRING_TIMEOUT_MS");
663        assert!(e.contains("timed out"), "{e}");
664        // The timeout hint must stay backend-agnostic — this crate
665        // is shared (tardigrada consumes it too), so it names only
666        // the shared CUORA_KEYRING_TIMEOUT_MS knob, never cuora's
667        // own CUORA_IDENTITY_BACKEND.
668        assert!(e.contains("CUORA_KEYRING_TIMEOUT_MS"), "{e}");
669        assert!(!e.contains("CUORA_IDENTITY_BACKEND"), "{e}");
670    }
