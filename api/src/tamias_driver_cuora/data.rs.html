1//! data — one `cuora-ops` handle per wire op.
2//!
3//! The driver is a *reader*: every `view`/`query` builds an
4//! [`Ops`] exactly like the CLI does (`--config`/`--env`/`--actor`
5//! become the `config`/`env`/`actor` request params, then the same
6//! `CUORA_CONFIG` → `./cuora.toml` → `~/.config/cuora/cuora.toml`
7//! search `cuora` verbs use) and answers from vault *metadata* —
8//! refs, versions, timestamps, vault kinds and routes.
9//!
10//! **Secret values never leave this process.** `list_records`
11//! decrypts records to reach their bookkeeping fields; the
12//! `SecretRecord::value` is dropped the moment the row is built.
13//! The `secret` view reads a record only to report `stored` +
14//! bookkeeping — it never calls `Ops::get_in`, so no materialization
15//! audit event fires and no value can render. There is no reveal
16//! verb: `cuora get`/`cuora run` stay the only value edges.
17//!
18//! Degradation: a config/env/actor that fails [`Ops::load`], or a
19//! vault that won't open, becomes an `{"error": …}` payload inside
20//! an `ok:true` response — hosts render the failure inline, the
21//! same contract `cuora-tui` used for its "vault not readable"
22//! line.
23
24use std::collections::BTreeSet;
25
26use cuora_core::{config::CacheScope, Env, Ref};
27use cuora_ops::{CheckReport, LoadOpts, Ops};
28use serde_json::{json, Map, Value};

30/// Everything a wire op needs, loaded once per request.
31pub struct Ctx {
32    /// `Ops::load` result — `Err` carries the load failure string.
33    ops: Result<Ops, String>,
34    /// `ops.check()` memoized — the check view's three tables share
35    /// one probe round (remote vaults aren't free).
36    check: Option<CheckReport>,
37}
38
39impl Ctx {
40    /// Build the handle from request params. `config`/`env`/`actor`
41    /// map to the CLI's `--config`/`--env`/`--actor`; everything
42    /// else falls through to the usual env-var/default resolution.
43    pub fn open(params: &Map<String, Value>) -> Self {
44        let get = |k: &str| params.get(k).and_then(Value::as_str).map(str::to_string);
45        let ops = Ops::load(LoadOpts {
46            config_path: get("config"),
47            env: get("env"),
48            actor: get("actor"),
49        })
50        .map_err(|e| e.to_string());
51        Ctx { ops, check: None }
52    }
53
54    /// The loaded ops, or the load failure.
55    pub fn ops(&self) -> Result<&Ops, String> {
56        self.ops.as_ref().map_err(|e| e.clone())
57    }
58
59    /// `cuora check`'s report, computed once per request.
60    pub fn check_report(&mut self) -> Result<&CheckReport, String> {
61        if self.check.is_none() {
62            let report = self.ops()?.check();
63            self.check = Some(report);
64        }
65        match self.check.as_ref() {
66            Some(r) => Ok(r),
67            None => Err("check report missing".to_string()),
68        }
69    }
70
71    /// Every env worth a tab: well-known first, then `[envs]`
72    /// customs sorted, then the selected env if unlisted — the same
73    /// order `cuora-tui` built its tabs.
74    pub fn envs(&self) -> Vec<Env> {
75        let mut names: Vec<Env> = Env::WELL_KNOWN
76            .iter()
77            .filter_map(|n| Env::new(n).ok())
78            .collect();
79        if let Ok(ops) = self.ops() {
80            let mut customs: Vec<Env> = ops
81                .config()
82                .envs
83                .keys()
84                .filter(|e| !e.is_well_known())
85                .cloned()
86                .collect();
87            customs.sort();
88            names.extend(customs);
89            if !names.iter().any(|e| e == ops.env()) {
90                names.push(ops.env().clone());
91            }
92        }
93        names
94    }
95
96    /// Vault names the config knows: `[vaults]` keys ∪ every env
97    /// route's target ∪ the implicit `local`.
98    pub fn vault_names(&self) -> Vec<String> {
99        let mut names = BTreeSet::new();
100        names.insert(cuora_core::config::LOCAL_VAULT.to_string());
101        if let Ok(ops) = self.ops() {
102            names.extend(ops.config().vaults.keys().cloned());
103            for route in ops.config().envs.values() {
104                names.insert(route.vault().to_string());
105            }
106            names.insert(ops.config().vault_for(ops.env()).to_string());
107        }
108        names.into_iter().collect()
109    }
110
111    /// Envs routing to `vault` (for the vaults table's envs column).
112    pub fn envs_for(&self, vault: &str) -> Vec<String> {
113        let mut out = Vec::new();
114        if let Ok(ops) = self.ops() {
115            for env in self.envs() {
116                if ops.config().vault_for(&env) == vault {
117                    out.push(env.as_str().to_string());
118                }
119            }
120        }
121        out
122    }
123}
124
125/// `{"error": …}` — the degradation marker every renderer knows.
126pub fn error_payload(msg: impl Into<String>) -> Value {
127    json!({"error": msg.into()})
128}
129
130/// The data payload for one query name. Unknown names and failed
131/// loads degrade to `{"error": …}` — never a wire error.
132pub fn query_data(ctx: &mut Ctx, name: &str, params: &Map<String, Value>) -> Value {
133    match name {
134        "summary" => summary(ctx),
135        "vaults" => vaults_rows(ctx),
136        "envs" => envs_rows(ctx),
137        "declared" => declared_rows(ctx),
138        "refs" => refs_rows(ctx, params),
139        "tree" => tree_payload(ctx),
140        "secret" => secret_payload(ctx, params),
141        "check_info" => check_info(ctx),
142        "check_vaults" => check_vaults_rows(ctx),
143        "check_secrets" => check_secrets_rows(ctx),
144        "check_problems" => check_problems_rows(ctx),
145        other => error_payload(format!("unknown query '{other}'")),
146    }
147}

151/// `path`/`url` for a vault config, or an honest "default" marker.
152fn location(cfg: Option<&cuora_core::VaultConfig>) -> String {
153    match cfg {
154        Some(c) => c
155            .path
156            .clone()
157            .or_else(|| c.url.clone())
158            .unwrap_or_else(|| "implicit defaults".to_string()),
159        None => "implicit defaults".to_string(),
160    }
161}
162
163/// `cache` scope rendered for a table cell.
164fn cache_desc(cfg: Option<&cuora_core::VaultConfig>) -> String {
165    match cfg.and_then(|c| c.cache.as_ref()) {
166        None => "off".to_string(),
167        Some(CacheScope::All(true)) => "all envs".to_string(),
168        Some(CacheScope::All(false)) => "off".to_string(),
169        Some(CacheScope::One(e)) => e.as_str().to_string(),
170        Some(CacheScope::Envs(es)) => es.iter().map(|e| e.as_str()).collect::<Vec<_>>().join(", "),
171    }
172}

174/// RFC 3339 like the TUI's detail pane; "—" when absent.
175fn ts(t: Option<time::OffsetDateTime>) -> String {
176    match t {
177        Some(t) => t
178            .format(&time::format_description::well_known::Rfc3339)
179            .unwrap_or_else(|_| t.to_string()),
180        None => "—".to_string(),
181    }
182}

184/// `summary` — the statusbar/dashboard kv: where config came from,
185/// the env context, and surface counts. Pure config — no vault I/O,
186/// safe to poll.
187fn summary(ctx: &Ctx) -> Value {
188    let ops = match ctx.ops() {
189        Ok(o) => o,
190        Err(e) => return error_payload(e),
191    };
192    let cfg = ops.config();
193    let mut out = Map::new();
194    out.insert(
195        "config".into(),
196        json!(ops
197            .config_path()
198            .map(|p| p.display().to_string())
199            .unwrap_or_else(|| "defaults (no cuora.toml found)".into())),
200    );
201    out.insert("env".into(), json!(ops.env().as_str()));
202    out.insert("actor".into(), json!(ops.actor().as_str()));
203    out.insert("vault".into(), json!(cfg.vault_for(ops.env())));
204    out.insert("vaults".into(), json!(ctx.vault_names().len()));
205    out.insert("envs".into(), json!(ctx.envs().len()));
206    out.insert("declared_secrets".into(), json!(cfg.secrets.len()));
207    out.insert(
208        "identity_backend".into(),
209        json!(cfg
210            .identity
211            .as_ref()
212            .map(|i| format!("{:?}", i.backend).to_lowercase())
213            .unwrap_or_else(|| "keyring".into())),
214    );
215    Value::Object(out)
216}

218/// `vaults` — every known vault: kind, location, cache scope, the
219/// envs routed to it, and whether it's a configured `[vaults]`
220/// entry or an implicit default. Config-only — no opens, no probes
221/// (the `check` view owns live readability).
222fn vaults_rows(ctx: &Ctx) -> Value {
223    let ops = match ctx.ops() {
224        Ok(o) => o,
225        Err(e) => return error_payload(e),
226    };
227    let rows: Vec<Value> = ctx
228        .vault_names()
229        .into_iter()
230        .map(|name| {
231            let cfg = ops.config().vault_config(&name);
232            let configured = cfg.is_some() || name == cuora_core::config::LOCAL_VAULT;
233            json!({
234                "name": name,
235                "kind": cfg.map(|c| c.kind.clone()).unwrap_or_else(|| "age".into()),
236                "location": location(cfg),
237                "cache": cache_desc(cfg),
238                "envs": ctx.envs_for(&name).join(", "),
239                "configured": if configured { "yes" } else { "routed only" },
240            })
241        })
242        .collect();
243    Value::Array(rows)
244}
245
246/// `envs` — the env → vault routing table (the TUI's tab strip).
247fn envs_rows(ctx: &Ctx) -> Value {
248    let ops = match ctx.ops() {
249        Ok(o) => o,
250        Err(e) => return error_payload(e),
251    };
252    let rows: Vec<Value> = ctx
253        .envs()
254        .into_iter()
255        .map(|env| {
256            let vault = ops.config().vault_for(&env).to_string();
257            let cfg = ops.config().vault_config(&vault);
258            json!({
259                "env": env.as_str(),
260                "well_known": if env.is_well_known() { "yes" } else { "" },
261                "vault": vault,
262                "kind": cfg.map(|c| c.kind.clone()).unwrap_or_else(|| "age".into()),
263                "location": location(cfg),
264            })
265        })
266        .collect();
267    Value::Array(rows)
268}

270/// `declared` — the `[secrets]` injection surface: env-var name →
271/// ref, resolved env + vault. Names and refs only.
272fn declared_rows(ctx: &Ctx) -> Value {
273    let ops = match ctx.ops() {
274        Ok(o) => o,
275        Err(e) => return error_payload(e),
276    };
277    let rows: Vec<Value> = ops
278        .config()
279        .secrets
280        .iter()
281        .map(|(var, rref)| match ops.resolve(rref) {
282            Ok((env, stripped)) => json!({
283                "var": var,
284                "ref": rref.to_string(),
285                "resolved_ref": stripped.to_string(),
286                "env": env.as_str(),
287                "vault": ops.config().vault_for(&env),
288            }),
289            Err(e) => json!({
290                "var": var,
291                "ref": rref.to_string(),
292                "error": e.to_string(),
293            }),
294        })
295        .collect();
296    Value::Array(rows)
297}

299/// `refs` — the TUI's core list: refs + bookkeeping for one env
300/// (`--param env=` picks it; default is the selected env). Values
301/// are dropped inside `list_records` — rows carry metadata only.
302fn refs_rows(ctx: &Ctx, _params: &Map<String, Value>) -> Value {
303    let ops = match ctx.ops() {
304        Ok(o) => o,
305        Err(e) => return error_payload(e),
306    };
307    let env = ops.env().clone();
308    match ops.list_records(Some(&env)) {
309        Ok((_, records)) => {
310            let vault = ops.config().vault_for(&env).to_string();
311            Value::Array(
312                records
313                    .into_iter()
314                    .map(|(rref, rec)| {
315                        // `rec.value` is a redacting Secret and is
316                        // never read — bookkeeping only.
317                        json!({
318                            "ref": rref.to_string(),
319                            "version": rec.version,
320                            "created": ts(Some(rec.created_at)),
321                            "rotated": ts(rec.rotated_at),
322                            "vault": vault,
323                        })
324                    })
325                    .collect(),
326            )
327        }
328        Err(e) => error_payload(format!("env {}: {e}", env.as_str())),
329    }
330}

332/// `tree` — env → refs hierarchy, the TUI's env tabs as a tree.
333/// Uses `list` (paths only, no per-record decrypt) so the browse
334/// stays cheap; per-env failures become the node's detail text.
335fn tree_payload(ctx: &Ctx) -> Value {
336    let mut items = Vec::new();
337    for env in ctx.envs() {
338        match ctx.ops().and_then(|ops| {
339            ops.list(Some(&env))
340                .map(|(_, refs)| refs)
341                .map_err(|e| e.to_string())
342        }) {
343            Ok(refs) => {
344                let children: Vec<pavo_protocol::TreeItem> = refs
345                    .iter()
346                    .map(|r| {
347                        let mut item = pavo_driver_kit::tree_item(r.path());
348                        item.id = Some(r.to_string());
349                        item
350                    })
351                    .collect();
352                let mut env_item = pavo_driver_kit::tree_item(env.as_str());
353                env_item.id = Some(env.as_str().to_string());
354                env_item.detail = Some(format!("{} refs", children.len()));
355                env_item.children = children;
356                items.push(env_item);
357            }
358            Err(e) => {
359                let mut item = pavo_driver_kit::tree_item(env.as_str());
360                item.id = Some(env.as_str().to_string());
361                item.detail = Some(format!("unreadable: {e}"));
362                items.push(item);
363            }
364        }
365    }
366    json!({"items": items})
367}

369/// `secret` — one ref's metadata pane (the TUI detail pane minus
370/// the reveal). `ref` param is required (`cuora://…` or legacy
371/// `arca://…`); env pins resolve through `Ops::resolve` like the
372/// CLI. Reads the record for bookkeeping only — the value is never
373/// exposed and no materialization audit fires.
374fn secret_payload(ctx: &Ctx, params: &Map<String, Value>) -> Value {
375    let raw = match params.get("ref").and_then(Value::as_str) {
376        Some(r) => r,
377        None => {
378            return error_payload(
379                "missing required param 'ref' (e.g. --param ref=cuora://billing/stripe)",
380            )
381        }
382    };
383    let rref = match Ref::new(raw) {
384        Ok(r) => r,
385        Err(e) => return error_payload(format!("bad ref '{raw}': {e}")),
386    };
387    let ops = match ctx.ops() {
388        Ok(o) => o,
389        Err(e) => return error_payload(e),
390    };
391    let (env, stripped) = match ops.resolve(&rref) {
392        Ok(pair) => pair,
393        Err(e) => return error_payload(e.to_string()),
394    };
395    let vault_name = ops.config().vault_for(&env).to_string();
396    let mut out = Map::new();
397    out.insert("ref".into(), json!(rref.to_string()));
398    out.insert("env".into(), json!(env.as_str()));
399    out.insert("vault".into(), json!(vault_name));
400    let vault = match ops.open_vault(&env) {
401        Ok(v) => v,
402        Err(e) => return error_payload(format!("env {}: {e}", env.as_str())),
403    };
404    match vault.get(&env, &stripped) {
405        Ok(Some(rec)) => {
406            out.insert("stored".into(), json!("yes"));
407            out.insert("version".into(), json!(rec.version));
408            out.insert("created".into(), json!(ts(Some(rec.created_at))));
409            out.insert("rotated".into(), json!(ts(rec.rotated_at)));
410        }
411        Ok(None) => {
412            out.insert("stored".into(), json!("no"));
413        }
414        Err(e) => return error_payload(format!("env {}: {e}", env.as_str())),
415    }
416    Value::Object(out)
417}

421/// `check_info` — the kv header for the check view.
422fn check_info(ctx: &mut Ctx) -> Value {
423    let report = match ctx.check_report() {
424        Ok(r) => r,
425        Err(e) => return error_payload(e),
426    };
427    let ok = report.vaults.iter().filter(|v| v.ok).count();
428    let resolved = report.secrets.iter().filter(|s| s.resolves).count();
429    json!({
430        "config": report.config_path.as_ref().map(|p| p.display().to_string())
431            .unwrap_or_else(|| "defaults".into()),
432        "env": report.env.as_str(),
433        "vault": report.vault,
434        "vaults_ok": format!("{ok}/{}", report.vaults.len()),
435        "secrets_resolved": format!("{resolved}/{}", report.secrets.len()),
436        "problems": report.problems.len(),
437    })
438}

440/// `check_vaults` — one row per probed vault (`cuora check`'s
441/// readability table).
442fn check_vaults_rows(ctx: &mut Ctx) -> Value {
443    let report = match ctx.check_report() {
444        Ok(r) => r,
445        Err(e) => return error_payload(e),
446    };
447    Value::Array(
448        report
449            .vaults
450            .iter()
451            .map(|v| {
452                json!({
453                    "name": v.name,
454                    "kind": v.kind,
455                    "path": v.path.as_ref().map(|p| p.display().to_string()).unwrap_or_default(),
456                    "ok": if v.ok { "yes" } else { "no" },
457                    "error": v.error.clone().unwrap_or_default(),
458                })
459            })
460            .collect(),
461    )
462}

464/// `check_secrets` — the declared `[secrets]` surface with
465/// resolution status.
466fn check_secrets_rows(ctx: &mut Ctx) -> Value {

479                    "env": s.env,
480                    "resolves": if s.resolves { "yes" } else { "no" },
