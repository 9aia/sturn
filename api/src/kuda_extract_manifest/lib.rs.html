1//! Manifest adapter — package manifests -> module graph. No parser,
2//! no language knowledge: reads manifest basenames declared by
3//! `LangDef` entries (Cargo.toml, package.json, go.mod, …).
4
5use kuda_core::{DepEdge, EdgeKind, LangDef, Symbol, SymbolKind};
6use kuda_extract::{Extraction, Extractor};
7use std::collections::HashSet;
8use std::path::Path;
9
10/// `excludes`: extra dir names from `[extract] exclude`.
11/// `resolve_git_deps`: from `[workspace] resolve_git_deps` — emit
12/// a Cargo `git = "<url>"` dep's target as the URL's repo basename
13/// so the edge lands on a sibling repo module.
14pub struct ManifestExtractor {
15    pub excludes: Vec<String>,
16    pub resolve_git_deps: bool,
17}
18
19impl Extractor for ManifestExtractor {
20    fn kind(&self) -> &'static str {
21        "manifest"
22    }
23
24    fn extract(&self, root: &Path, langs: &[LangDef]) -> Result<Extraction, String> {
25        let names: HashSet<&str> = langs
26            .iter()
27            .flat_map(|l| l.manifests.iter().map(String::as_str))
28            .collect();
29        let mut out = Extraction::default();
30        let mut stack = vec![root.to_path_buf()];
31        while let Some(dir) = stack.pop() {
32            let Ok(entries) = std::fs::read_dir(&dir) else {
33                continue;
34            };
35            for e in entries.flatten() {
36                let path = e.path();
37                let name = e.file_name();
38                let name = name.to_string_lossy();
39                if path.is_dir() {
40                    // skip noise dirs + mounted git worktrees (.git
41                    // as a *file* marks a worktree, not a repo dir)
42                    if !kuda_core::dir_excluded(name.as_ref(), &self.excludes)
43                        && !path.join(".git").is_file()
44                    {
45                        stack.push(path);
46                    }
47                    continue;
48                }
49                if !names.contains(name.as_ref()) {
50                    continue;
51                }
52                let rel = path.strip_prefix(root).unwrap_or(&path).to_path_buf();
53                let text = std::fs::read_to_string(&path).unwrap_or_default();
54                let (pkg, deps) = parse_manifest(name.as_ref(), &text, self.resolve_git_deps);
55                let module = pkg.unwrap_or_else(|| {
56                    path.parent()
57                        .and_then(|p| p.file_name())
58                        .map(|f| f.to_string_lossy().into_owned())
59                        .unwrap_or_else(|| name.clone().into_owned())
60                });
61                out.symbols.push(Symbol {
62                    name: module.clone(),
63                    kind: SymbolKind::Module,
64                    file: rel.clone(),
65                    line: 1,
66                    module: module.clone(),
67                });
68                for d in deps {
69                    out.edges.push(DepEdge {
70                        from: module.clone(),
71                        to: d,
72                        kind: EdgeKind::PackageDep,
73                    });
74                }
75                out.files.push(rel);
76            }
77        }
78        Ok(out)
79    }
80}

82/// Manifest basename -> (package name, declared deps). Format
83/// detection by filename — adapters are allowed to know concrete tech.
84fn parse_manifest(
85    basename: &str,
86    text: &str,
87    resolve_git_deps: bool,
88) -> (Option<String>, Vec<String>) {
89    match basename {
90        "Cargo.toml" => cargo(text, resolve_git_deps),
91        "package.json" => package_json(text),
92        "pyproject.toml" => pyproject(text),
93        "setup.py" => setup_py(text),
94        "go.mod" => go_mod(text),
95        _ => (None, Vec::new()),
96    }
97}

99/// PEP 508 requirement string -> bare package name
100/// ("neuralmind-shared[extra]>=1.0; python>=3.9" -> "neuralmind-shared").
101fn pep508_name(req: &str) -> String {
102    req.trim()
103        .split([' ', '[', '<', '>', '=', '~', '!', ';', '('])
104        .next()
105        .unwrap_or("")
106        .to_owned()
107}

109fn pyproject(text: &str) -> (Option<String>, Vec<String>) {
110    let Ok(t) = text.parse::<toml::Table>() else {
111        return (None, Vec::new());
112    };
113    let project = t.get("project").and_then(|p| p.as_table());
114    let poetry = t
115        .get("tool")
116        .and_then(|t| t.get("poetry"))
117        .and_then(|p| p.as_table());
118    let name = project
119        .and_then(|p| p.get("name"))
120        .and_then(|n| n.as_str())
121        .or_else(|| poetry.and_then(|p| p.get("name")).and_then(|n| n.as_str()))
122        .map(str::to_owned);
123    let mut deps = Vec::new();
124    // PEP 621: dependencies = ["a>=1", ...]; optional-dependencies = {grp=[...]}
125    for key in ["dependencies", "optional-dependencies"] {
126        match project.and_then(|p| p.get(key)) {
127            Some(toml::Value::Array(a)) => {
128                deps.extend(a.iter().filter_map(|v| v.as_str()).map(pep508_name))
129            }
130            Some(toml::Value::Table(tbl)) => {
131                for v in tbl.values().filter_map(|v| v.as_array()).flatten() {
132                    if let Some(s) = v.as_str() {
133                        deps.push(pep508_name(s));
134                    }
135                }
136            }
137            _ => {}
138        }
139    }
140    // Poetry: [tool.poetry.dependencies] + dev-dependencies table keys
141    // (skip `python`); group deps nest one level deeper.
142    let mut poetry_tables: Vec<&toml::map::Map<String, toml::Value>> = Vec::new();
143    for key in ["dependencies", "dev-dependencies"] {
144        poetry_tables.extend(poetry.and_then(|p| p.get(key)).and_then(|d| d.as_table()));
145    }
146    if let Some(groups) = poetry
147        .and_then(|p| p.get("group"))
148        .and_then(|g| g.as_table())
149    {
150        for g in groups.values() {
151            poetry_tables.extend(g.get("dependencies").and_then(|d| d.as_table()));
152        }
153    }
154    for tbl in poetry_tables {
155        deps.extend(tbl.keys().filter(|k| k.as_str() != "python").cloned());
156    }
157    // PEP 735 / uv: [dependency-groups] dev = ["pytest>=8", ...]
158    if let Some(groups) = t.get("dependency-groups").and_then(|g| g.as_table()) {
159        for v in groups.values().filter_map(|v| v.as_array()).flatten() {
160            if let Some(s) = v.as_str() {
161                deps.push(pep508_name(s));
162            }
163        }
164    }
165    deps.retain(|d| !d.is_empty());
166    (name, deps)
167}

169/// `name=` kwarg scan — setup.py deps live in code, not data; the
170/// package name alone is enough for module identity.
171fn setup_py(text: &str) -> (Option<String>, Vec<String>) {
172    let name = text
173        .split("name=")
174        .nth(1)
175        .and_then(|rest| rest.trim_start().strip_prefix(['\'', '"']))
176        .and_then(|rest| rest.split(['\'', '"']).next())
177        .map(str::to_owned)
178        .filter(|n| !n.is_empty());
179    (name, Vec::new())
180}

182fn cargo(text: &str, resolve_git_deps: bool) -> (Option<String>, Vec<String>) {
183    let Ok(t) = text.parse::<toml::Table>() else {
184        return (None, Vec::new());
185    };
186    let name = t
187        .get("package")
188        .and_then(|p| p.get("name"))
189        .and_then(|n| n.as_str())
190        .map(str::to_owned);
191    let mut deps = Vec::new();
192    for key in ["dependencies", "dev-dependencies", "build-dependencies"] {
193        if let Some(tbl) = t.get(key).and_then(|d| d.as_table()) {
194            deps.extend(
195                tbl.iter()
196                    .map(|(k, v)| cargo_dep_name(k, v, resolve_git_deps)),
197            );
198        }
199    }
200    (name, deps)
201}

203/// Cargo dep-table entry -> edge target. `package = "real"` renames
204/// win (with or without `git`); then, when `resolve_git_deps` is on,
205/// a `git = "<url>"` source yields the URL's repo basename (last
206/// path segment minus `.git`). `workspace = true` and everything
207/// else keep the table key.
208fn cargo_dep_name(key: &str, v: &toml::Value, resolve_git_deps: bool) -> String {
209    let Some(tbl) = v.as_table() else {
210        return key.to_owned();
211    };
212    if let Some(pkg) = tbl.get("package").and_then(|p| p.as_str()) {
213        return pkg.to_owned();
214    }
215    if resolve_git_deps && !tbl.contains_key("workspace") {
216        if let Some(url) = tbl.get("git").and_then(|g| g.as_str()) {
217            let base = url
218                .trim_end_matches('/')
219                .rsplit('/')
220                .next()
221                .unwrap_or_default();
222            let base = base.strip_suffix(".git").unwrap_or(base);
223            if !base.is_empty() {
224                return base.to_owned();
225            }
226        }
227    }
228    key.to_owned()
229}

231fn package_json(text: &str) -> (Option<String>, Vec<String>) {
232    let Ok(v) = serde_json::from_str::<serde_json::Value>(text) else {
233        return (None, Vec::new());
234    };
235    let name = v.get("name").and_then(|n| n.as_str()).map(str::to_owned);
236    let mut deps = Vec::new();
237    for key in ["dependencies", "devDependencies"] {
238        if let Some(tbl) = v.get(key).and_then(|d| d.as_object()) {
239            deps.extend(tbl.keys().cloned());
240        }
241    }
242    (name, deps)
243}

245fn go_mod(text: &str) -> (Option<String>, Vec<String>) {
246    let name = text
247        .lines()
248        .find_map(|l| l.strip_prefix("module ").map(str::trim).map(str::to_owned));
249    let deps = text
250        .lines()
251        .filter(|l| l.starts_with('\t') && l.contains(" v"))
252        .filter_map(|l| l.split_whitespace().next().map(str::to_owned))
253        .collect();
254    (name, deps)
255}

288    fn extract_walks_tree_and_names_modules() {
289        let root = tmp("walk");
290        let mk = |p: &str, text: &str| {
291            let f = root.join(p);
292            std::fs::create_dir_all(f.parent().unwrap()).unwrap();
293            std::fs::write(&f, text).unwrap();
294        };
295        mk(
296            "crates/a/Cargo.toml",
297            "[package]\nname = \"a\"\n[dependencies]\nb = { path = \"../b\" }\n",
298        );
299        // manifest without a package name -> falls back to dir name
300        mk(
301            "crates/noname/Cargo.toml",
302            "[dependencies]\nserde = \"1\"\n",
303        );
304        // manifest at repo root: parent dir-name fallback uses the basename
305        mk("pyproject.toml", "not [valid toml");
306        // unknown basename -> ignored entirely
307        mk("crates/a/random.txt", "x");
308        // excluded + noise dirs skipped
309        mk("target/nested/Cargo.toml", "[package]\nname = \"skipme\"\n");
310        mk("vendored/Cargo.toml", "[package]\nname = \"vend\"\n");
311        // a .git *file* marks a mounted worktree -> not walked
312        mk("wt/.git", "gitdir: ../.git/worktrees/wt");
313        mk("wt/Cargo.toml", "[package]\nname = \"wt\"\n");
314        // unparsable manifest still yields a module symbol
315        mk("bad/package.json", "{oops");
316        // unreadable dir -> read_dir fails, skipped quietly
317        mk("locked/Cargo.toml", "[package]\nname = \"locked\"\n");
318        #[cfg(unix)]
319        {
320            use std::os::unix::fs::PermissionsExt;
321            std::fs::set_permissions(root.join("locked"), std::fs::Permissions::from_mode(0o000))
322                .unwrap();
323        }
324
325        let x = ManifestExtractor {
326            excludes: vec!["vendored".into()],
327            resolve_git_deps: true,
328        };
329        let out = x.extract(&root, &[rust_lang()]).unwrap();
330        assert_eq!(x.kind(), "manifest");
331        let mods: Vec<&str> = out.symbols.iter().map(|s| s.module.as_str()).collect();
332        assert!(mods.contains(&"a"), "{mods:?}");
333        assert!(mods.contains(&"noname"), "{mods:?}");
334        // invalid toml -> no pkg name -> dir-name fallback (root has a name)
335        assert!(mods.iter().any(|m| !m.is_empty()));
336        assert!(!mods.contains(&"skipme"));
337        assert!(!mods.contains(&"vend"));
338        assert!(!mods.contains(&"wt"));
339        // dep edge recorded
340        assert!(out.edges.iter().any(|e| e.to == "b"));
341        assert!(out.files.iter().any(|f| f.ends_with("Cargo.toml")));
342        #[cfg(unix)]
343        {
344            use std::os::unix::fs::PermissionsExt;
345            std::fs::set_permissions(root.join("locked"), std::fs::Permissions::from_mode(0o755))
346                .unwrap();
347        }
348        let _ = std::fs::remove_dir_all(&root);
349    }

369    #[test]
370    fn parse_manifest_edge_cases() {
371        // unknown basename
372        assert_eq!(parse_manifest("Brewfile", "x", false), (None, vec![]));
373        // invalid bodies -> (None, [])
374        assert_eq!(
375            parse_manifest("Cargo.toml", "not [toml", false),
376            (None, vec![])
377        );
378        assert_eq!(
379            parse_manifest("package.json", "{bad", false),
380            (None, vec![])
381        );
382        assert_eq!(
383            parse_manifest("pyproject.toml", "not [toml", false),
384            (None, vec![])
385        );
386        // setup.py without name=
387        assert_eq!(
388            parse_manifest("setup.py", "setup(version='1')", false),
389            (None, vec![])
390        );
391        // setup.py name with double quotes
392        let (n, _) = parse_manifest("setup.py", "setup(name=\"dq\")", false);
393        assert_eq!(n.as_deref(), Some("dq"));
394        // go.mod: module line + requires
395        let (n, deps) = parse_manifest(
396            "go.mod",
397            "module github.com/org/mod\n\ngo 1.21\n\nrequire (\n\tdep.one v1.2.3\n\tdep.two v0.0.1\n)\n",
398            false,
399        );
400        assert_eq!(n.as_deref(), Some("github.com/org/mod"));
401        assert_eq!(deps, vec!["dep.one", "dep.two"]);
402        // go.mod without module line
403        let (n, _) = parse_manifest("go.mod", "go 1.21\n", false);
404        assert_eq!(n, None);
405        // package.json: devDependencies too, missing name
406        let (n, deps) = parse_manifest(
407            "package.json",
408            "{\"devDependencies\":{\"vitest\":\"1\"},\"dependencies\":{\"react\":\"18\"}}",
409            false,
410        );
411        assert_eq!(n, None);
412        for d in ["vitest", "react"] {
413            assert!(deps.iter().any(|x| x == d));
414        }
415        // cargo: build-dependencies + dev-dependencies tables
416        let (_, deps) = parse_manifest(
417            "Cargo.toml",
418            "[package]\nname=\"p\"\n[dev-dependencies]\nd = \"1\"\n[build-dependencies]\nb = \"1\"\n",
419            false,
420        );
421        assert!(deps.iter().any(|x| x == "d") && deps.iter().any(|x| x == "b"));
422        // cargo: dep entries as bare strings (non-table)
423        let (_, deps) = parse_manifest("Cargo.toml", "[dependencies]\nserde = \"1\"\n", true);
424        assert!(deps.iter().any(|x| x == "serde"));
425        // cargo: git dep with empty basename -> keeps key
426        let (_, deps) = parse_manifest(
427            "Cargo.toml",
428            "[dependencies]\nh = { git = \"https://github.com/org/.git\" }\n",
429            true,
430        );
431        assert!(deps.iter().any(|x| x == "h"), "{deps:?}");
432        // pyproject: optional-dependencies as table of arrays + empty dep names dropped
433        let (n, deps) = parse_manifest(
434            "pyproject.toml",
435            "[project.optional-dependencies]\ndev = [\"pytest\"]\n[project]\ndependencies = [\"\"]\n",
436            false,
437        );
438        assert_eq!(n, None);
439        assert!(deps.iter().any(|x| x == "pytest"));
440        assert!(!deps.iter().any(|x| x.is_empty()));
441        // pyproject: poetry dev-dependencies + no project section
442        let (_, deps) = parse_manifest(
443            "pyproject.toml",
444            "[tool.poetry.dev-dependencies]\npytest = \"^8\"\n",
445            false,
446        );
447        assert!(deps.iter().any(|x| x == "pytest"));
448    }
