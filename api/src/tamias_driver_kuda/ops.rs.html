1//! Engine ops — `cli.<verb>` wire queries dispatch into the `kuda`
2//! engine lib through this binary's own `run` verb: each call
3//! re-execs `tamias-driver-kuda run <argv>` with cwd set to the
4//! request's repo, so `kuda::dispatch` writes to the child's
5//! captured stdout and never touches the serve loop's ndjson stream
6//! (the engine's `emit()` exits on broken pipe — one more reason
7//! ops stay out of the serve process).
8//!
9//! `ops` answers the ops manifest as JSON — *derived from the
10//! engine's own clap tree* ([`kuda::command`]), so a verb can't
11//! exist on the CLI and be missing here: declared args become
12//! typed params (`arg`/`opt`/`flag`), doc strings ride along
13//! unchanged. The CLI's globals (`--config`, `--workspace`) ride
14//! the manifest's `globals` list once, not per-op.
15//!
16//! `watch`/`mcp`/`acp` can never be request/response ops — they
17//! never return, so they ride `surfaces` instead and reach users
18//! through the `bins[]` shim passthrough (`cli <argv>` →
19//! `kuda::run_from` with inherited stdio) or the feature-gated
20//! `mcp` surface — `tamias-driver-kuda cli watch` ≡ `kuda watch`.
21//! Hidden plumbing (`gen-cli-docs`) stays reachable through `run`
22//! passthrough only.
23//!
24//! `mutates` is honest: `init` scaffolds `kuda.toml` + `.kuda/`,
25//! `sync` rewrites cards + the sqlite index; `status`/`map`/
26//! `card`/`find`/`dependents`/`refs`/`doctor` only read.
27//! Streaming *flags* would join `SKIP_PARAMS` — none exist among
28//! bounded verbs today (`watch --poll-refs` rides the surface).
29
30use clap::{Arg, ArgAction, Command};
31use pavo_protocol::wire::{QueryResult, Response};
32use serde_json::{json, Map, Value};
33use std::path::Path;
34
35/// Unbounded surfaces — verbs that never return (`watch`'s
36/// debounced loop, `mcp`/`acp` stdio servers). They can never be
37/// request/response ops; they need inherited stdio.
38const SURFACES: &[&str] = &["watch", "mcp", "acp"];
39
40/// Params excluded from ops — flags that turn a bounded verb into
41/// a stream, or params that must never ride argv. None today.
42const SKIP_PARAMS: &[&str] = &[];
43
44/// Read-only verbs — everything else (`init`, `sync`) writes
45/// `kuda.toml`/`.kuda/` cards + the sqlite index. Mutators belong
46/// behind a deliberate trigger, not ambient view polling.
47const READ_ONLY: &[&str] = &[
48    "status",
49    "map",
50    "card",
51    "find",
52    "dependents",
53    "refs",
54    "doctor",
55];
56
57/// `(name, clap spec)` for every leaf verb. Hidden plumbing and
58/// clap's `help` are skipped — they stay reachable through `run`
59/// passthrough.
60fn verbs() -> Vec<(String, Command)> {
61    fn walk(prefix: &str, cmd: &Command, out: &mut Vec<(String, Command)>) {
62        for sub in cmd.get_subcommands() {
63            let name = sub.get_name();
64            if name == "help" || sub.is_hide_set() {
65                continue;
66            }
67            let dotted = match prefix {
68                "" => name.to_string(),
69                _ => format!("{prefix}.{name}"),
70            };
71            if sub.has_subcommands() {
72                walk(&dotted, sub, out);
73            } else {
74                out.push((dotted, sub.clone()));
75            }
76        }
77    }
78    let mut out = Vec::new();
79    walk("", &kuda::command(), &mut out);
80    out
81}
82
83fn op_cmd(name: &str) -> Option<Command> {
84    verbs()
85        .into_iter()
86        .find(|(n, _)| n == name && !SURFACES.contains(&n.as_str()))
87        .map(|(_, c)| c)
88}
89
90/// Args that can't ride the wire: streaming flags (above) and
91/// clap's own help/version actions. Globals are handled separately
92/// — listed once on the manifest, prepended in [`argv`], never
93/// per-op (callers filter `is_global_set()` themselves).
94fn skipped(a: &Arg) -> bool {
95    SKIP_PARAMS.contains(&a.get_id().as_str())
96        || matches!(
97            a.get_action(),
98            ArgAction::Help | ArgAction::HelpShort | ArgAction::HelpLong | ArgAction::Version
99        )
100}
101
102/// A clap arg → the manifest's typed param kind — `arg` =
103/// positional, `opt` = `--name <v>`, `flag` = `--name` — the
104/// contract `pavo-driver-kit`'s surface layer renders args from.
105fn kind(a: &Arg) -> &'static str {
106    if a.is_positional() {
107        "arg"
108    } else if matches!(
109        a.get_action(),
110        ArgAction::SetTrue | ArgAction::SetFalse | ArgAction::Count
111    ) {
112        "flag"
113    } else {
114        "opt"
115    }
116}

118fn param_json(a: &Arg) -> Value {
119    // Wire name = the flag name (`--poll-refs` → `poll-refs`), not
120    // the field id — generated surfaces spell params `--<name>`.
121    let id = a.get_id().as_str();
122    let name = match a.is_positional() {
123        true => id,
124        false => a.get_long().unwrap_or(id),
125    };
126    json!({
127        "name": name,
128        "kind": kind(a),
129        "doc": a.get_help().map(|d| d.to_string()).unwrap_or_default(),
130    })
131}

133/// The `ops` query/`ops` argv verb payload — self-describing so
134/// hosts and agents discover the engine surface without argv help.
135/// Globals (`--config`, `--workspace`) are accepted by every op —
136/// listed once on the manifest; the child gets them as argv flags.
137pub fn manifest() -> Value {
138    json!({
139        "engine": "kuda",
140        "invoke": "query `cli.<verb>` with the listed params, or argv `tamias-driver-kuda run <kuda argv>`",
141        "globals": kuda::command()
142            .get_arguments()
143            .filter(|a| a.is_global_set() && !skipped(a))
144            .map(param_json)
145            .collect::<Vec<_>>(),
146        "ops": verbs()
147            .iter()
148            .filter(|(n, _)| !SURFACES.contains(&n.as_str()))
149            .map(|(n, c)| json!({
150                "query": format!("cli.{n}"),
151                "doc": c.get_about().map(|d| d.to_string()).unwrap_or_default(),
152                "params": c.get_arguments()
153                    .filter(|a| !skipped(a) && !a.is_global_set())
154                    .map(param_json)
155                    .collect::<Vec<_>>(),
156                "mutates": !READ_ONLY.contains(&n.as_str()),
157            }))
158            .collect::<Vec<_>>(),
159        "surfaces": SURFACES,
160    })
161}

163/// Is `query` an engine op (`cli.<verb>`)? Returns the verb name.
164pub fn op_of(query: &str) -> Option<String> {
165    query
166        .strip_prefix("cli.")
167        .filter(|n| op_cmd(n).is_some())
168        .map(str::to_string)
169}

171/// A wire scalar → its argv token (numbers/bools stringify).
172fn scalar(v: &Value) -> Option<String> {
173    match v {
174        Value::String(s) => Some(s.clone()),
175        Value::Number(n) => Some(n.to_string()),
176        Value::Bool(b) => Some(b.to_string()),
177        _ => None,
178    }
179}
180
181/// All values of a param — arrays flatten, scalars wrap.
182fn values(v: &Value) -> Vec<String> {
183    match v {
184        Value::Array(xs) => xs.iter().filter_map(scalar).collect(),
185        other => scalar(other).into_iter().collect(),
186    }
187}

189/// Emit `--<long> <v>` / `--<long>` for one arg when `params`
190/// carries it — looked up by wire name (long flag), field id, and
191/// kebab-case id, so callers may spell params either way. Flags
192/// read truthiness; opts/args take every value.
193fn push_param(out: &mut Vec<String>, a: &Arg, params: &Map<String, Value>) {
194    let id = a.get_id().as_str();
195    let long = a.get_long().unwrap_or(id);
196    let kebab = id.replace('_', "-");
197    let raw = params
198        .get(long)
199        .or_else(|| params.get(id))
200        .or_else(|| params.get(kebab.as_str()));
201    let Some(raw) = raw else { return };
202    match kind(a) {
203        "flag" => {
204            if raw.as_bool().unwrap_or(false) {
205                out.push(format!("--{long}"));
206            }
207        }
208        _ => {
209            for s in values(raw) {
210                out.push(format!("--{long}"));
211                out.push(s);
212            }
213        }
214    }
215}

217/// `cli.<verb>` params → kuda argv (no argv\[0\] — `run` prepends
218/// it). Globals go first (clap accepts them either side of the
219/// verb), then the verb's named flags, then its positionals in
220/// schema order.
221fn argv(name: &str, cmd: &Command, params: &Map<String, Value>) -> Vec<String> {
222    let mut out: Vec<String> = Vec::new();
223    for a in kuda::command().get_arguments() {
224        if a.is_global_set() && !skipped(a) {
225            push_param(&mut out, a, params);
226        }
227    }
228    out.extend(name.split('.').map(String::from));
229    for a in cmd.get_arguments() {
230        if skipped(a) || a.is_positional() || a.is_global_set() {
231            continue;
232        }
233        push_param(&mut out, a, params);
234    }
235    // `trailing_var_arg`/`last` positionals only parse after a
236    // `--` separator — none today, kept generic.
237    if cmd.get_positionals().any(|a| a.is_last_set()) {
238        out.push("--".into());
239    }
240    let mut positionals: Vec<String> = Vec::new();
241    for a in cmd.get_positionals() {
242        if skipped(a) {
243            continue;
244        }
245        let id = a.get_id().as_str();
246        let raw = params
247            .get(id)
248            .or_else(|| params.get(id.replace('_', "-").as_str()));
249        if let Some(raw) = raw {
250            positionals.extend(values(raw));
251        }
252    }
253    out.extend(positionals);
254    out
255}

257/// Run one engine op against `root`: re-exec `run` in a child whose
258/// cwd is the request's repo, and hand back
259/// `{op, argv, code, stdout, stderr}` as the query payload.
260/// `ok:false` is reserved for unknown ops and spawn failures — a
261/// nonzero `code` is the engine's own exit status (e.g. `find`
262/// miss), honest data. The child's stdin is null; the repo comes
263/// from `root` (driver-resolved) — unbounded/host-interactive
264/// verbs (`watch`/`mcp`/`acp`) are surfaces, never ops, precisely
265/// because they need inherited stdio.
266pub fn run_op(root: &Path, name: &str, params: &Map<String, Value>) -> Response {
267    let name = name.strip_prefix("cli.").unwrap_or(name);
268    let Some(cmd) = op_cmd(name) else {
269        return Response::err(format!("unknown op: {name}"));
270    };
271    let args = argv(name, &cmd, params);
272    let out = std::env::current_exe().and_then(|exe| {
273        std::process::Command::new(exe)
274            .arg("run")
275            .args(&args)
276            .current_dir(root)
277            .output()
278    });
279    let data = match out {
280        Ok(o) => json!({
281            "op": name,
282            "argv": args,
283            "code": o.status.code().unwrap_or(-1),
284            "stdout": String::from_utf8_lossy(&o.stdout),
285            "stderr": String::from_utf8_lossy(&o.stderr),
286        }),
287        Err(e) => return Response::err(format!("run {name}: {e}")),
288    };
289    Response::ok_typed(&QueryResult {
290        query: format!("cli.{name}"),
291        data,
292    })
293}

301    #[test]
302    fn op_of_accepts_cli_prefix_only() {
303        assert_eq!(op_of("cli.sync").unwrap(), "sync");
304        assert_eq!(op_of("cli.doctor").unwrap(), "doctor");
305        assert!(op_of("cli.watch").is_none()); // unbounded surface
306        assert!(op_of("cli.mcp").is_none()); // stdio surface
307        assert!(op_of("cli.acp").is_none()); // stdio surface
308        assert!(op_of("cli.gen-cli-docs").is_none()); // hidden plumbing
309        assert!(op_of("sync").is_none()); // view query, not an op
310        assert!(op_of("cli.bogus").is_none());
311    }
312
313    #[test]
314    fn manifest_lists_every_non_surface_verb_as_cli_query() {
315        let m = manifest();
316        let ops = m["ops"].as_array().unwrap();
317        for (n, _) in verbs()
318            .iter()
319            .filter(|(n, _)| !SURFACES.contains(&n.as_str()))
320        {
321            assert!(
322                ops.iter().any(|o| o["query"] == json!(format!("cli.{n}"))),
323                "{n} missing from ops manifest"
324            );
325        }
326        assert_eq!(m["surfaces"], json!(SURFACES));
327        for o in ops {
328            for p in o["params"].as_array().unwrap() {
329                assert!(["arg", "opt", "flag"].contains(&p["kind"].as_str().unwrap()));
330                // streaming/stdin params never ride the wire
331                assert!(!SKIP_PARAMS.contains(&p["name"].as_str().unwrap()));
332                // globals are manifest-level, never per-op params
333                assert!(!m["globals"]
334                    .as_array()
335                    .unwrap()
336                    .iter()
337                    .any(|g| g["name"] == p["name"]));
338            }
339        }
340        // mutates honesty — the side-effect set is exactly these
341        let mutators: Vec<&str> = ops
342            .iter()
343            .filter(|o| o["mutates"] == json!(true))
344            .filter_map(|o| o["query"].as_str())
345            .collect();
346        assert_eq!(mutators, vec!["cli.init", "cli.sync"]);
347    }

362    fn argv_maps_positionals_and_named_params() {
363        let cmd = op_cmd("card").unwrap();
364        let mut p = Map::new();
365        p.insert("module".into(), json!("kuda-store"));
366        assert_eq!(argv("card", &cmd, &p), vec!["card", "kuda-store"]);
367
368        let cmd = op_cmd("refs").unwrap();
369        let mut p = Map::new();
370        p.insert("name".into(), json!("X"));
371        p.insert("docs".into(), json!(true));
372        // named flags emit before positionals — clap accepts
373        // interspersed args either way.
374        assert_eq!(argv("refs", &cmd, &p), vec!["refs", "--docs", "X"]);
375    }
