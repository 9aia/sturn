19/// The decoded filter flags — each maps onto the same
20/// `ViewFilter` field `filter = { … }` uses. Shared by `view`
21/// (merged over the named spec's filter) and `next` (merged over
22/// its open-item default): a set flag replaces that field, unset
23/// fields keep the base spec's value.
24#[derive(Debug, Default)]
25pub(crate) struct FilterPatch {
26    pub(crate) kind: Vec<String>,
27    pub(crate) status: Vec<Status>,
28    pub(crate) pri_min: Option<Priority>,
29    pub(crate) pri_max: Option<Priority>,
30    pub(crate) label: Vec<String>,
31    pub(crate) owner: Vec<Actor>,
32    pub(crate) claimed: bool,
33    pub(crate) unclaimed: bool,
34    pub(crate) epic: Option<String>,
35    pub(crate) milestone: Option<String>,
36    pub(crate) due: Option<time::Date>,
37    pub(crate) estimate: Option<u32>,
38    pub(crate) cycle: Option<String>,
39    pub(crate) live_only: bool,
40    pub(crate) superseded: bool,
41}

43/// The filter args → the patch — `view` and `next` share the
44/// flag set.
45pub(crate) fn filter_args(a: &Args) -> Result<FilterPatch, String> {
46    Ok(FilterPatch {
47        kind: a.many("kind"),
48        status: a.many_parsed("status")?,
49        pri_min: a.parsed("pri_min")?,
50        pri_max: a.parsed("pri_max")?,
51        label: a.many("label"),
52        owner: a.many_parsed("owner")?,
53        claimed: a.flag("claimed"),
54        unclaimed: a.flag("unclaimed"),
55        epic: a.opt("epic").map(str::to_string),
56        milestone: a.opt("milestone").map(str::to_string),
57        due: items::opt_date(a, "due")?,
58        estimate: a.parsed("est")?,
59        cycle: a.opt("cycle").map(str::to_string),
60        live_only: a.flag("live_only"),
61        superseded: a.flag("superseded"),
62    })
63}

65impl FilterPatch {
66    /// Fold every set flag into `f`; unset fields keep the
67    /// base filter's value.
68    pub(crate) fn apply(&self, f: &mut ViewFilter) {
69        if !self.kind.is_empty() {
70            f.kind = self.kind.clone();
71        }
72        if !self.status.is_empty() {
73            f.status = self.status.clone();
74        }
75        if let Some(p) = self.pri_min {
76            f.pri_min = Some(p);
77        }
78        if let Some(p) = self.pri_max {
79            f.pri_max = Some(p);
80        }
81        if !self.label.is_empty() {
82            f.label = self.label.clone();
83        }
84        if !self.owner.is_empty() {
85            f.owner = self.owner.clone();
86        }
87        if self.claimed {
88            f.claimed = Some(true);
89        }
90        if self.unclaimed {
91            f.claimed = Some(false);
92        }
93        if let Some(e) = &self.epic {
94            f.epic = Some(e.clone());
95        }
96        if let Some(m) = &self.milestone {
97            f.milestone = Some(m.clone());
98        }
99        if let Some(d) = self.due {
100            f.due = Some(d);
101        }
102        if let Some(e) = self.estimate {
103            f.estimate = Some(e);
104        }
105        if let Some(c) = &self.cycle {
106            f.cycle = Some(c.clone());
107        }
108        if self.live_only {
109            f.superseded = Some(false);
110        }
111        if self.superseded {
112            f.superseded = Some(true);
113        }
114    }
115
116    /// Any flag set?
117    pub(crate) fn any_set(&self) -> bool {
118        !self.kind.is_empty()
119            || !self.status.is_empty()
120            || self.pri_min.is_some()
121            || self.pri_max.is_some()
122            || !self.label.is_empty()
123            || !self.owner.is_empty()
124            || self.claimed
125            || self.unclaimed
126            || self.epic.is_some()
127            || self.milestone.is_some()
128            || self.due.is_some()
129            || self.estimate.is_some()
130            || self.cycle.is_some()
131            || self.live_only
132            || self.superseded
133    }
134}

153        source: a.parsed("source")?,
154        group_by: a.parsed("group_by")?,

161impl SpecPatchCli {
162    /// Fold every set flag over `spec`; unset fields keep the
163    /// named spec's value.
164    fn apply(&self, spec: &mut ViewSpec) {
165        self.filter.apply(&mut spec.filter);
166        if let Some(s) = self.source {
167            spec.source = s;
168        }
169        if let Some(g) = self.group_by {
170            spec.group_by = Some(g);
171        }
172        if !self.sort.is_empty() {
173            spec.sort = self.sort.clone();
174        }
175        if let Some(n) = self.limit {
176            spec.limit = Some(n);
177        }
178        if let Some(fmt) = self.format {
179            spec.format = fmt;
180        }
181    }
182
183    /// Any flag set? Bare `myco view` (no name, no flags) lists.
184    fn any_set(&self) -> bool {
185        self.filter.any_set()
186            || self.source.is_some()
187            || self.group_by.is_some()
188            || !self.sort.is_empty()
189            || self.limit.is_some()
190            || self.format.is_some()
191    }
192}

194/// One-line spec digest for `--list` — only set fields.
195fn spec_digest(s: &ViewSpec) -> String {
196    let f = &s.filter;
197    let mut parts = Vec::new();
198    let names = |v: &[String]| v.join(",");
199    if !f.kind.is_empty() {
200        parts.push(format!("kind=[{}]", names(&f.kind)));
201    }
202    if !f.status.is_empty() {
203        parts.push(format!(
204            "status=[{}]",
205            f.status
206                .iter()
207                .map(|s| s.token())
208                .collect::<Vec<_>>()
209                .join(",")
210        ));
211    }
212    if let Some(p) = f.pri_min {
213        parts.push(format!("pri_min={}", p.name()));
214    }
215    if let Some(p) = f.pri_max {
216        parts.push(format!("pri_max={}", p.name()));
217    }
218    if !f.label.is_empty() {
219        parts.push(format!("label=[{}]", f.label.join(",")));
220    }
221    if !f.owner.is_empty() {
222        parts.push(format!(
223            "owner=[{}]",
224            f.owner
225                .iter()
226                .map(|a| a.as_str())
227                .collect::<Vec<_>>()
228                .join(",")
229        ));
230    }
231    if let Some(c) = f.claimed {
232        parts.push(format!("claimed={c}"));
233    }
234    if let Some(e) = &f.epic {
235        parts.push(format!("epic={e}"));
236    }
237    if let Some(m) = &f.milestone {
238        parts.push(format!("milestone={m}"));
239    }
240    if let Some(d) = f.due {
241        parts.push(format!("due={d}"));
242    }
243    if let Some(e) = f.estimate {
244        parts.push(format!("estimate={e}"));
245    }
246    if let Some(c) = &f.cycle {
247        parts.push(format!("cycle={c}"));
248    }
249    if let Some(sup) = f.superseded {
250        parts.push(format!("superseded={sup}"));
251    }
252    let mut out = if parts.is_empty() {
253        "filter:{}".to_string()
254    } else {
255        format!("filter:{{{}}}", parts.join(" "))
256    };
257    if !s.source.is_default() {
258        out.push_str(&format!(" source:{}", s.source.name()));
259    }
260    if let Some(g) = s.group_by {
261        out.push_str(&format!(" group_by:{}", g.name()));
262    }
263    if !s.sort.is_empty() {
264        out.push_str(&format!(
265            " sort:{}",
266            s.sort
267                .iter()
268                .map(|k| k.to_string())
269                .collect::<Vec<_>>()
270                .join(",")
271        ));
272    }
273    if let Some(n) = s.limit {
274        out.push_str(&format!(" limit:{n}"));
275    }
276    out.push_str(&format!(" format:{}", s.format.name()));
277    out
278}

280/// The `--list` output.
281fn list_views(views: &[NamedView], json_out: bool) {
282    if json_out {
283        ops::print_json(&json!({
284            "views": views.iter().map(|v| json!({
285                "name": v.name,
286                "origin": match v.origin {
287                    Origin::Builtin => "builtin",
288                    Origin::Config => "config",
289                },
290                "spec": v.spec,
291            })).collect::<Vec<_>>(),
292        }));
293        return;
294    }
295    for v in views {
296        let origin = match v.origin {
297            Origin::Builtin => "builtin",
298            Origin::Config => "config",
299        };
300        println!("{:<10} {:<7} {}", v.name, origin, spec_digest(&v.spec));
301    }
302}

304/// `list` — compact lines; `<label> (n)` headers when the spec
305/// groups; children ride indented under their parent. A
306/// `repo:…` tag prefixes index-sourced items.
307pub(crate) fn render_list(groups: &[ViewGroup], interrupted: &BTreeSet<ItemId>) -> String {
308    fn push_item(
309        out: &mut String,
310        item: &Item,
311        depth: usize,
312        base: &str,
313        repo: Option<&str>,
314        interrupted: &BTreeSet<ItemId>,
315    ) {
316        out.push_str(base);
317        out.push_str(&"  ".repeat(depth));
318        if let Some(r) = repo {
319            out.push_str(r);
320            out.push(':');
321        }
322        out.push_str(&items::compact(item));
323        if interrupted.contains(&item.id) {
324            out.push_str(" ⚡");
325        }
326        out.push('\n');
327        for c in &item.children {
328            push_item(out, c, depth + 1, base, repo, interrupted);
329        }
330    }
331    let mut out = String::new();
332    for g in groups {
333        if g.items.is_empty() {
334            continue;
335        }
336        if let Some(label) = &g.label {
337            out.push_str(&format!("{label} ({})\n", g.items.len()));
338        }
339        let base = if g.label.is_some() { "  " } else { "" };
340        for i in &g.items {
341            push_item(&mut out, i.item, 0, base, i.repo, interrupted);
342        }
343    }
344    out.trim_end_matches('\n').to_string()
345}

347/// `board` — a status-grouped spec earns the lane chrome
348/// (capitalized Backlog | Open | … lanes via
349/// [`board::status_cards`]); any other grouping renders generic
350/// columns headed by the group labels.
351pub(crate) fn render_board(
352    spec: &ViewSpec,
353    groups: &[ViewGroup],
354    interrupted: &BTreeSet<ItemId>,
355    md: bool,
356    all: bool,
357) -> String {
358    if board::is_status_board(spec) {
359        let (headers, cards) = board::status_cards(groups, interrupted);
360        return board::render_columns(&headers, &cards, md, board::Fit::terminal(all));
361    }
362    let headers: Vec<String> = groups
363        .iter()
364        .map(|g| g.label.clone().unwrap_or_default())
365        .collect();
366    let columns: Vec<Vec<String>> = groups
367        .iter()
368        .map(|g| {
369            g.items
370                .iter()
371                .map(|i| board::card(i.item, i.repo, interrupted.contains(&i.item.id)))
372                .collect()
373        })
374        .collect();
375    board::render_columns(&headers, &columns, md, board::Fit::terminal(all))
376}

397/// Pick the renderer `spec.format` names — the shared tail of
398/// every eval-driven read.
399pub(crate) fn render_groups(
400    spec: &ViewSpec,
401    groups: &[ViewGroup],
402    interrupted: &BTreeSet<ItemId>,
403    md: bool,
404    all: bool,
405) -> String {
406    match spec.format {
407        ViewFormat::Board => render_board(spec, groups, interrupted, md, all),
408        // `repos` never reaches here — `run`/`w_view` route it to
409        // `emit_repos` before eval; the analytics formats route to
410        // `crate::analytics` the same way. The fallback keeps this
411        // total if a caller ever skips those branches.
412        ViewFormat::List | ViewFormat::Repos => render_list(groups, interrupted),
413        ViewFormat::Timeline | ViewFormat::Stats | ViewFormat::Burnup => String::new(),
414    }
415}

417/// One row of `format = "repos"` — a repo and its live-ledger
418/// counts (`None` + `error` when the rescan failed).
419pub(crate) struct RepoRow {
420    /// Registry / project name.
421    pub(crate) name: String,
422    /// The repo root's parent-dir name (`~/GitHub/9aia/myco` → `9aia`).
423    pub(crate) org: Option<String>,
424    /// Canonical repo-root path.
425    pub(crate) path: String,
426    /// `(total, open, claimed)` over live items — `claimed`
427    /// counts owned items (the `claimed` filter's rule).
428    pub(crate) counts: Option<(usize, usize, usize)>,
429    /// Terse rescan failure — the row shows `—` counts.
430    pub(crate) error: Option<String>,
431}

433/// `path`'s parent-dir name — the org bucket a repos row
434/// reports (`~/GitHub/9aia/myco` → `9aia`).
435fn org_of(path: &std::path::Path) -> Option<String> {
436    path.parent()
437        .and_then(|p| p.file_name())
438        .map(|n| n.to_string_lossy().into_owned())
439}

441/// `(total, open, claimed)` over a live-ledger iterator —
442/// children count like any item (`live`/`live_items` flatten).
443fn live_counts<'a>(items: impl Iterator<Item = &'a Item>) -> (usize, usize, usize) {
444    let mut c = (0, 0, 0);
445    for i in items {
446        c.0 += 1;
447        c.1 += usize::from(i.status == Status::Open);
448        c.2 += usize::from(i.owner.is_some());
449    }
450    c
451}

453/// [`org::RepoView`] → a repos row. An error row keeps its
454/// identity (`name`/`org`/`path`) and names the failure minus
455/// the redundant worktree-path prefix; counts stay `None` even
456/// when a stale snapshot rode along.
457pub(crate) fn repo_row(v: &org::RepoView) -> RepoRow {
458    RepoRow {
459        name: v.name.clone(),
460        org: org_of(v.path.as_std_path()),
461        path: v.path.to_string(),
462        counts: if v.error.is_none() {
463            v.snap.as_ref().map(|s| live_counts(s.live_items()))
464        } else {
465            None
466        },
467        error: v.error.as_ref().map(|e| {
468            e.strip_prefix(v.worktree.as_str())
469                .and_then(|r| r.strip_prefix(": "))
470                .unwrap_or(e)
471                .to_string()
472        }),
473    }
474}

500/// `repo org items open claimed err` — one record per row,
501/// `—` fills cells a missing/failed repo can't supply.
502fn render_repos(rows: &[RepoRow], md: bool) -> String {
503    let mut b = tabled::builder::Builder::default();
504    b.push_record(["repo", "org", "items", "open", "claimed", "err"]);
505    for r in rows {
506        let dash = "—".to_string();
507        let (total, open, claimed) = r
508            .counts
509            .map(|(t, o, c)| (t.to_string(), o.to_string(), c.to_string()))
510            .unwrap_or((dash.clone(), dash.clone(), dash));
511        b.push_record([
512            r.name.as_str(),
513            r.org.as_deref().unwrap_or("—"),
514            total.as_str(),
515            open.as_str(),
516            claimed.as_str(),
517            r.error.as_deref().unwrap_or(""),
518        ]);
519    }
520    let mut table = b.build();
521    if md {
522        table.with(tabled::settings::Style::markdown());
523    }
524    table.to_string()
525}

527/// `format = "repos"` — one row per repo instead of item
528/// groups. `index` rows come from the registered snapshots
529/// (a rescan failure shows the error + `—` counts); `repo`
530/// emits a single row for this worktree.
531fn emit_repos(
532    name: Option<&str>,
533    spec: &ViewSpec,
534    dir: Option<&std::path::Path>,
535    root: Option<&std::path::Path>,
536    out: Out,
537    json_out: bool,
538) -> Result<i32, String> {
539    let (rows, warnings) = match spec.source {
540        ViewSource::Index => {
541            let (views, w) = index_universe()?;
542            (views.iter().map(repo_row).collect::<Vec<_>>(), w)
543        }
544        ViewSource::Repo => {
545            let (proj, store) = items::open_read(dir, root)?;
546            let scan = digest::scan(&store)?;
547            (vec![repo_row_local(&proj, &scan)], Vec::new())
548        }
549    };
550    warn(&warnings);
551    if json_out {
552        ops::print_json(&json!({
553            "view": name,
554            "spec": spec,
555            "repos": rows.iter().map(row_json).collect::<Vec<_>>(),
556            "warnings": warnings,
557        }));
558        return Ok(0);
559    }
560    if rows.is_empty() {
561        println!("no repos registered — `myco index add <path>`");
562    } else {
563        ops::print_paged(&render_repos(&rows, out.md));
564    }
565    Ok(0)
566}

589/// The index-sourced universe — every registered repo's
590/// snapshot, freshness-checked + rebuilt by [`org::load_fresh`],
591/// items tagged with their repo's name. Rescan failures warn on
592/// stderr and contribute their last-good snapshot (or nothing).
593pub(crate) fn index_universe() -> Result<(Vec<org::RepoView>, Vec<String>), String> {
594    let index = indexcmd::open_index()?;
595    let views = org::load_fresh(&index)?;
596    let warnings = views
597        .iter()
598        .filter_map(|v| v.error.as_ref().map(|e| format!("{}: {e}", v.name)))
599        .collect();
600    Ok((views, warnings))
601}

603/// The `view` verb — `NAME` resolves builtin presets then config
604/// views; no NAME builds the spec from flags alone; bare `myco
605/// view` (or `--list`) enumerates the defined views.
606/// `spec.source` picks the universe: `repo` reads this
607/// worktree's ledger, `index` unions the registered repos'
608/// snapshots (the `org-*` presets — `myco org status|board`'s
609/// heirs).
610pub fn h(ctx: &crate::registry::Ctx, a: &Args) -> Result<i32, String> {
611    let name = a.opt("name").map(str::to_string);
612    let flags = spec_args(a)?;
613    let out = Out {
614        list: a.flag("list"),
615        md: a.flag("md"),
616        all: a.flag("all"),
617    };
618    run(
619        name,
620        flags,
621        out,
622        ctx.dir.clone(),
623        ctx.root.clone(),
624        ctx.json,
625    )
626}

628fn run(
629    name: Option<String>,
630    flags: SpecPatchCli,
631    out: Out,
632    dir: Option<PathBuf>,
633    root: Option<PathBuf>,
634    json_out: bool,
635) -> Result<i32, String> {
636    // The config root for named-view resolution — repo discovery
637    // when it works, else `--root`/cwd: an index-sourced view
638    // needs no store, so discovery failure isn't fatal (yet).
639    let opened = items::open_read(dir.as_deref(), root.as_deref());
640    let cfg_root = opened
641        .as_ref()
642        .map(|(p, _)| p.root.clone())
643        .ok()
644        .or_else(|| root.clone())
645        .or_else(|| std::env::current_dir().ok())
646        .unwrap_or_else(|| PathBuf::from("."));
647
648    // `default_view` in config names the landing view — bare
649    // `myco view` renders it instead of listing (a `~/myco.toml`
650    // pins the machine-global board outside any repo).
651    let name = match name {
652        None if !flags.any_set() && !out.list => {
653            myco_ops::config::default_view_for(&cfg_root, ops::CONFIG)
654        }
655        n => n,
656    };
657
658    if out.list || (name.is_none() && !flags.any_set()) {
659        let (views, _) = myco_ops::view::named_views_lenient(&cfg_root, ops::CONFIG)?;
660        list_views(&views, json_out);
661        return Ok(0);
662    }
663
664    let (mut spec, warnings) = match &name {
665        Some(n) => {
666            let (v, w) = myco_ops::view::resolve_lenient(&cfg_root, n, ops::CONFIG)?;
667            (v.spec, w)
668        }
669        None => (ViewSpec::default(), Vec::new()),
670    };
671    flags.apply(&mut spec);
672    spec.validate()?;
673    warn(&warnings);
674
675    // `repos` emits per-repo rows, not item groups — a repo
676    // source is a single row for this worktree.
677    if spec.format == ViewFormat::Repos {
678        return emit_repos(
679            name.as_deref(),
680            &spec,
681            dir.as_deref(),
682            root.as_deref(),
683            out,
684            json_out,
685        );
686    }
687
688    // The universe owners — `groups` borrows item refs out of
689    // them, so they must outlive it.
690    let (interrupted, empty_name, scan, views, kinds) = match spec.source {
691        ViewSource::Repo => {
692            let (_proj, store) = opened?;
693            let s = digest::scan(&store)?;
694            let kinds = myco_store::Store::kinds(&store).clone();
695            (
696                continuity::interrupted_set(&store, s.live())?,
697                s.name.clone(),
698                Some(s),
699                Vec::new(),
700                kinds,
701            )
702        }
703        ViewSource::Index => {
704            let (v, repo_warnings) = index_universe()?;
705            warn(&repo_warnings);
706            // snapshots carry no handoffs — the interrupted set
707            // is empty, not failed
708            (
709                BTreeSet::new(),
710                name.clone().unwrap_or_else(|| "index".into()),
711                None,
712                v,
713                myco_core::KindTable::prelude(),
714            )
715        }
716    };
717    // The analytics emits (timeline/stats/burnup) replace the
718    // group render — same filter, different projection. `cycles`
719    // come from this repo's config for both sources (the index
720    // universe has no config of its own).
721    if crate::analytics::is_emit(&spec) {
722        let cycles = myco_ops::config::cycles_for(&cfg_root, ops::CONFIG)?;
723        let emit_items = crate::analytics::emit_items(&spec, scan.as_ref(), &views, &kinds);
724        if json_out {
725            ops::print_json(&crate::analytics::emit_json(
726                name.as_deref(),
727                &spec,
728                &emit_items,
729                &cycles,
730                &warnings,
731            ));
732        } else {
733            let rendered = crate::analytics::render(&spec, &emit_items, &cycles);
734            if emit_items.is_empty() {
735                println!("{empty_name}: empty view — no items match");
736            } else {
737                ops::print_paged(&rendered);
738            }
739        }
740        return Ok(0);
741    }
742
743    // top-level items only — children ride with their parent
744    // (they can't match a filter on their own)
745    let groups = match spec.source {
746        ViewSource::Repo => eval(
747            scan.iter()
748                .flat_map(|s| s.items.iter().filter(|i| !s.archived.contains(&i.id))),
749            &spec,
750            &kinds,
751        ),
752        ViewSource::Index => eval_scoped(
753            views.iter().flat_map(|v| {
754                v.snap.iter().flat_map(|s| {
755                    s.items
756                        .iter()
757                        .filter(|i| !s.archived.contains(&i.id))
758                        .map(move |i| Scoped::tagged(s.name.as_str(), i))
759                })
760            }),
761            &spec,
762            &kinds,
763        ),
764    };
765
766    if json_out {
767        let payload = if board::is_status_board(&spec) {
768            // index snapshots carry no handoffs — omit the flag
769            // entirely rather than emitting `false` everywhere
770            let set = (spec.source == ViewSource::Repo).then_some(&interrupted);
771            board::groups_json(&groups, set)
772        } else {
773            view_json(name.as_deref(), &spec, &groups, &interrupted)
774        };
775        ops::print_json(&payload);
776        return Ok(0);
777    }
778
779    let rendered = render_groups(&spec, &groups, &interrupted, out.md, out.all);
780    if groups.iter().all(|g| g.items.is_empty()) {
781        println!("{empty_name}: empty view — no items match");
782    } else {
783        ops::print_paged(&rendered);
784    }
785    Ok(0)
786}

789#[allow(clippy::unwrap_used)]
790mod tests {
791    use super::*;
792    use myco_core::ItemKind;
793
794    #[test]
795    fn repo_config_shadow_redefines_the_named_board() {
796        // `[views.board]` in the repo's myco.toml replaces the
797        // preset `myco view board` resolves — the spec-level proof
798        // that config shadowing reaches the named view, and that a
799        // non-status spec drops the lane chrome.
800        let dir = tempfile::tempdir().unwrap();
801        std::fs::write(
802            dir.path().join("myco.toml"),
803            "[views.board]\nfilter = { kind = [\"issue\"] }\n",
804        )
805        .unwrap();
806        temp_env::with_vars(
807            [
808                ("MYCO_CONFIG", None::<&str>),
809                (
810                    "HOME",
811                    Some(dir.path().join("nohome").to_string_lossy().as_ref()),
812                ),
813                ("XDG_CONFIG_HOME", None::<&str>),
814            ],
815            || {
816                let (v, warnings) =
817                    myco_ops::view::resolve_lenient(dir.path(), "board", ops::CONFIG).unwrap();
818                assert_eq!(warnings.len(), 1);
819                let spec = v.spec;
820                assert_eq!(spec.filter.kind, vec!["issue".to_string()]);
821                // the shadowed spec isn't a status-axis board →
822                // the generic renderers/json, not the lane chrome
823                assert!(!board::is_status_board(&spec));
824                // flag merge still applies over the shadowed spec
825                let mut scoped = spec.clone();
826                SpecPatchCli {
827                    filter: FilterPatch {
828                        epic: Some("v2".into()),
829                        ..Default::default()
830                    },
831                    ..Default::default()
832                }
833                .apply(&mut scoped);
834                assert_eq!(scoped.filter.epic.as_deref(), Some("v2"));
835                // and eval honours the shadow — tasks drop out of
836                // every group, so the rendered board changes
837                let items = vec![
838                    Item::new(ItemId::new(ItemKind::Task, 1), "t"),
839                    Item::new(ItemId::new(ItemKind::Issue, 1), "i"),
840                ];
841                let ids: Vec<String> = eval(&items, &spec, myco_core::kind::prelude_table())
842                    .iter()
843                    .flat_map(|g| g.items.iter().map(|i| i.id.to_string()))
844                    .collect();
845                assert_eq!(ids, ["I-001"]);
846            },
847        );
848    }
