1//! `myco issue new|list|resolve|archive|comment` — one md file
2//! per issue under `issues/` (archived files in `issues/archive/`).
3
4use myco_core::{ItemId, ItemKind, Status};
5use myco_store::Store;
6use serde_json::json;
7
8use crate::items;
9use crate::registry::{Args, Ctx};

12/// The `issue new` verb.
13pub fn h_new(ctx: &Ctx, a: &Args) -> Result<i32, String> {
14    let (proj, store) = items::open(ctx.dir.as_deref(), ctx.root.as_deref())?;
15    items::new_file(
16        &proj,
17        &store,
18        &items::NewItem {
19            kind: ItemKind::Issue,
20            title: a.text("title")?.to_string(),
21            body: body_source(a)?.as_deref(),
22            meta: &items::meta_args(a)?,
23            template: templates::issue,
24            word: "issue",
25            json_out: ctx.json,
26            no_commit: ctx.no_commit,
27        },
28    )
29}
30
31/// `--body/--text` — shared body source for the `new` verbs.
32fn body_source(a: &Args) -> Result<Option<String>, String> {
33    items::body_args(a)
34}

36/// The `issue list` verb.
37pub fn h_list(ctx: &Ctx, a: &Args) -> Result<i32, String> {
38    let (_proj, store) = items::open_read(ctx.dir.as_deref(), ctx.root.as_deref())?;
39    items::list_files(
40        &store,
41        ItemKind::Issue,
42        a.flag("archived"),
43        &items::list_args(a)?,
44        ctx.json,
45    )
46}

48/// The `issue resolve` verb.
49pub fn h_resolve(ctx: &Ctx, a: &Args) -> Result<i32, String> {
50    let (proj, store) = items::open(ctx.dir.as_deref(), ctx.root.as_deref())?;
51    let id = a.req::<ItemId>("id")?;
52    items::require_kind(id, ItemKind::Issue, "issue resolve")?;
53    let mut file = items::item_for_edit(&store, id)?;
54    items::walk_to(&mut file.meta, Status::Done)?;
55    items::stamp_done(&mut file.meta);
56    if let Some(res) = a.opt("resolution") {
57        templates::append_to_section(&mut file.body, "## Resolution", res);
58        file.body = myco_core::canonical_body(&file.body).to_string();
59    }
60    let item = file.meta.clone();
61    store.save_item(&file).map_err(|e| e.to_string())?;
62    let committed = items::commit(
63        &proj.worktree,
64        &format!("chore: resolve {id}"),
65        None,
66        ctx.no_commit,
67    )?;
68    items::report(
69        json!({"item": item}),
70        committed,
71        ctx.json,
72        &format!("resolved {id}"),
73    )
74}

76/// The `issue archive` verb.
77pub fn h_archive(ctx: &Ctx, a: &Args) -> Result<i32, String> {
78    let (proj, store) = items::open(ctx.dir.as_deref(), ctx.root.as_deref())?;
79    let id = a.req::<ItemId>("id")?;
80    items::require_kind(id, ItemKind::Issue, "issue archive")?;
81    let file = items::item_for_edit(&store, id)?;
82    if file.meta.status != Status::Done {
83        return Err(format!(
84            "{id} must be done before archiving (status: {}) — resolve it first",
85            file.meta.status
86        ));
87    }
88    store.archive_item(id).map_err(|e| e.to_string())?;
89    // asked after the move so the impl reports the
90    // archive-scope path itself
91    let rel = items::item_rel_path(&store, &file.meta)?;
92    let committed = items::commit(
93        &proj.worktree,
94        &format!("chore: archive {id}"),
95        None,
96        ctx.no_commit,
97    )?;
98    items::report(
99        json!({"item": file.meta, "archived_to": rel}),
100        committed,
101        ctx.json,
102        &format!("archived {id} → {rel}"),
103    )
104}

106/// The `issue comment` verb.
107pub fn h_comment(ctx: &Ctx, a: &Args) -> Result<i32, String> {
108    let (proj, store) = items::open(ctx.dir.as_deref(), ctx.root.as_deref())?;
109    let id = a.req::<ItemId>("id")?;
110    items::require_kind(id, ItemKind::Issue, "issue comment")?;
111    let mut file = items::item_for_edit(&store, id)?;
112    let line = format!(
113        "- {} {}",
114        myco_core::date::render(items::today()),
115        a.text("text")?
116    );
117    templates::append_to_section(&mut file.body, "## Comments", &line);
118    file.body = myco_core::canonical_body(&file.body).to_string();
119    store.save_item(&file).map_err(|e| e.to_string())?;
120    let committed = items::commit(
121        &proj.worktree,
122        &format!("chore: comment {id}"),
123        None,
124        ctx.no_commit,
125    )?;
126    items::report(
127        json!({"item": file.meta}),
128        committed,
129        ctx.json,
130        &format!("commented on {id}"),
131    )
132}
