1//! `sealed:` packs — age-encrypted payloads tamias carries **opaquely**
2//! (tamias ADR-0004 / org P-027).
3//!
4//! A sealed pack's payload is ciphertext only: the engine, store,
5//! cache, registry, TUI, and logs treat every payload file as an
6//! opaque blob — digest, verify, and sign work unchanged because they
7//! already operate on bytes. Tamias has **no decrypt path**; opening
8//! is cuora's job, in the consumer's context, via the content-type
9//! driver.
10//!
11//! Layout (payload-relative):
12//!
13//! ```text
14//! sealed/
15//!   recipients            # age public keys, one per line (public by design)
16//!   registry-token.age    # age ciphertext of the secret bytes
17//! tamias.yaml             # manifest; `sealed:` maps logical → file
18//! ```
19//!
20//! This module owns the manifest-side [`Sealed`] block, its semantic
21//! checks, and [`check_ciphertext_only`] — the publish/install gate
22//! that refuses a sealed pack carrying stray plaintext.
23

35/// The only sealing format v1 understands — `age` file encryption
36/// (X25519 recipients + ChaCha20-Poly1305 STREAM), header line
37/// `age-encryption.org/v1`.
38pub const AGE_FORMAT: &str = "age/v1";
39
40/// age file magic — every age-encrypted file opens with this header
41/// line. The publish/install ciphertext gate keys on it.
42pub const AGE_MAGIC: &[u8] = b"age-encryption.org/v1";
43
44/// The `sealed:` manifest block — present means the pack is
45/// **ciphertext-only**. Orthogonal to `type:`: `type: secrets` is a
46/// convention paired with the cuora driver, not a requirement.
47///
48/// ```yaml
49/// sealed:
50///   format: age/v1
51///   recipients: sealed/recipients      # payload-relative path
52///   files:
53///     registry-token: sealed/registry-token.age
54/// ```
55#[derive(Clone, Debug, Default, PartialEq, Eq, Serialize, Deserialize, schemars::JsonSchema)]
56#[serde(default)]
57pub struct Sealed {
58    /// Sealing format — only `age/v1` exists in v1.
59    pub format: String,
60    /// Payload-relative path of the recipients file — age public keys,
61    /// one per line. Public by design; it rides the payload so the
62    /// pack digest (and any `tamias-sign` signature) covers it.
63    pub recipients: String,
64    /// Logical secret name → payload-relative `.age` file. The map is
65    /// the index cuora's driver resolves `cuora://` refs through.
66    pub files: std::collections::BTreeMap<String, String>,
67}
68
69/// A clean payload-relative path: `/`-separated, no empty/`.`/`..`
70/// segments, no leading `/`, no NUL/backslash. Same safety bar as
71/// `bins[].payload`, minus the `tools/bin/` prefix rule.
72fn check_rel_path(what: &str, path: &str) -> Result<(), Error> {
73    let bad = |why: &str| Err(Error::Failed(format!("sealed {what} '{path}' — {why}")));
74    if path.is_empty() || path.contains(['\0', '\\']) {
75        return bad("empty or contains NUL/backslash");
76    }
77    if path
78        .split('/')
79        .any(|c| c.is_empty() || c == "." || c == "..")
80    {
81        return bad("must be a clean relative path (no empty, '.', or '..' segments)");
82    }
83    Ok(())
84}
85
86impl Sealed {
87    /// Semantic checks the schema can't express — known format,
88    /// non-empty file map, valid logical names + payload-relative
89    /// `.age` paths, distinct declared paths, no manifest-name
90    /// collisions.
91    ///
92    /// # Errors
93    /// Fails on a malformed `sealed:` block.
94    pub fn check(&self) -> Result<(), Error> {
95        if self.format != AGE_FORMAT {
96            return Err(Error::Failed(format!(
97                "sealed format '{}' is not supported (only '{AGE_FORMAT}')",
98                self.format
99            )));
100        }
101        check_rel_path("recipients", &self.recipients)?;
102        if MANIFEST_NAMES.contains(&self.recipients.as_str()) {
103            return Err(Error::Failed(format!(
104                "sealed recipients path '{}' collides with a manifest filename",
105                self.recipients
106            )));
107        }
108        if self.files.is_empty() {
109            return Err(Error::Failed(
110                "sealed.files is empty — a sealed pack must map at least one secret".to_string(),
111            ));
112        }
113        let mut declared = BTreeSet::from([self.recipients.as_str()]);
114        for (name, path) in &self.files {
115            if !is_valid_name(name) {
116                return Err(Error::Failed(format!(
117                    "invalid sealed secret name '{name}' — expected [A-Za-z0-9][A-Za-z0-9._-]*"
118                )));
119            }
120            check_rel_path(&format!("files['{name}']"), path)?;
121            if !path.ends_with(".age") {
122                return Err(Error::Failed(format!(
123                    "sealed files['{name}'] '{path}' must end in .age"
124                )));
125            }
126            if MANIFEST_NAMES.contains(&path.as_str()) {
127                return Err(Error::Failed(format!(
128                    "sealed files['{name}'] '{path}' collides with a manifest filename"
129                )));
130            }
131            if !declared.insert(path.as_str()) {
132                return Err(Error::Failed(format!(
133                    "sealed path '{path}' is declared more than once"
134                )));
135            }
136        }
137        Ok(())
138    }
139}

141/// True when `bytes` opens with the age file magic — the ciphertext
142/// marker the sealed-pack gate validates against.
143#[must_use]
144pub fn is_age_ciphertext(bytes: &[u8]) -> bool {
145    bytes.starts_with(AGE_MAGIC)
146}
147
148/// True when the file at `path` carries the age magic header.
149///
150/// # Errors
151/// Propagates I/O failures opening/reading the file.
152pub fn has_age_magic(path: &Path) -> Result<bool, Error> {
153    let mut head = [0u8; 64];
154    let mut file = File::open(path).map_err(Error::io(format!("open {}", path.display())))?;
155    let n = file
156        .read(&mut head)
157        .map_err(Error::io(format!("read {}", path.display())))?;
158    Ok(is_age_ciphertext(&head[..n]))
159}

161/// The publish/install gate for sealed packs (ADR-0004 §2): every
162/// payload file — except the manifest spellings and the declared
163/// `sealed.recipients` target (a public-key list, plaintext by
164/// design) — must end in `.age` **and** carry the age magic. Declared
165/// `sealed.files` paths must exist. No warn-mode: a sealed pack with
166/// stray plaintext is a bug, not a preference.
167///
168/// No-op when the manifest has no `sealed:` block.
169///
170/// # Errors
171/// Fails naming the offending file, or when a declared path is
172/// missing from the payload.
173pub fn check_ciphertext_only(root: &Path, manifest: &Manifest) -> Result<(), Error> {
174    let Some(sealed) = &manifest.sealed else {
175        return Ok(());
176    };
177    let rel = |p: &Path| p.to_string_lossy().replace('\\', "/");
178    let files = payload_files(root)?;
179    let present: BTreeSet<String> = files.iter().map(|p| rel(p)).collect();
180    // Declared paths must actually ship — same spirit as the
181    // `bins[].payload` shipped-files gate.
182    for declared in std::iter::once(&sealed.recipients).chain(sealed.files.values()) {
183        if !present.contains(declared) {
184            return Err(Error::Failed(format!(
185                "sealed pack declares '{declared}' but it is not in the payload"
186            )));
187        }
188    }
189    for path in &files {
190        let rel_path = rel(path);
191        if MANIFEST_NAMES.contains(&rel_path.as_str()) || rel_path == sealed.recipients {
192            continue;
193        }
194        if !rel_path.ends_with(".age") {
195            return Err(Error::Failed(format!(
196                "sealed pack carries non-ciphertext payload file '{rel_path}' — \
197                 every payload file must be a .age file (or the manifest / recipients list)"
198            )));
199        }
200        if !has_age_magic(&root.join(path))? {
201            return Err(Error::Failed(format!(
202                "sealed pack payload file '{rel_path}' lacks the age magic header — \
203                 it is not age ciphertext"
204            )));
205        }
206    }
207    Ok(())
208}

217    fn sealed_manifest() -> Manifest {
218        Manifest {

234    /// Minimal age-shaped ciphertext — magic line + a body. The gate
235    /// only checks the header; real encryption is cuora's job.
236    fn fake_age(dir: &Path, rel: &str) {
237        let path = dir.join(rel);
238        std::fs::create_dir_all(path.parent().unwrap()).unwrap();
239        let mut bytes = AGE_MAGIC.to_vec();
240        bytes.extend_from_slice(b"\n-> X25519 fake\n\nBODY");
241        std::fs::write(path, bytes).unwrap();
242    }

244    #[test]
245    fn sealed_block_validates() {
246        assert!(sealed_manifest().validate().is_ok());
247        // unknown format
248        let mut m = sealed_manifest();
249        m.sealed.as_mut().unwrap().format = "sops/v1".into();
250        assert!(m.validate().is_err());
251        // empty files map
252        let mut m = sealed_manifest();
253        m.sealed.as_mut().unwrap().files.clear();
254        assert!(m.validate().is_err());
255        // bad logical name / bad path / non-.age path / traversal
256        for (name, path) in [
257            ("bad name", "sealed/a.age"),
258            ("ok", "../escape.age"),
259            ("ok", "sealed//a.age"),
260            ("ok", "plain.txt"),
261            ("ok", "tamias.yaml"),
262        ] {
263            let mut m = sealed_manifest();
264            m.sealed.as_mut().unwrap().files =
265                std::collections::BTreeMap::from([(name.into(), path.into())]);
266            assert!(m.validate().is_err(), "{name} -> {path} must fail");
267        }
268        // duplicate declared path
269        let mut m = sealed_manifest();
270        m.sealed
271            .as_mut()
272            .unwrap()
273            .files
274            .insert("other".into(), "sealed/registry-token.age".into());
275        assert!(m.validate().is_err());
276        // recipients path colliding with a files value
277        let mut m = sealed_manifest();
278        m.sealed.as_mut().unwrap().recipients = "sealed/registry-token.age".into();
279        assert!(m.validate().is_err());
280    }

320    #[test]
321    fn ciphertext_gate_rejects_stray_plaintext() {
322        let dir = tempfile::tempdir().unwrap();
323        let m = sealed_manifest();
324        std::fs::write(dir.path().join("tamias.yaml"), "name: secrets-x\n").unwrap();
325        std::fs::create_dir_all(dir.path().join("sealed")).unwrap();
326        std::fs::write(dir.path().join("sealed/recipients"), "age1pub...").unwrap();
327        fake_age(dir.path(), "sealed/registry-token.age");
328        // stray plaintext — no .age suffix
329        std::fs::write(dir.path().join("notes.txt"), "oops").unwrap();
330        let err = check_ciphertext_only(dir.path(), &m)
331            .unwrap_err()
332            .to_string();
333        assert!(err.contains("notes.txt"), "{err}");
334        // .age suffix but no magic (notes.txt removed — the gate reports
335        // the first offender in sorted order)
336        std::fs::remove_file(dir.path().join("notes.txt")).unwrap();
337        std::fs::write(dir.path().join("sealed/fake.age"), "not age").unwrap();
338        let err = check_ciphertext_only(dir.path(), &m)
339            .unwrap_err()
340            .to_string();
341        assert!(err.contains("sealed/fake.age"), "{err}");
342        // declared path missing from payload
343        let mut m2 = sealed_manifest();
344        m2.sealed
345            .as_mut()
346            .unwrap()
347            .files
348            .insert("ghost".into(), "sealed/ghost.age".into());
349        m2.validate().unwrap();
350        let err = check_ciphertext_only(dir.path(), &m2)
351            .unwrap_err()
352            .to_string();
353        assert!(err.contains("sealed/ghost.age"), "{err}");
354    }
