1//! ViewNode → flat render model for the ratatui canvas.
2//!
3//! [`build_view`] walks a resolved view's node tree (as raw JSON, same
4//! as `pavo-cli`'s `render.rs` — the protocol's forward-compat rule
5//! holds end-to-end: an unknown `kind` degrades through
6//! [`kv_fallback_json`] to a key/value listing, never an error) and
7//! produces plain rows the renderer styles. Selectable rows carry an
8//! index into [`Model::selections`]; the detail pane renders the
9//! matching [`Selection`].
10//!
11//! Traversal semantics deliberately mirror `pavo-cli/src/render.rs`:
12//! containers (`panel`/`row`/`column`) print their title then children
13//! indented two spaces per level (`row` does not attempt side-by-side
14//! layout), inline `rows`/`items`/`pairs` win over query-bound data,
15//! and a bound payload of `{"error": "…"}` renders as an error row
16//! instead of data.
17//!
18//! [`build_dashboard`] and [`build_note`] synthesize models for the
19//! chrome pseudo-views (contribution listing, error/info panes) that
20//! aren't driver views at all.
21

28/// Visual weight of a canvas row.
29#[derive(Clone, Copy, Debug, PartialEq, Eq)]
30pub enum RowStyle {
31    /// Body content.
32    Normal,
33    /// Column headers / emphasized structure.
34    Header,
35    /// Secondary text — separators, hints, edge lists.
36    Dim,
37    /// Errors — failed query bindings, resolve failures.
38    Error,
39    /// Node/panel titles.
40    Title,
41}

54/// Detail-pane content for one selectable row — title plus ordered
55/// `key → value` fields (empty key = plain line). Values may contain
56/// `\n`; the pane splits them.
57#[derive(Clone, Debug, Default)]
58pub struct Selection {
59    /// Heading — what was selected (view title, row id, node label…).
60    pub title: String,
61    /// Ordered detail fields.
62    pub fields: Vec<(String, String)>,
63    /// The [`pavo_core::commands::item_value`] JSON when the row is a
64    /// tree item — feeds `ViewSpec.item_route` interpolation and the
65    /// `{item.*}`/`{target.*}` command-template context on Enter.
66    /// `None` for non-item rows (tables, kv pairs, dashboard entries).
67    pub item: Option<Value>,
68}

70/// The flat canvas: rows to draw plus the selection table.
71#[derive(Clone, Debug, Default)]
72pub struct Model {
73    /// Display rows, top to bottom.
74    pub rows: Vec<Row>,
75    /// Detail content per selectable row (indexed by `Row::sel`).
76    pub selections: Vec<Selection>,
77    /// Indices into `rows` that are selectable, in display order —
78    /// j/k navigation walks this list.
79    pub selectables: Vec<usize>,
80    /// Every `graph` node's DOT dump, concatenated — `g` writes this
81    /// to a file. `None` when the view has no graphs.
82    pub graph_dot: Option<String>,
83    /// Interactive forms in the view (document order) — forms with a
84    /// `submit_op`. The app opens an edit session from
85    /// [`Model::form_marks`].
86    pub forms: Vec<FormSpec>,
87    /// Rows that belong to an interactive form: field rows
88    /// (`field: Some`) and the submit row (`field: None`). Enter on a
89    /// marked row opens the form editor instead of the detail pane.
90    pub form_marks: Vec<FormMark>,
91}

150/// Build the canvas model for a resolved view. `width` is the canvas
151/// width in columns — tables size their columns to fit (truncating
152/// cells with `…` when they can't). Convenience wrapper — no plugin
153/// renders; the app goes through [`build_view_with_plugins`].
154#[must_use]
155#[allow(dead_code)]
156pub fn build_view(root: &ViewNode, data: &Map<String, Value>, width: usize) -> Model {
157    build_view_with_plugins(root, data, width, &HashMap::new())
158}

160/// [`build_view`] with renderer-plugin substitution: nodes whose tree
161/// path (child indices joined by `.`, root = `""`) has a
162/// [`RenderModel`] in `renders` — produced by a driver `render` op for
163/// surface `"tui"` — emit the plugin's output instead of the built-in
164/// interpretation. Paths come from
165/// [`pavo_core::renderers::plugin_nodes`].
166#[must_use]
167pub fn build_view_with_plugins(
168    root: &ViewNode,
169    data: &Map<String, Value>,
170    width: usize,
171    renders: &HashMap<String, RenderModel>,
172) -> Model {
173    // ViewNode → JSON can't fail for a well-formed node; on the
174    // impossible error we render an empty doc rather than panic.
175    let value = serde_json::to_value(root).unwrap_or(Value::Null);
176    let mut b = Builder::default();
177    node_json_path(&value, data, 0, width, String::new(), renders, &mut b);
178    b.finish()
179}

181/// Build the dashboard pseudo-view: every view every driver offers,
182/// plus the merged chrome contributions (dashboard/navbar/statusbar/
183/// routes/sidebar/headerActions/notifications/jobs), grouped under
184/// dim section headers. Every entry is selectable — the app resolves
185/// view targets on Enter.
186///
187/// The platform slots (`sidebar`, `header_actions`, `notifications`,
188/// `jobs`) get no dedicated TUI chrome — there is no sidebar pane, no
189/// header action bar, and the feed ops (`host.jobs()` /
190/// `host.notifications()`) aren't fetched on the worker — so they
191/// render here as contribution listings like every other slot. Feed
192/// *data* lives behind the `pavo jobs`/`pavo notifications` CLI verbs
193/// (and `/api/jobs`, `/api/notifications` on the web bridge).
194#[must_use]
195pub fn build_dashboard(
196    views: &[ViewEntry],
197    chrome: &Chrome,
198    route_target: &dyn Fn(&str) -> Option<String>,
199    width: usize,
200) -> Model {
201    let _ = width; // no tables here — nothing to fit
202    let mut b = Builder::default();
203    b.row(
204        "dashboard — jump to any view or contribution",
205        RowStyle::Title,
206    );
207
208    b.row("views", RowStyle::Dim);
209    for v in views {
210        let text = format!("{}:{} — {}", v.driver, v.id, v.title);
211        let mut fields = vec![
212            ("view".to_string(), v.namespaced.clone()),
213            ("driver".to_string(), v.driver.clone()),
214            ("title".to_string(), v.title.clone()),
215            ("route".to_string(), v.route.clone()),
216        ];
217        if let Some(d) = &v.description {
218            fields.push(("description".to_string(), d.clone()));
219        }
220        if !v.queries.is_empty() {
221            fields.push(("queries".to_string(), v.queries.join(", ")));
222        }
223        b.sel_row(&text, RowStyle::Normal, v.namespaced.clone(), fields);
224    }
225    if views.is_empty() {
226        b.row("  (no views discovered)", RowStyle::Dim);
227    }
228
229    section(
230        &mut b,
231        "dashboard contributions",
232        &chrome.dashboard,
233        route_target,
234    );
235    section(&mut b, "navbar", &chrome.navbar, route_target);
236    section(&mut b, "statusbar", &chrome.statusbar, route_target);
237    section(&mut b, "routes", &chrome.routes, route_target);
238    section(&mut b, "sidebar", &chrome.sidebar, route_target);
239    section(
240        &mut b,
241        "header actions",
242        &chrome.header_actions,
243        route_target,
244    );
245    // Provider *metadata* only — the feeds themselves are wire ops
246    // the TUI worker doesn't call (see the fn docs).
247    section(&mut b, "notifications", &chrome.notifications, route_target);
248    section(&mut b, "jobs", &chrome.jobs, route_target);
249    b.finish()
250}

252/// A titled key/value note — error panes, route-info panes, the empty
253/// state. Every field is a selectable row.
254#[must_use]
255pub fn build_note(title: &str, fields: &[(String, String)], error: bool) -> Model {
256    let mut b = Builder::default();
257    b.row(
258        title,
259        if error {
260            RowStyle::Error
261        } else {
262            RowStyle::Title
263        },
264    );
265    for (k, v) in fields {
266        let text = if k.is_empty() {
267            v.clone()
268        } else {
269            format!("{k}: {v}")
270        };
271        b.sel_row(
272            &text,
273            if error {
274                RowStyle::Error
275            } else {
276                RowStyle::Normal
277            },
278            title.to_string(),
279            vec![(k.clone(), v.clone())],
280        );
281    }
282    b.finish()
283}

295#[derive(Default)]
296struct Builder {
297    rows: Vec<Row>,
298    selections: Vec<Selection>,
299    dots: Vec<String>,
300    forms: Vec<FormSpec>,
301    form_marks: Vec<FormMark>,
302}
303
304impl Builder {
305    fn row(&mut self, text: &str, style: RowStyle) {
306        self.rows.push(Row {
307            text: text.to_string(),
308            style,
309            sel: None,
310        });
311    }
312
313    /// Push a selectable row; returns nothing — `sel` indexes
314    /// `selections`.
315    fn sel_row(
316        &mut self,
317        text: &str,
318        style: RowStyle,
319        title: String,
320        fields: Vec<(String, String)>,
321    ) {
322        let sel = self.selections.len();
323        self.selections.push(Selection {
324            title,
325            fields,
326            item: None,
327        });
328        self.rows.push(Row {
329            text: text.to_string(),
330            style,
331            sel: Some(sel),
332        });
333    }
334
335    /// [`Builder::sel_row`] plus an item-context payload (tree items
336    /// carry their [`item_value`](pavo_core::item_value) JSON for route interpolation).
337    fn sel_row_with_item(
338        &mut self,
339        text: &str,
340        style: RowStyle,
341        title: String,
342        fields: Vec<(String, String)>,
343        item: Option<Value>,
344    ) {
345        let sel = self.selections.len();
346        self.selections.push(Selection {
347            title,
348            fields,
349            item,
350        });
351        self.rows.push(Row {
352            text: text.to_string(),
353            style,
354            sel: Some(sel),
355        });
356    }
357
358    /// Mark the most recently pushed row as belonging to an
359    /// interactive form (`field: None` = the submit row).
360    fn mark_form(&mut self, form: usize, field: Option<usize>) {
361        if let Some(row) = self.rows.len().checked_sub(1) {
362            self.form_marks.push(FormMark { row, form, field });
363        }
364    }
365
366    fn finish(self) -> Model {
367        let selectables = self
368            .rows
369            .iter()
370            .enumerate()
371            .filter_map(|(i, r)| r.sel.map(|_| i))
372            .collect();
373        Model {
374            rows: self.rows,
375            selections: self.selections,
376            selectables,
377            graph_dot: if self.dots.is_empty() {
378                None
379            } else {
380                Some(self.dots.join("\n"))
381            },
382            forms: self.forms,
383            form_marks: self.form_marks,
384        }
385    }
386}

388fn section(
389    b: &mut Builder,
390    name: &str,
391    entries: &[pavo_core::SlotEntry],
392    route_target: &dyn Fn(&str) -> Option<String>,
393) {
394    if entries.is_empty() {
395        return;
396    }
397    b.row(name, RowStyle::Dim);
398    for e in entries {
399        let c = &e.contribution;
400        let target = route_target(&c.route);
401        let text = format!("  {} — {} ({})", c.title, c.route, e.driver);
402        let mut fields = vec![
403            ("title".to_string(), c.title.clone()),
404            ("route".to_string(), c.route.clone()),
405            ("driver".to_string(), e.driver.clone()),
406            ("id".to_string(), c.id.clone()),
407        ];
408        if let Some(t) = &target {
409            fields.push(("resolves to".to_string(), t.clone()));
410        } else {
411            fields.push(("note".to_string(), "driver-owned route".to_string()));
412        }
413        // Selection title doubles as the jump target when resolvable.
414        b.sel_row(
415            &text,
416            RowStyle::Normal,
417            target.unwrap_or_else(|| c.title.clone()),
418            fields,
419        );
420    }
421}

425/// The node walker with a tree `path` (plugin-render lookup key) and
426/// the plugin `renders` map — a claimed node emits its
427/// [`RenderModel`] instead of the built-in dispatch.
428fn node_json_path(
429    v: &Value,
430    data: &Map<String, Value>,
431    depth: usize,
432    width: usize,
433    path: String,
434    renders: &HashMap<String, RenderModel>,
435    b: &mut Builder,
436) {
437    let ind = "  ".repeat(depth);
438    if let Some(model) = renders.get(&path) {
439        render_model_rows(model, depth, width, b);
440        return;
441    }
442    match node_kind_json(v).unwrap_or("") {
443        "panel" | "row" | "column" => {
444            if let Some(title) = scalar_str(v.get("title")) {
445                b.row(&format!("{ind}{title}"), RowStyle::Title);
446            }
447            if let Some(children) = v.get("children").and_then(Value::as_array) {
448                for (i, child) in children.iter().enumerate() {
449                    let child_path = if path.is_empty() {
450                        i.to_string()
451                    } else {
452                        format!("{path}.{i}")
453                    };
454                    node_json_path(child, data, depth + 1, width, child_path, renders, b);
455                }
456            }
457        }
458        "table" => table_json(v, data, depth, width, b),
459        "tree" => tree_json(v, data, depth, b),
460        "graph" => graph_json(v, data, depth, b),
461        "form" => form_json(v, depth, b),
462        "kv" => kv_json(v, data, depth, b),
463        "markdown" => markdown_json(v, depth, b),
464        "embed" => embed_json(v, depth, b),
465        // Unknown or missing kind — degrade to KV per the protocol's
466        // forward-compat rule.
467        _ => {
468            let fallback = kv_fallback_json(v);
469            match serde_json::from_value::<KvNode>(fallback) {
470                Ok(kv) => kv_node(&kv, depth, b),
471                Err(_) => b.row(&format!("{ind}(unrenderable node)"), RowStyle::Dim),
472            }
473        }
474    }
475}

477/// The node's bound query payload, if `query` names a key in `data`.
478fn bound<'a>(v: &'a Value, data: &'a Map<String, Value>) -> Option<(&'a str, &'a Value)> {
479    let query = v.get("query").and_then(Value::as_str)?;
480    data.get(query).map(|d| (query, d))
481}
482
483/// Emit the `[query 'q' error: …]` row when the bound payload is a
484/// degradation marker. Returns true when emitted (caller returns).
485fn bound_error(v: &Value, data: &Map<String, Value>, ind: &str, b: &mut Builder) -> bool {
486    let Some((query, payload)) = bound(v, data) else {
487        return false;
488    };
489    if let Some(err) = payload.get("error").and_then(Value::as_str) {
490        b.row(
491            &format!("{ind}[query '{query}' error: {err}]"),
492            RowStyle::Error,
493        );
494        return true;
495    }
496    false
497}

507struct Col {
508    key: String,
509    title: String,
510    align: String,
511}

513fn table_json(v: &Value, data: &Map<String, Value>, depth: usize, width: usize, b: &mut Builder) {
514    let ind = "  ".repeat(depth);
515    title_row(v, &ind, b);
516    if bound_error(v, data, &ind, b) {
517        return;
518    }
519    let mut columns: Vec<Col> = v
520        .get("columns")
521        .and_then(Value::as_array)
522        .map(|cols| {
523            cols.iter()
524                .map(|c| Col {
525                    key: c
526                        .get("key")
527                        .and_then(Value::as_str)
528                        .unwrap_or("")
529                        .to_string(),
530                    title: c
531                        .get("title")
532                        .and_then(Value::as_str)
533                        .or_else(|| c.get("key").and_then(Value::as_str))
534                        .unwrap_or("")
535                        .to_string(),
536                    align: c
537                        .get("align")
538                        .and_then(Value::as_str)
539                        .unwrap_or("left")
540                        .to_string(),
541                })
542                .collect()
543        })
544        .unwrap_or_default();
545
546    // Rows: inline `rows` win; otherwise the bound query payload —
547    // an array of arrays, or an array of objects keyed by column key.
548    let mut rows: Vec<Vec<String>> = Vec::new();
549    if let Some(inline) = v.get("rows").and_then(Value::as_array) {
550        for row in inline {
551            if let Some(cells) = row.as_array() {
552                rows.push(cells.iter().map(cell_str).collect());
553            }
554        }
555    } else if let Some((_, payload)) = bound(v, data) {
556        match payload {
557            Value::Array(items) => {
558                for item in items {
559                    match item {
560                        Value::Array(cells) => {
561                            rows.push(cells.iter().map(cell_str).collect());
562                        }
563                        Value::Object(map) => {
564                            rows.push(
565                                columns
566                                    .iter()
567                                    .map(|c| cell_str(map.get(&c.key).unwrap_or(&Value::Null)))
568                                    .collect(),
569                            );
570                        }
571                        other => rows.push(vec![cell_str(other)]),
572                    }
573                }
574            }
575            other => {
576                b.row(&format!("{ind}{}", cell_str(other)), RowStyle::Dim);
577                return;
578            }
579        }
580    }
581    if columns.is_empty() && !rows.is_empty() {
582        let n = rows.iter().map(Vec::len).max().unwrap_or(0);
583        columns = (1..=n)
584            .map(|i| Col {
585                key: i.to_string(),
586                title: i.to_string(),
587                align: "left".to_string(),
588            })
589            .collect();
590    }
591    if columns.is_empty() {
592        b.row(&format!("{ind}(empty table)"), RowStyle::Dim);
593        return;
594    }
595    for row in &mut rows {
596        row.resize(columns.len(), String::new());
597        row.truncate(columns.len());
598    }
599
600    // Column widths = max(header, cells); shrink the widest until the
601    // row fits `width` (cells truncate with '…' at emit time).
602    let avail = width.saturating_sub(depth * 2).max(columns.len() * 4);
603    let sep = 2usize;
604    let mut widths: Vec<usize> = columns.iter().map(|c| c.title.chars().count()).collect();
605    for row in &rows {
606        for (i, cell) in row.iter().enumerate() {
607            if i < widths.len() {
608                widths[i] = widths[i].max(cell.chars().count());
609            }
610        }
611    }
612    loop {
613        let total: usize = widths.iter().sum::<usize>() + sep * (widths.len() - 1);
614        if total <= avail {
615            break;
616        }
617        // Shrink the widest column that can still shrink.
618        let Some((idx, _)) = widths
619            .iter()
620            .enumerate()
621            .filter(|(_, w)| **w > 3)
622            .max_by_key(|(_, w)| *w)
623        else {
624            break;
625        };
626        widths[idx] -= 1;
627    }
628
629    let mut emit = |cells: &[String], style: RowStyle, sel: Option<Selection>| {
630        let parts: Vec<String> = cells
631            .iter()
632            .enumerate()
633            .map(|(i, cell)| fit(cell, widths[i], &columns[i].align))
634            .collect();
635        let text = format!("{ind}{}", parts.join(&" ".repeat(sep)));
636        if let Some(sel) = sel {
637            let sel_idx = b.selections.len();
638            b.selections.push(sel);
639            b.rows.push(Row {
640                text,
641                style,
642                sel: Some(sel_idx),
643            });
644        } else {
645            b.row(&text, style);
646        }
647    };
648
649    let header_cells: Vec<String> = columns.iter().map(|c| c.title.clone()).collect();
650    emit(&header_cells, RowStyle::Header, None);
651    let sep_cells: Vec<String> = widths.iter().map(|w| "-".repeat(*w)).collect();
652    emit(&sep_cells, RowStyle::Dim, None);
653    let col_meta: Vec<(String, String)> = columns
654        .iter()
655        .map(|c| (c.title.clone(), c.key.clone()))
656        .collect();
657    for row in &rows {
658        let fields: Vec<(String, String)> = col_meta
659            .iter()
660            .enumerate()
661            .map(|(i, (title, _))| (title.clone(), row.get(i).cloned().unwrap_or_default()))
662            .collect();
663        let node_title = v
664            .get("title")
665            .and_then(Value::as_str)
666            .unwrap_or("table")
667            .to_string();
668        emit(
669            row,
670            RowStyle::Normal,
671            Some(Selection {
672                title: format!("{node_title} row"),
673                fields,
674                item: None,
675            }),
676        );
677    }

682fn tree_json(v: &Value, data: &Map<String, Value>, depth: usize, b: &mut Builder) {
683    let ind = "  ".repeat(depth);
684    title_row(v, &ind, b);
685    if bound_error(v, data, &ind, b) {
686        return;
687    }
688    let mut items: Vec<TreeItem> = Vec::new();
689    let inline = v.get("items").and_then(Value::as_array);
690    let bound_items = bound(v, data).and_then(|(_, d)| {
691        d.get("items")
692            .and_then(Value::as_array)
693            .or_else(|| d.as_array())
694    });
695    let source = inline.or(bound_items);
696    if let Some(arr) = source {
697        for item in arr {
698            if let Ok(t) = serde_json::from_value::<TreeItem>(item.clone()) {
699                items.push(t);
700            }
701        }
702    }
703    if items.is_empty() {
704        if let Some((_, payload)) = bound(v, data) {
705            if inline.is_none() && !payload.is_array() {
706                b.row(&format!("{ind}{}", cell_str(payload)), RowStyle::Dim);
707                return;
708            }
709        }
710        b.row(&format!("{ind}(empty tree)"), RowStyle::Dim);
711        return;
712    }
713    let prefix = format!("{ind}  ");
714    tree_items(&items, &prefix, b);
715}
716
717fn tree_items(items: &[TreeItem], prefix: &str, b: &mut Builder) {
718    for (i, item) in items.iter().enumerate() {
719        let last = i + 1 == items.len();
720        let conn = if last { "└─ " } else { "├─ " };
721        let mut text = format!("{prefix}{conn}{}", item.label);
722        if let Some(detail) = &item.detail {
723            text.push_str(&format!("  — {detail}"));
724        }
725        let mut fields = vec![("label".to_string(), item.label.clone())];
726        if let Some(d) = &item.detail {
727            fields.push(("detail".to_string(), d.clone()));
728        }
729        if let Some(id) = &item.id {
730            fields.push(("id".to_string(), id.clone()));
731        }
732        if let Some(icon) = &item.icon {
733            fields.push(("icon".to_string(), icon.clone()));
734        }
735        if !item.children.is_empty() {
736            fields.push(("children".to_string(), item.children.len().to_string()));
737        }
738        if !item.props.is_empty() {
739            fields.push((
740                "props".to_string(),
741                pretty(&Value::Object(item.props.clone())),
742            ));
743        }
744        b.sel_row_with_item(
745            &text,
746            RowStyle::Normal,
747            item.label.clone(),
748            fields,
749            Some(pavo_core::item_value(item)),
750        );
751        let child_prefix = format!("{prefix}{}", if last { "   " } else { "│  " });
752        tree_items(&item.children, &child_prefix, b);
753    }
754}

758fn graph_json(v: &Value, data: &Map<String, Value>, depth: usize, b: &mut Builder) {
759    let ind = "  ".repeat(depth);
760    title_row(v, &ind, b);
761    if bound_error(v, data, &ind, b) {
762        return;
763    }
764    let (nodes, edges) = graph_parts(v, data);
765    if edges.is_empty() && nodes.is_empty() {
766        b.row(&format!("{ind}(empty graph)"), RowStyle::Dim);
767        return;
768    }
769    // id → label lookup so edges read as names where possible.
770    let mut labels: Map<String, Value> = Map::new();
771    for node in &nodes {
772        if let (Some(id), Some(label)) = (
773            node.get("id").and_then(Value::as_str),
774            node.get("label").and_then(Value::as_str),
775        ) {
776            labels.insert(id.to_string(), Value::from(label));
777        }
778    }
779    let disp = |id: &str| -> String {
780        labels
781            .get(id)
782            .and_then(Value::as_str)
783            .map(str::to_string)
784            .unwrap_or_else(|| id.to_string())
785    };
786
787    // Simple deterministic layered layout (longest-path layering +
788    // barycenter ordering) — edges between adjacent layers draw as
789    // box-drawing connectors; longer edges stay in the edge list.
790    let layout = ascii_layout(&nodes, &edges);
791    if !layout.is_empty() {
792        b.row(&format!("{ind}layout"), RowStyle::Dim);
793        for line in layout {
794            b.row(&format!("{ind}  {line}"), RowStyle::Normal);
795        }
796    }
797
798    b.row(&format!("{ind}nodes"), RowStyle::Dim);
799    for node in &nodes {
800        let id = node.get("id").and_then(Value::as_str).unwrap_or("");
801        let label = node.get("label").and_then(Value::as_str).unwrap_or(id);
802        let kind = node.get("kind").and_then(Value::as_str);
803        let mut text = format!("{ind}  {label}");
804        if let Some(k) = kind {
805            text.push_str(&format!(" [{k}]"));
806        }
807        if !id.is_empty() {
808            text.push_str(&format!(" ({id})"));
809        }
810        let mut fields = vec![
811            ("id".to_string(), id.to_string()),
812            ("label".to_string(), label.to_string()),
813        ];
814        if let Some(k) = kind {
815            fields.push(("kind".to_string(), k.to_string()));
816        }
817        if let Some(props) = node
818            .get("props")
819            .filter(|p| !p.as_object().is_some_and(Map::is_empty))
820        {
821            fields.push(("props".to_string(), pretty(props)));
822        }
823        b.sel_row(&text, RowStyle::Normal, label.to_string(), fields);
824    }
825
826    b.row(&format!("{ind}edges"), RowStyle::Dim);
827    for edge in &edges {
828        let from = edge.get("from").and_then(Value::as_str).unwrap_or("");
829        let to = edge.get("to").and_then(Value::as_str).unwrap_or("");
830        let mut text = format!("{ind}  {} -> {}", disp(from), disp(to));
831        if let Some(label) = edge.get("label").and_then(Value::as_str) {
832            text.push_str(&format!("  ({label})"));
833        }
834        let fields = vec![
835            ("from".to_string(), disp(from)),
836            ("to".to_string(), disp(to)),
837            (
838                "label".to_string(),
839                edge.get("label")
840                    .and_then(Value::as_str)
841                    .unwrap_or("")
842                    .to_string(),
843            ),
844        ];
845        b.sel_row(&text, RowStyle::Dim, format!("{from} -> {to}"), fields);
846    }
847
848    // DOT dump for the 'g' key — same shape as pavo-cli render_dot.
849    b.dots.push(dot_block(&nodes, &edges));
850}

852/// A graph node's effective `(nodes, edges)` — inline fields win,
853/// falling back to the bound query payload's `nodes`/`edges` keys.
854fn graph_parts(v: &Value, data: &Map<String, Value>) -> (Vec<Value>, Vec<Value>) {
855    let mut nodes = v
856        .get("nodes")
857        .and_then(Value::as_array)
858        .cloned()
859        .unwrap_or_default();
860    let mut edges = v
861        .get("edges")
862        .and_then(Value::as_array)
863        .cloned()
864        .unwrap_or_default();
865    if let Some((_, payload)) = bound(v, data) {
866        if payload.get("error").and_then(Value::as_str).is_none() {
867            if let Some(n) = payload.get("nodes").and_then(Value::as_array) {
868                if !n.is_empty() {
869                    nodes = n.clone();
870                }
871            }
872            if let Some(e) = payload.get("edges").and_then(Value::as_array) {
873                if !e.is_empty() {
874                    edges = e.clone();
875                }
876            }
877        }
878    }
879    (nodes, edges)
880}

882// ---------- graph ASCII layout ----------
883
884/// Render `(nodes, edges)` as a layered ASCII diagram: longest-path
885/// layering, one barycenter ordering sweep per direction, node boxes
886/// `┌─┐│x│└─┘`, and `│ ─ ┌ ┐ └ ┘` connectors for edges between
887/// adjacent layers. Edges spanning multiple layers (or cycles) are
888/// not drawn — they remain in the edge list below. Deterministic,
889/// allocation-light; fine for dozens of nodes at 60fps.
890fn ascii_layout(nodes: &[Value], edges: &[Value]) -> Vec<String> {
891    /// One grid cell.
892    #[derive(Clone, Copy)]
893    struct Cell(char);
894
895    let ids: Vec<String> = nodes
896        .iter()
897        .map(|n| {
898            n.get("id")
899                .and_then(Value::as_str)
900                .unwrap_or("")
901                .to_string()
902        })
903        .collect();
904    let index: HashMap<String, usize> = ids
905        .iter()
906        .enumerate()
907        .map(|(i, id)| (id.clone(), i))
908        .collect();
909    let labels: Vec<String> = nodes
910        .iter()
911        .enumerate()
912        .map(|(i, n)| {
913            n.get("label")
914                .and_then(Value::as_str)
915                .filter(|l| !l.is_empty())
916                .unwrap_or(ids[i].as_str())
917                .to_string()
918        })
919        .collect();
920    // (from_idx, to_idx) — unknown endpoints dropped.
921    let links: Vec<(usize, usize)> = edges
922        .iter()
923        .filter_map(|e| {
924            let f = e.get("from").and_then(Value::as_str)?;
925            let t = e.get("to").and_then(Value::as_str)?;
926            Some((*index.get(f)?, *index.get(t)?))
927        })
928        .collect();
929    let n = ids.len();
930    if n == 0 {
931        return Vec::new();
932    }
933
934    // Longest-path layering, bounded relaxation (cycle-safe).
935    let mut layer = vec![0usize; n];
936    for _ in 0..n {
937        let mut changed = false;
938        for &(f, t) in &links {
939            let want = layer[f] + 1;
940            if want > layer[t] {
941                layer[t] = want;
942                changed = true;
943            }
944        }
945        if !changed {
946            break;
947        }
948    }
949    let depth = layer.iter().copied().max().unwrap_or(0);
950
951    // Group node indices per layer (stable, input order).
952    let mut by_layer: Vec<Vec<usize>> = vec![Vec::new(); depth + 1];
953    for (i, &l) in layer.iter().enumerate() {
954        by_layer[l].push(i);
955    }
956
957    // Barycenter ordering: a few top-down + bottom-up sweeps sorting
958    // each layer by the mean position of its already-placed neighbors.
959    let mut pos = vec![0usize; n]; // index within its layer
960    let refresh_pos = |by_layer: &[Vec<usize>], pos: &mut [usize]| {
961        for layer_nodes in by_layer {
962            for (i, &node) in layer_nodes.iter().enumerate() {
963                pos[node] = i;
964            }
965        }
966    };
967    refresh_pos(&by_layer, &mut pos);
968    for sweep in 0..4 {
969        let forward = sweep % 2 == 0;
970        let order: Vec<usize> = if forward {
971            (1..=depth).collect()
972        } else {
973            (0..depth).rev().collect()
974        };
975        for li in order {
976            // Neighbor positions in adjacent layers (already placed
977            // this sweep): predecessors on the way down, successors
978            // on the way up.
979            let want_preds = forward;
980            by_layer[li].sort_by_key(|&node| {
981                // Mean position of adjacent-layer neighbors already
982                // placed this sweep (predecessors going down,
983                // successors going up); unconnected nodes keep place.
984                let mut sum = 0usize;
985                let mut cnt = 0usize;
986                for &(f, t) in &links {
987                    let other = if want_preds {
988                        (t == node).then_some(f)
989                    } else {
990                        (f == node).then_some(t)
991                    };
992                    if let Some(o) = other {
993                        if layer[o].abs_diff(layer[node]) == 1 {
994                            sum += pos[o];
995                            cnt += 1;
996                        }
997                    }
998                }
999                sum.checked_div(cnt).unwrap_or(pos[node])
1000            });
1001            refresh_pos(&by_layer, &mut pos);
1002        }
1003    }
1004
1005    // Geometry: per-layer column pitch, per-node box width, fixed
1006    // 4-row pitch (3 box rows + 1 connector gap).
1007    const BOX_H: usize = 3;
1008    const PITCH_Y: usize = 4;
1009    let widths: Vec<usize> = labels
1010        .iter()
1011        .map(|l| l.chars().count().max(1) + 4)
1012        .collect();
1013    let layer_pitch: Vec<usize> = by_layer
1014        .iter()
1015        .map(|ns| ns.iter().map(|&i| widths[i]).max().unwrap_or(0) + 3)
1016        .collect();
1017    // A layer's width is its node count times the per-node pitch.
1018    let layer_w: Vec<usize> = by_layer
1019        .iter()
1020        .enumerate()
1021        .map(|(li, ns)| ns.len() * layer_pitch[li])
1022        .collect();
1023    let mut layer_x = vec![0usize; depth + 1];
1024    for li in 1..=depth {
1025        layer_x[li] = layer_x[li - 1] + layer_w[li - 1];
1026    }
1027    let total_w = layer_x[depth] + layer_w[depth];
1028    let rows_n = depth * PITCH_Y + BOX_H;
1029
1030    // Node boxes.
1031    let mut grid = vec![vec![Cell(' '); total_w]; rows_n];
1032    let mut center = vec![(0usize, 0usize); n]; // (x_center, y_top)
1033    for (li, ns) in by_layer.iter().enumerate() {
1034        for (slot, &node) in ns.iter().enumerate() {
1035            let w = widths[node];
1036            let x = layer_x[li] + slot * layer_pitch[li] + layer_pitch[li].saturating_sub(w) / 2;
1037            let y = li * PITCH_Y;
1038            center[node] = (x + w / 2, y);
1039            let label = &labels[node];
1040            let label_len = label.chars().count();
1041            let pad = w.saturating_sub(2 + label_len);
1042            let left = pad / 2;
1043            let right = pad - left;
1044            let top: String = format!("┌{}┐", "─".repeat(w.saturating_sub(2)));
1045            let mid = format!("│{}{}{}│", " ".repeat(left), label, " ".repeat(right));
1046            let bot: String = format!("└{}┘", "─".repeat(w.saturating_sub(2)));
1047            for (dy, line) in [top, mid, bot].iter().enumerate() {
1048                for (dx, ch) in line.chars().enumerate() {
1049                    if x + dx < total_w && y + dy < rows_n {
1050                        grid[y + dy][x + dx] = Cell(ch);
1051                    }
1052                }
1053            }
1054        }
1055    }
1056
1057    // Edges between adjacent layers only. Two passes — horizontal
1058    // runs first, then stubs — so later edges don't erase earlier
1059    // corners on a shared gap row.
1060    let mut drawn: Vec<(usize, usize, usize)> = Vec::new(); // (fx, tx, gap_y)
1061    let mut long_edges = 0usize;
1062    for &(f, t) in &links {
1063        if layer[t] != layer[f] + 1 {
1064            long_edges += 1;
1065            continue;
1066        }
1067        let (fx, fy) = center[f];
1068        let (tx, _ty) = center[t];
1069        drawn.push((fx, tx, fy + BOX_H));
1070    }
1071    for &(fx, tx, gap_y) in &drawn {
1072        let (lo, hi) = if fx <= tx { (fx, tx) } else { (tx, fx) };
1073        if gap_y < rows_n {
1074            for cell in grid[gap_y].iter_mut().take(hi + 1).skip(lo) {
1075                *cell = Cell('─');
1076            }
1077        }
1078    }
1079    for &(fx, tx, gap_y) in &drawn {
1080        // Source stub: down out of the box, turning toward target.
1081        if gap_y < rows_n && fx < total_w {
1082            grid[gap_y][fx] = Cell(if tx > fx {
1083                '└'
1084            } else if tx < fx {
1085                '┘'
1086            } else {
1087                '│'
1088            });
1089        }
1090        // Target stub: horizontal arrives, turns down into the box.
1091        if gap_y < rows_n && tx < total_w {
1092            grid[gap_y][tx] = Cell(if fx < tx {
1093                '┐'
1094            } else if fx > tx {
1095                '┌'
1096            } else {
1097                '│'
1098            });
1099        }
1100    }
1101
1102    let mut out: Vec<String> = grid
1103        .iter()
1104        .map(|row| {
1105            let s: String = row.iter().map(|c| c.0).collect();
1106            s.trim_end().to_string()
1107        })
1108        .filter(|s| !s.trim().is_empty())
1109        .collect();
1110    if long_edges > 0 {
1111        out.push(format!(
1112            "({long_edges} edge(s) span layers — see edge list)"
1113        ));
1114    }
1115    out
1116}

1142                dot_str(to)

1144        }
1145    }
1146    out.push_str("}\n");
1147    out

1152fn form_json(v: &Value, depth: usize, b: &mut Builder) {
1153    let ind = "  ".repeat(depth);
1154    title_row(v, &ind, b);
1155    // Interactive = the driver declared a submit action. Forms
1156    // without `submit_op` stay a read-only field listing (the
1157    // renderer "may only preview" per the protocol docs).
1158    let submit_op = v.get("submit_op").and_then(Value::as_str);
1159    let Some(submit_op) = submit_op.filter(|s| !s.is_empty()) else {
1160        form_readonly(v, &ind, b);
1161        return;
1162    };
1163
1164    // Register the spec first so field/submit rows can mark it.
1165    let form_idx = b.forms.len();
1166    let fields: Vec<FormFieldSpec> = v
1167        .get("fields")
1168        .and_then(Value::as_array)
1169        .map(|arr| arr.iter().map(form_field_spec).collect())
1170        .unwrap_or_default();
1171    let title = v
1172        .get("title")
1173        .and_then(Value::as_str)
1174        .unwrap_or("form")
1175        .to_string();
1176    b.forms.push(FormSpec {
1177        id: v.get("id").and_then(Value::as_str).map(str::to_string),
1178        title: title.clone(),
1179        submit_op: submit_op.to_string(),
1180        submit_label: v
1181            .get("submit_label")
1182            .and_then(Value::as_str)
1183            .unwrap_or("Submit")
1184            .to_string(),
1185        fields,
1186    });
1187
1188    for (fi, f) in b.forms[form_idx].fields.clone().iter().enumerate() {
1189        let mut text = format!("{ind}- {}: {} [{}]", f.key, f.label, f.kind);
1190        if f.required {
1191            text.push_str(" (required)");
1192        }
1193        // Show the effective value: `value` (existing state) wins
1194        // over `default`; placeholder marks an empty editable field.
1195        let current = f.value.as_ref().or(f.default.as_ref());
1196        match current {
1197            Some(val) => text.push_str(&format!(" = {}", cell_str(val))),
1198            None if f.editable => {
1199                if let Some(p) = &f.placeholder {
1200                    text.push_str(&format!(" = {p} (placeholder)"));
1201                }
1202            }
1203            None => {}
1204        }
1205        if f.editable {
1206            text.push_str(" ✎");
1207        } else {
1208            text.push_str(" (read-only)");
1209        }
1210        if let Some(options) = (!f.options.is_empty()).then(|| f.options.join("|")) {
1211            text.push_str(&format!(" options={options}"));
1212        }
1213        if let Some(hint) = &f.hint {
1214            text.push_str(&format!(" — {hint}"));
1215        }
1216        let mut detail = vec![
1217            ("key".to_string(), f.key.clone()),
1218            ("label".to_string(), f.label.clone()),
1219            ("kind".to_string(), f.kind.clone()),
1220        ];
1221        if f.required {
1222            detail.push(("required".to_string(), "yes".to_string()));
1223        }
1224        if !f.editable {
1225            detail.push(("editable".to_string(), "no".to_string()));
1226        }
1227        for (k, val) in [("default", &f.default), ("value", &f.value)] {
1228            if let Some(val) = val {
1229                detail.push((k.to_string(), cell_str(val)));
1230            }
1231        }
1232        if let Some(hint) = &f.hint {
1233            detail.push(("hint".to_string(), hint.clone()));
1234        }
1235        let style = if f.editable {
1236            RowStyle::Normal
1237        } else {
1238            RowStyle::Dim
1239        };
1240        b.sel_row(&text, style, f.label.clone(), detail);
1241        if f.editable {
1242            b.mark_form(form_idx, Some(fi));
1243        }
1244    }
1245
1246    // The submit row — Enter opens the editor at the first editable
1247    // field; the actual `submit` op fires from inside the session.
1248    let label = b.forms[form_idx].submit_label.clone();
1249    let op = b.forms[form_idx].submit_op.clone();
1250    b.sel_row(
1251        &format!("{ind}{label} → {op}"),
1252        RowStyle::Dim,
1253        format!("{title} submit"),
1254        vec![
1255            ("form".to_string(), title),
1256            ("submit_op".to_string(), op),
1257            ("label".to_string(), label),
1258        ],
1259    );
1260    b.mark_form(form_idx, None);
1261}

1263/// Parse one raw field object into a [`FormFieldSpec`]. `editable`
1264/// defaults to true per the protocol (`editable:false` opts out).
1265fn form_field_spec(f: &Value) -> FormFieldSpec {
1266    FormFieldSpec {

1356fn kv_json(v: &Value, data: &Map<String, Value>, depth: usize, b: &mut Builder) {
1357    let ind = "  ".repeat(depth);
1358    title_row(v, &ind, b);
1359    if bound_error(v, data, &ind, b) {
1360        return;
1361    }
1362    let mut pairs: Vec<(String, Value, Option<String>)> = v
1363        .get("pairs")
1364        .and_then(Value::as_array)
1365        .map(|arr| {
1366            arr.iter()
1367                .filter_map(|p| {
1368                    Some((
1369                        p.get("key").and_then(Value::as_str)?.to_string(),
1370                        p.get("value").cloned().unwrap_or(Value::Null),
1371                        p.get("kind").and_then(Value::as_str).map(str::to_string),
1372                    ))
1373                })
1374                .collect()
1375        })
1376        .unwrap_or_default();
1377    // Query-bound kv: an object payload becomes the pairs.
1378    if pairs.is_empty() {
1379        if let Some((_, payload)) = bound(v, data) {
1380            if let Value::Object(map) = payload {
1381                pairs = map
1382                    .iter()
1383                    .map(|(k, val)| (k.clone(), val.clone(), None))
1384                    .collect();
1385            } else {
1386                b.row(&format!("{ind}{}", cell_str(payload)), RowStyle::Dim);
1387                return;
1388            }
1389        }
1390    }
1391    if pairs.is_empty() {
1392        b.row(&format!("{ind}(empty)"), RowStyle::Dim);
1393        return;
1394    }
1395    for (key, value, kind) in pairs {
1396        let mut text = format!("{ind}{key}: {}", cell_str(&value));
1397        if let Some(k) = &kind {
1398            text.push_str(&format!("  [{k}]"));
1399        }
1400        let fields = vec![
1401            ("key".to_string(), key.clone()),
1402            ("value".to_string(), detail_value(&value)),
1403        ];
1404        b.sel_row(&text, RowStyle::Normal, key.clone(), fields);
1405    }
1406}

1515/// `RenderModel::Rows` → canvas rows: header + separator + one
1516/// selectable row per body row (detail = column → cell).
1517fn plugin_rows(
1518    headers: &[String],
1519    aligns: &[pavo_protocol::Align],
1520    rows: &[Vec<String>],
1521    depth: usize,
1522    width: usize,
1523    b: &mut Builder,
1524) {
1525    let ind = "  ".repeat(depth);
1526    let ncols = headers
1527        .len()
1528        .max(rows.iter().map(Vec::len).max().unwrap_or(0));
1529    if ncols == 0 {
1530        b.row(&format!("{ind}(empty)"), RowStyle::Dim);
1531        return;
1532    }
1533    let align_at = |i: usize| -> &str {
1534        aligns
1535            .get(i)
1536            .map(|a| match a {
1537                pavo_protocol::Align::Left => "left",
1538                pavo_protocol::Align::Center => "center",
1539                pavo_protocol::Align::Right => "right",
1540            })
1541            .unwrap_or("left")
1542    };
1543    // Same width-fitting policy as table_json: shrink widest until fit.
1544    let avail = width.saturating_sub(depth * 2).max(ncols * 4);
1545    let sep = 2usize;
1546    let mut widths: Vec<usize> = (0..ncols)
1547        .map(|i| headers.get(i).map_or(0, |h| h.chars().count()))
1548        .collect();
1549    for row in rows {
1550        for (i, cell) in row.iter().enumerate() {
1551            if i < widths.len() {
1552                widths[i] = widths[i].max(cell.chars().count());
1553            }
1554        }
1555    }
1556    loop {
1557        let total: usize = widths.iter().sum::<usize>() + sep * (widths.len() - 1);
1558        if total <= avail {
1559            break;
1560        }
1561        let Some((idx, _)) = widths
1562            .iter()
1563            .enumerate()
1564            .filter(|(_, w)| **w > 3)
1565            .max_by_key(|(_, w)| *w)
1566        else {
1567            break;
1568        };
1569        widths[idx] -= 1;
1570    }
1571    let emit = |cells: &[String], style: RowStyle, sel: Option<Selection>, b: &mut Builder| {
1572        let parts: Vec<String> = cells
1573            .iter()
1574            .enumerate()
1575            .map(|(i, cell)| fit(cell, widths[i], align_at(i)))
1576            .collect();
1577        let text = format!("{ind}{}", parts.join(&" ".repeat(sep)));
1578        match sel {
1579            Some(sel) => {
1580                let idx = b.selections.len();
1581                b.selections.push(sel);
1582                b.rows.push(Row {
1583                    text,
1584                    style,
1585                    sel: Some(idx),
1586                });
1587            }
1588            None => b.row(&text, style),
1589        }
1590    };
1591    let header_cells: Vec<String> = (0..ncols)
1592        .map(|i| headers.get(i).cloned().unwrap_or_default())
1593        .collect();
1594    emit(&header_cells, RowStyle::Header, None, b);
1595    let sep_cells: Vec<String> = widths.iter().map(|w| "-".repeat(*w)).collect();
1596    emit(&sep_cells, RowStyle::Dim, None, b);
1597    for row in rows {
1598        let cells: Vec<String> = (0..ncols)
1599            .map(|i| row.get(i).cloned().unwrap_or_default())
1600            .collect();
1601        let fields: Vec<(String, String)> = (0..ncols)
1602            .map(|i| {
1603                (
1604                    headers.get(i).cloned().unwrap_or_else(|| format!("col{i}")),
1605                    cells[i].clone(),
1606                )
1607            })
1608            .collect();
1609        emit(
1610            &cells,
1611            RowStyle::Normal,
1612            Some(Selection {
1613                title: "plugin row".to_string(),
1614                fields,
1615                item: None,
1616            }),
1617            b,
1618        );
1619    }
1620}

1652/// Truncate to `width` chars with a trailing `…`, then pad to exactly
1653/// `width` per the alignment hint.
1654fn fit(text: &str, width: usize, align: &str) -> String {
1655    let truncated = if text.chars().count() > width {
1656        if width == 0 {
1657            String::new()
1658        } else if width == 1 {
1659            "…".to_string()
1660        } else {
1661            let mut t: String = text.chars().take(width - 1).collect();
1662            t.push('…');
1663            t
1664        }
1665    } else {
1666        text.to_string()
1667    };
1668    let len = truncated.chars().count();
1669    if len >= width {
1670        return truncated;
1671    }
1672    let fill = width - len;
1673    match align {
1674        "right" => format!("{}{truncated}", " ".repeat(fill)),
1675        "center" => {
1676            let left = fill / 2;
1677            format!("{}{truncated}{}", " ".repeat(left), " ".repeat(fill - left))
1678        }
1679        _ => format!("{truncated}{}", " ".repeat(fill)),
1680    }
1681}
