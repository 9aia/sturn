1//! Interaction actions (P-032) — the TUI's side of the `invoke`
2//! protocol.
3//!
4//! [`UiAction`] is the owned form of [`pavo_core::ScopedAction`] the
5//! worker ships to the app (scoped actions borrow the view result;
6//! the UI needs 'static data across frames). [`PaletteEntry`],
7//! [`PaletteSession`], [`ConfirmSession`] and [`TypeTargetSession`]
8//! are the overlay sessions; [`action_form_spec`] compiles an
9//! action's `params` into the existing [`FormSpec`]/[`FormSession`](crate::form::FormSession)
10//! machinery so auto-forms reuse the form editor verbatim.
11//!
12//! Dispatch flow ([`InvokeStep`]): params → auto-form; on submit the
13//! collected values are [`coerce_params`]-ed and
14//! [`pavo_core::validate_params`]-checked, then the danger gate runs
15//! — `none` invokes immediately, `confirm` opens [`ConfirmSession`],
16//! `type-target` opens [`TypeTargetSession`].
17

36/// An [`Action`] plus everything the TUI needs to dispatch it —
37/// owned clone of `pavo_core::ScopedAction` (which borrows the view
38/// result the worker collected it from).
39#[derive(Clone, Debug)]
40pub struct UiAction {
41    /// The action declaration.
42    pub action: Action,
43    /// Which declaration site it came from (keybinding + palette
44    /// grouping context).
45    pub scope: ActionScope,
46    /// Namespaced `driver:view` — the `invoke` op's `target` arg
47    /// (driver resolution + default view context).
48    pub target: String,
49    /// Bare view id — the invoke request's `view` context override.
50    pub view: Option<String>,
51    /// Tree item id for [`ActionScope::Item`] actions — becomes the
52    /// invoke `target` payload `{"id": …}`.
53    pub item_id: Option<String>,
54}
55
56impl UiAction {
57    /// The opaque `target` JSON for the invoke request: tree-item
58    /// actions carry `{"id": <item_id>}` (the protocol README's
59    /// example shape); everything else sends no target — the driver
60    /// resolves node/manifest actions from the `view` context.
61    #[must_use]
62    pub fn action_target(&self) -> Option<Value> {
63        self.item_id
64            .as_ref()
65            .map(|id| serde_json::json!({"id": id}))
66    }
67}

69/// Scope precedence for id collisions — item > node > manifest, per
70/// `pavo_core::actions::collect_actions` docs.
71fn scope_rank(scope: ActionScope) -> u8 {
72    match scope {
73        ActionScope::Item => 3,
74        ActionScope::Node => 2,
75        ActionScope::Manifest => 1,
76    }
77}
78
79/// Dedupe collected actions by `action.id`, keeping the
80/// highest-precedence scope (item > node > manifest). The surviving
81/// entry keeps the position of its id's first occurrence, so
82/// declaration order still drives display/keybinding order.
83#[must_use]
84pub fn dedupe_ui_actions(actions: Vec<UiAction>) -> Vec<UiAction> {
85    let mut out: Vec<UiAction> = Vec::with_capacity(actions.len());
86    let mut index: HashMap<String, usize> = HashMap::new();
87    for a in actions {
88        match index.get(&a.action.id) {
89            Some(&i) if scope_rank(out[i].scope) >= scope_rank(a.scope) => {}
90            Some(&i) => out[i] = a,
91            None => {
92                index.insert(a.action.id.clone(), out.len());
93                out.push(a);
94            }
95        }
96    }
97    out
98}

100/// What happens when an action is activated — the pure decision the
101/// app executes. `Form` means "open the auto-form first"; the other
102/// variants are the post-params danger gate.
103#[derive(Clone, Copy, Debug, PartialEq, Eq)]
104pub enum InvokeStep {
105    /// Params present — compile the auto-form; danger is re-checked
106    /// after the form submits.
107    Form,
108    /// No params, `danger: none` — invoke immediately.
109    Invoke,
110    /// `danger: confirm` — confirm modal, then invoke.
111    Confirm,
112    /// `danger: type-target` — type-the-target-id modal, then invoke.
113    TypeTarget,
114}
115
116/// The entry-point decision: params always win (the form collects
117/// them first; danger gates the collected invoke afterwards).
118#[must_use]
119pub fn plan_action(action: &Action) -> InvokeStep {
120    if !action.params.is_empty() {
121        return InvokeStep::Form;
122    }
123    danger_step(action.danger)
124}
125
126/// The post-form decision — same as [`plan_action`] minus the form
127/// branch (params were already collected).
128#[must_use]
129pub fn danger_step(danger: DangerLevel) -> InvokeStep {
130    match danger {
131        DangerLevel::None => InvokeStep::Invoke,
132        DangerLevel::Confirm => InvokeStep::Confirm,
133        DangerLevel::TypeTarget => InvokeStep::TypeTarget,
134    }
135}

161/// Extract the id a `type-target` session expects the user to type:
162/// a bare string target is itself the id; an object target's `"id"`
163/// string field wins. `None` → the session accepts any non-empty
164/// input (the target shape is driver-defined).
165#[must_use]
166pub fn target_id(target: &Option<Value>) -> Option<String> {
167    match target {
168        Some(Value::String(s)) if !s.is_empty() => Some(s.clone()),
169        Some(Value::Object(map)) => map
170            .get("id")
171            .and_then(Value::as_str)
172            .filter(|s| !s.is_empty())
173            .map(str::to_string),
174        _ => None,
175    }
176}

178/// Coerce a submitted form's `values` into invoke `params`: the
179/// `string[]` kind degrades to comma text in the form (per
180/// `param_form_field`), so split it back into a JSON array here.
181/// Everything else passes through — the form editor already coerces
182/// bool/number per field kind.
183#[must_use]
184pub fn coerce_params(action: &Action, values: Params) -> Params {
185    let mut out = values;
186    for p in &action.params {
187        if p.kind != pavo_protocol::ParamKind::StringList {
188            continue;
189        }
190        if let Some(Value::String(s)) = out.get(&p.name) {
191            let items: Vec<Value> = s
192                .split(',')
193                .map(str::trim)
194                .filter(|t| !t.is_empty())
195                .map(|t| Value::String(t.to_string()))
196                .collect();
197            out.insert(p.name.clone(), Value::Array(items));
198        }
199    }
200    out
201}

203/// Compile an action's `params` into a [`FormSpec`] for the existing
204/// form editor — [`param_form_field`] picks the widget kinds,
205/// [`FormFieldSpec`] carries them into [`crate::form::FormSession`].
206#[must_use]
207pub fn action_form_spec(action: &Action) -> FormSpec {
208    let fields = action
209        .params
210        .iter()
211        .map(|p| {
212            let f = param_form_field(p);
213            FormFieldSpec {
214                key: f.key,
215                label: f.label,
216                kind: kind_name(f.kind).to_string(),
217                required: f.required,
218                editable: f.editable,
219                options: f.options,
220                default: f.default,
221                value: f.value,
222                hint: f.hint,
223                placeholder: f.placeholder,
224            }
225        })
226        .collect();
227    FormSpec {
228        id: None,
229        title: action.label.clone(),
230        submit_op: "invoke".to_string(),
231        submit_label: "Invoke".to_string(),
232        fields,
233    }
234}

236/// [`FormFieldKind`] → the kind string [`FormFieldSpec`] carries
237/// (same spellings the canvas renderer and editor already use).
238fn kind_name(kind: FormFieldKind) -> &'static str {
239    match kind {
240        FormFieldKind::Text => "text",
241        FormFieldKind::Textarea => "textarea",
242        FormFieldKind::Number => "number",
243        FormFieldKind::Bool => "bool",
244        FormFieldKind::Select => "select",
245    }
246}

250/// One command-palette row — either a collected [`UiAction`] or a
251/// manifest `menus[]` entry (which itself resolves to an action or a
252/// route at activation time).
253#[derive(Clone, Debug)]
254pub enum PaletteEntry {
255    /// A declared/derived action — Enter runs the danger/param flow.
256    Action(UiAction),
257    /// A `contributes.menus[]` entry — `action` invokes (resolved
258    /// against collected actions by id), `route` navigates.
259    Menu(MenuEntry),
260}
261
262impl PaletteEntry {
263    /// Display label.
264    #[must_use]
265    pub fn label(&self) -> &str {
266        match self {
267            Self::Action(u) => &u.action.label,
268            Self::Menu(m) => &m.title,
269        }
270    }
271
272    /// Secondary line — description, action id, or route.
273    #[must_use]
274    pub fn detail(&self) -> String {
275        match self {
276            Self::Action(u) => u
277                .action
278                .description
279                .clone()
280                .unwrap_or_else(|| u.action.id.clone()),
281            Self::Menu(m) => m
282                .command
283                .clone()
284                .or_else(|| m.action.clone())
285                .or_else(|| m.route.clone())
286                .unwrap_or_else(|| m.id.clone()),
287        }
288    }
289
290    /// Extra fuzzy-match terms beyond the label.
291    fn haystack(&self) -> String {
292        let mut s = String::new();
293        match self {
294            Self::Action(u) => {
295                s.push_str(&u.action.id);
296                s.push(' ');
297                s.push_str(&u.action.label);
298                if let Some(d) = &u.action.description {
299                    s.push(' ');
300                    s.push_str(d);
301                }
302                for k in &u.action.keys {
303                    s.push(' ');
304                    s.push_str(k);
305                }
306            }
307            Self::Menu(m) => {
308                s.push_str(&m.id);
309                s.push(' ');
310                s.push_str(&m.title);
311                if let Some(c) = &m.command {
312                    s.push(' ');
313                    s.push_str(c);
314                }
315                for k in m.keywords.iter().chain(m.keys.iter()) {
316                    s.push(' ');
317                    s.push_str(k);
318                }
319            }
320        }
321        s
322    }
323
324    /// Grouping label for section headers (`None` = ungrouped).
325    #[must_use]
326    pub fn section(&self) -> Option<String> {
327        match self {
328            Self::Action(u) => Some(match u.scope {
329                ActionScope::Manifest => "actions".to_string(),
330                ActionScope::Node => "view".to_string(),
331                ActionScope::Item => "item".to_string(),
332            }),
333            Self::Menu(m) => m.section.clone(),
334        }
335    }
336}

338/// Case-insensitive subsequence match — the dmenu-style fuzzy
339/// filter. Empty query matches everything.
340#[must_use]
341pub fn fuzzy_match(query: &str, text: &str) -> bool {
342    if query.is_empty() {
343        return true;
344    }
345    let mut q = query.chars().flat_map(char::to_lowercase);
346    let mut want = q.next();
347    for c in text.chars().flat_map(char::to_lowercase) {
348        match want {
349            Some(w) if c == w => want = q.next(),
350            _ => {}
351        }
352        if want.is_none() {
353            return true;
354        }
355    }
356    want.is_none()
357}

359/// Build the palette's entry list: curated `menus[]` first (sorted
360/// by `priority`, then title), then the collected actions in
361/// declaration order.
362#[must_use]
363pub fn palette_entries(actions: &[UiAction], menus: &[MenuEntry]) -> Vec<PaletteEntry> {
364    let mut ms: Vec<&MenuEntry> = menus.iter().collect();
365    ms.sort_by(|a, b| {
366        a.priority
367            .cmp(&b.priority)
368            .then_with(|| a.title.cmp(&b.title))
369    });
370    ms.into_iter()
371        .map(|m| PaletteEntry::Menu(m.clone()))
372        .chain(actions.iter().cloned().map(PaletteEntry::Action))
373        .collect()
374}

378pub enum PaletteCmd {
379    /// Key consumed, nothing further.
380    None,

382    Close,
383    /// `Enter` — activate this entry (action or menu route).
384    Activate(Box<PaletteEntry>),

387/// The command-palette overlay: a [`LineEdit`] filter over
388/// [`palette_entries`], ↑↓/jk selection, Enter activates.
389#[derive(Clone, Debug)]
390pub struct PaletteSession {
391    /// The filter input.
392    pub edit: LineEdit,
393    /// All entries (pre-sorted — see [`palette_entries`]).
394    pub entries: Vec<PaletteEntry>,
395    /// Indices into `entries` matching the current filter.
396    pub matches: Vec<usize>,
397    /// Position within `matches`.
398    pub sel: usize,
399}
400
401impl PaletteSession {
402    /// Open over `entries`; the filter starts empty (everything
403    /// matches, in the given order).
404    #[must_use]
405    pub fn open(entries: Vec<PaletteEntry>) -> Self {
406        let mut s = Self {
407            edit: LineEdit::default(),
408            entries,
409            matches: Vec::new(),
410            sel: 0,
411        };
412        s.refilter();
413        s
414    }
415
416    /// Recompute `matches` from the filter text; selection resets
417    /// to the top.
418    fn refilter(&mut self) {
419        let q = self.edit.text.clone();
420        self.matches = self
421            .entries
422            .iter()
423            .enumerate()
424            .filter(|(_, e)| fuzzy_match(&q, &e.haystack()))
425            .map(|(i, _)| i)
426            .collect();
427        self.sel = 0;
428    }
429
430    /// The selected entry, when the filter matched anything.
431    #[must_use]
432    pub fn current(&self) -> Option<&PaletteEntry> {
433        self.matches.get(self.sel).map(|&i| &self.entries[i])
434    }
435
436    /// Move the selection `delta` rows through the matches (wraps).
437    fn move_sel(&mut self, delta: isize) {
438        let n = self.matches.len();
439        if n == 0 {
440            return;
441        }
442        self.sel = (self.sel as isize + delta).rem_euclid(n as isize) as usize;
443    }
444
445    /// Route one key into the session — always consumes (the overlay
446    /// owns the keyboard while open).
447    pub fn key(&mut self, code: KeyCode) -> PaletteCmd {
448        match code {
449            KeyCode::Esc => PaletteCmd::Close,
450            KeyCode::Enter => match self.current() {
451                Some(e) => PaletteCmd::Activate(Box::new(e.clone())),
452                None => PaletteCmd::None,
453            },
454            KeyCode::Up | KeyCode::Char('k') => {
455                self.move_sel(-1);
456                PaletteCmd::None
457            }
458            KeyCode::Down | KeyCode::Char('j') => {
459                self.move_sel(1);
460                PaletteCmd::None
461            }
462            KeyCode::PageUp => {
463                self.move_sel(-(self.matches.len() as isize).max(1));
464                PaletteCmd::None
465            }
466            KeyCode::PageDown => {
467                self.move_sel(self.matches.len() as isize);
468                PaletteCmd::None
469            }
470            KeyCode::Backspace => {
471                self.edit.backspace();
472                self.refilter();
473                PaletteCmd::None
474            }
475            KeyCode::Delete => {
476                self.edit.delete();
477                self.refilter();
478                PaletteCmd::None
479            }
480            KeyCode::Home => {
481                self.edit.home();
482                PaletteCmd::None
483            }
484            KeyCode::End => {
485                self.edit.end();
486                PaletteCmd::None
487            }
488            KeyCode::Left => {
489                self.edit.move_cursor(-1);
490                PaletteCmd::None
491            }
492            KeyCode::Right => {
493                self.edit.move_cursor(1);
494                PaletteCmd::None
495            }
496            KeyCode::Char(c) => {
497                self.edit.insert(c);
498                self.refilter();
499                PaletteCmd::None
500            }
501            _ => PaletteCmd::None,
502        }
503    }
504}

544/// What a type-target keypress asks the app to do.
545#[derive(Clone, Debug)]
546pub enum TypeTargetCmd {
547    /// Key consumed, nothing further.
548    None,
549    /// `Esc` — dismiss without invoking.
550    Close,
551    /// `Enter` — the typed value matched (or any non-empty value when
552    /// no expected id could be derived); proceed.
553    Submit,

556/// The `danger: type-target` modal — "Type &lt;id&gt; to confirm". Proceeds
557/// only on an exact match of the expected id; when the target JSON
558/// carries no derivable id, any non-empty input confirms (the value
559/// is still shown back in the prompt).
560#[derive(Clone, Debug)]
561pub struct TypeTargetSession {
562    /// The invoke waiting on confirmation.
563    pub pending: PendingInvoke,
564    /// The expected id (`None` → any non-empty input accepted).
565    pub expect: Option<String>,
566    /// The input buffer.
567    pub edit: LineEdit,
568    /// Set after a mismatched Enter — shown inline until the next
569    /// keystroke.
570    pub error: Option<String>,
571}
572
573impl TypeTargetSession {
574    /// Open over `pending`; derives `expect` from the target JSON.
575    #[must_use]
576    pub fn new(pending: PendingInvoke) -> Self {
577        let expect = target_id(&pending.target);
578        Self {
579            pending,
580            expect,
581            edit: LineEdit::default(),
582            error: None,
583        }
584    }
585
586    /// Route one key — the modal owns the keyboard while open.
587    pub fn key(&mut self, code: KeyCode) -> TypeTargetCmd {
588        match code {
589            KeyCode::Esc => TypeTargetCmd::Close,
590            KeyCode::Enter => {
591                let typed = self.edit.text.trim();
592                let ok = match &self.expect {
593                    Some(want) => typed == want,
594                    None => !typed.is_empty(),
595                };
596                if ok {
597                    TypeTargetCmd::Submit
598                } else {
599                    self.error = Some(match &self.expect {
600                        Some(want) => format!("type {want} exactly to confirm"),
601                        None => "type something to confirm".to_string(),
602                    });
603                    TypeTargetCmd::None
604                }
605            }
606            KeyCode::Backspace => {
607                self.edit.backspace();
608                self.error = None;
609                TypeTargetCmd::None
610            }
611            KeyCode::Delete => {
612                self.edit.delete();
613                self.error = None;
614                TypeTargetCmd::None
615            }
616            KeyCode::Home => {
617                self.edit.home();
618                TypeTargetCmd::None
619            }
620            KeyCode::End => {
621                self.edit.end();
622                TypeTargetCmd::None
623            }
624            KeyCode::Left => {
625                self.edit.move_cursor(-1);
626                TypeTargetCmd::None
627            }
628            KeyCode::Right => {
629                self.edit.move_cursor(1);
630                TypeTargetCmd::None
631            }
632            KeyCode::Char(c) => {
633                self.edit.insert(c);
634                self.error = None;
635                TypeTargetCmd::None
636            }
637            _ => TypeTargetCmd::None,
638        }
639    }
640}

644/// A [`BindingSource`] for a [`UiAction`] — the `UiAction` analogue
645/// of `pavo_core::commands::action_binding_source` (which takes the
646/// borrowing [`pavo_core::ScopedAction`]). The command is
647/// `action:<id>` locally, or `driver:<drv>:<id>` when the action
648/// declares a cross-driver `driver`.
649#[must_use]
650pub fn ui_binding_source(ui: &UiAction) -> BindingSource<'_> {
651    let command = match &ui.action.driver {
652        Some(driver) => Command::Driver {
653            driver: driver.clone(),
654            id: ui.action.id.clone(),
655        },
656        None => Command::Action {
657            id: ui.action.id.clone(),
658        },
659    };
660    BindingSource {
661        command,
662        label: Cow::Borrowed(ui.action.label.as_str()),
663        detail: ui
664            .action
665            .description
666            .as_deref()
667            .map(Cow::Borrowed)
668            .or(Some(Cow::Borrowed(ui.action.id.as_str()))),
669        section: Some(Cow::Borrowed(match ui.scope {
670            ActionScope::Manifest => "actions",
671            ActionScope::Node => "view",
672            ActionScope::Item => "item",
673        })),
674        keys: ui.action.keys.clone(),
675    }
676}

678/// The host builtins synthesized as palette entries — one
679/// [`MenuEntry`] per [`BUILTINS`] row so they flow through the same
680/// `menu_command` → `run_command` path as driver menus (command =
681/// `host.<name>`, section `"host"`, priority 1000 so driver menus
682/// and actions sort ahead of host chrome).
683///
684/// `host.editor` is included only when `$EDITOR` is set — the TUI
685/// can't hand the terminal to an editor mid-session, so the handler
686/// dumps the view JSON and reports the path instead of spawning.

688pub fn builtin_menu_entries() -> Vec<MenuEntry> {
689    BUILTINS
690        .iter()
691        .filter(|b| b.builtin != HostBuiltin::Editor || std::env::var_os("EDITOR").is_some())
692        .map(|b| MenuEntry {
693            id: b.id.to_string(),
694            title: b.label.to_string(),
695            icon: None,
696            action: None,
697            route: None,
698            command: Some(b.id.to_string()),
699            keys: b.keys.iter().map(|k| k.to_string()).collect(),
700            keywords: vec!["host".to_string(), b.builtin.name().to_string()],
701            section: Some("host".to_string()),
702            priority: 1000,
703        })
704        .collect()
705}

709/// What a filter-mode keypress asks the app to do.
710#[derive(Clone, Debug)]
711pub enum FilterCmd {
712    /// Key consumed, nothing further.
713    None,
714    /// `Esc` — close without applying.
715    Close,
716    /// `Enter` — apply: re-resolve the view with
717    /// `params[param] = text` (empty text clears the param).
718    Submit,
719}
720
721/// The `host.filter` overlay — a [`LineEdit`] bound to the current
722/// view's `filter_param` (convention `"q"`). Submitting re-resolves
723/// the view through the normal `Cmd::Resolve` path with the param
724/// set, so drivers see it as an ordinary view param.
725#[derive(Clone, Debug)]
726pub struct FilterSession {
727    /// The filter text input.
728    pub edit: LineEdit,
729    /// The view param the text binds to (`filter_param` or `"q"`).
730    pub param: String,
731    /// Namespaced `driver:view` being filtered (re-resolve target).
732    pub target: String,
733}
734
735impl FilterSession {
736    /// Open a filter session over `param`/`target`.
737    #[must_use]
738    pub fn new(param: String, target: String) -> Self {
739        Self {
740            edit: LineEdit::default(),
741            param,
742            target,
743        }
744    }
745
746    /// Route one key — the overlay owns the keyboard while open.
747    pub fn key(&mut self, code: KeyCode) -> FilterCmd {
748        match code {
749            KeyCode::Esc => FilterCmd::Close,
750            KeyCode::Enter => FilterCmd::Submit,
751            KeyCode::Backspace => {
752                self.edit.backspace();
753                FilterCmd::None
754            }
755            KeyCode::Delete => {
756                self.edit.delete();
757                FilterCmd::None
758            }
759            KeyCode::Home => {
760                self.edit.home();
761                FilterCmd::None
762            }
763            KeyCode::End => {
764                self.edit.end();
765                FilterCmd::None
766            }
767            KeyCode::Left => {
768                self.edit.move_cursor(-1);
769                FilterCmd::None
770            }
771            KeyCode::Right => {
772                self.edit.move_cursor(1);
773                FilterCmd::None
774            }
775            KeyCode::Char(c) => {
776                self.edit.insert(c);
777                FilterCmd::None
778            }
779            _ => FilterCmd::None,
780        }
781    }
782}
783
784/// Merge a filter submission into the current view params: non-empty
785/// trimmed text sets `params[param]`; empty text *removes* the param
786/// (clears the filter). All other params pass through untouched.
787#[must_use]
788pub fn filter_params(base: &Params, param: &str, text: &str) -> Params {
789    let mut out = base.clone();
790    let trimmed = text.trim();
791    if trimmed.is_empty() {
792        out.remove(param);
793    } else {
794        out.insert(param.to_string(), Value::String(trimmed.to_string()));
795    }
796    out
797}

799// ---------- key event → KeySpec ----------
800
801/// Map a crossterm key event to the [`pavo_core::KeySpec`] string the
802/// binding table matches — `None` for events no spec can name.
803///
804/// Rules:
805///
806/// - `Char(c)` with no ctrl/alt/cmd modifier → the bare char
807///   (case preserved — `"D"` ≠ `"d"`); `SPACE`/`+` map to the named
808///   specs `"space"`/`"plus"`. Shift alone doesn't add a modifier —
809///   the shifted char *is* the spec (`P` binds as `"P"`).
810/// - `Char(c)` with ctrl/alt/cmd → `"ctrl+alt+…+c"` (shift is
811///   ignored — `ctrl+shift+p` isn't representable; bind `"ctrl+p"`).
812/// - Named keys → their canonical lowercase spec (`enter`, `esc`,
813///   `tab`, `backspace`, `delete`, `insert`, `home`, `end`,
814///   `pageup`, `pagedown`, `up`, `down`, `left`, `right`, `f1`–`f12`);
815///   `BackTab` → `"shift+tab"`.
816#[must_use]
817pub fn key_event_spec(code: KeyCode, mods: KeyModifiers) -> Option<String> {
818    use KeyModifiers as M;
819    if let KeyCode::Char(c) = code {
820        let has_chord_mod = mods.intersects(M::CONTROL | M::ALT | M::SUPER | M::HYPER | M::META);
821        if !has_chord_mod {
822            return match c {
823                ' ' => Some("space".to_string()),
824                '+' => Some("plus".to_string()),
825                _ => Some(c.to_string()),
826            };
827        }
828        let mut spec = String::new();
829        if mods.contains(M::CONTROL) {
830            spec.push_str("ctrl+");
831        }
832        if mods.contains(M::ALT) {
833            spec.push_str("alt+");
834        }
835        if mods.contains(M::SUPER) || mods.contains(M::HYPER) || mods.contains(M::META) {
836            spec.push_str("cmd+");
837        }
838        spec.push(c);
839        return Some(spec);
840    }
841    let name = match code {
842        KeyCode::Enter => "enter",
843        KeyCode::Esc => "esc",
844        KeyCode::Tab => "tab",
845        KeyCode::BackTab => return Some("shift+tab".to_string()),
846        KeyCode::Backspace => "backspace",
847        KeyCode::Delete => "delete",
848        KeyCode::Insert => "insert",
849        KeyCode::Home => "home",
850        KeyCode::End => "end",
851        KeyCode::PageUp => "pageup",
852        KeyCode::PageDown => "pagedown",
853        KeyCode::Up => "up",
854        KeyCode::Down => "down",
855        KeyCode::Left => "left",
856        KeyCode::Right => "right",
857        KeyCode::F(n) => return Some(format!("f{n}")),
858        _ => return None,
859    };
860    Some(name.to_string())
861}
