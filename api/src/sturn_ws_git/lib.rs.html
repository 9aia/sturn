1//! Ws impl — reads/lists go straight to the ODB via `sturn-odb`
2//! (loose objects only, plumbing fallback for packs/fancy revs),
3//! `archive | tar` materializes. No index, no checkout, no lock
4//! contention.
5
6use std::io;
7use std::io::Write;
8use std::path::Path;
9use std::process::{Command, Stdio};
10use sturn_ws::Ws;
11
12/// ODB-backed workspaces. `tip` returns the ws ref commit — the WAL
13/// overlay (uncommitted ops visible in `read`) lands in M4.
14pub struct GitWs;
15
16fn git(repo: &Path, args: &[&str]) -> io::Result<String> {
17    let o = Command::new("git")
18        .arg("-C")
19        .arg(repo)
20        .args(args)
21        .stdout(Stdio::piped())
22        .stderr(Stdio::piped())
23        .output()?;
24    if o.status.success() {
25        Ok(String::from_utf8_lossy(&o.stdout).trim_end().to_string())
26    } else {
27        Err(io::Error::other(format!(
28            "git {}: {}",
29            args.join(" "),
30            String::from_utf8_lossy(&o.stderr).trim()
31        )))
32    }
33}
34
35fn ws_ref(name: &str) -> String {
36    format!("refs/sturn/ws/{name}")
37}
38
39impl Ws for GitWs {
40    fn read(&self, repo: &Path, rev: &str, path: &str) -> io::Result<Vec<u8>> {
41        if let Ok(Some(bytes)) = sturn_odb::read_at(repo, rev, path) {
42            return Ok(bytes);
43        }
44        let o = Command::new("git")
45            .arg("-C")
46            .arg(repo)
47            .args(["cat-file", "-p", &format!("{rev}:{path}")])
48            .output()?;
49        if o.status.success() {
50            Ok(o.stdout)
51        } else {
52            Err(io::Error::other(format!("{rev}:{path} not found")))
53        }
54    }
55
56    fn list(&self, repo: &Path, rev: &str) -> io::Result<Vec<String>> {
57        if let Ok(Some(paths)) = sturn_odb::list_tree(repo, rev) {
58            return Ok(paths);
59        }
60        let out = git(repo, &["ls-tree", "-r", "--name-only", rev])?;
61        Ok(out.lines().map(String::from).collect())
62    }
63
64    fn materialize(
65        &self,
66        repo: &Path,
67        rev: &str,
68        dir: &Path,
69        paths: &[String],
70    ) -> io::Result<()> {
71        std::fs::create_dir_all(dir)?;
72        let mut args = vec!["archive", rev];
73        if !paths.is_empty() {
74            args.push("--");
75            args.extend(paths.iter().map(|s| s.as_str()));
76        }
77        let mut arc = Command::new("git")
78            .arg("-C")
79            .arg(repo)
80            .args(&args)
81            .stdout(Stdio::piped())
82            .stderr(Stdio::piped())
83            .spawn()?;
84        let tar = Command::new("tar")
85            .args(["-x", "-C"])
86            .arg(dir)
87            .stdin(Stdio::from(arc.stdout.take().unwrap()))
88            .stderr(Stdio::piped())
89            .spawn()?;
90        let st = arc.wait()?;
91        let to = tar.wait_with_output()?;
92        if st.success() && to.status.success() {
93            Ok(())
94        } else {
95            Err(io::Error::other(format!(
96                "archive|tar failed: {}",
97                String::from_utf8_lossy(&to.stderr).trim()
98            )))
99        }
100    }
101
102    fn create(&self, repo: &Path, name: &str, base: &str) -> io::Result<()> {
103        git(repo, &["update-ref", &ws_ref(name), base])?;
104        Ok(())
105    }
106
107    fn entry(&self, repo: &Path, rev: &str, path: &str) -> io::Result<Option<String>> {
108        if let Ok(Some(sha)) = sturn_odb::entry_oid(repo, rev, path) {
109            return Ok(Some(sha));
110        }
111        let o = Command::new("git")
112            .arg("-C")
113            .arg(repo)
114            .args(["rev-parse", "-q", "--verify", &format!("{rev}:{path}")])
115            .output()?;
116        Ok(o.status
117            .success()
118            .then(|| String::from_utf8_lossy(&o.stdout).trim().to_string())
119            .filter(|s| !s.is_empty()))
120    }
121
122    fn blob(&self, repo: &Path, sha: &str) -> io::Result<Vec<u8>> {
123        if let Ok(Some((kind, bytes))) = sturn_odb::read_object(repo, sha) {
124            if kind == "blob" {
125                return Ok(bytes);
126            }
127        }
128        let o = Command::new("git")
129            .arg("-C")
130            .arg(repo)
131            .args(["cat-file", "-p", sha])
132            .output()?;
133        if o.status.success() {
134            Ok(o.stdout)
135        } else {
136            Err(io::Error::other(format!("blob {sha} not found")))
137        }
138    }
139
140    fn store(&self, repo: &Path, content: &[u8]) -> io::Result<String> {
141        if let Ok(sha) = sturn_odb::write_object(repo, "blob", content) {
142            return Ok(sha);
143        }
144        let mut c = Command::new("git")
145            .arg("-C")
146            .arg(repo)
147            .args(["hash-object", "-w", "--stdin"])
148            .stdin(Stdio::piped())
149            .stdout(Stdio::piped())
150            .stderr(Stdio::piped())
151            .spawn()?;
152        c.stdin.take().unwrap().write_all(content)?;
153        let o = c.wait_with_output()?;
154        if o.status.success() {
155            Ok(String::from_utf8_lossy(&o.stdout).trim().to_string())
156        } else {
157            Err(io::Error::other("hash-object failed"))
158        }
159    }
160
161    fn tip(&self, repo: &Path, name: &str) -> io::Result<String> {
162        if let Ok(Some(sha)) = sturn_odb::resolve(repo, &ws_ref(name)) {
163            return Ok(sha);
164        }
165        git(repo, &["rev-parse", &ws_ref(name)])
166    }
167
168    fn workspaces(&self, repo: &Path) -> io::Result<Vec<String>> {
169        let out = git(
170            repo,
171            &["for-each-ref", "--format=%(refname)", "refs/sturn/ws/"],
172        )?;
173        Ok(out
174            .lines()
175            .filter_map(|l| l.strip_prefix("refs/sturn/ws/").map(String::from))
176            .collect())
177    }
178}
