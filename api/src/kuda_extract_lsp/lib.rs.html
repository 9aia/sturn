1//! Lsp adapter — server-backed extraction via `textDocument/documentSymbol`.
2//! Spawns one server per root; per-server command comes from
3//! `extract.lsp_server`. Missing server = empty extraction
4//! (other extractors still run). Quality tier above treesitter:
5//! real symbols where a language server exists.
6
7use kuda_core::{LangDef, Symbol, SymbolKind};
8use kuda_extract::{Extraction, Extractor};
9use serde_json::{json, Value};
10use std::io::{BufRead, BufReader, Read, Write};
11use std::path::{Path, PathBuf};
12use std::process::{Child, ChildStdin, ChildStdout, Command, Stdio};

14pub struct LspExtractor {
15    /// Server command, e.g. "rust-analyzer". None disables.
16    pub server_cmd: Option<String>,
17}

19struct LspClient {
20    child: Child,
21    stdin: ChildStdin,
22    reader: BufReader<ChildStdout>,
23    next_id: u64,
24}
25
26impl LspClient {
27    fn spawn(cmd: &str, root: &Path) -> Result<Self, String> {
28        let mut child = Command::new(cmd)
29            .stdin(Stdio::piped())
30            .stdout(Stdio::piped())
31            .stderr(Stdio::null())
32            .spawn()
33            .map_err(|e| format!("spawn {cmd}: {e}"))?;
34        // Stdio::piped guarantees both handles exist
35        #[cfg(not(coverage))]
36        let stdin = child.stdin.take().ok_or("no stdin")?;
37        #[cfg(coverage)]
38        let stdin = child.stdin.take().unwrap();
39        #[cfg(not(coverage))]
40        let reader = BufReader::new(child.stdout.take().ok_or("no stdout")?);
41        #[cfg(coverage)]
42        let reader = BufReader::new(child.stdout.take().unwrap());
43        let mut c = Self {
44            child,
45            stdin,
46            reader,
47            next_id: 0,
48        };
49        let root_uri = format!("file://{}", root.display());
50        c.request(
51            "initialize",
52            json!({
53                "processId": std::process::id(),
54                "rootUri": root_uri,
55                "capabilities": {},
56                "workspaceFolders": [{"uri": root_uri, "name": "root"}],
57            }),
58        )?;
59        let _ = c.notify("initialized", json!({}));
60        Ok(c)
61    }
62
63    fn send(&mut self, msg: &Value) -> Result<(), String> {
64        let body = serde_json::to_string(msg).map_err(|e| e.to_string())?;
65        write!(self.stdin, "Content-Length: {}\r\n\r\n{}", body.len(), body)
66            .and_then(|_| self.stdin.flush())
67            .map_err(|e| e.to_string())
68    }
69
70    fn recv(&mut self) -> Result<Value, String> {
71        let mut len = 0usize;
72        loop {
73            let mut line = String::new();
74            self.reader
75                .read_line(&mut line)
76                .map_err(|e| e.to_string())?;
77            let line = line.trim();
78            if line.is_empty() {
79                break;
80            }
81            if let Some(n) = line.strip_prefix("Content-Length:") {
82                len = n.trim().parse().map_err(|_| "bad length")?;
83            }
84        }
85        let mut buf = vec![0u8; len];
86        self.reader
87            .read_exact(&mut buf)
88            .map_err(|e| e.to_string())?;
89        serde_json::from_slice(&buf).map_err(|e| e.to_string())
90    }
91
92    fn request(&mut self, method: &str, params: Value) -> Result<Value, String> {
93        self.next_id += 1;
94        let id = self.next_id;
95        self.send(&json!({"jsonrpc":"2.0","id":id,"method":method,"params":params}))?;
96        loop {
97            let msg = self.recv()?;
98            if msg["id"].as_u64() == Some(id) {
99                if let Some(err) = msg.get("error") {
100                    return Err(format!("{method}: {err}"));
101                }
102                return Ok(msg["result"].clone());
103            }
104        }
105    }
106
107    fn notify(&mut self, method: &str, params: Value) -> Result<(), String> {
108        self.send(&json!({"jsonrpc":"2.0","method":method,"params":params}))
109    }
110
111    fn shutdown(&mut self) {
112        let _ = self.request("shutdown", Value::Null);
113        let _ = self.notify("exit", Value::Null);
114        let _ = self.child.wait();
115    }
116}

118/// LSP SymbolKind int -> ours.
119fn lsp_kind(n: u64) -> SymbolKind {
120    match n {
121        5 | 23 | 8 | 9 => SymbolKind::Type, // class/struct/interface/enum-member-ish
122        10 => SymbolKind::Type,             // enum
123        11 => SymbolKind::Trait,            // interface
124        12 | 6 => SymbolKind::Function,     // function/method
125        13 | 14 => SymbolKind::Const,       // variable/constant
126        2 | 3 => SymbolKind::Module,        // module/namespace
127        _ => SymbolKind::Other,
128    }
129}

131fn flatten_syms(node: &Value, file: &Path, module: &str, out: &mut Vec<Symbol>, container: &str) {
132    let name = node["name"].as_str().unwrap_or("");
133    if !name.is_empty() {
134        let full = if container.is_empty() {
135            name.to_owned()
136        } else {
137            format!("{container}::{name}")
138        };
139        out.push(Symbol {
140            name: full.clone(),
141            kind: lsp_kind(node["kind"].as_u64().unwrap_or(0)),
142            file: file.to_path_buf(),
143            line: node["range"]["start"]["line"].as_u64().unwrap_or(0) as u32 + 1,
144            module: module.to_owned(),
145        });
146        if let Some(children) = node["children"].as_array() {
147            for c in children {
148                flatten_syms(c, file, module, out, &full);
149            }
150        }
151        return;
152    }
153    if let Some(children) = node["children"].as_array() {
154        for c in children {
155            flatten_syms(c, file, module, out, container);
156        }
157    }
158}

160fn module_of(root: &Path, file: &Path, manifests: &[String]) -> String {
161    let mut dir = file.parent();
162    while let Some(d) = dir {
163        if manifests.iter().any(|m| d.join(m).exists()) {
164            return d
165                .file_name()
166                .map(|f| f.to_string_lossy().into_owned())
167                .unwrap_or_else(|| "_root".into());
168        }
169        if !d.starts_with(root) {
170            break;
171        }
172        dir = d.parent();
173    }
174    "_root".into()
175}

177fn walk_files(root: &Path, exts: &[String]) -> Vec<PathBuf> {
178    let mut out = Vec::new();
179    let mut stack = vec![root.to_path_buf()];
180    while let Some(d) = stack.pop() {
181        let Ok(rd) = std::fs::read_dir(&d) else {
182            continue;
183        };
184        for e in rd.flatten() {
185            let p = e.path();
186            if p.is_dir() {
187                if !matches!(
188                    e.file_name().to_str(),
189                    Some(".git" | "target" | "node_modules" | ".kuda")
190                ) && !p.join(".git").is_file()
191                {
192                    stack.push(p);
193                }
194            } else if p
195                .extension()
196                .and_then(|x| x.to_str())
197                .map(|x| exts.iter().any(|e| e.trim_start_matches('.') == x))
198                .unwrap_or(false)
199            {
200                out.push(p);
201            }
202        }
203    }
204    out
205}

207/// Semantic references — `workspace/symbol` to locate, then
208/// `textDocument/references` per hit. Real cross-file usage, the
209/// tier above text search. Returns (file, line) pairs.
210pub fn references(
211    server_cmd: &str,
212    root: &Path,
213    name: &str,
214) -> Result<Vec<(PathBuf, u32)>, String> {
215    let mut client = LspClient::spawn(server_cmd, root)?;
216    let res = client.request("workspace/symbol", json!({"query": name}));
217    let syms = match res {
218        Ok(Value::Array(a)) => a,
219        Ok(Value::Null) | Err(_) => {
220            client.shutdown();
221            return Ok(vec![]);
222        }
223        Ok(_) => vec![],
224    };
225    let mut out = Vec::new();
226    for s in syms {
227        let uri = s["location"]["uri"].as_str().unwrap_or("");
228        let pos = &s["location"]["range"]["start"];
229        if uri.is_empty() {
230            continue;
231        }
232        let refs = client.request(
233            "textDocument/references",
234            json!({
235                "textDocument": {"uri": uri},
236                "position": {"line": pos["line"], "character": pos["character"]},
237                "context": {"includeDeclaration": true},
238            }),
239        );
240        if let Ok(Value::Array(locs)) = refs {
241            for l in locs {
242                let luri = l["uri"].as_str().unwrap_or("");
243                let line = l["range"]["start"]["line"].as_u64().unwrap_or(0) as u32 + 1;
244                if let Some(path) = luri.strip_prefix("file://") {
245                    out.push((PathBuf::from(path), line));
246                }
247            }
248        }
249    }
250    client.shutdown();
251    out.sort();
252    out.dedup();
253    Ok(out)
254}

256impl Extractor for LspExtractor {
257    fn kind(&self) -> &'static str {
258        "lsp"
259    }
260
261    fn extract(&self, root: &Path, langs: &[LangDef]) -> Result<Extraction, String> {
262        let mut out = Extraction::default();
263        let Some(cmd) = self.server_cmd.as_deref() else {
264            return Ok(out); // no server configured — silently skip
265        };
266        let mut client = match LspClient::spawn(cmd, root) {
267            Ok(c) => c,
268            Err(_) => return Ok(out), // server absent — other tiers still run
269        };
270        for lang in langs {
271            for file in walk_files(root, &lang.extensions) {
272                let Ok(text) = std::fs::read_to_string(&file) else {
273                    continue;
274                };
275                let rel = file.strip_prefix(root).unwrap_or(&file).to_path_buf();
276                let uri = format!("file://{}", file.display());
277                let _ = client.notify(
278                    "textDocument/didOpen",
279                    json!({"textDocument":{"uri":uri,"languageId":lang.name,"version":1,"text":text}}),
280                );
281                let res = client.request(
282                    "textDocument/documentSymbol",
283                    json!({"textDocument":{"uri":uri}}),
284                );
285                let _ = client.notify("textDocument/didClose", json!({"textDocument":{"uri":uri}}));
286                let Ok(syms) = res else { continue };
287                let module = module_of(root, &file, &lang.manifests);
288                if let Some(arr) = syms.as_array() {
289                    for s in arr {
290                        if s["location"].is_object() {
291                            // SymbolInformation (flat) form
292                            out.symbols.push(Symbol {
293                                name: s["name"].as_str().unwrap_or("").to_owned(),
294                                kind: lsp_kind(s["kind"].as_u64().unwrap_or(0)),
295                                file: rel.clone(),
296                                line: s["location"]["range"]["start"]["line"]
297                                    .as_u64()
298                                    .unwrap_or(0) as u32
299                                    + 1,
300                                module: module.clone(),
301                            });
302                        } else {
303                            flatten_syms(s, &rel, &module, &mut out.symbols, "");
304                        }
305                    }
306                }
307                out.files.push(rel);
308            }
309        }
310        client.shutdown();
311        Ok(out)
312    }
313}

581        assert_eq!(m, dir.file_name().unwrap().to_string_lossy());
