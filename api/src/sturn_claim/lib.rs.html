1//! Claims — advisory locks folded out of the WAL. Publish before
2//! editing; contenders learn in milliseconds instead of at merge.
3//! Pure functions over the event log: same log, same claims,
4//! deterministic everywhere it's computed.
5//!
6//! Claim records carry a typed resource URI `scheme:path` —
7//! `fs:src/**` globs, `dev:display0`, `net:13000`, `sess:T-012`,
8//! `cap:role`. A bare glob with no scheme is `fs:` — old WALs and
9//! scripts keep working unchanged.
10
11use sturn_core::{Event, Op};
12
13/// A live claim after folding the WAL.
14#[derive(Debug, Clone)]
15pub struct Active {
16    pub ws: String,
17    pub actor: String,
18    /// Canonical resource URI — bare globs carry `fs:` here.
19    pub uri: String,
20    /// Unix seconds — past this, the claim is gone.
21    pub expires: u64,
22    /// Fencing token — a Lamport-style epoch derived from the WAL,
23    /// never stored: `1 + max` fence over earlier claims on
24    /// overlapping URIs. Any later overlapping claim outranks it,
25    /// so a holder compares on `check` to detect a usurper.
26    pub fence: u64,
27}
28
29/// Fold events into live claims at `now`: Claim upserts on
30/// (ws, uri), Release removes, expired TTLs drop. Callers fold
31/// *all* workspaces' WALs for swarm-wide claims.
32pub fn active(events: &[Event], now: u64) -> Vec<Active> {
33    let fences = fences(events);
34    let mut out: Vec<Active> = vec![];
35    for (i, ev) in events.iter().enumerate() {
36        match &ev.op {
37            Op::Claim { glob, ttl_secs } => {
38                let uri = normalize(glob);
39                out.retain(|c| !(c.ws == ev.ws && c.uri == uri));
40                out.push(Active {
41                    ws: ev.ws.clone(),
42                    actor: ev.actor.clone(),
43                    uri,
44                    expires: ev.ts + ttl_secs,
45                    fence: fences[i],
46                });
47            }
48            Op::Release { glob } => {
49                let uri = normalize(glob);
50                out.retain(|c| !(c.ws == ev.ws && c.uri == uri));
51            }
52            // An actor leaving a ws drops all its claims there —
53            // dead agents can't hold the swarm hostage.
54            Op::Presence { status } if status == "left" => {
55                out.retain(|c| !(c.ws == ev.ws && c.actor == ev.actor));
56            }
57            _ => {}
58        }
59    }
60    out.retain(|c| c.expires > now);
61    out
62}

64/// Fencing tokens per Claim event index (0 for non-claims).
65/// Claim order: `ts` first — claims race in wall-clock time —
66/// then fold position. Same-ws ties resolve by WAL order, which
67/// *is* append order; cross-ws ties land on the concatenation
68/// order — deterministic, if arbitrary. Expired and released
69/// claims still count as predecessors: the epoch never repeats
70/// for a resource. Cross-actor clock skew blurs ordering at the
71/// edges — the token is advisory, like the claim itself.
72fn claim_glob(e: &Event) -> &str {
73    match &e.op {
74        Op::Claim { glob, .. } => glob,
75        _ => "",
76    }
77}

79fn fences(events: &[Event]) -> Vec<u64> {
80    let mut order: Vec<usize> = events
81        .iter()
82        .enumerate()
83        .filter(|(_, e)| matches!(e.op, Op::Claim { .. }))
84        .map(|(i, _)| i)
85        .collect();
86    order.sort_by(|&a, &b| (events[a].ts, a).cmp(&(events[b].ts, b)));
87    let mut out = vec![0u64; events.len()];
88    for (n, &i) in order.iter().enumerate() {
89        let uri = normalize(claim_glob(&events[i]));
90        let mut f = 1;
91        for &j in &order[..n] {
92            if overlaps(&uri, &normalize(claim_glob(&events[j]))) {
93                f = f.max(out[j] + 1);
94            }
95        }
96        out[i] = f;
97    }
98    out
99}

101/// URI scheme shape — ALPHA *(ALPHA / DIGIT / "+" / "-" / ".").
102/// Anything else before the first `:` is a path char, not a scheme.
103fn scheme_ok(s: &str) -> bool {
104    s.chars().next().is_some_and(|c| c.is_ascii_alphabetic())
105        && s.chars()
106            .all(|c| c.is_ascii_alphanumeric() || matches!(c, '+' | '-' | '.'))
107}

109/// Split a claim URI into `(scheme, path)`. No valid `scheme:`
110/// prefix → `fs` (a bare glob is a filesystem claim).
111pub fn split(uri: &str) -> (&str, &str) {
112    match uri.find(':') {
113        Some(i) if scheme_ok(&uri[..i]) => (&uri[..i], &uri[i + 1..]),
114        _ => ("fs", uri),
115    }
116}

118/// Canonical URI — bare globs gain `fs:`, schemes lowercase so
119/// `fs:` and `FS:` name the same resource.
120pub fn normalize(uri: &str) -> String {
121    let (s, p) = split(uri);
122    format!("{}:{p}", s.to_ascii_lowercase())
123}

125/// Glob match: `*` = any run within a segment, `**` = any run
126/// across segments, else literal. Recursive, no deps.
127pub fn glob_match(pat: &str, path: &str) -> bool {
128    fn m(p: &[u8], s: &[u8]) -> bool {
129        match p.first() {
130            None => s.is_empty(),
131            Some(b'*') if p.get(1) == Some(&b'*') => {
132                (0..=s.len()).any(|i| m(&p[2..], &s[i..]))
133            }
134            Some(b'*') => {
135                let end = s.iter().position(|&c| c == b'/').unwrap_or(s.len());
136                (0..=end).any(|i| m(&p[1..], &s[i..]))
137            }
138            Some(c) => !s.is_empty() && s[0] == *c && m(&p[1..], &s[1..]),
139        }
140    }
141    m(pat.as_bytes(), path.as_bytes())
142}

144/// Literal stem — the part before the first wildcard.
145fn stem(p: &str) -> &str {
146    match p.find(['*', '?', '[']) {
147        Some(i) => p[..i].trim_end_matches('/'),
148        None => p,
149    }
150}

152/// Do two fs paths/patterns overlap? Both may be globs —
153/// approximate intersection: match each against the other and
154/// against the other's literal stem. Conservative — it may
155/// over-report on exotic globs, and that's the right failure
156/// direction for a lock.
157pub fn covers(a: &str, b: &str) -> bool {
158    glob_match(a, b)
159        || glob_match(b, a)
160        || glob_match(a, stem(b))
161        || glob_match(b, stem(a))
162        || stem(a) == stem(b)
163}

165/// `net:` path as an inclusive port range — `13000` or
166/// `13000-13100`; a bare port is a range of one. Anything that
167/// isn't numeric (subpaths like `13000/tcp`) isn't a range and
168/// falls back to opaque matching.
169fn port_range(p: &str) -> Option<(u16, u16)> {
170    match p.split_once('-') {
171        Some((a, b)) => {
172            let (a, b) = (a.parse().ok()?, b.parse().ok()?);
173            (a <= b).then_some((a, b))
174        }
175        None if !p.is_empty() => p.parse().map(|p| (p, p)).ok(),
176        None => None,
177    }
178}

180/// Opaque-scheme overlap: equality, or the shorter URI is a prefix
181/// of the longer ending at a `:`/`/` boundary — a bare `dev:`
182/// claims the whole scheme while `net:130` stays clear of
183/// `net:13000`.
184fn opaque_covers(a: &str, b: &str) -> bool {
185    if a == b {
186        return true;
187    }
188    let (s, l) = if a.len() <= b.len() { (a, b) } else { (b, a) };
189    l.starts_with(s)
190        && (s.ends_with(':')
191            || s.ends_with('/')
192            || l[s.len()..].starts_with(':')
193            || l[s.len()..].starts_with('/'))
194}

196/// Do two claim URIs overlap? Different schemes never conflict;
197/// `fs` uses glob intersection; `net` uses port-range
198/// intersection when both paths parse, else falls back with every
199/// other (opaque) scheme to exact or boundary-prefix match.
200pub fn overlaps(a: &str, b: &str) -> bool {
201    let (sa, pa) = split(a);
202    let (sb, pb) = split(b);
203    if !sa.eq_ignore_ascii_case(sb) {
204        return false;
205    }
206    if sa.eq_ignore_ascii_case("fs") {
207        covers(pa, pb)
208    } else if sa.eq_ignore_ascii_case("net") {
209        match (port_range(pa), port_range(pb)) {
210            (Some((alo, ahi)), Some((blo, bhi))) => alo <= bhi && blo <= ahi,
211            _ => opaque_covers(&normalize(a), &normalize(b)),
212        }
213    } else {
214        opaque_covers(&normalize(a), &normalize(b))
215    }
216}

218/// Live claims that overlap `uri` — what `sturn claim` warns about.
219/// Overlap is symmetric: an existing `src/**` claim conflicts with
220/// a request for `src/foo.rs`, and vice versa.
221pub fn conflicts<'a>(claims: &'a [Active], uri: &str) -> Vec<&'a Active> {
222    claims.iter().filter(|c| overlaps(&c.uri, uri)).collect()
223}
