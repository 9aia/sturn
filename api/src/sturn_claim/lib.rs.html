1//! Claims — advisory locks folded out of the WAL. Publish before
2//! editing; contenders learn in milliseconds instead of at merge.
3//! Pure functions over the event log: same log, same claims,
4//! deterministic everywhere it's computed.
5
6use sturn_core::{Event, Op};
7
8/// A live claim after folding the WAL.
9#[derive(Debug, Clone)]
10pub struct Active {
11    pub ws: String,
12    pub actor: String,
13    pub glob: String,
14    /// Unix seconds — past this, the claim is gone.
15    pub expires: u64,
16}
17
18/// Fold events into live claims at `now`: Claim upserts on
19/// (ws, glob), Release removes, expired TTLs drop. Callers fold
20/// *all* workspaces' WALs for swarm-wide claims.
21pub fn active(events: &[Event], now: u64) -> Vec<Active> {
22    let mut out: Vec<Active> = vec![];
23    for ev in events {
24        match &ev.op {
25            Op::Claim { glob, ttl_secs } => {
26                out.retain(|c| !(c.ws == ev.ws && c.glob == *glob));
27                out.push(Active {
28                    ws: ev.ws.clone(),
29                    actor: ev.actor.clone(),
30                    glob: glob.clone(),
31                    expires: ev.ts + ttl_secs,
32                });
33            }
34            Op::Release { glob } => {
35                out.retain(|c| !(c.ws == ev.ws && c.glob == *glob));
36            }
37            // An actor leaving a ws drops all its claims there —
38            // dead agents can't hold the swarm hostage.
39            Op::Presence { status } if status == "left" => {
40                out.retain(|c| !(c.ws == ev.ws && c.actor == ev.actor));
41            }
42            _ => {}
43        }
44    }
45    out.retain(|c| c.expires > now);
46    out
47}

49/// Glob match: `*` = any run within a segment, `**` = any run
50/// across segments, else literal. Recursive, no deps.
51pub fn glob_match(pat: &str, path: &str) -> bool {
52    fn m(p: &[u8], s: &[u8]) -> bool {
53        match p.first() {
54            None => s.is_empty(),
55            Some(b'*') if p.get(1) == Some(&b'*') => {
56                (0..=s.len()).any(|i| m(&p[2..], &s[i..]))
57            }
58            Some(b'*') => {
59                let end = s.iter().position(|&c| c == b'/').unwrap_or(s.len());
60                (0..=end).any(|i| m(&p[1..], &s[i..]))
61            }
62            Some(c) => !s.is_empty() && s[0] == *c && m(&p[1..], &s[1..]),
63        }
64    }
65    m(pat.as_bytes(), path.as_bytes())
66}

68/// Literal stem — the part before the first wildcard.
69fn stem(p: &str) -> &str {
70    match p.find(['*', '?', '[']) {
71        Some(i) => p[..i].trim_end_matches('/'),
72        None => p,
73    }
74}

76/// Do two patterns/paths overlap? Both may be globs — approximate
77/// intersection: match each against the other and against the
78/// other's literal stem. Conservative — it may over-report on
79/// exotic globs, and that's the right failure direction for a lock.
80pub fn covers(a: &str, b: &str) -> bool {
81    glob_match(a, b)
82        || glob_match(b, a)
83        || glob_match(a, stem(b))
84        || glob_match(b, stem(a))
85        || stem(a) == stem(b)
86}

88/// Live claims that overlap `glob` — what `sturn claim` warns about.
89/// Overlap is symmetric: an existing `src/**` claim conflicts with a
90/// request for `src/foo.rs`, and vice versa.
91pub fn conflicts<'a>(claims: &'a [Active], glob: &str) -> Vec<&'a Active> {
92    claims.iter().filter(|c| covers(&c.glob, glob)).collect()
93}
